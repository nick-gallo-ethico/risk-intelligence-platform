Note: Full Azure Front Door API integration would require @azure/arm-cdn SDK. For initial implementation, we'll create a service that:

1. Validates domain format
2. Generates required DNS CNAME record
3. Stores domain in TenantBranding
4. Provides verification status

```typescript
import { Injectable, Logger, BadRequestException } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";

export interface CustomDomainConfig {
  domain: string;
  status: "pending" | "active" | "failed";
  cnameRecord: {
    name: string;
    value: string;
  };
  verificationToken: string;
}

@Injectable()
export class DomainRoutingService {
  private readonly logger = new Logger(DomainRoutingService.name);
  private readonly FRONT_DOOR_HOSTNAME =
    process.env.AZURE_FRONTDOOR_HOSTNAME || "ethico.azurefd.net";

  constructor(private readonly prisma: PrismaService) {}

  /**
   * Configure a custom domain for a tenant.
   * Returns the CNAME record that must be configured in DNS.
   */
  async configureDomain(
    organizationId: string,
    domain: string,
  ): Promise<CustomDomainConfig> {
    // Validate domain format
    if (!this.isValidDomain(domain)) {
      throw new BadRequestException("Invalid domain format");
    }

    // Check if domain is already in use by another tenant
    const existing = await this.prisma.tenantBranding.findFirst({
      where: {
        customDomain: domain,
        NOT: { organizationId },
      },
    });

    if (existing) {
      throw new BadRequestException(
        "Domain is already in use by another organization",
      );
    }

    // Generate verification token
    const verificationToken = this.generateVerificationToken(organizationId);

    // Update branding with custom domain
    await this.prisma.tenantBranding.upsert({
      where: { organizationId },
      create: {
        organizationId,
        customDomain: domain,
      },
      update: {
        customDomain: domain,
      },
    });

    this.logger.log(
      `Custom domain configured for org ${organizationId}: ${domain}`,
    );

    return {
      domain,
      status: "pending",
      cnameRecord: {
        name: domain.split(".")[0], // e.g., "ethics" from "ethics.acme.com"
        value: this.FRONT_DOOR_HOSTNAME,
      },
      verificationToken,
    };
  }

  /**
   * Get DNS configuration instructions for a custom domain.
   */
  getDnsInstructions(
    domain: string,
    organizationId: string,
  ): {
    cnameRecord: { name: string; type: "CNAME"; value: string };
    txtRecord: { name: string; type: "TXT"; value: string };
  } {
    const verificationToken = this.generateVerificationToken(organizationId);

    return {
      cnameRecord: {
        name: domain,
        type: "CNAME",
        value: this.FRONT_DOOR_HOSTNAME,
      },
      txtRecord: {
        name: `_dnsauth.${domain}`,
        type: "TXT",
        value: verificationToken,
      },
    };
  }

  /**
   * Verify custom domain DNS configuration.
   * In production, this would perform actual DNS lookup.
   */
  async verifyDomain(
    organizationId: string,
  ): Promise<{ verified: boolean; message: string }> {
    const branding = await this.prisma.tenantBranding.findUnique({
      where: { organizationId },
    });

    if (!branding?.customDomain) {
      throw new BadRequestException("No custom domain configured");
    }

    // In production: Perform DNS lookup to verify CNAME and TXT records
    // For now, return pending status with instructions
    return {
      verified: false,
      message: `Please configure CNAME record pointing ${branding.customDomain} to ${this.FRONT_DOOR_HOSTNAME}. DNS propagation may take up to 48 hours.`,
    };
  }

  /**
   * Remove custom domain configuration.
   */
  async removeDomain(organizationId: string): Promise<void> {
    await this.prisma.tenantBranding.update({
      where: { organizationId },
      data: { customDomain: null },
    });
  }

  /**
   * Get custom domain status for a tenant.
   */
  async getDomainStatus(
    organizationId: string,
  ): Promise<CustomDomainConfig | null> {
    const branding = await this.prisma.tenantBranding.findUnique({
      where: { organizationId },
    });

    if (!branding?.customDomain) {
      return null;
    }

    return {
      domain: branding.customDomain,
      status: "pending", // Would be tracked in a separate field in production
      cnameRecord: {
        name: branding.customDomain,
        value: this.FRONT_DOOR_HOSTNAME,
      },
      verificationToken: this.generateVerificationToken(organizationId),
    };
  }

  private isValidDomain(domain: string): boolean {
    const domainRegex =
      /^(?:[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?\.)+[a-zA-Z]{2,}$/;
    return domainRegex.test(domain);
  }

  private generateVerificationToken(organizationId: string): string {
    // Simple hash for verification token
    const crypto = require("crypto");
    return crypto
      .createHash("sha256")
      .update(`ethico-verify-${organizationId}`)
      .digest("hex")
      .slice(0, 32);
  }
}
```

```typescript
import { DomainRoutingService } from './services/domain-routing.service';

// Inject in constructor

@Post('custom-domain')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN)
async configureCustomDomain(
  @TenantId() orgId: string,
  @Body('domain') domain: string,
) {
  return this.domainRoutingService.configureDomain(orgId, domain);
}

@Get('custom-domain')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN, Role.COMPLIANCE_OFFICER)
async getCustomDomainStatus(@TenantId() orgId: string) {
  return this.domainRoutingService.getDomainStatus(orgId);
}

@Post('custom-domain/verify')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN)
async verifyCustomDomain(@TenantId() orgId: string) {
  return this.domainRoutingService.verifyDomain(orgId);
}

@Delete('custom-domain')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN)
async removeCustomDomain(@TenantId() orgId: string) {
  await this.domainRoutingService.removeDomain(orgId);
  return { success: true };
}
```
