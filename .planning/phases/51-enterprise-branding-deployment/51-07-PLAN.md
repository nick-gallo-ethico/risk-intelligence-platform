```typescript
import { Injectable, Logger, BadRequestException } from "@nestjs/common";
import * as csstree from "css-tree";

@Injectable()
export class CssSanitizerService {
  private readonly logger = new Logger(CssSanitizerService.name);
  private readonly MAX_CSS_LENGTH = 50000;

  // Properties that could be used for security exploits
  private readonly FORBIDDEN_PROPERTIES = new Set([
    "behavior", // IE expression
    "expression", // IE expression
    "-moz-binding", // XUL binding
    "position", // Could overlay UI
    "z-index", // Could overlay UI
    "cursor", // Could confuse users
  ]);

  // At-rules that could load external resources or exfiltrate data
  private readonly FORBIDDEN_AT_RULES = new Set([
    "import", // External CSS loading
    "charset", // Not needed for inline
    "namespace", // Not needed
  ]);

  // Functions that could load external resources
  private readonly FORBIDDEN_FUNCTIONS = new Set([
    "url", // Data exfiltration risk
    "expression", // IE expression (XSS)
    "javascript", // Direct XSS
    "-webkit-image-set", // External images
    "image-set", // External images
  ]);

  sanitize(css: string): string {
    if (!css) return "";

    // Length check
    if (css.length > this.MAX_CSS_LENGTH) {
      throw new BadRequestException(
        `CSS exceeds maximum length of ${this.MAX_CSS_LENGTH} characters`,
      );
    }

    try {
      const ast = csstree.parse(css, {
        parseCustomProperty: true,
        parseValue: true,
      });

      csstree.walk(ast, {
        enter: (node, item, list) => {
          // Remove forbidden at-rules (@import, @charset)
          if (
            node.type === "Atrule" &&
            this.FORBIDDEN_AT_RULES.has(node.name.toLowerCase())
          ) {
            if (list) list.remove(item);
            return;
          }

          // Remove forbidden properties
          if (node.type === "Declaration") {
            if (this.FORBIDDEN_PROPERTIES.has(node.property.toLowerCase())) {
              if (list) list.remove(item);
              return;
            }
          }

          // Remove forbidden functions (url(), etc.)
          if (node.type === "Function") {
            if (this.FORBIDDEN_FUNCTIONS.has(node.name.toLowerCase())) {
              if (list) list.remove(item);
              return;
            }
          }
        },
      });

      const result = csstree.generate(ast);
      this.logger.debug(
        `Sanitized CSS: ${css.length} -> ${result.length} chars`,
      );
      return result;
    } catch (error) {
      this.logger.warn(`CSS parse error: ${error.message}`);
      throw new BadRequestException("Invalid CSS syntax");
    }
  }

  /**
   * Validate CSS without sanitizing - returns issues found
   */
  validate(css: string): { valid: boolean; issues: string[] } {
    const issues: string[] = [];

    if (css.length > this.MAX_CSS_LENGTH) {
      issues.push(
        `CSS exceeds maximum length of ${this.MAX_CSS_LENGTH} characters`,
      );
    }

    // Check for url() functions
    if (/url\s*\(/i.test(css)) {
      issues.push("url() functions are not allowed (external resource risk)");
    }

    // Check for @import
    if (/@import/i.test(css)) {
      issues.push("@import is not allowed (external CSS loading)");
    }

    // Check for expression()
    if (/expression\s*\(/i.test(css)) {
      issues.push("expression() is not allowed (security risk)");
    }

    // Check for javascript: in values
    if (/javascript\s*:/i.test(css)) {
      issues.push("javascript: URLs are not allowed");
    }

    return {
      valid: issues.length === 0,
      issues,
    };
  }
}
```
