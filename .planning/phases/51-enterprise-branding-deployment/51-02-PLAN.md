Create FontUploadService that validates font files using magic bytes:

```typescript
import { Injectable, BadRequestException, Logger } from "@nestjs/common";
import { fileTypeFromBuffer } from "file-type";
import { StorageService } from "../../storage/storage.service";
import { PrismaService } from "../../prisma/prisma.service";
import { CustomFont } from "../types/branding.types";

const ALLOWED_FONT_TYPES = new Map([
  ["font/woff2", ".woff2"],
  ["font/woff", ".woff"],
  ["font/ttf", ".ttf"],
  ["font/otf", ".otf"],
  ["application/font-woff2", ".woff2"],
  ["application/font-woff", ".woff"],
  ["application/x-font-ttf", ".ttf"],
  ["application/vnd.ms-opentype", ".otf"],
]);

@Injectable()
export class FontUploadService {
  private readonly logger = new Logger(FontUploadService.name);
  private readonly MAX_FONT_SIZE = 5 * 1024 * 1024; // 5MB

  constructor(
    private readonly storageService: StorageService,
    private readonly prisma: PrismaService,
  ) {}

  async validateFontFile(buffer: Buffer, filename: string): Promise<void> {
    // Size check
    if (buffer.length > this.MAX_FONT_SIZE) {
      throw new BadRequestException(`Font file exceeds maximum size of 5MB`);
    }

    // Magic byte validation
    const type = await fileTypeFromBuffer(buffer);
    if (!type || !ALLOWED_FONT_TYPES.has(type.mime)) {
      throw new BadRequestException(
        `Invalid font type. Allowed: woff2, woff, ttf, otf`,
      );
    }
  }

  async uploadFont(
    organizationId: string,
    file: Buffer,
    fontFamily: string,
    fontWeight: string = "400",
    fontStyle: string = "normal",
  ): Promise<CustomFont> {
    // Generate safe filename
    const safeFontFamily = fontFamily.replace(/[^a-zA-Z0-9-]/g, "-");
    const path = `fonts/${safeFontFamily}-${fontWeight}-${fontStyle}.woff2`;

    // Upload to storage
    const result = await this.storageService.uploadFile({
      organizationId,
      path,
      buffer: file,
      contentType: "font/woff2",
      metadata: { fontFamily, fontWeight, fontStyle },
    });

    return {
      fontFamily,
      fontWeight,
      fontStyle,
      url: result.url,
    };
  }

  async addFontToBranding(
    organizationId: string,
    font: CustomFont,
  ): Promise<void> {
    const branding = await this.prisma.tenantBranding.findUnique({
      where: { organizationId },
    });

    const existingFonts = (branding?.customFonts as CustomFont[]) || [];
    const updatedFonts = [...existingFonts, font];

    await this.prisma.tenantBranding.upsert({
      where: { organizationId },
      create: {
        organizationId,
        customFonts: updatedFonts,
      },
      update: {
        customFonts: updatedFonts,
      },
    });
  }

  async removeFontFromBranding(
    organizationId: string,
    fontFamily: string,
    fontWeight: string,
    fontStyle: string,
  ): Promise<void> {
    const branding = await this.prisma.tenantBranding.findUnique({
      where: { organizationId },
    });

    const existingFonts = (branding?.customFonts as CustomFont[]) || [];
    const updatedFonts = existingFonts.filter(
      (f) =>
        !(
          f.fontFamily === fontFamily &&
          f.fontWeight === fontWeight &&
          f.fontStyle === fontStyle
        ),
    );

    await this.prisma.tenantBranding.update({
      where: { organizationId },
      data: { customFonts: updatedFonts },
    });
  }
}
```

Add font upload endpoint to BrandingController:

```typescript
import { FileInterceptor } from "@nestjs/platform-express";
import { FontUploadService } from "./services/font-upload.service";

@Controller("api/branding")
export class BrandingController {
  constructor(
    private readonly brandingService: BrandingService,
    private readonly fontUploadService: FontUploadService,
  ) {}

  // ... existing endpoints ...

  @Post("fonts")
  @UseInterceptors(FileInterceptor("file"))
  @UseGuards(JwtAuthGuard, TenantGuard)
  @Roles(Role.SYSTEM_ADMIN, Role.COMPLIANCE_OFFICER)
  async uploadFont(
    @UploadedFile() file: Express.Multer.File,
    @TenantId() orgId: string,
    @Body("fontFamily") fontFamily: string,
    @Body("fontWeight") fontWeight: string = "400",
    @Body("fontStyle") fontStyle: string = "normal",
  ) {
    // Validate
    await this.fontUploadService.validateFontFile(
      file.buffer,
      file.originalname,
    );

    // Upload
    const font = await this.fontUploadService.uploadFont(
      orgId,
      file.buffer,
      fontFamily,
      fontWeight,
      fontStyle,
    );

    // Add to branding
    await this.fontUploadService.addFontToBranding(orgId, font);

    return { success: true, font };
  }

  @Delete("fonts")
  @UseGuards(JwtAuthGuard, TenantGuard)
  @Roles(Role.SYSTEM_ADMIN, Role.COMPLIANCE_OFFICER)
  async removeFont(
    @TenantId() orgId: string,
    @Body("fontFamily") fontFamily: string,
    @Body("fontWeight") fontWeight: string,
    @Body("fontStyle") fontStyle: string,
  ) {
    await this.fontUploadService.removeFontFromBranding(
      orgId,
      fontFamily,
      fontWeight,
      fontStyle,
    );
    return { success: true };
  }
}
```
