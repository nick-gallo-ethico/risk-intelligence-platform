```typescript
import { Injectable, Logger, BadRequestException } from "@nestjs/common";
import { JSDOM } from "jsdom";
import DOMPurify from "dompurify";

@Injectable()
export class HtmlSanitizerService {
  private readonly logger = new Logger(HtmlSanitizerService.name);
  private readonly purify: DOMPurify.DOMPurifyI;
  private readonly MAX_HTML_LENGTH = 10000;

  // Allowed tags for footer content
  private readonly ALLOWED_TAGS = [
    "p",
    "a",
    "span",
    "div",
    "br",
    "hr",
    "strong",
    "b",
    "em",
    "i",
    "u",
    "ul",
    "ol",
    "li",
    "h1",
    "h2",
    "h3",
    "h4",
    "h5",
    "h6",
    "img",
  ];

  // Allowed attributes
  private readonly ALLOWED_ATTR = [
    "href",
    "target",
    "rel",
    "title",
    "alt",
    "src",
    "class",
    "style",
  ];

  constructor() {
    // Initialize DOMPurify with JSDOM for server-side usage
    const window = new JSDOM("").window;
    this.purify = DOMPurify(window as unknown as Window);
  }

  sanitize(html: string): string {
    if (!html) return "";

    // Length check
    if (html.length > this.MAX_HTML_LENGTH) {
      throw new BadRequestException(
        `HTML exceeds maximum length of ${this.MAX_HTML_LENGTH} characters`,
      );
    }

    // Sanitize with DOMPurify
    const clean = this.purify.sanitize(html, {
      ALLOWED_TAGS: this.ALLOWED_TAGS,
      ALLOWED_ATTR: this.ALLOWED_ATTR,
      ALLOW_DATA_ATTR: false,
      // Force all links to open in new tab with noopener
      ADD_ATTR: ["target"],
      FORBID_TAGS: [
        "script",
        "style",
        "iframe",
        "object",
        "embed",
        "form",
        "input",
      ],
      FORBID_ATTR: ["onerror", "onload", "onclick", "onmouseover"],
    });

    // Post-process: ensure all links have target="_blank" rel="noopener noreferrer"
    const processed = clean.replace(/<a\s+([^>]*?)>/gi, (match, attrs) => {
      if (!attrs.includes("target=")) {
        attrs += ' target="_blank"';
      }
      if (!attrs.includes("rel=")) {
        attrs += ' rel="noopener noreferrer"';
      }
      return `<a ${attrs}>`;
    });

    this.logger.debug(
      `Sanitized HTML: ${html.length} -> ${processed.length} chars`,
    );
    return processed;
  }

  /**
   * Validate HTML without sanitizing - for preview purposes
   */
  validate(html: string): { valid: boolean; issues: string[] } {
    const issues: string[] = [];

    if (html.length > this.MAX_HTML_LENGTH) {
      issues.push(
        `HTML exceeds maximum length of ${this.MAX_HTML_LENGTH} characters`,
      );
    }

    // Check for script tags
    if (/<script/i.test(html)) {
      issues.push("Script tags are not allowed");
    }

    // Check for iframe
    if (/<iframe/i.test(html)) {
      issues.push("Iframe tags are not allowed");
    }

    // Check for inline event handlers
    if (/on\w+\s*=/i.test(html)) {
      issues.push("Inline event handlers are not allowed");
    }

    return {
      valid: issues.length === 0,
      issues,
    };
  }
}
```

```tsx
// apps/frontend/src/components/ethics/custom-footer.tsx
"use client";

import { useTenantBranding } from "@/hooks/useTenantBranding";

export function CustomFooter() {
  const { branding, isLoading } = useTenantBranding();

  if (isLoading || !branding?.footerHtml) return null;

  // HTML is pre-sanitized on the server, safe to render
  return (
    <div
      className="custom-footer px-4 py-6 text-sm text-muted-foreground"
      dangerouslySetInnerHTML={{ __html: branding.footerHtml }}
    />
  );
}
```
