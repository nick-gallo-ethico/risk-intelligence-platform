Create seed script that configures enterprise branding for Acme Co. demo tenant:

```typescript
import { PrismaClient } from "@prisma/client";

const prisma = new PrismaClient();

export async function seedPhase51() {
  console.log("Phase 51: Seeding enterprise branding for Acme Co...");

  // Find Acme Co. organization
  const acme = await prisma.organization.findFirst({
    where: { slug: "acme" },
  });

  if (!acme) {
    console.log("Acme Co. not found, skipping Phase 51 seed");
    return;
  }

  // Configure enterprise branding
  await prisma.tenantBranding.upsert({
    where: { organizationId: acme.id },
    create: {
      organizationId: acme.id,
      mode: "FULL_WHITE_LABEL",
      logoUrl: "/demo/acme-logo.svg",
      primaryColor: "210 100% 45%", // Acme blue
      theme: "LIGHT",
      colorPalette: {
        background: "0 0% 100%",
        foreground: "222 47% 11%",
        primary: "210 100% 45%",
        primaryForeground: "210 40% 98%",
        secondary: "210 40% 96%",
        secondaryForeground: "222 47% 11%",
        muted: "210 40% 96%",
        mutedForeground: "215 16% 47%",
        accent: "210 40% 96%",
        accentForeground: "222 47% 11%",
        destructive: "0 84% 60%",
        destructiveForeground: "210 40% 98%",
      },
      typography: {
        fontFamily: '"Inter", -apple-system, BlinkMacSystemFont, sans-serif',
        headingFontFamily:
          '"Inter", -apple-system, BlinkMacSystemFont, sans-serif',
      },
      heroImageUrl: "/demo/acme-hero.jpg",
      showPoweredBy: false, // Enterprise removes Ethico branding
      footerHtml: `
        <div class="text-center">
          <p><strong>Acme Corporation</strong> - Ethics & Compliance</p>
          <p>
            <a href="https://acme.example.com/privacy">Privacy Policy</a> |
            <a href="https://acme.example.com/terms">Terms of Use</a> |
            <a href="mailto:ethics@acme.example.com">Contact Us</a>
          </p>
          <p class="text-xs mt-2">Confidential reporting available 24/7</p>
        </div>
      `,
      customCss: `
        .ethics-portal-header {
          background: linear-gradient(135deg, hsl(210 100% 45%) 0%, hsl(220 100% 35%) 100%);
        }
        .report-form-card {
          border-radius: 12px;
          box-shadow: 0 4px 24px rgba(0, 0, 0, 0.08);
        }
      `,
      customDomain: "ethics.acme.example.com",
      emailSenderDomain: "notifications.acme.example.com",
      emailDomainStatus: "verified",
      customFonts: [
        {
          fontFamily: "Acme Sans",
          fontWeight: "400",
          fontStyle: "normal",
          url: "/demo/fonts/acme-sans-regular.woff2",
        },
        {
          fontFamily: "Acme Sans",
          fontWeight: "700",
          fontStyle: "normal",
          url: "/demo/fonts/acme-sans-bold.woff2",
        },
      ],
    },
    update: {
      // Update with same values for idempotency
      mode: "FULL_WHITE_LABEL",
      showPoweredBy: false,
      heroImageUrl: "/demo/acme-hero.jpg",
      // ... other fields ...
    },
  });

  console.log("Phase 51: Enterprise branding configured for Acme Co.");
}
```
