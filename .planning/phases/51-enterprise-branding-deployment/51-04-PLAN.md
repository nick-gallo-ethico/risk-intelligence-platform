```typescript
import { Injectable, Logger, BadRequestException } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";
import {
  EmailDomainDnsRecords,
  EmailDomainStatus,
} from "../types/branding.types";

@Injectable()
export class EmailDomainService {
  private readonly logger = new Logger(EmailDomainService.name);

  constructor(private readonly prisma: PrismaService) {}

  /**
   * Configure a custom email sender domain for a tenant.
   * Generates DNS records that must be configured by the tenant.
   */
  async configureDomain(
    organizationId: string,
    domain: string,
  ): Promise<{ domain: string; dnsRecords: EmailDomainDnsRecords }> {
    // Validate domain format
    if (!this.isValidDomain(domain)) {
      throw new BadRequestException("Invalid domain format");
    }

    // Generate DNS records
    const dnsRecords = this.generateDnsRecords(domain, organizationId);

    // Update branding with pending status
    await this.prisma.tenantBranding.upsert({
      where: { organizationId },
      create: {
        organizationId,
        emailSenderDomain: domain,
        emailDomainStatus: "pending",
      },
      update: {
        emailSenderDomain: domain,
        emailDomainStatus: "pending",
      },
    });

    this.logger.log(
      `Email domain configured for org ${organizationId}: ${domain}`,
    );

    return { domain, dnsRecords };
  }

  /**
   * Generate required DNS records for email domain authentication.
   */
  generateDnsRecords(
    domain: string,
    organizationId: string,
  ): EmailDomainDnsRecords {
    // Short org ID for DNS records
    const shortOrgId = organizationId.slice(0, 8);

    return {
      spfRecord: {
        type: "TXT",
        name: "@",
        value:
          "v=spf1 include:spf.protection.outlook.com include:spf.ethico.com -all",
      },
      dkimRecords: {
        selector1: {
          type: "CNAME",
          name: `selector1._domainkey`,
          value: `selector1-${shortOrgId}._domainkey.ethico-mail.azurecomm-${shortOrgId}.net`,
        },
        selector2: {
          type: "CNAME",
          name: `selector2._domainkey`,
          value: `selector2-${shortOrgId}._domainkey.ethico-mail.azurecomm-${shortOrgId}.net`,
        },
      },
    };
  }

  /**
   * Verify domain DNS configuration.
   * In production, this would use DNS lookup to verify records.
   * For now, we simulate verification with a status update.
   */
  async verifyDomain(
    organizationId: string,
  ): Promise<{ status: EmailDomainStatus; message: string }> {
    const branding = await this.prisma.tenantBranding.findUnique({
      where: { organizationId },
    });

    if (!branding?.emailSenderDomain) {
      throw new BadRequestException("No email domain configured");
    }

    // In production: Perform actual DNS lookup verification
    // For now: Simulate verification (would be replaced with real DNS checks)
    // const spfValid = await this.verifySpfRecord(branding.emailSenderDomain);
    // const dkimValid = await this.verifyDkimRecords(branding.emailSenderDomain);

    // Update status based on verification
    // For initial implementation, we'll set it to 'verified' after manual check
    // Real implementation would use Azure Communication Services SDK

    return {
      status: (branding.emailDomainStatus as EmailDomainStatus) || "pending",
      message:
        "Domain verification is pending. Please ensure DNS records are configured correctly.",
    };
  }

  /**
   * Get current domain configuration and status.
   */
  async getDomainStatus(organizationId: string): Promise<{
    domain: string | null;
    status: EmailDomainStatus | null;
    dnsRecords: EmailDomainDnsRecords | null;
  }> {
    const branding = await this.prisma.tenantBranding.findUnique({
      where: { organizationId },
      select: { emailSenderDomain: true, emailDomainStatus: true },
    });

    if (!branding?.emailSenderDomain) {
      return { domain: null, status: null, dnsRecords: null };
    }

    return {
      domain: branding.emailSenderDomain,
      status: branding.emailDomainStatus as EmailDomainStatus,
      dnsRecords: this.generateDnsRecords(
        branding.emailSenderDomain,
        organizationId,
      ),
    };
  }

  /**
   * Remove custom email domain configuration.
   */
  async removeDomain(organizationId: string): Promise<void> {
    await this.prisma.tenantBranding.update({
      where: { organizationId },
      data: {
        emailSenderDomain: null,
        emailDomainStatus: null,
      },
    });
  }

  private isValidDomain(domain: string): boolean {
    const domainRegex =
      /^(?:[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?\.)+[a-zA-Z]{2,}$/;
    return domainRegex.test(domain);
  }
}
```

Add email domain endpoints to BrandingController:

```typescript
import { EmailDomainService } from './services/email-domain.service';

// Inject EmailDomainService in constructor

@Post('email-domain')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN)
async configureEmailDomain(
  @TenantId() orgId: string,
  @Body('domain') domain: string,
) {
  const result = await this.emailDomainService.configureDomain(orgId, domain);
  return result;
}

@Get('email-domain')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN, Role.COMPLIANCE_OFFICER)
async getEmailDomainStatus(@TenantId() orgId: string) {
  return this.emailDomainService.getDomainStatus(orgId);
}

@Post('email-domain/verify')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN)
async verifyEmailDomain(@TenantId() orgId: string) {
  return this.emailDomainService.verifyDomain(orgId);
}

@Delete('email-domain')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN)
async removeEmailDomain(@TenantId() orgId: string) {
  await this.emailDomainService.removeDomain(orgId);
  return { success: true };
}
```
