Create PoweredByBadge component that conditionally renders based on branding config:

```tsx
// apps/frontend/src/components/ethics/powered-by-badge.tsx
"use client";

import { useTenantBranding } from "@/hooks/useTenantBranding";

export function PoweredByBadge() {
  const { branding, isLoading } = useTenantBranding();

  // Don't render while loading to avoid flash
  if (isLoading) return null;

  // Don't render if explicitly disabled
  if (branding?.showPoweredBy === false) return null;

  return (
    <div className="flex items-center justify-center py-4 text-sm text-muted-foreground">
      <span>Powered by</span>
      <a
        href="https://www.ethico.com"
        target="_blank"
        rel="noopener noreferrer"
        className="ml-1 font-semibold hover:text-primary transition-colors"
      >
        Ethico
      </a>
    </div>
  );
}
```
