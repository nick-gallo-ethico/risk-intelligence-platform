Add hero image upload endpoint to BrandingController:

```typescript
import { FileInterceptor } from '@nestjs/platform-express';

const ALLOWED_IMAGE_TYPES = ['image/jpeg', 'image/png', 'image/webp'];
const MAX_IMAGE_SIZE = 10 * 1024 * 1024; // 10MB

@Post('hero-image')
@UseInterceptors(FileInterceptor('file'))
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN, Role.COMPLIANCE_OFFICER)
async uploadHeroImage(
  @UploadedFile() file: Express.Multer.File,
  @TenantId() orgId: string,
) {
  // Validate file type
  if (!ALLOWED_IMAGE_TYPES.includes(file.mimetype)) {
    throw new BadRequestException('Invalid image type. Allowed: JPEG, PNG, WebP');
  }

  // Validate file size
  if (file.size > MAX_IMAGE_SIZE) {
    throw new BadRequestException('Image exceeds maximum size of 10MB');
  }

  // Upload to storage
  const result = await this.storageService.uploadFile({
    organizationId: orgId,
    path: 'branding/hero-image' + this.getExtension(file.mimetype),
    buffer: file.buffer,
    contentType: file.mimetype,
  });

  // Update branding config
  await this.brandingService.updateHeroImage(orgId, result.url);

  return { success: true, heroImageUrl: result.url };
}

@Delete('hero-image')
@UseGuards(JwtAuthGuard, TenantGuard)
@Roles(Role.SYSTEM_ADMIN, Role.COMPLIANCE_OFFICER)
async removeHeroImage(@TenantId() orgId: string) {
  await this.brandingService.updateHeroImage(orgId, null);
  return { success: true };
}

private getExtension(mimeType: string): string {
  const map: Record<string, string> = {
    'image/jpeg': '.jpg',
    'image/png': '.png',
    'image/webp': '.webp',
  };
  return map[mimeType] || '.jpg';
}
```

async updateHeroImage(organizationId: string, heroImageUrl: string | null): Promise<void> {
  await this.prisma.tenantBranding.upsert({
    where: { organizationId },
    create: { organizationId, heroImageUrl },
    update: { heroImageUrl },
  });

  // Invalidate cache
  await this.invalidateCache(organizationId);
}
