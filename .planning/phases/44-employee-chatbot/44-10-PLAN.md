import {
  Controller,
  Get,
  Post,
  Put,
  Delete,
  Patch,
  Body,
  Param,
  Query,
  UseGuards,
} from "@nestjs/common";
import { JwtAuthGuard } from "../auth/guards/jwt-auth.guard";
import { RolesGuard } from "../auth/guards/roles.guard";
import { Roles } from "../auth/decorators/roles.decorator";
import { CurrentUser } from "../auth/decorators/current-user.decorator";
import { TenantId } from "../auth/decorators/tenant.decorator";
import { Role } from "../auth/interfaces/role.interface";
import { FaqService } from "./services/faq.service";
import { EscalationService } from "./services/escalation.service";
import {
  CreateFaqDto,
  UpdateFaqDto,
  AssignInquiryDto,
  ResolveInquiryDto,
} from "./dto";
import { FaqStatus } from "@prisma/client";

/**
 * ChatbotController provides REST endpoints for FAQ and inquiry management.
 *
 * Endpoints:
 * - GET /api/chatbot/faq - List FAQ entries
 * - POST /api/chatbot/faq - Create FAQ entry
 * - GET /api/chatbot/faq/:id - Get FAQ entry
 * - PUT /api/chatbot/faq/:id - Update FAQ entry
 * - DELETE /api/chatbot/faq/:id - Archive FAQ entry
 * - POST /api/chatbot/faq/:id/helpful - Mark FAQ as helpful
 *
 * - GET /api/chatbot/inquiries - List pending inquiries
 * - GET /api/chatbot/inquiries/:id - Get inquiry details
 * - PATCH /api/chatbot/inquiries/:id/assign - Assign inquiry
 * - PATCH /api/chatbot/inquiries/:id/resolve - Resolve inquiry
 */
@Controller("api/chatbot")
@UseGuards(JwtAuthGuard, RolesGuard)
export class ChatbotController {
  constructor(
    private readonly faqService: FaqService,
    private readonly escalationService: EscalationService,
  ) {}

  // ==================== FAQ Endpoints ====================

  @Get("faq")
  @Roles(Role.COMPLIANCE_OFFICER, Role.SYSTEM_ADMIN)
  async listFaq(
    @TenantId() organizationId: string,
    @Query("status") status?: FaqStatus,
    @Query("category") category?: string,
    @Query("limit") limit?: string,
    @Query("offset") offset?: string,
  ) {
    return this.faqService.findAll(organizationId, {
      status,
      category,
      limit: limit ? parseInt(limit, 10) : undefined,
      offset: offset ? parseInt(offset, 10) : undefined,
    });
  }

  @Post("faq")
  @Roles(Role.COMPLIANCE_OFFICER, Role.SYSTEM_ADMIN)
  async createFaq(
    @TenantId() organizationId: string,
    @CurrentUser("id") userId: string,
    @Body() dto: CreateFaqDto,
  ) {
    return this.faqService.create(organizationId, userId, dto);
  }

  @Get("faq/:id")
  @Roles(Role.COMPLIANCE_OFFICER, Role.SYSTEM_ADMIN)
  async getFaq(@TenantId() organizationId: string, @Param("id") id: string) {
    return this.faqService.findById(id, organizationId);
  }

  @Put("faq/:id")
  @Roles(Role.COMPLIANCE_OFFICER, Role.SYSTEM_ADMIN)
  async updateFaq(
    @TenantId() organizationId: string,
    @CurrentUser("id") userId: string,
    @Param("id") id: string,
    @Body() dto: UpdateFaqDto,
  ) {
    return this.faqService.update(id, organizationId, userId, dto);
  }

  @Delete("faq/:id")
  @Roles(Role.COMPLIANCE_OFFICER, Role.SYSTEM_ADMIN)
  async archiveFaq(
    @TenantId() organizationId: string,
    @Param("id") id: string,
  ) {
    await this.faqService.archive(id, organizationId);
    return { success: true };
  }

  @Post("faq/:id/helpful")
  async markFaqHelpful(
    @TenantId() organizationId: string,
    @Param("id") id: string,
  ) {
    await this.faqService.markHelpful(id, organizationId);
    return { success: true };
  }

  // ==================== Inquiry Endpoints ====================

  @Get("inquiries")
  @Roles(Role.COMPLIANCE_OFFICER, Role.SYSTEM_ADMIN)
  async listInquiries(
    @TenantId() organizationId: string,
    @Query("limit") limit?: string,
    @Query("offset") offset?: string,
  ) {
    return this.escalationService.getPendingInquiries(organizationId, {
      limit: limit ? parseInt(limit, 10) : undefined,
      offset: offset ? parseInt(offset, 10) : undefined,
    });
  }

  @Patch("inquiries/:id/assign")
  @Roles(Role.COMPLIANCE_OFFICER, Role.SYSTEM_ADMIN)
  async assignInquiry(
    @TenantId() organizationId: string,
    @Param("id") id: string,
    @Body() dto: AssignInquiryDto,
  ) {
    return this.escalationService.assignInquiry(
      id,
      organizationId,
      dto.assigneeId,
    );
  }

  @Patch("inquiries/:id/resolve")
  @Roles(Role.COMPLIANCE_OFFICER, Role.SYSTEM_ADMIN)
  async resolveInquiry(
    @TenantId() organizationId: string,
    @Param("id") id: string,
    @Body() dto: ResolveInquiryDto,
  ) {
    return this.escalationService.resolveInquiry(
      id,
      organizationId,
      dto.resolution,
    );
  }
}

 * - Case status lookup with rate limiting

  imports: [PrismaModule, forwardRef(() => AiModule)],

  exports: [FaqService, ConsentService, CaseStatusService, EscalationService],
