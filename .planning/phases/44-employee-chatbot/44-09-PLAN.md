2. Create a wrapper client component if the layout is a server component, or add directly if it's already a client component:

If layout is a server component, create a client wrapper:

```typescript
// apps/frontend/src/components/employee/employee-chatbot-wrapper.tsx
'use client';

import { ChatbotWidget } from '@/components/chatbot';
import { useAuth } from '@/contexts/auth-context'; // or your auth hook

export function EmployeeChatbotWrapper() {
  const { token } = useAuth(); // Get JWT token from auth context

  if (!token) return null;

  return <ChatbotWidget token={token} />;
}
```

Then in the layout:

```tsx
import { EmployeeChatbotWrapper } from "@/components/employee/employee-chatbot-wrapper";

export default function EmployeePortalLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <>
      {/* Existing layout content */}
      {children}

      {/* Chatbot Widget - authenticated mode */}
      <EmployeeChatbotWrapper />
    </>
  );
}
```

If the layout is already a client component:

```tsx
"use client";

import { ChatbotWidget } from "@/components/chatbot";
import { useAuth } from "@/contexts/auth-context";

export default function EmployeePortalLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  const { token } = useAuth();

  return (
    <>
      {/* Existing layout content */}
      {children}

      {/* Chatbot Widget - authenticated mode */}
      {token && <ChatbotWidget token={token} />}
    </>
  );
}
```
