```prisma
model FaqEntry {
  id               String      @id @default(uuid())
  organizationId   String      @map("organization_id")
  question         String      // The FAQ question (used for text matching)
  questionVector   Unsupported("vector(1536)")? @map("question_vector") // For future semantic matching
  answer           String      // The curated answer
  relatedPolicies  Json?       @map("related_policies") // Array of {policyId, title, section, excerpt}
  category         String?     // Optional grouping category
  tags             String[]    @default([]) // Searchable tags
  priority         Int         @default(0)  // Higher priority shown first in matches
  status           FaqStatus   @default(ACTIVE)
  viewCount        Int         @default(0) @map("view_count")
  helpfulCount     Int         @default(0) @map("helpful_count")
  createdById      String      @map("created_by_id")
  updatedById      String?     @map("updated_by_id")
  createdAt        DateTime    @default(now()) @map("created_at")
  updatedAt        DateTime    @updatedAt @map("updated_at")
  organization     Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)

  @@index([organizationId, status])
  @@index([organizationId, category])
  @@map("faq_entries")
}

enum FaqStatus {
  ACTIVE
  DRAFT
  ARCHIVED
}
```

2. ChatbotConsentLog model (append-only for GDPR):

```prisma
model ChatbotConsentLog {
  id               String   @id @default(uuid())
  organizationId   String   @map("organization_id")
  sessionId        String   @map("session_id")  // Links anonymous session to consent
  consentType      String   @map("consent_type") // AI_USE, DATA_RETENTION, TERMS_OF_USE
  consentVersion   String   @map("consent_version") // Version of consent text shown
  consentTextShown String   @map("consent_text_shown") // Actual text displayed (for audit)
  consentGiven     Boolean  @map("consent_given") // true/false
  ipAddress        String?  @map("ip_address")
  userAgent        String?  @map("user_agent")
  capturedAt       DateTime @default(now()) @map("captured_at")
  // No updatedAt - this is append-only
  organization     Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)

  @@index([organizationId, sessionId])
  @@index([sessionId, capturedAt])
  @@map("chatbot_consent_logs")
}
```

```typescript
import {
  IsString,
  IsOptional,
  IsArray,
  IsInt,
  IsEnum,
  Min,
  MaxLength,
  ValidateNested,
} from "class-validator";
import { Type } from "class-transformer";
import { FaqStatus } from "@prisma/client";

export class RelatedPolicyDto {
  @IsString()
  policyId: string;

  @IsString()
  @MaxLength(255)
  title: string;

  @IsOptional()
  @IsString()
  @MaxLength(255)
  section?: string;

  @IsOptional()
  @IsString()
  @MaxLength(500)
  excerpt?: string;
}

export class CreateFaqDto {
  @IsString()
  @MaxLength(1000)
  question: string;

  @IsString()
  @MaxLength(10000)
  answer: string;

  @IsOptional()
  @IsArray()
  @ValidateNested({ each: true })
  @Type(() => RelatedPolicyDto)
  relatedPolicies?: RelatedPolicyDto[];

  @IsOptional()
  @IsString()
  @MaxLength(100)
  category?: string;

  @IsOptional()
  @IsArray()
  @IsString({ each: true })
  tags?: string[];

  @IsOptional()
  @IsInt()
  @Min(0)
  priority?: number;

  @IsOptional()
  @IsEnum(FaqStatus)
  status?: FaqStatus;
}

export class UpdateFaqDto {
  @IsOptional()
  @IsString()
  @MaxLength(1000)
  question?: string;

  @IsOptional()
  @IsString()
  @MaxLength(10000)
  answer?: string;

  @IsOptional()
  @IsArray()
  @ValidateNested({ each: true })
  @Type(() => RelatedPolicyDto)
  relatedPolicies?: RelatedPolicyDto[];
```
