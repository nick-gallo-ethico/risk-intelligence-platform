```typescript
import { useCallback, useEffect, useRef, useState } from "react";
import { io, Socket } from "socket.io-client";

export interface ChatMessage {
  id: string;
  role: "user" | "assistant";
  content: string;
  timestamp: Date;
  isStreaming?: boolean;
}

export interface ConsentInfo {
  version: string;
  text: string;
  type: string;
}

interface UseChatbotOptions {
  /** Tenant slug for anonymous connection (Ethics Portal) */
  tenantSlug?: string;
  /** JWT token for authenticated connection (Employee Portal) */
  token?: string;
  /** Auto-connect on mount */
  autoConnect?: boolean;
}

interface UseChatbotReturn {
  messages: ChatMessage[];
  isConnected: boolean;
  isStreaming: boolean;
  consentRequired: boolean;
  consentInfo: ConsentInfo | null;
  sessionId: string | null;
  send: (message: string) => void;
  acceptConsent: () => void;
  stop: () => void;
  reset: () => void;
  error: string | null;
}

const CHATBOT_URL = process.env.NEXT_PUBLIC_API_URL || "";

/**
 * Hook for chatbot WebSocket communication.
 *
 * Supports two modes:
 * - Anonymous (tenantSlug): For Ethics Portal, creates session-based connection
 * - Authenticated (token): For Employee Portal, uses JWT auth
 */
export function useChatbot(options: UseChatbotOptions = {}): UseChatbotReturn {
  const { tenantSlug, token, autoConnect = true } = options;

  const socketRef = useRef<Socket | null>(null);
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [isConnected, setIsConnected] = useState(false);
  const [isStreaming, setIsStreaming] = useState(false);
  const [consentRequired, setConsentRequired] = useState(false);
  const [consentInfo, setConsentInfo] = useState<ConsentInfo | null>(null);
  const [sessionId, setSessionId] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const streamingMessageRef = useRef<string>("");

  // Connect to WebSocket
  useEffect(() => {
    if (!autoConnect) return;
    if (!tenantSlug && !token) return;

    const socket = io(`${CHATBOT_URL}/chatbot`, {
      auth: token ? { token } : undefined,
      query: tenantSlug ? { tenant: tenantSlug } : undefined,
      reconnection: true,
      reconnectionAttempts: 5,
      reconnectionDelay: 1000,
    });

    socketRef.current = socket;

    socket.on("connect", () => {
      setIsConnected(true);
      setError(null);
    });

    socket.on("disconnect", () => {
      setIsConnected(false);
    });

    socket.on("connect_error", (err) => {
      setError(`Connection failed: ${err.message}`);
      setIsConnected(false);
    });

    socket.on(
      "message_start",
      (data: { conversationId: string; sessionId: string }) => {
        setSessionId(data.sessionId);
        setIsStreaming(true);
        streamingMessageRef.current = "";

        // Add empty assistant message for streaming
        const newMessage: ChatMessage = {
          id: `msg-${Date.now()}`,
          role: "assistant",
          content: "",
          timestamp: new Date(),
          isStreaming: true,
        };
        setMessages((prev) => [...prev, newMessage]);
      },
    );

    socket.on("text_delta", (data: { text: string }) => {
      streamingMessageRef.current += data.text;

      // Update the last message with new content
      setMessages((prev) => {
        const updated = [...prev];
        const lastIdx = updated.length - 1;
        if (lastIdx >= 0 && updated[lastIdx].isStreaming) {
          updated[lastIdx] = {
            ...updated[lastIdx],
            content: streamingMessageRef.current,
          };
        }
        return updated;
      });
    });

    socket.on("message_complete", () => {
      setIsStreaming(false);
      // Mark message as no longer streaming
      setMessages((prev) => {
        const updated = [...prev];
        const lastIdx = updated.length - 1;
        if (lastIdx >= 0) {
          updated[lastIdx] = {
            ...updated[lastIdx],
            isStreaming: false,
          };
        }
        return updated;
      });
    });

    socket.on(
      "consent_required",
      (data: { sessionId: string; consentText: ConsentInfo }) => {
        setSessionId(data.sessionId);
        setConsentRequired(true);
        setConsentInfo(data.consentText);
      },
    );

    socket.on("consent_accepted", () => {
      setConsentRequired(false);
    });

    socket.on("error", (data: { message: string }) => {
      setError(data.message);
      setIsStreaming(false);
    });

    socket.on("stopped", () => {
      setIsStreaming(false);
    });

    return () => {
      socket.disconnect();
      socketRef.current = null;
    };
  }, [tenantSlug, token, autoConnect]);

  const send = useCallback(
    (message: string) => {
      if (!socketRef.current || !isConnected) return;

      // Add user message immediately
      const userMessage: ChatMessage = {
        id: `msg-${Date.now()}`,
        role: "user",
        content: message,
        timestamp: new Date(),
      };
      setMessages((prev) => [...prev, userMessage]);

      // Clear any previous error
      setError(null);

      // Send to server
      socketRef.current.emit("chat", { message, sessionId });
    },
    [isConnected, sessionId],
  );

  const acceptConsent = useCallback(() => {
    if (!socketRef.current || !consentInfo || !sessionId) return;

    socketRef.current.emit("accept_consent", {
      sessionId,
      consentVersion: consentInfo.version,
      consentTextShown: consentInfo.text,
    });
  }, [consentInfo, sessionId]);

  const stop = useCallback(() => {
    if (socketRef.current && isStreaming) {
      socketRef.current.emit("stop");
    }
  }, [isStreaming]);

  const reset = useCallback(() => {
    setMessages([]);
    setError(null);
    streamingMessageRef.current = "";
  }, []);

  return {
    messages,
    isConnected,
    isStreaming,
    consentRequired,
    consentInfo,
    sessionId,
    send,
    acceptConsent,
    stop,
    reset,
    error,
  };
}
```

  export function ChatbotPanel({
  messages,
  isStreaming,
  isConnected,
  error,
  onSend,
  onStop,
  onClose,
  }: ChatbotPanelProps) {
  const [input, setInput] = useState('');
  const scrollRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLInputElement>(null);

useEffect(() => {

scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
}
}, [messages]);

// Focus input on mount
useEffect(() => {
inputRef.current?.focus();
}, []);

const handleSubmit = (e: React.FormEvent) => {
e.preventDefault();
if (!input.trim() || isStreaming || !isConnected) return;
onSend(input.trim());
setInput('');
};

return (
<div className="flex h-full flex-col">
{/_ Header _/}
<div className="flex items-center justify-between border-b bg-muted/30 px-4 py-3">
<div className="flex items-center gap-2">
<div className="flex h-8 w-8 items-center justify-center rounded-full bg-primary">
<Bot className="h-4 w-4 text-primary-foreground" />

<p className="text-xs text-muted-foreground">
{isConnected ? 'Online' : 'Connecting...'}
</p>
</div>
</div>
<Button variant="ghost" size="icon" onClick={onClose}>
<X className="h-4 w-4" />
</Button>
</div>

      {/* Messages */}
      <ScrollArea className="flex-1 p-4" ref={scrollRef}>
        {messages.length === 0 && (
          <div className="flex flex-col items-center justify-center py-8 text-center">
            <Bot className="mb-4 h-12 w-12 text-muted-foreground/50" />
            <p className="text-sm font-medium text-muted-foreground">

            </p>
            <p className="mt-1 text-xs text-muted-foreground">
              Ask me about policies, disclosures, or check your report status.
            </p>

        )}

        <div className="space-y-4">
          {messages.map((message) => (

              <div

                  message.role === 'user'
                    ? 'bg-primary'
                    : 'bg-muted',

              >
                {message.role === 'user' ? (
                  <User className="h-4 w-4 text-primary-foreground" />

                  <Bot className="h-4 w-4 text-muted-foreground" />
