<objective>
Persist MFA verification status in JWT payload to make it session-bound rather than per-request.

Purpose: SEC-09 - Currently MFA verification is checked per-request or stored in a way that can be bypassed. By adding mfaVerified to the JWT payload and issuing a new token after successful MFA verification, the verification becomes cryptographically bound to the session.

Output: MFA verification persisted in JWT with proper guard enforcement
</objective>

   ```typescript
   /**
    * Verify TOTP code during login.
    * Issues new JWT with mfaVerified: true on success.
    *
    * @param userId - User ID
    * @param code - TOTP code from authenticator app
    * @param currentPayload - Current JWT payload to preserve fields
    * @returns New access token with mfaVerified: true
    */
   async verifyMfaLogin(
     userId: string,
     code: string,
     currentPayload: AccessTokenPayload,
   ): Promise<{ accessToken: string }> {
     const user = await this.prisma.user.findUniqueOrThrow({
       where: { id: userId },
     });

     if (!user.mfaEnabled || !user.mfaSecret) {
       throw new BadRequestException('MFA is not enabled for this user');
     }

     const isValid = this.totp.verify({ token: code, secret: user.mfaSecret });

     if (!isValid) {
       // Log failed attempt for security monitoring
       this.logger.warn(`Failed MFA attempt for user ${userId}`);
       await this.auditService.log({
         action: 'MFA_VERIFY_FAILED',
         category: AuditActionCategory.AUTHENTICATION,
         entityType: 'USER',
         entityId: userId,
         actorType: ActorType.USER,
         actorId: userId,
         organizationId: user.organizationId,
       });
       throw new UnauthorizedException('Invalid verification code');
     }

     // MFA verified - issue new token with mfaVerified: true
     const signingKey = this.jwtKeyService.getSigningKey();

     const newAccessToken = this.jwtService.sign(
       {
         sub: userId,
         email: currentPayload.email,
         organizationId: currentPayload.organizationId,
         role: currentPayload.role,
         sessionId: currentPayload.sessionId,
         type: 'access',
         mfaVerified: true, // SEC-09: Session-bound MFA verification
       },
       {
         secret: signingKey,
         algorithm: 'RS256',
         expiresIn: '15m',
       },
     );

     // Log successful MFA - don't log email, only userId (SEC-13)
     this.logger.log(`MFA verified for user ${userId}`);
     await this.auditService.log({
       action: 'MFA_VERIFY_SUCCESS',
       category: AuditActionCategory.AUTHENTICATION,
       entityType: 'USER',
       entityId: userId,
       actorType: ActorType.USER,
       actorId: userId,
       organizationId: user.organizationId,
     });

     return { accessToken: newAccessToken };
   }
   ```

   ```typescript
   import {
     Injectable,
     CanActivate,
     ExecutionContext,
     UnauthorizedException,
   } from "@nestjs/common";
   import { PrismaService } from "../../prisma/prisma.service";
   import { RequestUser } from "../interfaces/jwt-payload.interface";

   /**
    * Guard that ensures MFA verification for users with MFA enabled.
    *
    * SEC-09: Checks mfaVerified field in JWT payload, which is set to true
    * only after successful TOTP verification. This makes MFA session-bound.
    *
    * Use after JwtAuthGuard:
    * @UseGuards(JwtAuthGuard, MfaGuard)
    */
   @Injectable()
   export class MfaGuard implements CanActivate {
     constructor(private prisma: PrismaService) {}

     async canActivate(context: ExecutionContext): Promise<boolean> {
       const request = context.switchToHttp().getRequest();
       const user: RequestUser = request.user;

       if (!user) {
         throw new UnauthorizedException("Authentication required");
       }

       // Check if user has MFA enabled
       const dbUser = await this.prisma.user.findUnique({
         where: { id: user.id },
         select: { mfaEnabled: true },
       });

       // If MFA is not enabled, allow access
       if (!dbUser?.mfaEnabled) {
         return true;
       }

       // MFA is enabled - check if verified in this session
       if (!user.mfaVerified) {
         throw new UnauthorizedException("MFA verification required");
       }

       return true;
     }
   }
   ```
