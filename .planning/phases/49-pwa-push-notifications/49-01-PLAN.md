    Add the PushSubscription model after the NotificationPreference model (around line 2380). Follow existing patterns for tenant isolation:

    ```prisma
    /// Web Push subscription for browser push notifications
    /// Stores VAPID subscription data with tenant and user scoping
    model PushSubscription {
      id             String   @id @default(uuid())
      organizationId String   @map("organization_id")
      userId         String   @map("user_id")

      /// Web Push endpoint URL (unique per subscription)
      endpoint       String   @unique
      /// P-256 ECDH public key (base64url encoded)
      p256dhKey      String   @map("p256dh_key")
      /// Authentication secret (base64url encoded)
      authKey        String   @map("auth_key")

      /// Device metadata for subscription management
      userAgent      String?  @map("user_agent")
      deviceType     String?  @map("device_type")

      /// Cache version for tenant-scoped cache busting
      cacheVersion   Int      @default(1) @map("cache_version")

      createdAt      DateTime @default(now()) @map("created_at")
      expiresAt      DateTime? @map("expires_at")

      organization   Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
      user           User         @relation(fields: [userId], references: [id], onDelete: Cascade)

      @@unique([organizationId, userId, endpoint])
      @@index([organizationId])
      @@index([userId])
      @@map("push_subscriptions")
    }
    ```

    Also add the relation arrays to User and Organization models:
