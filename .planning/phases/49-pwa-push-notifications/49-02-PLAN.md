    ```typescript
    /**
     * WebPushService - Web Push Notification Delivery
     *
     * Sends push notifications using VAPID authentication via web-push library.
     * Handles subscription expiration (410 responses) by removing stale subscriptions.
     */

    import { Injectable, Logger } from "@nestjs/common";
    import { ConfigService } from "@nestjs/config";
    import * as webpush from "web-push";
    import { PrismaService } from "../../prisma/prisma.service";
    import { PushPayload } from "../entities/notification.types";

    export interface PushResult {
      endpoint: string;
      success: boolean;
      error?: string;
    }

    interface PushSubscriptionData {
      id: string;
      endpoint: string;
      p256dhKey: string;
      authKey: string;
    }

    @Injectable()
    export class WebPushService {
      private readonly logger = new Logger(WebPushService.name);
      private readonly vapidConfigured: boolean;

      constructor(
        private readonly configService: ConfigService,
        private readonly prisma: PrismaService,
      ) {
        const vapidSubject = this.configService.get<string>("VAPID_SUBJECT");
        const vapidPublicKey = this.configService.get<string>("VAPID_PUBLIC_KEY");
        const vapidPrivateKey = this.configService.get<string>("VAPID_PRIVATE_KEY");

        if (vapidSubject && vapidPublicKey && vapidPrivateKey) {
          webpush.setVapidDetails(vapidSubject, vapidPublicKey, vapidPrivateKey);
          this.vapidConfigured = true;
          this.logger.log("VAPID credentials configured for web push");
        } else {
          this.vapidConfigured = false;
          this.logger.warn("VAPID credentials not configured - push notifications disabled");
        }
      }

      /**
       * Check if push notifications are configured and available.
       */
      isConfigured(): boolean {
        return this.vapidConfigured;
      }

      /**
       * Get the public VAPID key for client-side subscription.
       */
      getPublicKey(): string | null {
        return this.configService.get<string>("VAPID_PUBLIC_KEY") || null;
      }

      /**
       * Send push notification to all subscriptions for a user.
       *
       * @param organizationId - Organization ID
       * @param userId - User ID to notify
       * @param payload - Push notification payload
       * @returns Array of results per subscription
       */
      async sendToUser(
        organizationId: string,
        userId: string,
        payload: PushPayload,
      ): Promise<PushResult[]> {
        if (!this.vapidConfigured) {
          this.logger.debug("Push notification skipped - VAPID not configured");
          return [];
        }

        const subscriptions = await this.prisma.pushSubscription.findMany({
          where: { organizationId, userId },
          select: { id: true, endpoint: true, p256dhKey: true, authKey: true },
        });

        if (subscriptions.length === 0) {
          this.logger.debug(`No push subscriptions for user ${userId}`);
          return [];
        }

        const results = await Promise.allSettled(
          subscriptions.map((sub) => this.sendToSubscription(sub, payload)),
        );

        return results.map((result, index) => ({
          endpoint: subscriptions[index].endpoint,
          success: result.status === "fulfilled",
          error: result.status === "rejected" ? result.reason?.message : undefined,
        }));
      }

      /**
       * Send push notification to a single subscription.
       * Removes subscription on 410 (Gone) response.
       */
      private async sendToSubscription(
        subscription: PushSubscriptionData,
        payload: PushPayload,
      ): Promise<void> {
        try {
          await webpush.sendNotification(
            {
              endpoint: subscription.endpoint,
              keys: {
                p256dh: subscription.p256dhKey,
                auth: subscription.authKey,
              },
            },
            JSON.stringify(payload),
            {
              TTL: 86400, // 24 hours
              urgency: payload.urgent ? "high" : "normal",
            },
          );
          this.logger.debug(`Push sent to ${subscription.endpoint.slice(0, 50)}...`);
        } catch (error: any) {
          // Handle expired/invalid subscriptions
          if (error.statusCode === 410 || error.statusCode === 404) {
            await this.removeExpiredSubscription(subscription.id);
            this.logger.log(`Removed expired subscription ${subscription.id}`);
          }
          throw error;
        }
      }

      /**
       * Remove an expired subscription from the database.
       */
      private async removeExpiredSubscription(subscriptionId: string): Promise<void> {
        try {
          await this.prisma.pushSubscription.delete({
            where: { id: subscriptionId },
          });
        } catch (error) {
          this.logger.error(`Failed to remove subscription ${subscriptionId}`, error);
        }
      }
    }
    ```

    ```typescript
    /**
     * PushSubscriptionService - Push Subscription Management
     *
     * Manages Web Push subscriptions per user with tenant isolation.
     * Handles creation, listing, and deletion of subscriptions.
     */

    import { Injectable, Logger, NotFoundException } from "@nestjs/common";
    import { PrismaService } from "../../prisma/prisma.service";
    import { CreatePushSubscriptionDto, PushSubscriptionResponseDto } from "../dto/push-subscription.dto";

    @Injectable()
    export class PushSubscriptionService {
      private readonly logger = new Logger(PushSubscriptionService.name);

      constructor(private readonly prisma: PrismaService) {}

      /**
       * Create or update a push subscription.
       * If endpoint already exists for this user, updates the keys.
       */
      async createOrUpdate(
        organizationId: string,
        userId: string,
        dto: CreatePushSubscriptionDto,
      ): Promise<PushSubscriptionResponseDto> {
        const subscription = await this.prisma.pushSubscription.upsert({
          where: {
            endpoint: dto.endpoint,
          },
          create: {
            organizationId,
            userId,
            endpoint: dto.endpoint,
            p256dhKey: dto.keys.p256dh,
            authKey: dto.keys.auth,
            userAgent: dto.userAgent,
            deviceType: dto.deviceType,
          },
          update: {
            p256dhKey: dto.keys.p256dh,
            authKey: dto.keys.auth,
            userAgent: dto.userAgent,
            deviceType: dto.deviceType,
          },
        });

        this.logger.log(`Push subscription ${subscription.id} created/updated for user ${userId}`);

        return {
          id: subscription.id,
          endpoint: subscription.endpoint,
          deviceType: subscription.deviceType || undefined,
          createdAt: subscription.createdAt,
        };
      }

      /**
       * List all push subscriptions for a user.
       */
      async findAllForUser(
        organizationId: string,
        userId: string,
      ): Promise<PushSubscriptionResponseDto[]> {
        const subscriptions = await this.prisma.pushSubscription.findMany({
          where: { organizationId, userId },
          select: {
            id: true,
            endpoint: true,
            deviceType: true,
            createdAt: true,
          },
          orderBy: { createdAt: "desc" },
        });

        return subscriptions.map((sub) => ({
          id: sub.id,
          endpoint: sub.endpoint,
          deviceType: sub.deviceType || undefined,
          createdAt: sub.createdAt,
        }));
      }

      /**
       * Delete a push subscription.
       */
      async delete(
        organizationId: string,
        userId: string,
        subscriptionId: string,
      ): Promise<void> {
        const subscription = await this.prisma.pushSubscription.findFirst({
          where: { id: subscriptionId, organizationId, userId },
        });

        if (!subscription) {
          throw new NotFoundException("Push subscription not found");
        }

        await this.prisma.pushSubscription.delete({
          where: { id: subscriptionId },
        });

        this.logger.log(`Push subscription ${subscriptionId} deleted`);
      }

      /**
       * Delete all push subscriptions for a user (used on logout).
       */
      async deleteAllForUser(
        organizationId: string,
        userId: string,
      ): Promise<number> {
        const result = await this.prisma.pushSubscription.deleteMany({
          where: { organizationId, userId },
        });

        this.logger.log(`Deleted ${result.count} push subscriptions for user ${userId}`);
        return result.count;
      }
    }
    ```

    ```typescript
    /**
     * PushSubscriptionController - Push Subscription API
     *
     * REST endpoints for managing Web Push subscriptions.
     * All endpoints require authentication and scope to current tenant.
     */

    import {
      Controller,
      Get,
      Post,
      Delete,
      Body,
      Param,
      HttpCode,
      HttpStatus,
      UseGuards,
    } from "@nestjs/common";
    import {
      ApiTags,
      ApiOperation,
      ApiResponse,
      ApiBearerAuth,
    } from "@nestjs/swagger";
    import { JwtAuthGuard } from "../../auth/guards/jwt-auth.guard";
    import { TenantGuard } from "../../auth/guards/tenant.guard";
    import { CurrentUser } from "../../auth/decorators/current-user.decorator";
    import { TenantId } from "../../auth/decorators/tenant-id.decorator";
    import { User } from "@prisma/client";
    import { PushSubscriptionService } from "../services/push-subscription.service";
    import { WebPushService } from "../services/web-push.service";
    import {
      CreatePushSubscriptionDto,
      PushSubscriptionResponseDto,
      VapidPublicKeyResponseDto,
    } from "../dto/push-subscription.dto";

    @ApiTags("Push Subscriptions")
    @Controller("api/v1/push-subscriptions")
    @UseGuards(JwtAuthGuard, TenantGuard)
    @ApiBearerAuth()
    export class PushSubscriptionController {
      constructor(
        private readonly subscriptionService: PushSubscriptionService,
        private readonly webPushService: WebPushService,
      ) {}

      @Get("vapid-public-key")
      @ApiOperation({ summary: "Get VAPID public key for client-side subscription" })
      @ApiResponse({ status: 200, type: VapidPublicKeyResponseDto })
      getVapidPublicKey(): VapidPublicKeyResponseDto {
        return {
          publicKey: this.webPushService.getPublicKey(),
          configured: this.webPushService.isConfigured(),
        };
      }

      @Post()
      @ApiOperation({ summary: "Create or update a push subscription" })
      @ApiResponse({ status: 201, type: PushSubscriptionResponseDto })
      async create(
        @TenantId() organizationId: string,
        @CurrentUser() user: User,
        @Body() dto: CreatePushSubscriptionDto,
      ): Promise<PushSubscriptionResponseDto> {
        return this.subscriptionService.createOrUpdate(organizationId, user.id, dto);
      }

      @Get()
      @ApiOperation({ summary: "List all push subscriptions for current user" })
      @ApiResponse({ status: 200, type: [PushSubscriptionResponseDto] })
      async findAll(
        @TenantId() organizationId: string,
        @CurrentUser() user: User,
      ): Promise<PushSubscriptionResponseDto[]> {
        return this.subscriptionService.findAllForUser(organizationId, user.id);
      }

      @Delete(":id")
      @HttpCode(HttpStatus.NO_CONTENT)
      @ApiOperation({ summary: "Delete a push subscription" })
      @ApiResponse({ status: 204 })
      async delete(
        @TenantId() organizationId: string,
        @CurrentUser() user: User,
        @Param("id") subscriptionId: string,
      ): Promise<void> {
        await this.subscriptionService.delete(organizationId, user.id, subscriptionId);
      }
    }
    ```
