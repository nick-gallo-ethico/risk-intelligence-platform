    /**
     * Dispatches a push notification.
     * Creates notification record and sends via WebPushService.
     *
     * @param params - Push dispatch parameters
     * @returns Created notification ID or null if skipped
     */
    async dispatchPush(params: DispatchPushParams): Promise<string | null> {
      const {
        organizationId,
        recipientUserId,
        category,
        title,
        body,
        entityType,
        entityId,
        url,
        urgent,
      } = params;

      // Check if push is configured
      if (!this.webPushService.isConfigured()) {
        this.logger.debug("Push notification skipped - VAPID not configured");
        return null;
      }

      // Create notification record
      const notification = await this.prisma.notification.create({
        data: {
          organizationId,
          userId: recipientUserId,
          channel: NotificationChannel.PUSH,
          type: category as NotificationType,
          status: NotificationStatus.QUEUED,
          title,
          body,
          entityType,
          entityId,
        },
      });

      // Build push payload
      const pushPayload: PushPayload = {
        title,
        body,
        category,
        entityType,
        entityId,
        url: url || this.buildEntityUrl(entityType, entityId),
        urgent,
        tag: entityType && entityId ? `${entityType}-${entityId}` : undefined,
      };

      // Send push notification
      const results = await this.webPushService.sendToUser(
        organizationId,
        recipientUserId,
        pushPayload,
      );

      // Update notification status based on results
      const anySuccess = results.some((r) => r.success);
      await this.prisma.notification.update({
        where: { id: notification.id },
        data: {
          status: anySuccess
            ? NotificationStatus.DELIVERED
            : NotificationStatus.FAILED,
        },
      });

      if (anySuccess) {
        this.logger.debug(
          `Push notification ${notification.id} delivered to user ${recipientUserId}`,
        );
      } else {
        this.logger.warn(
          `Push notification ${notification.id} failed for user ${recipientUserId}`,
        );
      }

      return notification.id;
    }

    /**
     * Build URL for entity navigation.
     */
    private buildEntityUrl(entityType?: string, entityId?: string): string {
      const baseUrl = process.env.APP_URL || "https://app.ethico.com";

      if (!entityType || !entityId) return baseUrl;

      const routes: Record<string, string> = {
        case: `/cases/${entityId}`,
        investigation: `/investigations/${entityId}`,
        campaign: `/campaigns/${entityId}`,
        policy: `/policies/${entityId}`,
        disclosure: `/disclosures/${entityId}`,
      };

      const path = routes[entityType.toLowerCase()] || "";
      return `${baseUrl}${path}`;
    }
