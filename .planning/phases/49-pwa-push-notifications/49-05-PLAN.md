    /**
     * Custom Service Worker for Ethics Portal
     *
     * This file is injected into the generated service worker by @ducanh2912/next-pwa.
     * It handles:
     * - Push notification display and click handling
     * - Tenant-scoped cache management
     * - Cache clearing on logout
     */

    // ===== PUSH NOTIFICATION HANDLING =====

    /**
     * Handle incoming push notifications.
     * Displays notification using the Notifications API.
     */
    self.addEventListener("push", (event) => {
      if (!event.data) {
        console.log("[SW] Push received but no data");
        return;
      }

      let data;
      try {
        data = event.data.json();
      } catch (e) {
        console.error("[SW] Failed to parse push data:", e);
        return;
      }

      const options = {
        body: data.body || "",
        icon: "/icons/icon-192.png",
        badge: "/icons/icon-72.png",
        // Tag allows replacing notifications for same entity
        tag: data.tag || `notification-${Date.now()}`,
        // renotify: true makes device vibrate/sound even if replacing
        renotify: true,
        // Data passed to notificationclick event
        data: {
          url: data.url || "/",
          entityType: data.entityType,
          entityId: data.entityId,
        },
        // Require interaction for urgent notifications (won't auto-dismiss)
        requireInteraction: data.urgent === true,
        // Vibration pattern for mobile (vibrate, pause, vibrate)
        vibrate: data.urgent ? [200, 100, 200] : [100],
        // Actions for notification buttons (optional)
        actions: data.actions || [],
      };

      // waitUntil keeps the service worker alive until notification shown
      event.waitUntil(self.registration.showNotification(data.title, options));
    });

    /**
     * Handle notification click.
     * Opens the URL associated with the notification.
     */
    self.addEventListener("notificationclick", (event) => {
      event.notification.close();

      const url = event.notification.data?.url || "/";

      // Handle action button clicks if any
      if (event.action) {
        console.log("[SW] Notification action clicked:", event.action);
        // Could handle specific actions here
      }

      // Focus existing window or open new one
      event.waitUntil(
        clients.matchAll({ type: "window", includeUncontrolled: true }).then((windowClients) => {
          // Try to find existing window with same URL
          for (const client of windowClients) {
            if (client.url === url && "focus" in client) {
              return client.focus();
            }
          }

          // Try to find any window from our app
          for (const client of windowClients) {
            if (client.url.startsWith(self.location.origin) && "focus" in client) {
              // Navigate existing window to the URL
              return client.focus().then((focusedClient) => {
                if (focusedClient && "navigate" in focusedClient) {
                  return focusedClient.navigate(url);
                }
              });
            }
          }

          // No existing window, open new one
          return clients.openWindow(url);
        }),
      );
    });

    // ===== TENANT-SCOPED CACHE MANAGEMENT =====

    /**
     * Current tenant ID for cache scoping.
     * Set via postMessage from the main thread.
     */
    let currentTenant = null;

    /**
     * Get tenant-scoped cache name.
     */
    function getTenantCacheName(tenant, cacheType) {
      if (!tenant) {
        return `default-${cacheType}-v1`;
      }
      return `tenant-${tenant}-${cacheType}-v1`;
    }

    /**
     * Handle messages from main thread.
     */
    self.addEventListener("message", (event) => {
      if (!event.data || !event.data.type) return;

      switch (event.data.type) {
        case "SET_TENANT":
          // Store tenant for cache scoping
          currentTenant = event.data.tenant;
          console.log("[SW] Tenant set to:", currentTenant);
          break;

        case "CLEAR_TENANT_CACHE":
          // Clear all caches for a tenant (used on logout)
          const tenant = event.data.tenant;
          if (tenant) {
            event.waitUntil(
              caches.keys().then((cacheNames) => {
                const tenantCaches = cacheNames.filter((name) =>
                  name.startsWith(`tenant-${tenant}`),
                );
                console.log("[SW] Clearing tenant caches:", tenantCaches);
                return Promise.all(
                  tenantCaches.map((cacheName) => caches.delete(cacheName)),
                );
              }),
            );
          }
          break;

        case "CLEAR_ALL_CACHES":
          // Clear all caches (nuclear option)
          event.waitUntil(
            caches.keys().then((cacheNames) => {
              console.log("[SW] Clearing all caches:", cacheNames);
              return Promise.all(
                cacheNames.map((cacheName) => caches.delete(cacheName)),
              );
            }),
          );
          break;

        case "SKIP_WAITING":
          // Activate new service worker immediately
          self.skipWaiting();
          break;
      }
    });

    // ===== SERVICE WORKER LIFECYCLE =====

    /**
     * Claim all clients on activation.
     * Ensures new service worker takes control immediately.
     */
    self.addEventListener("activate", (event) => {
      event.waitUntil(
        Promise.all([
          // Claim all clients
          self.clients.claim(),
          // Clean up old caches (non-tenant-scoped legacy caches)
          caches.keys().then((cacheNames) => {
            return Promise.all(
              cacheNames
                .filter((name) => {
                  // Keep tenant-scoped and workbox caches
                  return (
                    !name.startsWith("tenant-") &&
                    !name.startsWith("workbox-") &&
                    !name.includes("-v1")
                  );
                })
                .map((cacheName) => {
                  console.log("[SW] Deleting old cache:", cacheName);
                  return caches.delete(cacheName);
                }),
            );
          }),
        ]),
      );
    });

    console.log("[SW] Custom service worker loaded");
    ```

    /**
     * Service Worker Bridge
     *
     * Utilities for communicating with the service worker from the main thread.
     * Used for:
     * - Setting tenant context for cache scoping
     * - Clearing tenant caches on logout
     * - Triggering service worker updates
     */

    /**
     * Send a message to the active service worker.
     */
    export async function sendToServiceWorker(
      message: Record<string, unknown>,
    ): Promise<void> {
      if (!("serviceWorker" in navigator)) {
        console.warn("[SW Bridge] Service worker not supported");
        return;
      }

      const registration = await navigator.serviceWorker.ready;
      if (registration.active) {
        registration.active.postMessage(message);
      }
    }

    /**
     * Set the current tenant for service worker cache scoping.
     * Call this after login or tenant switch.
     *
     * @param tenantSlug - The tenant's slug/identifier
     */
    export async function setTenantForServiceWorker(
      tenantSlug: string,
    ): Promise<void> {
      await sendToServiceWorker({
        type: "SET_TENANT",
        tenant: tenantSlug,
      });
    }

    /**
     * Clear all caches for a specific tenant.
     * Call this on logout to prevent data leakage.
     *
     * @param tenantSlug - The tenant's slug/identifier
     */
    export async function clearTenantCache(tenantSlug: string): Promise<void> {
      await sendToServiceWorker({
        type: "CLEAR_TENANT_CACHE",
        tenant: tenantSlug,
      });
    }

    /**
     * Clear all service worker caches.
     * Use sparingly - only for complete cache invalidation scenarios.
     */
    export async function clearAllCaches(): Promise<void> {
      await sendToServiceWorker({
        type: "CLEAR_ALL_CACHES",
      });
    }

    /**
     * Force the waiting service worker to become active.
     * Use after detecting a new service worker version.
     */
    export async function skipWaiting(): Promise<void> {
      await sendToServiceWorker({
        type: "SKIP_WAITING",
      });
    }

    /**
     * Check if a service worker update is available.
     * Returns true if there's a waiting service worker.
     */
    export async function checkForUpdate(): Promise<boolean> {
      if (!("serviceWorker" in navigator)) {
        return false;
      }

      try {
        const registration = await navigator.serviceWorker.getRegistration();
        if (registration) {
          // Check for updates
          await registration.update();
          return registration.waiting !== null;
        }
      } catch (error) {
        console.error("[SW Bridge] Failed to check for update:", error);
      }

      return false;
    }

    /**
     * Register a callback for when a new service worker is available.
     */
    export function onUpdateAvailable(callback: () => void): () => void {
      if (!("serviceWorker" in navigator)) {
        return () => {};
      }

      const handleStateChange = () => {
        navigator.serviceWorker.getRegistration().then((registration) => {
          if (registration?.waiting) {
            callback();
          }
        });
      };

      // Check for updates on load
      navigator.serviceWorker.ready.then((registration) => {
        registration.addEventListener("updatefound", () => {
          const newWorker = registration.installing;
          if (newWorker) {
            newWorker.addEventListener("statechange", handleStateChange);
          }
        });
      });

      // Return cleanup function
      return () => {
        // Cleanup is handled automatically when listeners are garbage collected
      };
    }
    ```
