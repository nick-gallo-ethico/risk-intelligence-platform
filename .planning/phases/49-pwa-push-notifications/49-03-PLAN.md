    /**
     * Push Subscription Client Library
     *
     * Manages Web Push subscription lifecycle:
     * - Permission request with user gesture
     * - Subscription creation via pushManager.subscribe
     * - Backend registration via API
     * - Unsubscription and cleanup
     */

    const API_BASE = "/api/v1/push-subscriptions";

    export type PushPermissionState = "granted" | "denied" | "default" | "unsupported";

    export interface PushSubscriptionStatus {
      supported: boolean;
      permission: PushPermissionState;
      subscribed: boolean;
      configured: boolean;
    }

    /**
     * Convert URL-safe base64 to Uint8Array for applicationServerKey.
     */
    function urlBase64ToUint8Array(base64String: string): Uint8Array {
      const padding = "=".repeat((4 - (base64String.length % 4)) % 4);
      const base64 = (base64String + padding).replace(/-/g, "+").replace(/_/g, "/");
      const rawData = window.atob(base64);
      return Uint8Array.from(rawData, (char) => char.charCodeAt(0));
    }

    /**
     * Convert ArrayBuffer to base64 string for API transport.
     */
    function arrayBufferToBase64(buffer: ArrayBuffer | null): string {
      if (!buffer) return "";
      const bytes = new Uint8Array(buffer);
      let binary = "";
      for (let i = 0; i < bytes.byteLength; i++) {
        binary += String.fromCharCode(bytes[i]);
      }
      return window.btoa(binary);
    }

    /**
     * Check if Push API is supported in this browser.
     */
    export function isPushSupported(): boolean {
      return (
        "serviceWorker" in navigator &&
        "PushManager" in window &&
        "Notification" in window
      );
    }

    /**
     * Get current push permission state.
     */
    export function getPushPermission(): PushPermissionState {
      if (!isPushSupported()) return "unsupported";
      return Notification.permission as PushPermissionState;
    }

    /**
     * Get VAPID public key from backend.
     */
    export async function getVapidPublicKey(): Promise<{
      publicKey: string | null;
      configured: boolean;
    }> {
      const response = await fetch(`${API_BASE}/vapid-public-key`, {
        credentials: "include",
      });

      if (!response.ok) {
        throw new Error("Failed to get VAPID public key");
      }

      return response.json();
    }

    /**
     * Get comprehensive push subscription status.
     */
    export async function getSubscriptionStatus(): Promise<PushSubscriptionStatus> {
      if (!isPushSupported()) {
        return {
          supported: false,
          permission: "unsupported",
          subscribed: false,
          configured: false,
        };
      }

      const permission = getPushPermission();

      try {
        const vapidInfo = await getVapidPublicKey();

        // Check if we have an active subscription
        const registration = await navigator.serviceWorker.ready;
        const subscription = await registration.pushManager.getSubscription();

        return {
          supported: true,
          permission,
          subscribed: subscription !== null,
          configured: vapidInfo.configured,
        };
      } catch {
        return {
          supported: true,
          permission,
          subscribed: false,
          configured: false,
        };
      }
    }

    /**
     * Subscribe to push notifications.
     * MUST be called from a user gesture (click handler).
     *
     * @returns true if subscription succeeded, false otherwise
     */
    export async function subscribeToPush(): Promise<boolean> {
      if (!isPushSupported()) {
        console.warn("Push notifications not supported");
        return false;
      }

      // Request notification permission
      const permission = await Notification.requestPermission();
      if (permission !== "granted") {
        console.log("Push permission denied");
        return false;
      }

      try {
        // Get VAPID public key from backend
        const { publicKey, configured } = await getVapidPublicKey();
        if (!configured || !publicKey) {
          console.warn("Push notifications not configured on server");
          return false;
        }

        // Get service worker registration
        const registration = await navigator.serviceWorker.ready;

        // Subscribe with VAPID key
        const subscription = await registration.pushManager.subscribe({
          userVisibleOnly: true,
          applicationServerKey: urlBase64ToUint8Array(publicKey),
        });

        // Register subscription with backend
        const response = await fetch(API_BASE, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          credentials: "include",
          body: JSON.stringify({
            endpoint: subscription.endpoint,
            keys: {
              p256dh: arrayBufferToBase64(subscription.getKey("p256dh")),
              auth: arrayBufferToBase64(subscription.getKey("auth")),
            },
            userAgent: navigator.userAgent,
            deviceType: detectDeviceType(),
          }),
        });

        if (!response.ok) {
          throw new Error("Failed to register subscription with backend");
        }

        console.log("Push subscription created successfully");
        return true;
      } catch (error) {
        console.error("Failed to subscribe to push:", error);
        return false;
      }
    }

    /**
     * Unsubscribe from push notifications.
     */
    export async function unsubscribeFromPush(): Promise<boolean> {
      if (!isPushSupported()) return false;

      try {
        const registration = await navigator.serviceWorker.ready;
        const subscription = await registration.pushManager.getSubscription();

        if (subscription) {
          // Unsubscribe locally
          await subscription.unsubscribe();

          // Note: Backend subscription will be removed on next failed push (410)
          // or can be explicitly deleted via API if we stored the subscription ID
        }

        return true;
      } catch (error) {
        console.error("Failed to unsubscribe from push:", error);
        return false;
      }
    }

    /**
     * Detect device type based on user agent.
     */
    function detectDeviceType(): "mobile" | "tablet" | "desktop" {
      const ua = navigator.userAgent.toLowerCase();

      if (/mobile|iphone|ipod|android.*mobile|windows phone/i.test(ua)) {
        return "mobile";
      }
      if (/ipad|android(?!.*mobile)|tablet/i.test(ua)) {
        return "tablet";
      }
      return "desktop";
    }

    "use client";

    import { useState, useEffect, useCallback } from "react";
    import {
      isPushSupported,
      getPushPermission,
      getSubscriptionStatus,
      subscribeToPush,
      unsubscribeFromPush,
      type PushSubscriptionStatus,
      type PushPermissionState,
    } from "@/lib/push-subscription";

    export interface UsePushSubscriptionResult {
      /** Whether push is supported in this browser */
      supported: boolean;
      /** Current permission state */
      permission: PushPermissionState;
      /** Whether user has an active subscription */
      subscribed: boolean;
      /** Whether push is configured on server */
      configured: boolean;
      /** Whether status is being loaded */
      loading: boolean;
      /** Subscribe to push notifications (call from user gesture) */
      subscribe: () => Promise<boolean>;
      /** Unsubscribe from push notifications */
      unsubscribe: () => Promise<boolean>;
      /** Refresh subscription status */
      refresh: () => Promise<void>;
    }

    /**
     * React hook for managing push notification subscription.
     *
     * Usage:
     * ```tsx
     * const { supported, subscribed, subscribe, unsubscribe } = usePushSubscription();
     *
     * <button onClick={subscribe} disabled={!supported || subscribed}>
     *   Enable notifications
     * </button>
     * ```
     */
    export function usePushSubscription(): UsePushSubscriptionResult {
      const [status, setStatus] = useState<PushSubscriptionStatus>({
        supported: false,
        permission: "default",
        subscribed: false,
        configured: false,
      });
      const [loading, setLoading] = useState(true);

      const refresh = useCallback(async () => {
        setLoading(true);
        try {
          const newStatus = await getSubscriptionStatus();
          setStatus(newStatus);
        } catch (error) {
          console.error("Failed to get push subscription status:", error);
        } finally {
          setLoading(false);
        }
      }, []);

      useEffect(() => {
        // Quick check for basic support
        if (!isPushSupported()) {
          setStatus((prev) => ({ ...prev, supported: false }));
          setLoading(false);
          return;
        }

        // Initial permission check (synchronous)
        setStatus((prev) => ({
          ...prev,
          supported: true,
          permission: getPushPermission(),
        }));

        // Full status check (async)
        refresh();
      }, [refresh]);

      const subscribe = useCallback(async (): Promise<boolean> => {
        const success = await subscribeToPush();
        if (success) {
          await refresh();
        }
        return success;
      }, [refresh]);

      const unsubscribe = useCallback(async (): Promise<boolean> => {
        const success = await unsubscribeFromPush();
        if (success) {
          await refresh();
        }
        return success;
      }, [refresh]);

      return {
        supported: status.supported,
        permission: status.permission,
        subscribed: status.subscribed,
        configured: status.configured,
        loading,
        subscribe,
        unsubscribe,
        refresh,
      };
    }
    ```

    "use client";

    import { useState } from "react";
    import { Bell, BellOff, X, Loader2 } from "lucide-react";
    import { Button } from "@/components/ui/button";
    import {
      Card,
      CardContent,
      CardDescription,
      CardFooter,
      CardHeader,
      CardTitle,
    } from "@/components/ui/card";
    import { usePushSubscription } from "@/hooks/use-push-subscription";
    import { cn } from "@/lib/utils";

    export interface PushPermissionPromptProps {
      /** When to show the prompt */
      trigger?: "auto" | "manual";
      /** Custom title */
      title?: string;
      /** Custom description */
      description?: string;
      /** Called when user subscribes successfully */
      onSubscribed?: () => void;
      /** Called when user dismisses the prompt */
      onDismissed?: () => void;
      /** Custom class name */
      className?: string;
    }

    /**
     * Push notification permission prompt component.
     *
     * Shows a card prompting the user to enable push notifications.
     * Only shows when:
     * - Push is supported
     * - Push is configured on server
     * - User hasn't subscribed yet
     * - Permission is not denied
     *
     * Best practice: Show after a meaningful user action (e.g., submitting a report)
     * rather than on page load.
     */
    export function PushPermissionPrompt({
      trigger = "manual",
      title = "Stay informed",
      description = "Get notified about case updates, SLA warnings, and campaign reminders.",
      onSubscribed,
      onDismissed,
      className,
    }: PushPermissionPromptProps) {
      const {
        supported,
        permission,
        subscribed,
        configured,
        loading,
        subscribe,
      } = usePushSubscription();

      const [dismissed, setDismissed] = useState(false);
      const [subscribing, setSubscribing] = useState(false);

      // Don't show if:
      // - Still loading
      // - Not supported
      // - Not configured
      // - Already subscribed
      // - Permission denied
      // - User dismissed
      if (
        loading ||
        !supported ||
        !configured ||
        subscribed ||
        permission === "denied" ||
        dismissed
      ) {
        return null;
      }

      const handleSubscribe = async () => {
        setSubscribing(true);
        try {
          const success = await subscribe();
          if (success) {
            onSubscribed?.();
          }
        } finally {
          setSubscribing(false);
        }
      };

      const handleDismiss = () => {
        setDismissed(true);
        onDismissed?.();
      };

      return (
        <Card className={cn("relative", className)}>
          <Button
            variant="ghost"
            size="icon"
            className="absolute right-2 top-2 h-6 w-6"
            onClick={handleDismiss}
            aria-label="Dismiss"
          >
            <X className="h-4 w-4" />
          </Button>

          <CardHeader className="pb-3">
            <div className="flex items-center gap-2">
              <Bell className="h-5 w-5 text-primary" />
              <CardTitle className="text-lg">{title}</CardTitle>
            </div>
            <CardDescription>{description}</CardDescription>
          </CardHeader>

          <CardFooter className="gap-2">
            <Button
              variant="default"
              size="sm"
              onClick={handleSubscribe}
              disabled={subscribing}
            >
              {subscribing ? (
                <>
                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
                  Enabling...
                </>
              ) : (
                <>
                  <Bell className="mr-2 h-4 w-4" />
                  Enable notifications
                </>
              )}
            </Button>
            <Button variant="ghost" size="sm" onClick={handleDismiss}>
              Not now
            </Button>
          </CardFooter>
        </Card>
      );
    }

    /**
     * Minimal push notification toggle for settings pages.
     */
    export function PushNotificationToggle({
      className,
    }: {
      className?: string;
    }) {
      const {
        supported,
        permission,
        subscribed,
        configured,
        loading,
        subscribe,
        unsubscribe,
      } = usePushSubscription();

      const [toggling, setToggling] = useState(false);

      if (!supported) {
        return (
          <div className={cn("flex items-center gap-2 text-muted-foreground", className)}>
            <BellOff className="h-4 w-4" />
            <span className="text-sm">Push notifications not supported in this browser</span>
          </div>
        );
      }

      if (!configured) {
        return (
          <div className={cn("flex items-center gap-2 text-muted-foreground", className)}>
            <BellOff className="h-4 w-4" />
            <span className="text-sm">Push notifications not configured</span>
          </div>
        );
      }

      if (permission === "denied") {
        return (
          <div className={cn("flex items-center gap-2 text-muted-foreground", className)}>
            <BellOff className="h-4 w-4" />
            <span className="text-sm">
              Push notifications blocked. Enable in browser settings.
            </span>
          </div>
        );
      }

      const handleToggle = async () => {
        setToggling(true);
        try {
          if (subscribed) {
            await unsubscribe();
          } else {
            await subscribe();
          }
        } finally {
          setToggling(false);
        }
      };

      return (
        <Button
          variant={subscribed ? "outline" : "default"}
          size="sm"
          onClick={handleToggle}
          disabled={loading || toggling}
          className={className}
        >
          {loading || toggling ? (
            <Loader2 className="mr-2 h-4 w-4 animate-spin" />
          ) : subscribed ? (
            <BellOff className="mr-2 h-4 w-4" />
          ) : (
            <Bell className="mr-2 h-4 w-4" />
          )}
          {subscribed ? "Disable push" : "Enable push"}
        </Button>
      );
    }
