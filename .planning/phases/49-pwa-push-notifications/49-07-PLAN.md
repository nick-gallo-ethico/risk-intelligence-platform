          findMany: jest.fn(),

          providers: [
            WebPushService,
            { provide: ConfigService, useValue: mockConfigService },
            { provide: PrismaService, useValue: mockPrismaService },
          ],
        }).compile();

        service = module.get<WebPushService>(WebPushService);

        expect(service.getPublicKey()).toBe("test-public-key");

    ```typescript
    import { Test, TestingModule } from "@nestjs/testing";
    import { NotFoundException } from "@nestjs/common";
    import { PushSubscriptionService } from "./push-subscription.service";
    import { PrismaService } from "../../prisma/prisma.service";

    describe("PushSubscriptionService", () => {
      let service: PushSubscriptionService;

      const mockPrismaService = {
        pushSubscription: {
          upsert: jest.fn(),
          findMany: jest.fn(),
          findFirst: jest.fn(),
          delete: jest.fn(),
          deleteMany: jest.fn(),
        },
      };

      beforeEach(async () => {
        jest.clearAllMocks();

        const module: TestingModule = await Test.createTestingModule({
          providers: [
            PushSubscriptionService,
            { provide: PrismaService, useValue: mockPrismaService },
          ],
        }).compile();

        service = module.get<PushSubscriptionService>(PushSubscriptionService);
      });

      it("should be defined", () => {
        expect(service).toBeDefined();
      });

      describe("createOrUpdate", () => {
        it("should create new subscription", async () => {
          const mockSubscription = {
            id: "sub-1",
            endpoint: "https://push.example.com/1",
            deviceType: "desktop",
            createdAt: new Date(),
          };

          mockPrismaService.pushSubscription.upsert.mockResolvedValue(mockSubscription);

          const result = await service.createOrUpdate("org-1", "user-1", {
            endpoint: "https://push.example.com/1",
            keys: { p256dh: "key", auth: "auth" },
            deviceType: "desktop",
          });

          expect(result.id).toBe("sub-1");
          expect(result.endpoint).toBe("https://push.example.com/1");
        });
      });

      describe("findAllForUser", () => {
        it("should return all subscriptions for user", async () => {
          const mockSubscriptions = [
            { id: "sub-1", endpoint: "https://push.example.com/1", deviceType: "desktop", createdAt: new Date() },
            { id: "sub-2", endpoint: "https://push.example.com/2", deviceType: "mobile", createdAt: new Date() },
          ];

          mockPrismaService.pushSubscription.findMany.mockResolvedValue(mockSubscriptions);

          const results = await service.findAllForUser("org-1", "user-1");

          expect(results).toHaveLength(2);
        });
      });

      describe("delete", () => {
        it("should delete subscription when found", async () => {
          mockPrismaService.pushSubscription.findFirst.mockResolvedValue({ id: "sub-1" });
          mockPrismaService.pushSubscription.delete.mockResolvedValue({});

          await expect(service.delete("org-1", "user-1", "sub-1")).resolves.not.toThrow();
          expect(mockPrismaService.pushSubscription.delete).toHaveBeenCalledWith({
            where: { id: "sub-1" },
          });
        });

        it("should throw NotFoundException when subscription not found", async () => {
          mockPrismaService.pushSubscription.findFirst.mockResolvedValue(null);

          await expect(service.delete("org-1", "user-1", "sub-1")).rejects.toThrow(
            NotFoundException,
          );
        });
      });

      describe("deleteAllForUser", () => {
        it("should delete all subscriptions for user", async () => {
          mockPrismaService.pushSubscription.deleteMany.mockResolvedValue({ count: 3 });

          const count = await service.deleteAllForUser("org-1", "user-1");

          expect(count).toBe(3);
        });
      });
    });
    ```
