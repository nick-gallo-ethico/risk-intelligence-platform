    Add a function to clear all data for a specific tenant. Add after the existing `cleanupExpiredDrafts` function:

    ```typescript
    /**
     * Clear all offline data for a specific tenant.
     * Call this on logout to ensure tenant isolation.
     *
     * @param tenantSlug - The tenant slug to clear data for
     * @returns Number of records deleted
     */
    export async function clearTenantOfflineData(tenantSlug: string): Promise<number> {
      await db.initEncryption();

      const [draftsDeleted, submissionsDeleted] = await Promise.all([
        db.drafts.where("tenantSlug").equals(tenantSlug).delete(),
        db.pendingSubmissions.where("tenantSlug").equals(tenantSlug).delete(),
      ]);

      return draftsDeleted + submissionsDeleted;
    }

    /**
     * Clear all offline data (all tenants).
     * Use sparingly - only for complete data wipe scenarios.
     *
     * @returns Number of records deleted
     */
    export async function clearAllOfflineData(): Promise<number> {
      await db.initEncryption();

      const [draftsDeleted, submissionsDeleted] = await Promise.all([
        db.drafts.clear().then(() => db.drafts.count()),
        db.pendingSubmissions.clear().then(() => db.pendingSubmissions.count()),
      ]);

      // clear() doesn't return count, but we can return 0 to indicate success
      return 0;
    }
    ```

    Also export the db instance for direct access if needed (it may already be exported):
    ```typescript

    ```
