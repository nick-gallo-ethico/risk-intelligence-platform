```typescript
import { PrismaClient, Severity } from "@prisma/client";
import { addDays, subDays, subHours } from "date-fns";

export async function seedPhase41(prisma: PrismaClient): Promise<void> {
  console.log("Seeding Phase 41: SLA Monitoring & Escalation...");

  // Find Acme Co organization
  const acme = await prisma.organization.findFirst({
    where: { name: { contains: "Acme" } },
  });

  if (!acme) {
    console.log("Acme organization not found, skipping Phase 41 seed");
    return;
  }

  // 1. Configure SLA settings for Acme
  await prisma.organization.update({
    where: { id: acme.id },
    data: {
      caseSlaConfig: {
        enabled: true,
        defaultDays: 14,
        warningThresholdPercent: 80,
        criticalThresholdHours: 48,
        severityOverrides: {
          HIGH: 7,
          MEDIUM: 14,
          LOW: 30,
        },
      },
    },
  });
  console.log("  - Configured Acme SLA settings");

  // 2. Create cases at various SLA states for demo
  const systemUser = await prisma.user.findFirst({
    where: { organizationId: acme.id, role: "SYSTEM_ADMIN" },
  });

  if (!systemUser) {
    console.log("System user not found, skipping case creation");
    return;
  }

  // Case approaching SLA warning (created 10 days ago for 14-day SLA = ~71%)
  const warningCase = await prisma.case.create({
    data: {
      organizationId: acme.id,
      referenceNumber: `SLA-WARN-${Date.now()}`,
      status: "IN_PROGRESS",
      sourceChannel: "WEB_FORM",
      caseType: "REPORT",
      reporterType: "ANONYMOUS",
      severity: Severity.MEDIUM,
      details: "Demo case approaching SLA warning threshold",
      createdAt: subDays(new Date(), 10),
      createdById: systemUser.id,
      updatedById: systemUser.id,
    },
  });
  console.log(`  - Created warning-state case: ${warningCase.referenceNumber}`);

  // Case past SLA (created 20 days ago for 14-day SLA = breached)
  const breachedCase = await prisma.case.create({
    data: {
      organizationId: acme.id,
      referenceNumber: `SLA-BREACH-${Date.now()}`,
      status: "IN_PROGRESS",
      sourceChannel: "PHONE",
      caseType: "REPORT",
      reporterType: "CONFIDENTIAL",
      severity: Severity.HIGH,
      details: "Demo case that has breached SLA",
      createdAt: subDays(new Date(), 20),
      slaDueDate: subDays(new Date(), 13), // Due 13 days ago
      slaState: {
        lastStatus: "breached",
        lastNotifiedAt: subDays(new Date(), 5).toISOString(),
        lastNotificationType: "breach",
      },
      createdById: systemUser.id,
      updatedById: systemUser.id,
    },
  });
  console.log(`  - Created breached case: ${breachedCase.referenceNumber}`);

  // Case critically overdue (breached 72+ hours ago)
  const criticalCase = await prisma.case.create({
    data: {
      organizationId: acme.id,
      referenceNumber: `SLA-CRIT-${Date.now()}`,
      status: "IN_PROGRESS",
      sourceChannel: "EMAIL",
      caseType: "REPORT",
      reporterType: "ANONYMOUS",
      severity: Severity.HIGH,
      details: "Demo case requiring critical escalation",
      createdAt: subDays(new Date(), 25),
      slaDueDate: subDays(new Date(), 18), // Due 18 days ago (critical)
      slaState: {
        lastStatus: "critical",
        lastNotifiedAt: subDays(new Date(), 2).toISOString(),
        lastNotificationType: "critical",
      },
      createdById: systemUser.id,
      updatedById: systemUser.id,
    },
  });
  console.log(`  - Created critical case: ${criticalCase.referenceNumber}`);

  // 3. Create a sample escalation rule
  const escalationRule = await prisma.ruleDefinition.create({
    data: {
      organizationId: acme.id,
      name: "High Severity Unassigned Escalation",
      description:
        "Escalate HIGH severity cases that remain unassigned for 4+ hours to Compliance Officer",
      triggerEvent: "sla.warning",
      conditions: {
        all: [
          { fact: "severity", operator: "equal", value: "HIGH" },
          { fact: "isUnassigned", operator: "equal", value: true },
          { fact: "hoursUnassigned", operator: "greaterThan", value: 4 },
        ],
      },
      actions: [
        {
          type: "escalate_to_role",
          params: {
            role: "COMPLIANCE_OFFICER",
            notifyOriginalAssignee: true,
          },
        },
      ],
      priority: 10,
      isActive: true,
      createdById: systemUser.id,
      updatedById: systemUser.id,
    },
  });
  console.log(`  - Created escalation rule: ${escalationRule.name}`);

  // 4. Create SLA breach escalation rule
  const breachRule = await prisma.ruleDefinition.create({
    data: {
      organizationId: acme.id,
      name: "SLA Breach Escalation",
      description: "Escalate any breached case to Compliance Officer",
      triggerEvent: "sla.breached",
      conditions: {
        all: [{ fact: "slaEvent.type", operator: "equal", value: "breached" }],
      },
      actions: [
        {
          type: "escalate_to_role",
          params: {
            role: "COMPLIANCE_OFFICER",
            notifyOriginalAssignee: true,
          },
        },
      ],
      priority: 5,
      isActive: true,
      createdById: systemUser.id,
      updatedById: systemUser.id,
    },
  });
  console.log(`  - Created breach escalation rule: ${breachRule.name}`);

  console.log("Phase 41 seeding complete.");
}
```
