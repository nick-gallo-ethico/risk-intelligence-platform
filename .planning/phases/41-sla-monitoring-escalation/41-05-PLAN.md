```typescript
import { Controller, Get, Patch, Body, UseGuards } from "@nestjs/common";
import { JwtAuthGuard } from "../../auth/guards/jwt-auth.guard";
import { RolesGuard } from "../../auth/guards/roles.guard";
import { Roles } from "../../auth/decorators/roles.decorator";
import { TenantId } from "@common/decorators/tenant.decorator";
import { CurrentUser } from "@common/decorators/current-user.decorator";
import { SlaConfigService } from "./sla-config.service";
import { UpdateCaseSlaConfigDto } from "./dto/sla-config.dto";

@Controller("api/sla")
@UseGuards(JwtAuthGuard, RolesGuard)
export class SlaConfigController {
  constructor(private readonly slaConfigService: SlaConfigService) {}

  /**
   * Get current SLA configuration for the organization.
   */
  @Get("config")
  @Roles("SYSTEM_ADMIN", "COMPLIANCE_OFFICER")
  async getConfig(@TenantId() organizationId: string) {
    return this.slaConfigService.getConfig(organizationId);
  }

  /**
   * Update SLA configuration for the organization.
   */
  @Patch("config")
  @Roles("SYSTEM_ADMIN", "COMPLIANCE_OFFICER")
  async updateConfig(
    @TenantId() organizationId: string,
    @Body() dto: UpdateCaseSlaConfigDto,
  ) {
    return this.slaConfigService.updateConfig(organizationId, dto);
  }
}
```

```typescript
import { ArrowLeft } from 'lucide-react';

import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
import Link from 'next/link';

const fetcher = (url: string) => fetch(url).then((res) => res.json());

export default function SlaSettingsPage() {
  const router = useRouter();
  const { data: config, error, mutate } = useSWR('/api/sla/config', fetcher);

  const handleSave = async (values: any) => {
    const response = await fetch('/api/sla/config', {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(values),
    });

    if (response.ok) {
      mutate();
    }

    return response.ok;
  };

  return (
    <div className="container mx-auto py-6 space-y-6">
      <PageHeader
        title="SLA Configuration"
        description="Configure case SLA monitoring thresholds and escalation settings"

        </Button>
      </PageHeader>

        <TabsList>
          <TabsTrigger value="thresholds">SLA Thresholds</TabsTrigger>
          <TabsTrigger value="escalation">Escalation Rules</TabsTrigger>
        </TabsList>

            <CardHeader>
              <CardTitle>Case SLA Thresholds</CardTitle>
```

```typescript
'use client';

import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';
import {
  Form,
  FormControl,
  FormDescription,
  FormField,
  FormItem,
  FormLabel,
  FormMessage,
} from '@/components/ui/form';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { Switch } from '@/components/ui/switch';
import { toast } from 'sonner';
import { useState } from 'react';

const slaConfigSchema = z.object({
  enabled: z.boolean(),
  defaultDays: z.number().min(1).max(365),
  warningThresholdPercent: z.number().min(50).max(99),
  criticalThresholdHours: z.number().min(1).max(168),
  severityOverrides: z.object({
    HIGH: z.number().min(1).max(365).optional(),
    MEDIUM: z.number().min(1).max(365).optional(),
    LOW: z.number().min(1).max(365).optional(),
  }).optional(),
});

type SlaConfigFormValues = z.infer<typeof slaConfigSchema>;

interface SlaConfigFormProps {
  initialConfig: SlaConfigFormValues;
  onSave: (values: SlaConfigFormValues) => Promise<boolean>;
}

export function SlaConfigForm({ initialConfig, onSave }: SlaConfigFormProps) {
  const [isSaving, setIsSaving] = useState(false);

  const form = useForm<SlaConfigFormValues>({
    resolver: zodResolver(slaConfigSchema),
    defaultValues: initialConfig,
  });

  const handleSubmit = async (values: SlaConfigFormValues) => {
    setIsSaving(true);
    try {
      const success = await onSave(values);
      if (success) {
        toast.success('SLA configuration saved');
      } else {
        toast.error('Failed to save SLA configuration');
      }
    } catch (error) {
      toast.error('Failed to save SLA configuration');
    } finally {
      setIsSaving(false);
    }
  };

  return (
    <Form {...form}>
      <form onSubmit={form.handleSubmit(handleSubmit)} className="space-y-6">
        <FormField
          control={form.control}
          name="enabled"
          render={({ field }) => (
            <FormItem className="flex flex-row items-center justify-between rounded-lg border p-4">
              <div className="space-y-0.5">
                <FormLabel className="text-base">Enable SLA Monitoring</FormLabel>
                <FormDescription>
                  Monitor case SLAs and send notifications
                </FormDescription>
              </div>
              <FormControl>
                <Switch checked={field.value} onCheckedChange={field.onChange} />
              </FormControl>
            </FormItem>
          )}
        />

        <div className="grid gap-4 md:grid-cols-2">
          <FormField
            control={form.control}
            name="defaultDays"
            render={({ field }) => (
              <FormItem>
                <FormLabel>Default SLA (days)</FormLabel>
                <FormControl>
                  <Input
                    type="number"
                    {...field}
                    onChange={(e) => field.onChange(parseInt(e.target.value))}
                  />
                </FormControl>
                <FormDescription>
                  Default case resolution time in days
                </FormDescription>
                <FormMessage />
              </FormItem>
            )}
          />

          <FormField
            control={form.control}
            name="warningThresholdPercent"
            render={({ field }) => (
              <FormItem>
                <FormLabel>Warning Threshold (%)</FormLabel>
                <FormControl>
                  <Input
                    type="number"
                    {...field}
                    onChange={(e) => field.onChange(parseInt(e.target.value))}
                  />
                </FormControl>
                <FormDescription>
                  Send warning when this % of SLA time is used (default: 80%)
                </FormDescription>
                <FormMessage />
              </FormItem>
            )}
          />

          <FormField
            control={form.control}
            name="criticalThresholdHours"
            render={({ field }) => (
              <FormItem>
                <FormLabel>Critical Threshold (hours)</FormLabel>
                <FormControl>
                  <Input
                    type="number"
                    {...field}
                    onChange={(e) => field.onChange(parseInt(e.target.value))}
                  />
                </FormControl>
                <FormDescription>
                  Escalate to compliance officer after this many hours past breach
                </FormDescription>
                <FormMessage />
              </FormItem>
            )}
          />
        </div>

        <div className="space-y-4">
          <h4 className="font-medium">Severity Overrides</h4>
          <p className="text-sm text-muted-foreground">
            Override default SLA days based on case severity
          </p>

          <div className="grid gap-4 md:grid-cols-3">
            <FormField
              control={form.control}
              name="severityOverrides.HIGH"
              render={({ field }) => (
                <FormItem>
                  <FormLabel>HIGH Severity (days)</FormLabel>
                  <FormControl>
                    <Input
                      type="number"
                      placeholder="7"
                      {...field}
                      value={field.value ?? ''}
                      onChange={(e) =>
                        field.onChange(e.target.value ? parseInt(e.target.value) : undefined)
                      }
                    />
                  </FormControl>
                  <FormMessage />
                </FormItem>
              )}
            />

            <FormField
              control={form.control}
              name="severityOverrides.MEDIUM"
              render={({ field }) => (
                <FormItem>
                  <FormLabel>MEDIUM Severity (days)</FormLabel>
                  <FormControl>
                    <Input
                      type="number"
                      placeholder="14"
                      {...field}
                      value={field.value ?? ''}
                      onChange={(e) =>
                        field.onChange(e.target.value ? parseInt(e.target.value) : undefined)
                      }
                    />
                  </FormControl>
                  <FormMessage />
                </FormItem>
              )}
            />

            <FormField
              control={form.control}
              name="severityOverrides.LOW"
              render={({ field }) => (
                <FormItem>
                  <FormLabel>LOW Severity (days)</FormLabel>
                  <FormControl>
                    <Input
                      type="number"
                      placeholder="30"
                      {...field}
                      value={field.value ?? ''}
                      onChange={(e) =>
                        field.onChange(e.target.value ? parseInt(e.target.value) : undefined)
                      }
                    />
                  </FormControl>
                  <FormMessage />
                </FormItem>
              )}
            />
          </div>
        </div>

        <Button type="submit" disabled={isSaving}>
          {isSaving ? 'Saving...' : 'Save Configuration'}
        </Button>
      </form>
    </Form>
  );
}
```
