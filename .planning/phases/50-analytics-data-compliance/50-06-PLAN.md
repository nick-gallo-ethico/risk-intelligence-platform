```typescript
import { SecretClient } from '@azure/keyvault-secrets';

@Injectable()
export class KeyManagementService {
private client: SecretClient;

constructor(private configService: ConfigService) {
const vaultUrl = configService.get('AZURE_KEY_VAULT_URL');
const credential = new DefaultAzureCredential();
this.client = new SecretClient(vaultUrl, credential);
}

async storeKey(keyId: string, keyData: Buffer): Promise<string> {
// Store key in Azure Key Vault
await this.client.setSecret(keyId, keyData.toString('base64'));
return keyId;
}

async getKey(keyId: string): Promise<Buffer | null> {
// Retrieve key from Key Vault
try {
const secret = await this.client.getSecret(keyId);
return Buffer.from(secret.value!, 'base64');
} catch (error) {
if (error.code === 'SecretNotFound') return null;
throw error;
}
}

async deleteKey(keyId: string): Promise<void> {
// Delete key (soft delete + purge for GDPR)
await this.client.beginDeleteSecret(keyId);
// Note: May need to purge immediately for GDPR compliance
}
}
````

**crypto-shredding.service.ts (CryptoShreddingService):**
```typescript
import { createCipheriv, createDecipheriv, randomBytes } from 'crypto';

@Injectable()
export class CryptoShreddingService {
  private readonly algorithm = 'aes-256-gcm';
  private readonly PII_FIELDS = {
    Case: ['reporterName', 'reporterEmail', 'reporterPhone'],
    RIU: ['reporterName', 'reporterEmail', 'reporterPhone', 'details'],
  };

  constructor(
    private readonly keyManagement: KeyManagementService,
    private readonly prisma: PrismaService,
  ) {}

  async encryptPiiFields(
    entityType: 'Case' | 'RIU',
    entityId: string,
    data: Record<string, string>,
  ): Promise<{ encryptedData: Record<string, string>; keyId: string }> {
    // Generate per-record key
    const dataKey = randomBytes(32);
    const keyId = `${entityType.toLowerCase()}-${entityId}`;

    // Store key in Key Vault
    await this.keyManagement.storeKey(keyId, dataKey);

    // Encrypt each PII field
    const encryptedData: Record<string, string> = {};
    for (const [field, value] of Object.entries(data)) {
      if (value) {
        const iv = randomBytes(16);
        const cipher = createCipheriv(this.algorithm, dataKey, iv);
        const encrypted = Buffer.concat([cipher.update(value, 'utf8'), cipher.final()]);
        const authTag = cipher.getAuthTag();
        encryptedData[field] = Buffer.concat([iv, authTag, encrypted]).toString('base64');
      }
    }

    return { encryptedData, keyId };
  }

  async decryptPiiFields(
    keyId: string,
    encryptedData: Record<string, string>,
  ): Promise<Record<string, string>> {
    const dataKey = await this.keyManagement.getKey(keyId);
    if (!dataKey) {
      throw new Error('Encryption key not found - data may have been shredded');
    }

    const decryptedData: Record<string, string> = {};
    for (const [field, encryptedValue] of Object.entries(encryptedData)) {
      const buffer = Buffer.from(encryptedValue, 'base64');
      const iv = buffer.subarray(0, 16);
      const authTag = buffer.subarray(16, 32);
      const encrypted = buffer.subarray(32);
      const decipher = createDecipheriv(this.algorithm, dataKey, iv);
      decipher.setAuthTag(authTag);
      decryptedData[field] = decipher.update(encrypted) + decipher.final('utf8');
    }

    return decryptedData;
  }

  async shredRecord(entityType: 'Case' | 'RIU', entityId: string): Promise<void> {
    const keyId = `${entityType.toLowerCase()}-${entityId}`;

    // Delete key from Key Vault
    await this.keyManagement.deleteKey(keyId);

    // Mark record as shredded
    if (entityType === 'Case') {
      await this.prisma.case.update({
        where: { id: entityId },
        data: { piiShredded: true, piiShreddedAt: new Date() },
      });
    } else {
      await this.prisma.riskIntelligenceUnit.update({
        where: { id: entityId },
        data: { piiShredded: true, piiShreddedAt: new Date() },
      });
    }
  }
}
````

```typescript
@Injectable()
export class DataErasureService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly cryptoShredding: CryptoShreddingService,
    private readonly auditService: AuditService,
  ) {}

async createRequest(
organizationId: string,
userId: string,
dto: CreateErasureRequestDto,
): Promise<ErasureRequest> {
// Create erasure request record
const request = await this.prisma.erasureRequest.create({
data: {
organizationId,
requestedById: userId,
subjectEmail: dto.subjectEmail,
subjectType: dto.subjectType,
entityType: dto.entityType,
entityIds: dto.entityIds,
status: 'PENDING',
},
});

    // Log to audit trail
    await this.auditService.log({
      organizationId,
      entityType: 'ERASURE_REQUEST',
      entityId: request.id,
      action: 'ERASURE_REQUESTED',
      actionDescription: `GDPR erasure requested for ${dto.entityIds.length} ${dto.entityType} records`,
      actorUserId: userId,
    });

    return request;

}

async processRequest(requestId: string): Promise<void> {
const request = await this.prisma.erasureRequest.findUnique({
where: { id: requestId },
});

    if (!request || request.status !== 'PENDING') return;

    await this.prisma.erasureRequest.update({
      where: { id: requestId },
      data: { status: 'PROCESSING' },
    });

    try {
      // Shred each entity
      for (const entityId of request.entityIds) {
        await this.cryptoShredding.shredRecord(
          request.entityType as 'Case' | 'RIU',
          entityId,
        );
      }

      await this.prisma.erasureRequest.update({
        where: { id: requestId },
        data: { status: 'COMPLETED', processedAt: new Date() },
      });
    } catch (error) {
      await this.prisma.erasureRequest.update({
        where: { id: requestId },
        data: {
          status: 'FAILED',
          errorMessage: error.message,
        },
      });
      throw error;
    }

}
}
````
