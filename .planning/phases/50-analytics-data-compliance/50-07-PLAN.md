@Injectable()
export class RetentionPolicyService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
    private readonly logger: Logger,
  ) {}

  async createPolicy(
    organizationId: string,
    userId: string,
    dto: CreateRetentionPolicyDto,
  ): Promise<RetentionPolicy> {
    return this.prisma.retentionPolicy.create({
      data: {
        organizationId,
        entityType: dto.entityType,
        archiveAfterDays: dto.archiveAfterDays,
        deleteAfterDays: dto.deleteAfterDays,
        excludeStatuses: dto.excludeStatuses || [],
        createdById: userId,
      },
    });
  }

  async getPolicy(organizationId: string, entityType: string) {
    return this.prisma.retentionPolicy.findUnique({
      where: { organizationId_entityType: { organizationId, entityType } },
    });
  }

  async executeRetention(organizationId: string): Promise<void> {
    const policies = await this.prisma.retentionPolicy.findMany({
      where: { organizationId, isActive: true },
    });

    for (const policy of policies) {
      await this.executePolicy(policy);
    }
  }

  private async executePolicy(policy: RetentionPolicy): Promise<void> {
    const now = new Date();

    // Archive records past archiveAfterDays
    if (policy.archiveAfterDays) {
      const archiveThreshold = new Date(now.getTime() - policy.archiveAfterDays * 24 * 60 * 60 * 1000);
      const archived = await this.archiveRecords(policy, archiveThreshold);
      if (archived > 0) {
        await this.logRetentionAction(policy, 'ARCHIVED', archived, archiveThreshold);
      }
    }

    // Delete records past deleteAfterDays (only if already archived)
    if (policy.deleteAfterDays) {
      const deleteThreshold = new Date(now.getTime() - policy.deleteAfterDays * 24 * 60 * 60 * 1000);
      const deleted = await this.deleteRecords(policy, deleteThreshold);
      if (deleted > 0) {
        await this.logRetentionAction(policy, 'DELETED', deleted, deleteThreshold);
      }
    }
  }

  private async archiveRecords(policy: RetentionPolicy, threshold: Date): Promise<number> {
    // Archive based on entity type, excluding specified statuses
    if (policy.entityType === 'Case') {
      const result = await this.prisma.case.updateMany({
        where: {
          organizationId: policy.organizationId,
          isArchived: false,
          createdAt: { lt: threshold },
          status: { notIn: policy.excludeStatuses as any },
        },
        data: { isArchived: true, archivedAt: new Date() },
      });
      return result.count;
    }
    // Similar for RIU, etc.
    return 0;
  }

@Processor('retention', { concurrency: 1 })
export class RetentionProcessor extends WorkerHost {
  constructor(
    private readonly retentionService: RetentionPolicyService,
    private readonly prisma: PrismaService,
    private readonly logger: Logger,
  ) {
    super();
  }

async process(job: Job): Promise<void> {
if (job.name === 'execute-retention') {
this.logger.log('Starting nightly retention processing');

      // Get all organizations with active retention policies
      const organizations = await this.prisma.organization.findMany({
        where: { isActive: true },
        select: { id: true },
      });

      let processed = 0;
      for (const org of organizations) {
        try {
          await this.retentionService.executeRetention(org.id);
          processed++;
        } catch (error) {
          this.logger.error(`Retention failed for org ${org.id}: ${error.message}`);
        }
      }

      this.logger.log(`Retention processing complete: ${processed}/${organizations.length} orgs`);
    }

}
}
