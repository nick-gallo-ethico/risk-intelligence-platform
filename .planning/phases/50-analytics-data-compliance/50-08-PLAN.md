**virus-scan.service.ts (VirusScanService):**

```typescript
import NodeClam from "clamscan";

export interface ScanResult {
  clean: boolean;
  filename: string;
  threats: string[];
  scannedAt: Date;
  scanDurationMs: number;
}

@Injectable()
export class VirusScanService implements OnModuleInit {
  private clamav: NodeClam | null = null;
  private initialized = false;
  private readonly logger = new Logger(VirusScanService.name);

  constructor(private readonly configService: ConfigService) {}

  async onModuleInit(): Promise<void> {
    if (this.configService.get("VIRUS_SCAN_ENABLED") === "false") {
      this.logger.warn("Virus scanning disabled by configuration");
      return;
    }

    try {
      this.clamav = await new NodeClam().init({
        clamdscan: {
          socket: this.configService.get("CLAMAV_SOCKET"),
          host: this.configService.get("CLAMAV_HOST", "localhost"),
          port: parseInt(this.configService.get("CLAMAV_PORT", "3310")),
          timeout: 60000, // 60 seconds for large files
          localFallback: true,
        },
        preference: "clamdscan",
      });
      this.initialized = true;
      this.logger.log("ClamAV virus scanner initialized");
    } catch (error) {
      this.logger.error(`Failed to initialize ClamAV: ${error.message}`);
      // Don't throw - allow graceful degradation if ClamAV unavailable
    }
  }

  async scanBuffer(buffer: Buffer, filename: string): Promise<ScanResult> {
    const startTime = Date.now();

    if (!this.initialized || !this.clamav) {
      // Fallback behavior when ClamAV unavailable
      const fallback = this.configService.get("VIRUS_SCAN_FALLBACK", "ALLOW");
      if (fallback === "REJECT") {
        throw new Error("Virus scanning unavailable - uploads rejected");
      }
      this.logger.warn(
        `Virus scan skipped for ${filename} - ClamAV not available`,
      );
      return {
        clean: true, // Allow through when ClamAV down (configurable)
        filename,
        threats: [],
        scannedAt: new Date(),
        scanDurationMs: Date.now() - startTime,
      };
    }

    try {
      const { isInfected, viruses } = await this.clamav.scanBuffer(buffer);

      const result: ScanResult = {
        clean: !isInfected,
        filename,
        threats: viruses || [],
        scannedAt: new Date(),
        scanDurationMs: Date.now() - startTime,
      };

      if (isInfected) {
        this.logger.warn(
          `Virus detected in ${filename}: ${viruses?.join(", ")}`,
        );
      }

      return result;
    } catch (error) {
      this.logger.error(`Virus scan failed for ${filename}: ${error.message}`);
      throw new Error(`Virus scan failed: ${error.message}`);
    }
  }

  async isHealthy(): Promise<boolean> {
    if (!this.initialized || !this.clamav) {
      return false;
    }

    try {
      // Use ping or version check
      await this.clamav.ping();
      return true;
    } catch {
      return false;
    }
  }
}
```

```typescript
@Injectable()
export class StorageService {
  constructor(
    private readonly storageProvider: StorageProvider,
    private readonly virusScanService: VirusScanService,
    private readonly auditService: AuditService,
    private readonly logger: Logger,
  ) {}

async uploadFile(
organizationId: string,
userId: string,
file: Express.Multer.File,
options?: UploadOptions,
): Promise<UploadResult> {
// 1. Scan file for viruses
const scanResult = await this.virusScanService.scanBuffer(
file.buffer,
file.originalname,
);

    // 2. Reject infected files
    if (!scanResult.clean) {
      await this.auditService.log({
        organizationId,
        entityType: 'ATTACHMENT',
        entityId: null,
        action: 'UPLOAD_REJECTED_VIRUS',
        actionDescription: `Upload rejected - virus detected: ${scanResult.threats.join(', ')}`,
        actorUserId: userId,
        context: {
          filename: file.originalname,
          threats: scanResult.threats,
        },
      });

      throw new BadRequestException(
        `File rejected: virus detected (${scanResult.threats.join(', ')})`
      );
    }

    // 3. Log successful scan
    this.logger.log(
      `File ${file.originalname} scanned clean in ${scanResult.scanDurationMs}ms`
    );

    // 4. Proceed with upload (existing code)
    const result = await this.storageProvider.upload(
      organizationId,
      file,
      options,
    );

    return result;

}
}
````

**clamav.health.ts (ClamavHealthIndicator):**
```typescript
import { Injectable } from '@nestjs/common';
import {
  HealthIndicator,
  HealthIndicatorResult,
  HealthCheckError,
} from '@nestjs/terminus';
import { VirusScanService } from '../../storage/services/virus-scan.service';

@Injectable()
export class ClamavHealthIndicator extends HealthIndicator {
  constructor(private readonly virusScanService: VirusScanService) {
    super();
  }

  async isHealthy(key: string): Promise<HealthIndicatorResult> {
    const isHealthy = await this.virusScanService.isHealthy();

    const result = this.getStatus(key, isHealthy, {
      status: isHealthy ? 'connected' : 'disconnected',
    });

    if (isHealthy) {
      return result;
    }

    throw new HealthCheckError('ClamAV check failed', result);
  }
}
````
