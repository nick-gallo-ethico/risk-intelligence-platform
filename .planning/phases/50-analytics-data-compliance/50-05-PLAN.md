**benchmarking.module.ts:**
```typescript
@Module({
  imports: [
    BullModule.registerQueue({ name: 'benchmarking' }),
  ],
  providers: [
    BenchmarkingService,
    BenchmarkingProcessor,
  ],
  exports: [BenchmarkingService],
})
export class BenchmarkingModule {
  constructor(
    @InjectQueue('benchmarking') private benchmarkQueue: Queue,
  ) {
    // Schedule nightly benchmark computation at 3 AM
    this.benchmarkQueue.add('compute-benchmarks',
      { period: this.getCurrentPeriod() },
      {
        repeat: { pattern: '0 3 * * *' },
        jobId: 'nightly-benchmark-computation',
      }
    );
  }

  private getCurrentPeriod(): string {
    const now = new Date();
    return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;
  }
}
````
