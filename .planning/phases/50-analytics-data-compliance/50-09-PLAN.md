export async function seedAcmePhase50(prisma: PrismaClient) {
  const acmeOrg = await prisma.organization.findFirst({
    where: { slug: "acme-corp" },
  });

  if (!acmeOrg) {
    console.log("Acme org not found, skipping Phase 50 seed");
    return;
  }

  // 1. Seed fact tables with historical data
  // Generate 12 months of daily fact data
  const factData = generateFactData(acmeOrg.id, 365);
  await prisma.factCaseDaily.createMany({ data: factData.cases });
  await prisma.factRiuDaily.createMany({ data: factData.rius });
  await prisma.factCampaignDaily.createMany({ data: factData.campaigns });

  // 2. Create benchmark cohort with Acme
  const cohort = await prisma.benchmarkCohort.create({
    data: {
      name: "Healthcare 1000-5000 employees",
      industry: "HEALTHCARE",
      employeeRange: "1000-5000",
      region: "NORTH_AMERICA",
      organizationCount: 12, // Acme + simulated peers
    },
  });

  // Add benchmark metrics
  await prisma.benchmarkMetric.createMany({
    data: [
      {
        cohortId: cohort.id,
        metricType: "avg_days_to_close",
        period: "2026-Q1",
        value: 18.5,
        percentile25: 12,
        percentile50: 18.5,
        percentile75: 28,
        percentile90: 42,
        sampleSize: 12,
      },
      {
        cohortId: cohort.id,
        metricType: "cases_per_1000_employees",
        period: "2026-Q1",
        value: 4.2,
        percentile25: 2.8,
        percentile50: 4.2,
        percentile75: 6.1,
        percentile90: 8.5,
        sampleSize: 12,
      },
    ],
  });

  // 3. Create retention policies for Acme
  const systemAdmin = await prisma.user.findFirst({
    where: { organizationId: acmeOrg.id, role: "SYSTEM_ADMIN" },
  });

  if (systemAdmin) {
    await prisma.retentionPolicy.createMany({
      data: [
        {
          organizationId: acmeOrg.id,
          entityType: "Case",
          archiveAfterDays: 365,
          deleteAfterDays: null,
          excludeStatuses: ["NEW", "IN_INVESTIGATION", "PENDING_REMEDIATION"],
          createdById: systemAdmin.id,
        },
        {
          organizationId: acmeOrg.id,
          entityType: "RIU",
          archiveAfterDays: 365,
          deleteAfterDays: null,
          excludeStatuses: ["PENDING_QA"],
          createdById: systemAdmin.id,
        },
        {
          organizationId: acmeOrg.id,
          entityType: "AuditLog",
          archiveAfterDays: 365,
          deleteAfterDays: 2555,
          excludeStatuses: [],
          createdById: systemAdmin.id,
        },
      ],
    });
  }

  // 4. Create sample scheduled report
  // (Uses existing SavedReport from prior phases)

  console.log("Phase 50 seed complete: facts, benchmarks, retention policies");
}
