```typescript
import { Injectable } from "@nestjs/common";
import {
  HealthIndicator,
  HealthIndicatorResult,
  HealthCheckError,
} from "@nestjs/terminus";
import { PrismaService } from "../../prisma/prisma.service";

/**
 * Prisma/PostgreSQL health indicator.
 * Verifies database connectivity by executing SELECT 1.
 */
@Injectable()
export class PrismaHealthIndicator extends HealthIndicator {
  constructor(private readonly prisma: PrismaService) {
    super();
  }

  /**
   * Check database health by executing a simple query.
   *
   * @param key - The key to use in the health check result
   * @returns Health indicator result
   * @throws HealthCheckError if database is unreachable
   */
  async isHealthy(key: string): Promise<HealthIndicatorResult> {
    try {
      await this.prisma.$queryRaw`SELECT 1`;
      return this.getStatus(key, true);
    } catch (error) {
      const message = error instanceof Error ? error.message : "Unknown error";
      throw new HealthCheckError(
        "Database check failed",
        this.getStatus(key, false, { message }),
      );
    }
  }
}
```

```typescript
import { Injectable, Inject, Optional } from "@nestjs/common";
import {
  HealthIndicator,
  HealthIndicatorResult,
  HealthCheckError,
} from "@nestjs/terminus";
import Redis from "ioredis";

/**
 * Redis health indicator.
 * Verifies Redis connectivity via PING command.
 */
@Injectable()
export class RedisHealthIndicator extends HealthIndicator {
  constructor(
    @Optional() @Inject("REDIS_CLIENT") private readonly redis?: Redis,
  ) {
    super();
  }

  /**
   * Check Redis health by sending PING.
   *
   * @param key - The key to use in the health check result
   * @returns Health indicator result
   * @throws HealthCheckError if Redis is unreachable
   */
  async isHealthy(key: string): Promise<HealthIndicatorResult> {
    // If Redis client not configured, skip check
    if (!this.redis) {
      return this.getStatus(key, true, { status: "not_configured" });
    }

    try {
      const result = await this.redis.ping();
      const isHealthy = result === "PONG";
      return this.getStatus(key, isHealthy);
    } catch (error) {
      const message = error instanceof Error ? error.message : "Unknown error";
      throw new HealthCheckError(
        "Redis check failed",
        this.getStatus(key, false, { message }),
      );
    }
  }
}
```

```typescript
import { Injectable, Optional } from "@nestjs/common";
import {
  HealthIndicator,
  HealthIndicatorResult,
  HealthCheckError,
} from "@nestjs/terminus";
import { ElasticsearchService } from "@nestjs/elasticsearch";

/**
 * Elasticsearch health indicator.
 * Verifies cluster health status.
 */
@Injectable()
export class ElasticsearchHealthIndicator extends HealthIndicator {
  constructor(@Optional() private readonly esService?: ElasticsearchService) {
    super();
  }

  /**
   * Check Elasticsearch cluster health.
   * Considers 'green' and 'yellow' as healthy (yellow = replicas not assigned, common in dev).
   *
   * @param key - The key to use in the health check result
   * @returns Health indicator result
   * @throws HealthCheckError if Elasticsearch is unreachable
   */
  async isHealthy(key: string): Promise<HealthIndicatorResult> {
    // If Elasticsearch not configured, skip check
    if (!this.esService) {
      return this.getStatus(key, true, { status: "not_configured" });
    }

    try {
      const health = await this.esService.cluster.health();
      // 'green' = all good, 'yellow' = no replicas (fine for single-node)
      const isHealthy = ["green", "yellow"].includes(health.status);

      return this.getStatus(key, isHealthy, {
        status: health.status,
        numberOfNodes: health.number_of_nodes,
        activeShards: health.active_shards,
      });
    } catch (error) {
      const message = error instanceof Error ? error.message : "Unknown error";
      throw new HealthCheckError(
        "Elasticsearch check failed",
        this.getStatus(key, false, { message }),
      );
    }
  }
}
```

```typescript
import { Controller, Get } from "@nestjs/common";
import { ApiTags, ApiOperation, ApiResponse } from "@nestjs/swagger";
import {
  HealthCheck,
  HealthCheckService,
  HealthCheckResult,
} from "@nestjs/terminus";
import { PrismaHealthIndicator } from "./indicators/prisma.health";
import { RedisHealthIndicator } from "./indicators/redis.health";
import { ElasticsearchHealthIndicator } from "./indicators/elasticsearch.health";

/**
 * Health check controller providing deep dependency verification.
 *
 * ENDPOINTS:
 * - GET /health - Full health check (database, Redis, Elasticsearch)
 * - GET /health/liveness - Simple liveness probe (app is running)
 * - GET /health/readiness - Readiness probe (app can serve traffic)
 *
 * RESPONSE FORMAT:
 * {
 *   "status": "ok" | "error",
 *   "info": { ... healthy components },
 *   "error": { ... unhealthy components },
 *   "details": { ... all component details }
 * }
 *
 * HTTP STATUS:
 * - 200: All checks passed
 * - 503: One or more checks failed
 */
@ApiTags("Health")
@Controller("health")
export class HealthController {
  constructor(
    private health: HealthCheckService,
    private prismaHealth: PrismaHealthIndicator,
    private redisHealth: RedisHealthIndicator,
    private esHealth: ElasticsearchHealthIndicator,
  ) {}

  @Get()
  @HealthCheck()
  @ApiOperation({
    summary: "Deep health check",
    description:
      "Checks database, Redis, and Elasticsearch connectivity. Returns 503 if any dependency is unhealthy.",
  })
  @ApiResponse({
    status: 200,
    description: "All dependencies healthy",
  })
  @ApiResponse({
    status: 503,
    description: "One or more dependencies unhealthy",
  })
  check(): Promise<HealthCheckResult> {
    return this.health.check([
      () => this.prismaHealth.isHealthy("database"),
      () => this.redisHealth.isHealthy("redis"),
      () => this.esHealth.isHealthy("elasticsearch"),
    ]);
  }

  @Get("liveness")
  @ApiOperation({
    summary: "Liveness probe",
    description:
      "Simple check that the application is running. Use for Kubernetes liveness probes.",
  })
  @ApiResponse({
    status: 200,
    description: "Application is alive",
  })
  liveness(): { status: string; timestamp: string } {
    return {
      status: "ok",
      timestamp: new Date().toISOString(),
    };
  }

  @Get("readiness")
  @HealthCheck()
  @ApiOperation({
    summary: "Readiness probe",
    description:
      "Checks if the application can serve traffic (database must be connected). Use for Kubernetes readiness probes.",
  })
  @ApiResponse({
    status: 200,
    description: "Application ready to serve traffic",
  })
  @ApiResponse({
    status: 503,
    description: "Application not ready (database unavailable)",
  })
  readiness(): Promise<HealthCheckResult> {
    // Readiness only requires database - Redis and ES are optional
    return this.health.check([() => this.prismaHealth.isHealthy("database")]);
  }
}
```
