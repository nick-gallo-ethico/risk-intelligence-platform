Add a magic-byte validation method using dynamic import for ESM package:

```typescript
/**
 * Validates file content against expected type using magic bytes.
 * Returns true if file content matches expected extension.
 */
async validateMagicBytes(buffer: Buffer, expectedExt: string): Promise<{ valid: boolean; detectedMime?: string; error?: string }> {
  try {
    // Dynamic import for ESM package in CommonJS context
    const { fileTypeFromBuffer } = await import('file-type');
    const detected = await fileTypeFromBuffer(buffer);

    // Plain text files have no magic bytes - allow if extension is text-based
    if (!detected) {
      const textExtensions = ['.txt', '.csv', '.json', '.xml', '.md', '.html'];
      if (textExtensions.includes(expectedExt.toLowerCase())) {
        return { valid: true };
      }
      return { valid: false, error: 'Could not detect file type from content' };
    }

    // Map extensions to allowed MIME types
    const extToMime: Record<string, string[]> = {
      '.pdf': ['application/pdf'],
      '.docx': ['application/vnd.openxmlformats-officedocument.wordprocessingml.document'],
      '.xlsx': ['application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'],
      '.pptx': ['application/vnd.openxmlformats-officedocument.presentationml.presentation'],
      '.png': ['image/png'],
      '.jpg': ['image/jpeg'],
      '.jpeg': ['image/jpeg'],
      '.gif': ['image/gif'],
      '.webp': ['image/webp'],
      '.zip': ['application/zip'],
    };

    const allowedMimes = extToMime[expectedExt.toLowerCase()];
    if (!allowedMimes) {
      return { valid: false, error: `Extension ${expectedExt} not in allowed list` };
    }

    if (allowedMimes.includes(detected.mime)) {
      return { valid: true, detectedMime: detected.mime };
    }

    return {
      valid: false,
      detectedMime: detected.mime,
      error: `File content (${detected.mime}) does not match extension (${expectedExt})`
    };
  } catch (error) {
    this.logger.error(`Magic byte validation failed: ${error instanceof Error ? error.message : 'Unknown error'}`);
    return { valid: false, error: 'Magic byte validation failed' };
  }
}
```

```typescript
const DANGEROUS_EXTENSIONS = [
  ".exe",
  ".bat",
  ".cmd",
  ".sh",
  ".ps1",
  ".msi",
  ".dll",
  ".scr",
];
const ALLOWED_EXTENSIONS = [
  ".pdf",
  ".docx",
  ".xlsx",
  ".pptx",
  ".png",
  ".jpg",
  ".jpeg",
  ".gif",
  ".webp",
  ".txt",
  ".csv",
  ".json",
  ".xml",
  ".md",
  ".zip",
];

const fileFilter = (req: any, file: Express.Multer.File, callback: any) => {
  const ext = path.extname(file.originalname).toLowerCase();

  if (DANGEROUS_EXTENSIONS.includes(ext)) {
    return callback(
      new BadRequestException(`Dangerous file extension: ${ext}`),
      false,
    );
  }

  if (!ALLOWED_EXTENSIONS.includes(ext)) {
    return callback(
      new BadRequestException(`File extension not allowed: ${ext}`),
      false,
    );
  }

  callback(null, true);
};
```
