import { Injectable, Logger, BadRequestException } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { AuditService } from "../../audit/audit.service";
import { AssociationConfig, BaseAssociationEntity } from "./association.types";

/**
 * Abstract base class for association services.
 *
 * Provides common CRUD, event emission, and audit logging for:
 * - PersonCaseAssociationService
 * - CaseCaseAssociationService
 * - PersonPersonAssociationService
 * - PersonRiuAssociationService
 *
 * @template TCreate - DTO type for creating associations
 * @template TEntity - Entity type returned from Prisma
 * @template TLabel - Union type of valid labels for this association type
 */
@Injectable()
export abstract class BaseAssociationService<
  TCreate extends { notes?: string },
  TEntity extends BaseAssociationEntity,
  TLabel extends string,
> {
  protected abstract readonly config: AssociationConfig<TLabel>;
  protected readonly logger: Logger;

  constructor(
    protected readonly prisma: PrismaService,
    protected readonly eventEmitter: EventEmitter2,
    protected readonly auditService: AuditService,
  ) {
    this.logger = new Logger(this.constructor.name);
  }

  /**
   * Get the Prisma model delegate for this association type.
   * Subclasses implement this to return the correct model.
   */
  protected abstract getPrismaModel(): any;

  /**
   * Validate the create DTO. Subclasses implement specific validation.
   * @throws BadRequestException if validation fails
   */
  protected abstract validateCreate(dto: TCreate): void;

  /**
   * Build the Prisma create data from DTO and user context.
   * Subclasses implement to map DTO fields to Prisma input.
   */
  protected abstract buildCreateData(
    dto: TCreate,
    userId: string,
    organizationId: string,
  ): any;

  /**
   * Build a human-readable description for audit logging.
   */
  protected abstract buildAuditDescription(
    action: "created" | "deleted",
    entity: TEntity,
  ): string;

  /**
   * Create a new association.
   */
  async create(
    dto: TCreate,
    userId: string,
    organizationId: string,
  ): Promise<TEntity> {
    // Validate (subclass-specific)
    this.validateCreate(dto);

    // Check for duplicates (if applicable)
    await this.checkDuplicates(dto, organizationId);

    // Create the association
    const entity = await this.getPrismaModel().create({
      data: this.buildCreateData(dto, userId, organizationId),
    });

    // Emit event
    this.emitEvent("created", entity, organizationId);

    // Log audit
    await this.logAudit("created", entity, userId, organizationId);

    this.logger.log(`Created ${this.config.entityName}: ${entity.id}`);
    return entity;
  }

  /**
   * Delete an association by ID.
   */
  async delete(
    id: string,
    userId: string,
    organizationId: string,
  ): Promise<void> {
    const entity = await this.findByIdOrThrow(id, organizationId);

    await this.getPrismaModel().delete({
      where: { id, organizationId },
    });

    this.emitEvent("deleted", entity, organizationId);
    await this.logAudit("deleted", entity, userId, organizationId);

    this.logger.log(`Deleted ${this.config.entityName}: ${id}`);
  }

  /**
   * Find by ID or throw NotFoundException.
   */
  async findByIdOrThrow(id: string, organizationId: string): Promise<TEntity> {
    const entity = await this.getPrismaModel().findFirst({
      where: { id, organizationId },
    });

    if (!entity) {
      throw new BadRequestException(
        `${this.config.entityName} not found: ${id}`,
      );
    }

    return entity;
  }

  /**
   * Check for duplicate associations. Override in subclass if needed.
   * Default implementation does nothing.
   */
  protected async checkDuplicates(
    dto: TCreate,
    organizationId: string,
  ): Promise<void> {
    // Override in subclass to implement duplicate checking
  }

  /**
   * Emit event for association changes.
   */
  protected emitEvent(
    action: "created" | "deleted",
    entity: TEntity,
    organizationId: string,
  ): void {
    this.eventEmitter.emit(`${this.config.eventPrefix}.${action}`, {
      entity,
      organizationId,
      entityType: this.config.entityType,
    });
  }

  /**
   * Log audit entry for association changes.
   */
  protected async logAudit(
    action: "created" | "deleted",
    entity: TEntity,
    userId: string,
    organizationId: string,
  ): Promise<void> {
    try {
      await this.auditService.log({
        entityType: this.config.entityType,
        entityId: entity.id,
        action,
        actionDescription: this.buildAuditDescription(action, entity),
        actorUserId: userId,
        organizationId,
        changes: { entity },
      });
    } catch (error) {
      this.logger.error(`Failed to log audit for ${this.config.entityName}`, {
        error: error.message,
        entityId: entity.id,
      });
      // Don't throw - audit failure shouldn't break the operation
    }
  }
}
