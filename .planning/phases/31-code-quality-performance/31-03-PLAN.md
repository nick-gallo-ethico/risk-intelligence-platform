2. Add class properties and implement OnModuleInit:

```typescript
@Injectable()
export class SearchService implements OnModuleInit {
  private readonly logger = new Logger(SearchService.name);
  private searchBreaker: CircuitBreaker;

  constructor(
    @Inject("ELASTICSEARCH_CLIENT")
    private readonly esClient: Client,
    private readonly configService: ConfigService,
  ) {}

  onModuleInit() {
    const cbConfig = this.configService.get("elasticsearch.circuitBreaker");

    this.searchBreaker = new CircuitBreaker(
      (params: SearchRequest) => this.executeSearch(params),
      {
        timeout: cbConfig?.timeout || 5000,
        errorThresholdPercentage: cbConfig?.errorThresholdPercentage || 50,
        resetTimeout: cbConfig?.resetTimeout || 30000,
        volumeThreshold: cbConfig?.volumeThreshold || 5,
        name: "elasticsearch-search",
      },
    );

    // Graceful fallback when circuit is open
    this.searchBreaker.fallback(() => ({
      hits: { hits: [], total: { value: 0, relation: "eq" } },
      _fallback: true,
      _message: "Search temporarily unavailable. Please try again later.",
    }));

    // Logging for circuit state changes
    this.searchBreaker.on("open", () => {
      this.logger.warn(
        "Search circuit breaker OPENED - returning fallback results",
      );
    });
    this.searchBreaker.on("halfOpen", () => {
      this.logger.info("Search circuit breaker HALF-OPEN - testing recovery");
    });
    this.searchBreaker.on("close", () => {
      this.logger.info(
        "Search circuit breaker CLOSED - normal operation resumed",
      );
    });
  }

  /**
   * Internal search execution (wrapped by circuit breaker)
   */
  private async executeSearch(params: SearchRequest): Promise<SearchResponse> {
    return this.esClient.search(params);
  }

  /**
   * Public search method - uses circuit breaker
   */
  async search(params: SearchRequest): Promise<SearchResponse> {
    try {
      return await this.searchBreaker.fire(params);
    } catch (error) {
      this.logger.error("Search failed after circuit breaker", {
        error: error.message,
      });
      // Return fallback on unexpected errors
      return {
        hits: { hits: [], total: { value: 0, relation: "eq" } },
        _fallback: true,
        _message: "Search error occurred",
      } as any;
    }
  }
}
```
