export class ConflictDetectionService {
  constructor(
    private readonly rules: ConflictRulesService,
    private readonly matching: ConflictMatchingService,
  ) {}

  async detectConflicts(disclosure: Disclosure): Promise<Conflict[]> {
    const rules = await this.rules.getActiveRules(disclosure.organizationId);
    const matches = await this.matching.findPotentialMatches(disclosure);
    return this.evaluateRules(rules, matches);
  }
}
