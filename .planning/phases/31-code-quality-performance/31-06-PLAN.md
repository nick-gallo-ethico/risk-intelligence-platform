export class AiOrchestrationService {
  constructor(
    private readonly aiService: AiService,
    private readonly contextLoader: ContextLoaderService,
    private readonly skillsRegistry: SkillsRegistryService,
  ) {}

  /**
   * Execute AI chat with full context loading and skill routing.
   */
  async executeChat(
    message: string,
    entityType: string,
    entityId: string,
    userId: string,
    organizationId: string,
  ): Promise<AiResponse> {
    // Context loading
    const context = await this.contextLoader.loadForEntity(
      entityType,
      entityId,
      organizationId,
    );

    // Skill detection
    const skill = this.skillsRegistry.detectSkill(message);

export class AiController {
  constructor(private readonly orchestrationService: AiOrchestrationService) {}

  @Post("chat")
  async chat(@Body() dto, @CurrentUser() user, @TenantId() orgId) {
    return this.orchestrationService.executeChat(
      dto.message,
      dto.entityType,
      dto.entityId,
      user.id,
      orgId,
    );
  }

  @Post("actions/:id/execute")
  async executeAction(
    @Param("id") id,
    @Body() dto,
    @CurrentUser() user,
    @TenantId() orgId,
  ) {
    return this.orchestrationService.executeAction(
      id,
      dto.confirmed,
      user.id,
      orgId,
    );
  }
}
