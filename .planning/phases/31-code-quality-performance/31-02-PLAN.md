1. Create `apps/frontend/src/config/env.ts`:

/**
 * Centralized environment configuration.
 * All environment variables accessed from this file.
 * Build fails if required variables are missing.
 */

// Required in all environments
const API_URL = process.env.NEXT_PUBLIC_API_URL;
const WS_URL = process.env.NEXT_PUBLIC_WS_URL;

// Validate required variables (fail at build time, not runtime)
if (typeof window === 'undefined') {
// Server-side validation during build
if (!API_URL) {
console.warn('Warning: NEXT_PUBLIC_API_URL not set, using development default');
}
}

export const config = {
/\*\*

- Backend API base URL (e.g., https://api.example.com)
- Used for all REST API calls
  \*/
  apiUrl: API_URL || 'http://localhost:3001',

/\*\*

- WebSocket server URL (e.g., wss://api.example.com)
- Used for real-time features (AI chat, notifications)
  \*/
  wsUrl: WS_URL || 'http://localhost:3001',

/\*\*

- Environment name for conditional logic
  \*/
  isDevelopment: process.env.NODE_ENV === 'development',
  isProduction: process.env.NODE_ENV === 'production',
  } as const;

// Type export for consumers
export type Config = typeof config;

````

NEXT_PUBLIC_WS_URL=http://localhost:3001
````
