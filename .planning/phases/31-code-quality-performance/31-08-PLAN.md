```typescript
import { Injectable, Logger, OnModuleInit } from "@nestjs/common";
import { ConfigService } from "@nestjs/config";
import {
  createPrivateKey,
  createPublicKey,
  KeyObject,
  generateKeyPairSync,
} from "crypto";

/**
 * JWT key pair with metadata for rotation
 */
export interface JwtKeyPair {
  kid: string; // Key ID (used in JWT header)
  privateKey: KeyObject; // For signing
  publicKey: KeyObject; // For verification
  createdAt: Date;
  expiresAt: Date; // Keys valid until this date
}

/**
 * JwtKeyService manages RS256 key pairs for JWT signing and verification.
 *
 * Features:
 * - RS256 asymmetric encryption
 * - Key rotation with overlap period (old keys valid until token expiry)
 * - Key ID (kid) in JWT header for key selection during verification
 *
 * Key rotation flow:
 * 1. Generate new key pair with new kid
 * 2. Start signing with new key
 * 3. Keep old key valid for refresh token lifetime (7 days)
 * 4. After overlap period, old key can be removed
 */
@Injectable()
export class JwtKeyService implements OnModuleInit {
  private readonly logger = new Logger(JwtKeyService.name);

  private currentKey: JwtKeyPair;
  private previousKey: JwtKeyPair | null = null;

  // Token lifetimes for calculating key expiry overlap
  private readonly accessTokenLifetime = 15 * 60 * 1000; // 15 minutes
  private readonly refreshTokenLifetime = 7 * 24 * 60 * 60 * 1000; // 7 days

  constructor(private readonly configService: ConfigService) {}

  async onModuleInit() {
    await this.loadOrGenerateKeys();
  }

  /**
   * Load keys from config or generate new ones.
   */
  private async loadOrGenerateKeys(): Promise<void> {
    const privateKeyPem = this.configService.get<string>(
      "JWT_RS256_PRIVATE_KEY",
    );
    const publicKeyPem = this.configService.get<string>("JWT_RS256_PUBLIC_KEY");
    const keyId = this.configService.get<string>("JWT_RS256_KEY_ID", "key-1");

    if (privateKeyPem && publicKeyPem) {
      // Load from environment (production)
      this.currentKey = {
        kid: keyId,
        privateKey: createPrivateKey(privateKeyPem),
        publicKey: createPublicKey(publicKeyPem),
        createdAt: new Date(),
        expiresAt: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000), // 1 year
      };
      this.logger.log(`Loaded RS256 key pair with kid: ${keyId}`);
    } else {
      // Generate for development (NOT for production)
      this.logger.warn(
        "No RS256 keys configured - generating ephemeral keys (dev only)",
      );
      this.currentKey = this.generateKeyPair("dev-key-1");
    }
  }

  /**
   * Generate a new key pair (for development or rotation).
   */
  private generateKeyPair(kid: string): JwtKeyPair {
    const { privateKey, publicKey } = generateKeyPairSync("rsa", {
      modulusLength: 2048,
      publicKeyEncoding: { type: "spki", format: "pem" },
      privateKeyEncoding: { type: "pkcs8", format: "pem" },
    });

    return {
      kid,
      privateKey: createPrivateKey(privateKey),
      publicKey: createPublicKey(publicKey),
      createdAt: new Date(),
      expiresAt: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000),
    };
  }

  /**
   * Get the current signing key and kid.
   * Used by JwtService for token signing.
   */
  getSigningKey(): { privateKey: KeyObject; kid: string } {
    return {
      privateKey: this.currentKey.privateKey,
      kid: this.currentKey.kid,
    };
  }

  /**
   * Get key for verification by kid.
   * Returns null if key not found or expired.
   */
  getKeyById(kid: string): JwtKeyPair | null {
    if (this.currentKey.kid === kid) {
      return this.currentKey;
    }

    if (
      this.previousKey?.kid === kid &&
      this.previousKey.expiresAt > new Date()
    ) {
      return this.previousKey;
    }

    return null;
  }

  /**
   * Get all valid verification keys.
   * Returns current key and previous key if still valid.
   */
  getVerificationKeys(): JwtKeyPair[] {
    const keys = [this.currentKey];

    if (this.previousKey && this.previousKey.expiresAt > new Date()) {
      keys.push(this.previousKey);
    }

    return keys;
  }

  /**
   * Rotate to a new key pair.
   * Called by admin endpoint or scheduled job.
   */
  async rotateKey(): Promise<void> {
    const newKid = `key-${Date.now()}`;
    const newKey = this.generateKeyPair(newKid);

    // Move current to previous with overlap period
    this.previousKey = {
      ...this.currentKey,
      expiresAt: new Date(Date.now() + this.refreshTokenLifetime),
    };

    // Set new current key
    this.currentKey = newKey;

    this.logger.log(
      `Rotated to new key: ${newKid}, previous key valid until: ${this.previousKey.expiresAt}`,
    );
  }

  /**
   * Get public key PEM for JWKS endpoint (future use).
   */
  getPublicKeyPem(): string {
    return this.currentKey.publicKey.export({
      type: "spki",
      format: "pem",
    }) as string;
  }
}
```

```typescript
@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
  constructor(
    private readonly configService: ConfigService,
    private readonly jwtKeyService: JwtKeyService,
  ) {
    const algorithm = configService.get<string>("jwt.algorithm", "RS256");

    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      ignoreExpiration: false,
      algorithms: [algorithm],
      secretOrKeyProvider: (request, rawJwtToken, done) => {
        if (algorithm === "RS256") {
          // Extract kid from token header
          const decoded = jwt.decode(rawJwtToken, { complete: true });
          const kid = (decoded?.header as any)?.kid;

          if (!kid) {
            return done(new Error("Token missing kid header"), null);
          }

          const key = this.jwtKeyService.getKeyById(kid);
          if (!key) {
            return done(new Error(`Unknown key: ${kid}`), null);
          }

          return done(null, key.publicKey);
        }

        // HS256 fallback
        return done(null, configService.get<string>("jwt.secret"));
      },
    });
  }

  async validate(payload: JwtPayload) {
    return {
      userId: payload.sub,
      email: payload.email,
      organizationId: payload.organizationId,
      roles: payload.roles,
    };
  }
}
```
