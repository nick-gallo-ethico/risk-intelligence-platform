   ```typescript
   async getMyTasks(params: FetchTasksParams): Promise<TaskFetchResult> {
     // Fetch from all sources in parallel
     const [caseTasks, workflowTasks] = await Promise.all([
       this.taskCaseFetcherService.fetchAllCaseTasks(params),
       this.taskWorkflowFetcherService.fetchAllWorkflowTasks(params),
     ]);

     // Combine and sort
     const allTasks = [...caseTasks, ...workflowTasks];
     const sortedTasks = this.taskSorterService.sortByPriorityAndDueDate(allTasks);
     const filteredTasks = this.taskSorterService.applyFilters(sortedTasks, params.filters);

     // Paginate
     return this.taskSorterService.paginateResults(filteredTasks, params.limit, params.offset);
   }
   ```
