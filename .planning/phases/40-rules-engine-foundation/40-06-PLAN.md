import { Injectable, Logger, NotFoundException } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";
import { RulesEngineService } from "../engine/rules-engine.service";
import type {
  RuleConditions,
  RuleAction,
  RuleTestResult,
  RuleTestSample,
} from "../types/rule.types";

/**
 * Options for rule testing.
 */
export interface TestRuleOptions {
  /** Maximum number of historical cases to test against */
  limit?: number;
  /** Only test against cases created after this date */
  dateFrom?: Date;
  /** Only test against cases in specific categories */
  categoryIds?: string[];
  /** Only test against cases with specific severities */
  severities?: string[];
}

/**
 * RuleTesterService enables preview/testing of rules against historical data.
 *
 * Key features:
 * - Dry-run evaluation (no actions executed)
 * - Match rate calculation
 * - Sample case previews with predicted outcomes
 * - Test results saved to rule definition
 *
 * This implements RULE-07 from requirements.
 */
@Injectable()
export class RuleTesterService {
  private readonly logger = new Logger(RuleTesterService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly rulesEngine: RulesEngineService,
  ) {}

  /**
   * Test a persisted rule against historical cases.
   *
   * @param ruleId - ID of the rule to test
   * @param organizationId - Tenant ID
   * @param options - Test options
   * @returns Test results with match statistics and samples
   */
  async testRule(
    ruleId: string,
    organizationId: string,
    options: TestRuleOptions = {},
  ): Promise<RuleTestResult> {
    const rule = await this.prisma.ruleDefinition.findFirst({
      where: {
        id: ruleId,
        organizationId,
      },
    });

    if (!rule) {
      throw new NotFoundException(`Rule ${ruleId} not found`);
    }

    const conditions = rule.conditions as RuleConditions;
    const actions = rule.actions as RuleAction[];

    return this.testRuleDefinition(
      { conditions, actions, priority: rule.priority },
      organizationId,
      options,
    );
  }

  /**
   * Test a rule definition (not yet persisted) against historical cases.
   * Useful for validating rules during creation before saving.
   *
   * @param ruleDefinition - Rule conditions and actions
   * @param organizationId - Tenant ID
   * @param options - Test options
   * @returns Test results
   */
  async testRuleDefinition(
    ruleDefinition: {
      conditions: RuleConditions;
      actions: RuleAction[];
      priority?: number;
    },
    organizationId: string,
    options: TestRuleOptions = {},
  ): Promise<RuleTestResult> {
    const startTime = Date.now();
    const limit = options.limit || 100;

    // Load historical cases for simulation
    const historicalCases = await this.loadHistoricalCases(
      organizationId,
      limit,
      options,
    );

    if (historicalCases.length === 0) {
      return {
        totalCases: 0,
        matchedCases: 0,
        matchRate: 0,
        samples: [],
        testedAt: new Date(),
      };
    }

    const samples: RuleTestSample[] = [];
    let matchedCount = 0;

    // Evaluate each historical case
    for (const caseRecord of historicalCases) {
      const facts = this.buildFactsFromCase(caseRecord);

      // Evaluate rule WITHOUT executing actions (dry run)
      const wouldMatch = await this.rulesEngine.evaluateRule(
        {
          conditions: ruleDefinition.conditions,
          priority: ruleDefinition.priority,
        },
        facts,
        { dryRun: true },
      );

      if (wouldMatch) {
        matchedCount++;
      }

      // Collect samples (first 10 of each type)
      const matchedSamples = samples.filter((s) => s.wouldMatch).length;
      const unmatchedSamples = samples.filter((s) => !s.wouldMatch).length;

      if (
        (wouldMatch && matchedSamples < 10) ||
        (!wouldMatch && unmatchedSamples < 10)
      ) {
        samples.push({
          caseId: caseRecord.id,
          referenceNumber: caseRecord.referenceNumber,
          wouldMatch,
          currentAssignee: caseRecord.assignedToId,
          predictedAssignee: wouldMatch
            ? this.predictAssignee(ruleDefinition.actions)
            : null,
          caseDetails: {
            severity: caseRecord.severity,
            categoryName: caseRecord.category?.name,
            locationName: caseRecord.primaryRiu?.location?.name,
            createdAt: caseRecord.createdAt,
          },
        });
      }
    }

    const executionTimeMs = Date.now() - startTime;

    this.logger.log(
      `Rule test completed: ${matchedCount}/${historicalCases.length} matches ` +
        `(${((matchedCount / historicalCases.length) * 100).toFixed(1)}%) in ${executionTimeMs}ms`,
    );

    return {
      totalCases: historicalCases.length,
      matchedCases: matchedCount,
      matchRate: (matchedCount / historicalCases.length) * 100,
      samples: samples.slice(0, 20), // Limit to 20 total samples
      testedAt: new Date(),
    };
  }

  /**
   * Test a rule and save results to the rule definition.
   */
  async testAndSaveResults(
    ruleId: string,
    organizationId: string,
    options: TestRuleOptions = {},
  ): Promise<RuleTestResult> {
    const result = await this.testRule(ruleId, organizationId, options);

    // Save test results to rule definition
    await this.prisma.ruleDefinition.update({
      where: { id: ruleId },
      data: {
        lastTestedAt: result.testedAt,
        testResults: result as any,
      },
    });

    return result;
  }

  /**
   * Load historical cases for testing.
   */
  private async loadHistoricalCases(
    organizationId: string,
    limit: number,
    options: TestRuleOptions,
  ): Promise<HistoricalCase[]> {
    return this.prisma.case.findMany({
      where: {
        organizationId,
        createdAt: options.dateFrom ? { gte: options.dateFrom } : undefined,
        categoryId: options.categoryIds?.length
          ? { in: options.categoryIds }
          : undefined,
        severity: options.severities?.length
          ? { in: options.severities }
          : undefined,
      },
      take: limit,
      orderBy: { createdAt: "desc" },
      select: {
        id: true,
        referenceNumber: true,
        severity: true,
        categoryId: true,
        sourceChannel: true,
        priority: true,
        assignedToId: true,
        createdAt: true,
        category: {
          select: {
            id: true,
            name: true,
            parentId: true,
          },
        },
        primaryRiu: {
          select: {
            locationId: true,
            location: {
              select: {
                id: true,
                name: true,
                regionId: true,
              },
            },
          },
        },
      },
    });
  }

  /**
   * Build facts from a historical case for rule evaluation.
   */
  private buildFactsFromCase(
    caseRecord: HistoricalCase,
  ): Record<string, unknown> {
    return {
      caseId: caseRecord.id,
      referenceNumber: caseRecord.referenceNumber,
      severity: caseRecord.severity,
      categoryId: caseRecord.categoryId,
      categoryName: caseRecord.category?.name,
      parentCategoryId: caseRecord.category?.parentId,
      sourceChannel: caseRecord.sourceChannel,
      priority: caseRecord.priority,
      locationId: caseRecord.primaryRiu?.locationId,
      locationName: caseRecord.primaryRiu?.location?.name,
      regionId: caseRecord.primaryRiu?.location?.regionId,
      // Nested structure for path-based conditions
      case: {
        id: caseRecord.id,
        severity: caseRecord.severity,
        categoryId: caseRecord.categoryId,
        sourceChannel: caseRecord.sourceChannel,
        priority: caseRecord.priority,
      },
      location: {
        id: caseRecord.primaryRiu?.locationId,
        name: caseRecord.primaryRiu?.location?.name,
        regionId: caseRecord.primaryRiu?.location?.regionId,
      },
    };
  }

  /**
   * Predict assignee based on rule actions.
   * Returns a description of what would happen.
   */
  private predictAssignee(actions: RuleAction[]): string | null {
    for (const action of actions) {
      switch (action.type) {
        case "assign_user":
          return `User: ${action.params.userId}`;
        case "assign_team":
          return `Team: ${action.params.teamId}`;
        case "round_robin":
          return `Round-robin: Team ${action.params.teamId}`;
        default:
          continue;
      }
    }
    return null;
  }
}

/**
 * Internal type for historical case data.
 */
interface HistoricalCase {
  id: string;
  referenceNumber: string;
  severity: string;
  categoryId: string | null;
  sourceChannel: string;
  priority: string | null;
  assignedToId: string | null;
  createdAt: Date;
  category: {
    id: string;
    name: string;
    parentId: string | null;
  } | null;
  primaryRiu: {
    locationId: string | null;
    location: {
      id: string;
      name: string;
      regionId: string | null;
    } | null;
  } | null;
}

import { RuleTesterService, TestRuleOptions } from './testing/rule-tester.service';
import { TestRuleDto } from './dto/test-rule.dto';

// Add to constructor:
constructor(
  private readonly rulesService: RulesService,
  private readonly ruleTesterService: RuleTesterService,
) {}

// Add endpoints:

/**
 * Test a rule against historical cases (dry run).
 */
@Post(':id/test')
@Roles(Role.SYSTEM_ADMIN, Role.COMPLIANCE_OFFICER)
@UseGuards(JwtAuthGuard, RolesGuard)
async testRule(
  @Param('id') id: string,
  @Body() dto: TestRuleDto,
  @TenantId() organizationId: string,
) {
  const options: TestRuleOptions = {
    limit: dto.limit,
    dateFrom: dto.dateFrom,
    categoryIds: dto.categoryIds,
    severities: dto.severities,
  };

  return this.ruleTesterService.testAndSaveResults(id, organizationId, options);
}

/**
 * Get test results for a rule (from last test run).
 */
@Get(':id/test-results')
@Roles(Role.SYSTEM_ADMIN, Role.COMPLIANCE_OFFICER)
@UseGuards(JwtAuthGuard, RolesGuard)
async getTestResults(
  @Param('id') id: string,
  @TenantId() organizationId: string,
) {
  const rule = await this.rulesService.findOne(id, organizationId);
  if (!rule) {
    throw new NotFoundException(`Rule ${id} not found`);
  }

  return {
    lastTestedAt: rule.lastTestedAt,
    testResults: rule.testResults,
  };
}

import { Test, TestingModule } from "@nestjs/testing";
import { NotFoundException } from "@nestjs/common";
import { RuleTesterService } from "./rule-tester.service";
import { PrismaService } from "../../prisma/prisma.service";
import { RulesEngineService } from "../engine/rules-engine.service";

describe("RuleTesterService", () => {

  let rulesEngineService: jest.Mocked<RulesEngineService>;

  const mockRule = {
    id: "rule-1",
    organizationId: "org-1",
    name: "Route HIGH to CCO",
    priority: 1,
    conditions: {
      all: [{ fact: "severity", operator: "equal", value: "HIGH" }],
    },
    actions: [{ type: "assign_user", params: { userId: "user-cco" } }],
  };

  const mockCases = [
    {
      id: "case-1",
      referenceNumber: "CASE-001",
      severity: "HIGH",
      categoryId: "cat-1",
      sourceChannel: "hotline",
      priority: null,
      assignedToId: "user-1",

      severity: "LOW",
      categoryId: "cat-2",
      sourceChannel: "web",
      priority: null,
      assignedToId: null,
      createdAt: new Date("2026-01-02"),
      category: { id: "cat-2", name: "Inquiry", parentId: null },
      primaryRiu: null,
    },
  ];

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        RuleTesterService,
        {
          provide: PrismaService,
          useValue: {
            ruleDefinition: {
              findFirst: jest.fn(),
              update: jest.fn(),
            },
            case: {
              findMany: jest.fn(),
            },
          },
        },
        {
          provide: RulesEngineService,
          useValue: {
            evaluateRule: jest.fn(),
          },
        },
      ],
    }).compile();

    service = module.get<RuleTesterService>(RuleTesterService);
    prismaService = module.get(PrismaService);
    rulesEngineService = module.get(RulesEngineService);
  });

  it("should be defined", () => {
    expect(service).toBeDefined();
  });

  describe("testRule", () => {
    it("should throw NotFoundException if rule not found", async () => {
      prismaService.ruleDefinition.findFirst.mockResolvedValue(null);

      await expect(service.testRule("rule-1", "org-1")).rejects.toThrow(
        NotFoundException,
      );
    });

    it("should return empty results if no historical cases", async () => {
      prismaService.ruleDefinition.findFirst.mockResolvedValue(mockRule as any);
      prismaService.case.findMany.mockResolvedValue([]);

      const result = await service.testRule("rule-1", "org-1");

      expect(result.totalCases).toBe(0);
      expect(result.matchedCases).toBe(0);
      expect(result.matchRate).toBe(0);
    });

    it("should calculate match rate correctly", async () => {
      prismaService.ruleDefinition.findFirst.mockResolvedValue(mockRule as any);
      prismaService.case.findMany.mockResolvedValue(mockCases as any);

        .mockResolvedValueOnce(false);

      prismaService.case.findMany.mockResolvedValue(mockCases as any);

      rulesEngineService.evaluateRule.mockResolvedValue(true);

      expect(prismaService.ruleDefinition.update).toHaveBeenCalledWith({
        where: { id: "rule-1" },
        data: expect.objectContaining({
          lastTestedAt: expect.any(Date),
          testResults: expect.any(Object),
        }),
      });
    });
  });
});
