```typescript
import { Injectable, Logger } from "@nestjs/common";
import { OnEvent } from "@nestjs/event-emitter";
import { PrismaService } from "../../prisma/prisma.service";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { InvestigationStatusChangedEvent } from "../../events/events/investigation.events";
import { CaseStatusChangedEvent } from "../../events/events/case.events";

/**
 * Closed investigation statuses that indicate completion.
 */
const CLOSED_STATUSES = ["CLOSED", "COMPLETED", "RESOLVED"];

/**
 * Case statuses where auto-derivation should apply.
 * Don't auto-derive if case is already closed/archived.
 */
const DERIVABLE_CASE_STATUSES = [
  "OPEN",
  "IN_PROGRESS",
  "PENDING",
  "UNDER_INVESTIGATION",
];

/**
 * InvestigationStatusListener derives case status from investigation states.
 *
 * Rule: When ALL investigations on a case are closed, the case automatically
 * moves to PENDING_REVIEW status (unless manually overridden).
 *
 * This implements RULE-06 from the requirements.
 */
@Injectable()
export class InvestigationStatusListener {
  private readonly logger = new Logger(InvestigationStatusListener.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  /**
   * Handle investigation.status_changed events.
   * Check if all investigations are now closed and update case status if so.
   */
  @OnEvent(InvestigationStatusChangedEvent.eventName, { async: true })
  async handleInvestigationStatusChanged(
    event: InvestigationStatusChangedEvent,
  ): Promise<void> {
    // Only process when investigation moves to a closed status
    if (!this.isClosedStatus(event.newStatus)) {
      this.logger.debug(
        `Investigation ${event.investigationId} moved to ${event.newStatus}, not a closed status`,
      );
      return;
    }

    this.logger.debug(
      `Investigation ${event.investigationId} closed, checking case ${event.caseId}`,
    );

    try {
      // Get the parent case and check if it's in a derivable status
      const parentCase = await this.prisma.case.findFirst({
        where: {
          id: event.caseId,
          organizationId: event.organizationId,
        },
        select: {
          id: true,
          status: true,
          referenceNumber: true,
        },
      });

      if (!parentCase) {
        this.logger.warn(
          `Case ${event.caseId} not found for investigation ${event.investigationId}`,
        );
        return;
      }

      // Don't auto-derive if case is already in a terminal status
      if (!DERIVABLE_CASE_STATUSES.includes(parentCase.status)) {
        this.logger.debug(
          `Case ${event.caseId} is in ${parentCase.status}, not auto-deriving`,
        );
        return;
      }

      // Check if ALL investigations for this case are now closed
      const allInvestigationsClosed = await this.checkAllInvestigationsClosed(
        event.caseId,
        event.organizationId,
      );

      if (allInvestigationsClosed) {
        await this.deriveCaseStatus(parentCase, event.organizationId);
      } else {
        this.logger.debug(
          `Case ${event.caseId} still has open investigations, not auto-deriving`,
        );
      }
    } catch (error) {
      this.logger.error(
        `Error processing investigation status change: ${error}`,
      );
      // Don't throw - status derivation failure shouldn't block other listeners
    }
  }

  /**
   * Check if a status is considered "closed".
   */
  private isClosedStatus(status: string): boolean {
    return CLOSED_STATUSES.includes(status.toUpperCase());
  }

  /**
   * Check if all investigations on a case are in closed status.
   */
  private async checkAllInvestigationsClosed(
    caseId: string,
    organizationId: string,
  ): Promise<boolean> {
    // Get all investigations for this case
    const investigations = await this.prisma.investigation.findMany({
      where: {
        caseId,
        organizationId,
      },
      select: { status: true },
    });

    // If no investigations, don't auto-derive (edge case)
    if (investigations.length === 0) {
      return false;
    }

    // Check if ALL are in closed status
    return investigations.every((inv) => this.isClosedStatus(inv.status));
  }

  /**
   * Derive case status to PENDING_REVIEW.
   */
  private async deriveCaseStatus(
    parentCase: { id: string; status: string; referenceNumber: string },
    organizationId: string,
  ): Promise<void> {
    const previousStatus = parentCase.status;
    const newStatus = "PENDING_REVIEW";

    // Update case status
    await this.prisma.case.update({
      where: { id: parentCase.id },
      data: {
        status: newStatus,
        statusChangedAt: new Date(),
        statusRationale: "All investigations completed (auto-derived)",
      },
    });

    // Log to audit
    await this.prisma.auditLog.create({
      data: {
        organizationId,
        entityType: "CASE",
        entityId: parentCase.id,
        action: "status_changed",
        changes: {
          status: { old: previousStatus, new: newStatus },
        },
        actorType: "SYSTEM",
        metadata: {
          reason: "All investigations completed",
          autoDerivation: true,
        },
      },
    });

    // Emit status changed event
    this.eventEmitter.emit(
      CaseStatusChangedEvent.eventName,
      new CaseStatusChangedEvent({
        organizationId,
        caseId: parentCase.id,
        previousStatus,
        newStatus,
        rationale: "All investigations completed (auto-derived)",
        actorUserId: "SYSTEM",
      }),
    );

    this.logger.log(
      `Auto-derived case ${parentCase.referenceNumber} to ${newStatus} ` +
        `(all investigations closed)`,
    );
  }
}
```

```typescript
import { Test, TestingModule } from "@nestjs/testing";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { InvestigationStatusListener } from "./investigation-status.listener";
import { PrismaService } from "../../prisma/prisma.service";
import { InvestigationStatusChangedEvent } from "../../events/events/investigation.events";

describe("InvestigationStatusListener", () => {
  let listener: InvestigationStatusListener;
  let prismaService: jest.Mocked<PrismaService>;
  let eventEmitter: jest.Mocked<EventEmitter2>;

  const mockEvent = new InvestigationStatusChangedEvent({
    organizationId: "org-1",
    investigationId: "inv-1",
    caseId: "case-1",
    previousStatus: "IN_PROGRESS",
    newStatus: "CLOSED",
    actorUserId: "user-1",
  });

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        InvestigationStatusListener,
        {
          provide: PrismaService,
          useValue: {
            case: {
              findFirst: jest.fn(),
              update: jest.fn(),
            },
            investigation: {
              findMany: jest.fn(),
            },
            auditLog: {
              create: jest.fn(),
            },
          },
        },
        {
          provide: EventEmitter2,
          useValue: { emit: jest.fn() },
        },
      ],
    }).compile();

    listener = module.get<InvestigationStatusListener>(
      InvestigationStatusListener,
    );
    prismaService = module.get(PrismaService);
    eventEmitter = module.get(EventEmitter2);
  });

  it("should be defined", () => {
    expect(listener).toBeDefined();
  });

  describe("handleInvestigationStatusChanged", () => {
    it("should skip if new status is not closed", async () => {
      const openEvent = new InvestigationStatusChangedEvent({
        ...mockEvent,
        newStatus: "IN_PROGRESS",
      });

      await listener.handleInvestigationStatusChanged(openEvent);

      expect(prismaService.case.findFirst).not.toHaveBeenCalled();
    });

    it("should skip if case is not in derivable status", async () => {
      prismaService.case.findFirst.mockResolvedValue({
        id: "case-1",
        status: "CLOSED", // Already closed
        referenceNumber: "CASE-001",
      } as any);

      await listener.handleInvestigationStatusChanged(mockEvent);

      expect(prismaService.investigation.findMany).not.toHaveBeenCalled();
    });

    it("should not derive if some investigations are still open", async () => {
      prismaService.case.findFirst.mockResolvedValue({
        id: "case-1",
        status: "IN_PROGRESS",
        referenceNumber: "CASE-001",
      } as any);

      prismaService.investigation.findMany.mockResolvedValue([
        { status: "CLOSED" },
        { status: "IN_PROGRESS" }, // Still open
      ] as any);

      await listener.handleInvestigationStatusChanged(mockEvent);

      expect(prismaService.case.update).not.toHaveBeenCalled();
    });

    it("should derive case to PENDING_REVIEW when all investigations closed", async () => {
      prismaService.case.findFirst.mockResolvedValue({
        id: "case-1",
        status: "IN_PROGRESS",
        referenceNumber: "CASE-001",
      } as any);

      prismaService.investigation.findMany.mockResolvedValue([
        { status: "CLOSED" },
        { status: "CLOSED" },
      ] as any);

      prismaService.case.update.mockResolvedValue({} as any);
      prismaService.auditLog.create.mockResolvedValue({} as any);

      await listener.handleInvestigationStatusChanged(mockEvent);

      expect(prismaService.case.update).toHaveBeenCalledWith({
        where: { id: "case-1" },
        data: expect.objectContaining({
          status: "PENDING_REVIEW",
          statusRationale: expect.stringContaining(
            "All investigations completed",
          ),
        }),
      });
    });

    it("should emit CaseStatusChangedEvent after derivation", async () => {
      prismaService.case.findFirst.mockResolvedValue({
        id: "case-1",
        status: "IN_PROGRESS",
        referenceNumber: "CASE-001",
      } as any);

      prismaService.investigation.findMany.mockResolvedValue([
        { status: "CLOSED" },
      ] as any);

      prismaService.case.update.mockResolvedValue({} as any);
      prismaService.auditLog.create.mockResolvedValue({} as any);

      await listener.handleInvestigationStatusChanged(mockEvent);

      expect(eventEmitter.emit).toHaveBeenCalledWith(
        "case.status_changed",
        expect.objectContaining({
          caseId: "case-1",
          previousStatus: "IN_PROGRESS",
          newStatus: "PENDING_REVIEW",
        }),
      );
    });

    it("should create audit log entry for auto-derivation", async () => {
      prismaService.case.findFirst.mockResolvedValue({
        id: "case-1",
        status: "IN_PROGRESS",
        referenceNumber: "CASE-001",
      } as any);

      prismaService.investigation.findMany.mockResolvedValue([
        { status: "COMPLETED" }, // Also a closed status
      ] as any);

      prismaService.case.update.mockResolvedValue({} as any);
      prismaService.auditLog.create.mockResolvedValue({} as any);

      await listener.handleInvestigationStatusChanged(mockEvent);

      expect(prismaService.auditLog.create).toHaveBeenCalledWith({
        data: expect.objectContaining({
          entityType: "CASE",
          entityId: "case-1",
          action: "status_changed",
          actorType: "SYSTEM",
          metadata: expect.objectContaining({
            autoDerivation: true,
          }),
        }),
      });
    });

    it("should handle COMPLETED and RESOLVED as closed statuses", async () => {
      prismaService.case.findFirst.mockResolvedValue({
        id: "case-1",
        status: "UNDER_INVESTIGATION",
        referenceNumber: "CASE-001",
      } as any);

      prismaService.investigation.findMany.mockResolvedValue([
        { status: "COMPLETED" },
        { status: "RESOLVED" },
      ] as any);

      prismaService.case.update.mockResolvedValue({} as any);
      prismaService.auditLog.create.mockResolvedValue({} as any);

      const completedEvent = new InvestigationStatusChangedEvent({
        ...mockEvent,
        newStatus: "COMPLETED",
      });

      await listener.handleInvestigationStatusChanged(completedEvent);

      expect(prismaService.case.update).toHaveBeenCalled();
    });
  });
});
```
