```typescript
import { Injectable, Logger } from "@nestjs/common";
import { PrismaService } from "../../../prisma/prisma.service";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { CaseAssignedEvent } from "../../../events/events/case.events";
import type {
  RuleActionExecutor,
  ActionContext,
  ActionResult,
} from "./base.action";

/**
 * Parameters for round-robin team action.
 */
interface RoundRobinParams {
  teamId: string;
  /** Optional: only consider users with these roles */
  roleFilter?: string[];
  /** Optional: skip users with more than N open cases */
  maxOpenCases?: number;
}

/**
 * RoundRobinTeamAction distributes assignments fairly across team members.
 *
 * Algorithm:
 * 1. Get all active team members (optionally filtered by role)
 * 2. Find the last user who was assigned via this team's round-robin
 * 3. Assign to the next user in the list (wrapping around)
 * 4. Skip users with too many open cases if maxOpenCases is set
 *
 * Distribution is tracked via RuleExecutionLog to ensure consistency
 * across distributed system restarts.
 */
@Injectable()
export class RoundRobinTeamAction implements RuleActionExecutor {
  readonly type = "round_robin";
  private readonly logger = new Logger(RoundRobinTeamAction.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  async execute(
    params: Record<string, unknown>,
    context: ActionContext,
  ): Promise<ActionResult> {
    const teamId = params.teamId as string;
    const roleFilter = params.roleFilter as string[] | undefined;
    const maxOpenCases = params.maxOpenCases as number | undefined;

    if (!teamId) {
      return {
        success: false,
        actionType: this.type,
        details: {},
        error: "Missing teamId parameter",
      };
    }

    try {
      // Verify team exists in same org
      const team = await this.prisma.team.findFirst({
        where: {
          id: teamId,
          organizationId: context.organizationId,
        },
        select: { id: true, name: true },
      });

      if (!team) {
        return {
          success: false,
          actionType: this.type,
          details: { teamId },
          error: "Team not found",
        };
      }

      // Get eligible team members
      const members = await this.getEligibleMembers(
        teamId,
        context.organizationId,
        roleFilter,
        maxOpenCases,
      );

      if (members.length === 0) {
        this.logger.warn(`No eligible team members found for team ${teamId}`);
        return {
          success: false,
          actionType: this.type,
          details: { teamId, teamName: team.name },
          error: "No eligible team members",
        };
      }

      // Find the next assignee using round-robin
      const nextAssignee = await this.findNextAssignee(
        members,
        teamId,
        context.organizationId,
      );

      // Get current assignee for event
      const currentCase = await this.prisma.case.findUnique({
        where: { id: context.entityId },
        select: { assignedToId: true },
      });

      // Update case assignment
      await this.prisma.case.update({
        where: { id: context.entityId },
        data: {
          assignedToId: nextAssignee.id,
          assignedTeamId: teamId,
        },
      });

      // Emit assignment event
      this.eventEmitter.emit(
        CaseAssignedEvent.eventName,
        new CaseAssignedEvent({
          organizationId: context.organizationId,
          caseId: context.entityId,
          previousAssigneeId: currentCase?.assignedToId || null,
          newAssigneeId: nextAssignee.id,
          actorUserId: "SYSTEM",
        }),
      );

      this.logger.log(
        `Round-robin assigned case ${context.entityId} to ${nextAssignee.firstName} ${nextAssignee.lastName} ` +
          `(team: ${team.name}, rule: ${context.triggeredByRuleId})`,
      );

      return {
        success: true,
        actionType: this.type,
        details: {
          teamId,
          teamName: team.name,
          userId: nextAssignee.id,
          userName: `${nextAssignee.firstName} ${nextAssignee.lastName}`,
          reason: `Round-robin team assignment (rule: ${context.triggeredByRuleId})`,
          memberIndex: members.findIndex((m) => m.id === nextAssignee.id),
          totalMembers: members.length,
        },
      };
    } catch (error) {
      this.logger.error(`Round-robin assignment failed: ${error}`);
      return {
        success: false,
        actionType: this.type,
        details: { teamId },
        error: error instanceof Error ? error.message : "Unknown error",
      };
    }
  }

  /**
   * Get eligible team members for assignment.
   */
  private async getEligibleMembers(
    teamId: string,
    organizationId: string,
    roleFilter?: string[],
    maxOpenCases?: number,
  ): Promise<TeamMember[]> {
    // Base query for active team members
    const members = await this.prisma.user.findMany({
      where: {
        organizationId,
        teamId,
        isActive: true,
        ...(roleFilter?.length && {
          role: { in: roleFilter as any },
        }),
      },
      select: {
        id: true,
        firstName: true,
        lastName: true,
        createdAt: true,
      },
      orderBy: { createdAt: "asc" }, // Consistent ordering for round-robin
    });

    // If maxOpenCases is set, filter out users with too many cases
    if (maxOpenCases !== undefined && maxOpenCases > 0) {
      const eligibleMembers: TeamMember[] = [];

      for (const member of members) {
        const openCaseCount = await this.prisma.case.count({
          where: {
            organizationId,
            assignedToId: member.id,
            status: { in: ["OPEN", "IN_PROGRESS", "PENDING"] },
          },
        });

        if (openCaseCount < maxOpenCases) {
          eligibleMembers.push(member);
        }
      }

      return eligibleMembers;
    }

    return members;
  }

  /**
   * Find the next user in round-robin sequence.
   *
   * Uses RuleExecutionLog to track last assignment per team, ensuring
   * consistent distribution even across system restarts.
   */
  private async findNextAssignee(
    members: TeamMember[],
    teamId: string,
    organizationId: string,
  ): Promise<TeamMember> {
    // Find the last round-robin assignment for this team
    const lastAssignment = await this.prisma.ruleExecutionLog.findFirst({
      where: {
        organizationId,
        matched: true,
        actionsTaken: {
          path: ["$[*].actionType"],
          array_contains: "round_robin",
        },
      },
      orderBy: { executedAt: "desc" },
      select: { actionsTaken: true },
    });

    // Parse last assigned user from action results
    let lastUserId: string | null = null;
    if (lastAssignment?.actionsTaken) {
      const actions = lastAssignment.actionsTaken as ActionResult[];
      const roundRobinAction = actions.find(
        (a) =>
          a.actionType === "round_robin" &&
          a.success &&
          (a.details as any)?.teamId === teamId,
      );
      if (roundRobinAction) {
        lastUserId = (roundRobinAction.details as any)?.userId;
      }
    }

    // Determine next index
    let nextIndex = 0;
    if (lastUserId) {
      const lastIndex = members.findIndex((m) => m.id === lastUserId);
      if (lastIndex >= 0) {
        nextIndex = (lastIndex + 1) % members.length;
      }
    }

    return members[nextIndex];
  }
}

interface TeamMember {
  id: string;
  firstName: string;
  lastName: string;
  createdAt: Date;
}
```

```typescript
import { Test, TestingModule } from "@nestjs/testing";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { RoundRobinTeamAction } from "./round-robin-team.action";
import { PrismaService } from "../../../prisma/prisma.service";
import type { ActionContext } from "./base.action";

describe("RoundRobinTeamAction", () => {
  let action: RoundRobinTeamAction;
  let prismaService: jest.Mocked<PrismaService>;
  let eventEmitter: jest.Mocked<EventEmitter2>;

  const mockTeam = { id: "team-1", name: "Investigators" };
  const mockMembers = [
    {
      id: "user-1",
      firstName: "Alice",
      lastName: "Smith",
      createdAt: new Date("2024-01-01"),
    },
    {
      id: "user-2",
      firstName: "Bob",
      lastName: "Jones",
      createdAt: new Date("2024-01-02"),
    },
    {
      id: "user-3",
      firstName: "Carol",
      lastName: "Davis",
      createdAt: new Date("2024-01-03"),
    },
  ];

  const mockContext: ActionContext = {
    organizationId: "org-1",
    entityType: "CASE",
    entityId: "case-123",
    triggeredByRuleId: "rule-1",
    actorType: "SYSTEM",
  };

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        RoundRobinTeamAction,
        {
          provide: PrismaService,
          useValue: {
            team: { findFirst: jest.fn() },
            user: { findMany: jest.fn() },
            case: {
              findUnique: jest.fn(),
              update: jest.fn(),
              count: jest.fn(),
            },
            ruleExecutionLog: { findFirst: jest.fn() },
          },
        },
        {
          provide: EventEmitter2,
          useValue: { emit: jest.fn() },
        },
      ],
    }).compile();

    action = module.get<RoundRobinTeamAction>(RoundRobinTeamAction);
    prismaService = module.get(PrismaService);
    eventEmitter = module.get(EventEmitter2);
  });

  it("should be defined", () => {
    expect(action).toBeDefined();
    expect(action.type).toBe("round_robin");
  });

  describe("execute", () => {
    it("should fail if teamId is missing", async () => {
      const result = await action.execute({}, mockContext);

      expect(result.success).toBe(false);
      expect(result.error).toContain("Missing teamId");
    });

    it("should fail if team not found", async () => {
      prismaService.team.findFirst.mockResolvedValue(null);

      const result = await action.execute({ teamId: "team-1" }, mockContext);

      expect(result.success).toBe(false);
      expect(result.error).toContain("Team not found");
    });

    it("should fail if no eligible members", async () => {
      prismaService.team.findFirst.mockResolvedValue(mockTeam as any);
      prismaService.user.findMany.mockResolvedValue([]);

      const result = await action.execute({ teamId: "team-1" }, mockContext);

      expect(result.success).toBe(false);
      expect(result.error).toContain("No eligible team members");
    });

    it("should assign to first member when no prior assignments", async () => {
      prismaService.team.findFirst.mockResolvedValue(mockTeam as any);
      prismaService.user.findMany.mockResolvedValue(mockMembers as any);
      prismaService.ruleExecutionLog.findFirst.mockResolvedValue(null);
      prismaService.case.findUnique.mockResolvedValue({
        assignedToId: null,
      } as any);
      prismaService.case.update.mockResolvedValue({} as any);

      const result = await action.execute({ teamId: "team-1" }, mockContext);

      expect(result.success).toBe(true);
      expect(result.details.userId).toBe("user-1");
      expect(result.details.userName).toBe("Alice Smith");
      expect(result.details.memberIndex).toBe(0);
    });

    it("should assign to next member in sequence", async () => {
      prismaService.team.findFirst.mockResolvedValue(mockTeam as any);
      prismaService.user.findMany.mockResolvedValue(mockMembers as any);
      // Last assignment was to user-1
      prismaService.ruleExecutionLog.findFirst.mockResolvedValue({
        actionsTaken: [
          {
            actionType: "round_robin",
            success: true,
            details: { teamId: "team-1", userId: "user-1" },
          },
        ],
      } as any);
      prismaService.case.findUnique.mockResolvedValue({
        assignedToId: null,
      } as any);
      prismaService.case.update.mockResolvedValue({} as any);

      const result = await action.execute({ teamId: "team-1" }, mockContext);

      expect(result.success).toBe(true);
      expect(result.details.userId).toBe("user-2");
      expect(result.details.userName).toBe("Bob Jones");
      expect(result.details.memberIndex).toBe(1);
    });

    it("should wrap around to first member after last", async () => {
      prismaService.team.findFirst.mockResolvedValue(mockTeam as any);
      prismaService.user.findMany.mockResolvedValue(mockMembers as any);
      // Last assignment was to user-3 (last in list)
      prismaService.ruleExecutionLog.findFirst.mockResolvedValue({
        actionsTaken: [
          {
            actionType: "round_robin",
            success: true,
            details: { teamId: "team-1", userId: "user-3" },
          },
        ],
      } as any);
      prismaService.case.findUnique.mockResolvedValue({
        assignedToId: null,
      } as any);
      prismaService.case.update.mockResolvedValue({} as any);

      const result = await action.execute({ teamId: "team-1" }, mockContext);

      expect(result.success).toBe(true);
      expect(result.details.userId).toBe("user-1"); // Wrapped around
      expect(result.details.memberIndex).toBe(0);
    });

    it("should emit CaseAssignedEvent on success", async () => {
      prismaService.team.findFirst.mockResolvedValue(mockTeam as any);
      prismaService.user.findMany.mockResolvedValue(mockMembers as any);
      prismaService.ruleExecutionLog.findFirst.mockResolvedValue(null);
      prismaService.case.findUnique.mockResolvedValue({
        assignedToId: null,
      } as any);
      prismaService.case.update.mockResolvedValue({} as any);

      await action.execute({ teamId: "team-1" }, mockContext);

      expect(eventEmitter.emit).toHaveBeenCalledWith(
        "case.assigned",
        expect.objectContaining({
          caseId: "case-123",
          newAssigneeId: "user-1",
        }),
      );
    });
  });
});
```
