```typescript
import { PrismaClient } from "@prisma/client";

/**
 * Phase 40: Rules Engine Foundation - Acme Co. Demo Data
 *
 * Seeds realistic routing rules for the demo tenant:
 * - Route HIGH severity cases to CCO
 * - Route Fraud category to Legal team
 * - Round-robin distribution for general cases
 * - Auto-escalation triggers (for Phase 41)
 */
export async function seedAcmePhase40(prisma: PrismaClient): Promise<void> {
  console.log("Seeding Phase 40: Rules Engine Foundation...");

  // Get Acme organization
  const acme = await prisma.organization.findUnique({
    where: { slug: "acme" },
  });

  if (!acme) {
    console.log("Acme organization not found, skipping Phase 40 seed");
    return;
  }

  // Get existing users for assignment rules
  const ccoUser = await prisma.user.findFirst({
    where: {
      organizationId: acme.id,
      role: "COMPLIANCE_OFFICER",
    },
  });

  const investigators = await prisma.user.findMany({
    where: {
      organizationId: acme.id,
      role: "INVESTIGATOR",
      isActive: true,
    },
    take: 3,
  });

  // Get a team for round-robin (create if needed)
  let investigationTeam = await prisma.team.findFirst({
    where: {
      organizationId: acme.id,
      name: "Investigation Team",
    },
  });

  if (!investigationTeam && investigators.length > 0) {
    investigationTeam = await prisma.team.create({
      data: {
        organizationId: acme.id,
        name: "Investigation Team",
        description: "Primary investigation team for case routing",
      },
    });

    // Associate investigators with team
    await prisma.user.updateMany({
      where: {
        id: { in: investigators.map((i) => i.id) },
      },
      data: {
        teamId: investigationTeam.id,
      },
    });
  }

  // Get fraud category for category-based routing
  const fraudCategory = await prisma.category.findFirst({
    where: {
      organizationId: acme.id,
      name: { contains: "Fraud", mode: "insensitive" },
    },
  });

  // Create demo routing rules
  const rules = [
    {
      name: "Route HIGH/CRITICAL to CCO",
      description:
        "High and critical severity cases go directly to the Chief Compliance Officer for immediate attention",
      priority: 10,
      isActive: true,
      triggerEvent: "case.created",
      conditions: {
        any: [
          { fact: "severity", operator: "severityEquals", value: "HIGH" },
          { fact: "severity", operator: "severityEquals", value: "CRITICAL" },
        ],
      },
      actions: ccoUser
        ? [{ type: "assign_user", params: { userId: ccoUser.id } }]
        : [],
    },
    {
      name: "Route Fraud to Legal Review",
      description:
        "Fraud-related cases are routed to legal team for compliance review",
      priority: 20,
      isActive: fraudCategory && ccoUser,
      triggerEvent: "case.created",
      conditions: {
        all: [
          {
            fact: "categoryId",
            operator: "categoryEquals",
            value: fraudCategory?.id || "cat-fraud",
          },
        ],
      },
      actions: ccoUser
        ? [{ type: "assign_user", params: { userId: ccoUser.id } }]
        : [],
    },
    {
      name: "Round-Robin General Cases",
      description:
        "Distribute general cases evenly across the investigation team",
      priority: 100,
      isActive: investigationTeam !== null,
      triggerEvent: "case.created",
      conditions: {
        all: [
          {
            fact: "severity",
            operator: "severityIn",
            value: ["LOW", "MEDIUM"],
          },
        ],
      },
      actions: investigationTeam
        ? [
            {
              type: "round_robin",
              params: {
                teamId: investigationTeam.id,
                maxOpenCases: 10,
              },
            },
          ]
        : [],
    },
    {
      name: "Hotline Reports to Triage",
      description:
        "Cases from hotline channel get priority routing to triage lead",
      priority: 15,
      isActive: false, // Inactive - for testing
      triggerEvent: "case.created",
      conditions: {
        all: [{ fact: "sourceChannel", operator: "equal", value: "hotline" }],
      },
      actions: [
        {
          type: "assign_team",
          params: { teamId: investigationTeam?.id || "team-triage" },
        },
      ],
    },
  ];

  // Get admin user for createdBy
  const adminUser = await prisma.user.findFirst({
    where: {
      organizationId: acme.id,
      role: "SYSTEM_ADMIN",
    },
  });

  if (!adminUser) {
    console.log("No admin user found, skipping rule creation");
    return;
  }

  // Upsert rules
  for (const rule of rules) {
    if (rule.actions.length === 0) {
      console.log(`Skipping rule "${rule.name}" - no valid actions`);
      continue;
    }

    await prisma.ruleDefinition.upsert({
      where: {
        // Use a composite unique check via findFirst + create/update
        id: `seed-rule-${rule.name.toLowerCase().replace(/\s+/g, "-")}`,
      },
      create: {
        id: `seed-rule-${rule.name.toLowerCase().replace(/\s+/g, "-")}`,
        organizationId: acme.id,
        name: rule.name,
        description: rule.description,
        priority: rule.priority,
        isActive: Boolean(rule.isActive),
        triggerEvent: rule.triggerEvent,
        conditions: rule.conditions,
        actions: rule.actions,
        createdById: adminUser.id,
      },
      update: {
        name: rule.name,
        description: rule.description,
        priority: rule.priority,
        isActive: Boolean(rule.isActive),
        conditions: rule.conditions,
        actions: rule.actions,
      },
    });

    console.log(`Created/updated rule: ${rule.name}`);
  }

  // Create some sample execution logs for demo purposes
  const activeRules = await prisma.ruleDefinition.findMany({
    where: {
      organizationId: acme.id,
      isActive: true,
    },
    take: 2,
  });

  const recentCases = await prisma.case.findMany({
    where: {
      organizationId: acme.id,
    },
    take: 5,
    orderBy: { createdAt: "desc" },
  });

  // Create execution logs for demo
  for (let i = 0; i < Math.min(5, recentCases.length); i++) {
    const caseRecord = recentCases[i];
    const rule = activeRules[i % activeRules.length];

    if (!rule) continue;

    const existingLog = await prisma.ruleExecutionLog.findFirst({
      where: {
        organizationId: acme.id,
        entityId: caseRecord.id,
        ruleId: rule.id,
      },
    });

    if (!existingLog) {
      await prisma.ruleExecutionLog.create({
        data: {
          organizationId: acme.id,
          ruleId: rule.id,
          entityType: "CASE",
          entityId: caseRecord.id,
          facts: {
            caseId: caseRecord.id,
            severity: caseRecord.severity,
            categoryId: caseRecord.categoryId,
            sourceChannel: caseRecord.sourceChannel,
          },
          matched: i % 2 === 0, // Alternate match/no-match for demo
          actionsTaken:
            i % 2 === 0
              ? [
                  {
                    success: true,
                    actionType: "assign_user",
                    details: { userId: adminUser.id },
                  },
                ]
              : null,
          executionTimeMs: Math.floor(Math.random() * 50) + 5,
        },
      });
    }
  }

  console.log("Phase 40 seed complete");
}
```
