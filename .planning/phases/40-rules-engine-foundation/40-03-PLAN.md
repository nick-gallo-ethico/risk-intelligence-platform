import { Injectable, Logger } from "@nestjs/common";
import { OnEvent } from "@nestjs/event-emitter";
import { PrismaService } from "../../prisma/prisma.service";
import { RulesEngineService } from "../engine/rules-engine.service";
import { CaseCreatedEvent } from "../../events/events/case.events";
import type { ActionContext } from "../engine/actions/base.action";

/**
 * CaseRoutingListener evaluates routing rules when new cases are created.
 *
 * This listener:
 * 1. Receives case.created events
 * 2. Builds facts from case data
 * 3. Evaluates active routing rules
 * 4. Executes matching rule actions (assignment)
 * 5. Logs all evaluations to audit trail
 *
 * CRITICAL: Uses { async: true } to avoid blocking case creation response.
 */
@Injectable()
export class CaseRoutingListener {
  private readonly logger = new Logger(CaseRoutingListener.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly rulesEngine: RulesEngineService,
  ) {}

  /**
   * Handle case.created events by evaluating routing rules.
   *
   * @param event - CaseCreatedEvent with case details
   */
  @OnEvent(CaseCreatedEvent.eventName, { async: true })
  async handleCaseCreated(event: CaseCreatedEvent): Promise<void> {
    this.logger.debug(`Evaluating routing rules for case ${event.caseId}`);

    try {
      // Skip if case already has an assignee (avoid re-routing)
      const existingCase = await this.prisma.case.findUnique({
        where: { id: event.caseId },
        select: { assignedToId: true, assignedTeamId: true },
      });

      if (existingCase?.assignedToId || existingCase?.assignedTeamId) {
        this.logger.debug(
          `Case ${event.caseId} already assigned, skipping routing rules`,
        );
        return;
      }

      // Load additional case data for facts
      const caseData = await this.loadCaseData(
        event.caseId,
        event.organizationId,
      );
      if (!caseData) {
        this.logger.warn(`Case ${event.caseId} not found, skipping routing`);
        return;
      }

      // Build facts from case data
      const facts = this.buildFacts(event, caseData);

      // Evaluate routing rules
      const result = await this.rulesEngine.evaluate(
        event.organizationId,
        "case.created",
        facts,
      );

      // Log evaluation regardless of match (audit completeness)
      if (result.matchedRuleId) {
        // Execute actions if matched
        const context: ActionContext = {
          organizationId: event.organizationId,
          entityType: "CASE",
          entityId: event.caseId,
          triggeredByRuleId: result.matchedRuleId,
          actorType: "SYSTEM",
        };

        const actionResults = await this.rulesEngine.executeActions(
          result.triggeredActions,
          context,
        );

        // Log execution with action results
        await this.rulesEngine.logExecution(
          event.organizationId,
          result.matchedRuleId,
          "CASE",
          event.caseId,
          result,
          actionResults.actions,
        );

        this.logger.log(
          `Routing rule "${result.matchedRuleName}" applied to case ${event.caseId}. ` +
            `Actions: ${actionResults.actions.length}, Success: ${actionResults.allSuccessful}`,
        );
      } else {
        this.logger.debug(`No routing rules matched for case ${event.caseId}`);

        // Log no-match evaluation for audit trail (find first active rule to log against)
        const firstActiveRule = await this.prisma.ruleDefinition.findFirst({
          where: {
            organizationId: event.organizationId,
            triggerEvent: "case.created",
            isActive: true,
          },
          select: { id: true },
        });

        if (firstActiveRule) {
          await this.rulesEngine.logExecution(
            event.organizationId,
            firstActiveRule.id,
            "CASE",
            event.caseId,
            result,
          );
        }
      }
    } catch (error) {
      this.logger.error(
        `Error evaluating routing rules for case ${event.caseId}: ${error}`,
      );
      // Don't throw - routing failure shouldn't block case creation
    }
  }

  /**
   * Load additional case data needed for fact building.
   */
  private async loadCaseData(
    caseId: string,
    organizationId: string,
  ): Promise<CaseFactData | null> {
    const caseRecord = await this.prisma.case.findFirst({
      where: {
        id: caseId,
        organizationId,
      },
      select: {
        id: true,
        referenceNumber: true,
        severity: true,
        categoryId: true,
        sourceChannel: true,
        priority: true,
        category: {
          select: {
            id: true,
            name: true,
            parentId: true,
          },
        },
        primaryRiu: {
          select: {
            locationId: true,
            location: {
              select: {
                id: true,
                name: true,
                regionId: true,
              },
            },
          },
        },
      },
    });

    if (!caseRecord) return null;

    return {
      caseId: caseRecord.id,
      referenceNumber: caseRecord.referenceNumber,
      severity: caseRecord.severity,
      categoryId: caseRecord.categoryId,
      categoryName: caseRecord.category?.name,
      parentCategoryId: caseRecord.category?.parentId,
      sourceChannel: caseRecord.sourceChannel,
      priority: caseRecord.priority,
      locationId: caseRecord.primaryRiu?.locationId,
      locationName: caseRecord.primaryRiu?.location?.name,
      regionId: caseRecord.primaryRiu?.location?.regionId,
    };
  }

  /**
   * Build facts object for rule evaluation.
   */
  private buildFacts(
    event: CaseCreatedEvent,
    caseData: CaseFactData,
  ): Record<string, unknown> {
    return {
      // Core case facts (flat for easy rule authoring)
      caseId: caseData.caseId,
      referenceNumber: caseData.referenceNumber,
      severity: caseData.severity,
      categoryId: caseData.categoryId,
      categoryName: caseData.categoryName,
      parentCategoryId: caseData.parentCategoryId,
      sourceChannel: caseData.sourceChannel,
      priority: caseData.priority,

      // Location facts
      locationId: caseData.locationId,
      locationName: caseData.locationName,
      regionId: caseData.regionId,

      // Nested structure for path-based conditions
      case: {
        id: caseData.caseId,
        severity: caseData.severity,
        categoryId: caseData.categoryId,
        sourceChannel: caseData.sourceChannel,
        priority: caseData.priority,
      },
      location: {
        id: caseData.locationId,
        name: caseData.locationName,
        regionId: caseData.regionId,
      },
    };
  }
}

/**
 * Internal type for case data used in fact building.
 */
interface CaseFactData {
  caseId: string;
  referenceNumber: string;
  severity: string;
  categoryId: string | null;
  categoryName?: string;
  parentCategoryId?: string | null;
  sourceChannel: string;
  priority?: string | null;
  locationId?: string | null;
  locationName?: string;
  regionId?: string | null;
}

import { Test, TestingModule } from "@nestjs/testing";
import { CaseRoutingListener } from "./case-routing.listener";
import { PrismaService } from "../../prisma/prisma.service";
import { RulesEngineService } from "../engine/rules-engine.service";
import { CaseCreatedEvent } from "../../events/events/case.events";

describe("CaseRoutingListener", () => {
  let listener: CaseRoutingListener;
  let prismaService: jest.Mocked<PrismaService>;
  let rulesEngineService: jest.Mocked<RulesEngineService>;

  const mockCase = {
    id: "case-123",
    referenceNumber: "CASE-2026-001",
    severity: "HIGH",
    categoryId: "cat-fraud",
    sourceChannel: "hotline",
    priority: "HIGH",
    assignedToId: null,
    assignedTeamId: null,
    category: {
      id: "cat-fraud",
      name: "Fraud",
      parentId: null,
    },
    primaryRiu: {
      locationId: "loc-ny",
      location: {
        id: "loc-ny",
        name: "New York",
        regionId: "reg-east",
      },
    },
  };

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        CaseRoutingListener,
        {
          provide: PrismaService,
          useValue: {
            case: {
              findUnique: jest.fn(),
              findFirst: jest.fn(),
            },
            ruleDefinition: {
              findFirst: jest.fn(),
            },
          },
        },
        {
          provide: RulesEngineService,
          useValue: {
            evaluate: jest.fn(),
            executeActions: jest.fn(),
            logExecution: jest.fn(),
          },
        },
      ],
    }).compile();

    listener = module.get<CaseRoutingListener>(CaseRoutingListener);
    prismaService = module.get(PrismaService);
    rulesEngineService = module.get(RulesEngineService);
  });

  it("should be defined", () => {
    expect(listener).toBeDefined();
  });

  describe("handleCaseCreated", () => {
    it("should skip routing if case already assigned", async () => {
      prismaService.case.findUnique.mockResolvedValue({
        assignedToId: "user-123",
        assignedTeamId: null,
      } as any);

      const event = new CaseCreatedEvent({
        organizationId: "org-1",
        caseId: "case-123",
        referenceNumber: "CASE-001",
        sourceChannel: "hotline",
        severity: "HIGH",
      });

      await listener.handleCaseCreated(event);

      expect(rulesEngineService.evaluate).not.toHaveBeenCalled();
    });

    it("should evaluate rules for unassigned case", async () => {
      prismaService.case.findUnique.mockResolvedValue({
        assignedToId: null,
        assignedTeamId: null,
      } as any);

      prismaService.case.findFirst.mockResolvedValue(mockCase as any);

      rulesEngineService.evaluate.mockResolvedValue({
        matched: true,
        matchedRuleId: "rule-1",
        matchedRuleName: "Route HIGH to CCO",
        triggeredActions: [
          { type: "assign_user", params: { userId: "user-cco" } },
        ],
        executionTimeMs: 15,
        facts: {},
      });

      rulesEngineService.executeActions.mockResolvedValue({
        ruleId: "rule-1",
        actions: [{ success: true, actionType: "assign_user", details: {} }],
        allSuccessful: true,
      });

      const event = new CaseCreatedEvent({
        organizationId: "org-1",
        caseId: "case-123",
        referenceNumber: "CASE-001",
        sourceChannel: "hotline",
        severity: "HIGH",
      });

      await listener.handleCaseCreated(event);

      expect(rulesEngineService.evaluate).toHaveBeenCalledWith(
        "org-1",
        "case.created",
        expect.objectContaining({
          caseId: "case-123",
          severity: "HIGH",
        }),
      );

      expect(rulesEngineService.executeActions).toHaveBeenCalled();
      expect(rulesEngineService.logExecution).toHaveBeenCalled();
    });

    it("should log evaluation even when no rules match", async () => {
      prismaService.case.findUnique.mockResolvedValue({
        assignedToId: null,
        assignedTeamId: null,
      } as any);

      prismaService.case.findFirst.mockResolvedValue(mockCase as any);

      rulesEngineService.evaluate.mockResolvedValue({
        matched: false,
        triggeredActions: [],
        executionTimeMs: 5,
        facts: {},
      });

      prismaService.ruleDefinition.findFirst.mockResolvedValue({
        id: "rule-1",
      } as any);

      const event = new CaseCreatedEvent({
        organizationId: "org-1",
        caseId: "case-123",
        referenceNumber: "CASE-001",
        sourceChannel: "hotline",
        severity: "LOW",
      });

      await listener.handleCaseCreated(event);

      expect(rulesEngineService.executeActions).not.toHaveBeenCalled();
      expect(rulesEngineService.logExecution).toHaveBeenCalled();
    });
  });
});
