```typescript
import { Injectable, Logger } from "@nestjs/common";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { Engine, Rule } from "json-rules-engine";
import { PrismaService } from "../../prisma/prisma.service";
import { AuditService } from "../../audit/audit.service";
import { AutoClearRejectRule, DisclosureType, RiuStatus } from "@prisma/client";

export interface AutoClearRejectResult {
  evaluated: boolean;
  matchedRule: { id: string; name: string; action: string } | null;
  actionTaken: "AUTO_CLEARED" | "AUTO_REJECTED" | "FLAGGED_FOR_REVIEW" | "NONE";
}

@Injectable()
export class DisclosureAutomationService {
  private readonly logger = new Logger(DisclosureAutomationService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  /**
   * Evaluates a disclosure against auto-clear/reject rules.
   * Called after disclosure submission to determine automation action.
   */
  async evaluateDisclosure(
    disclosureId: string,
    organizationId: string,
    disclosureType: DisclosureType,
    answers: Record<string, unknown>,
    userId: string,
  ): Promise<AutoClearRejectResult> {
    // Load active rules for this disclosure type, ordered by priority
    const rules = await this.prisma.autoClearRejectRule.findMany({
      where: {
        organizationId,
        isActive: true,
        disclosureTypes: { has: disclosureType },
      },
      orderBy: { priority: "asc" },
    });

    if (rules.length === 0) {
      return { evaluated: true, matchedRule: null, actionTaken: "NONE" };
    }

    // Evaluate each rule in priority order (first match wins)
    for (const rule of rules) {
      const matched = await this.evaluateRule(rule, answers);

      if (matched) {
        const result = await this.executeAction(
          rule,
          disclosureId,
          organizationId,
          userId,
        );
        return {
          evaluated: true,
          matchedRule: { id: rule.id, name: rule.name, action: rule.action },
          actionTaken: result,
        };
      }
    }

    return { evaluated: true, matchedRule: null, actionTaken: "NONE" };
  }

  private async evaluateRule(
    rule: AutoClearRejectRule,
    answers: Record<string, unknown>,
  ): Promise<boolean> {
    const engine = new Engine();
    const conditions = rule.conditions as { all?: unknown[]; any?: unknown[] };

    engine.addRule(
      new Rule({
        conditions: conditions as any,
        event: { type: "rule-matched" },
      }),
    );

    // Build facts from answers
    const facts = this.buildFacts(answers);
    const { events } = await engine.run(facts);

    return events.length > 0;
  }

  private buildFacts(
    answers: Record<string, unknown>,
  ): Record<string, unknown> {
    // Common disclosure facts
    return {
      ...answers,
      hasItemsToDisclose: Boolean(answers.hasItemsToDisclose),
      confirmNoDisclosure: Boolean(answers.confirmNoDisclosure),
      disclosureValue: Number(answers.disclosureValue) || 0,
      disclosureCount: Array.isArray(answers.items) ? answers.items.length : 0,
    };
  }

  private async executeAction(
    rule: AutoClearRejectRule,
    disclosureId: string,
    organizationId: string,
    userId: string,
  ): Promise<"AUTO_CLEARED" | "AUTO_REJECTED" | "FLAGGED_FOR_REVIEW"> {
    if (rule.requiresReview) {
      // Flag for human review instead of auto-action
      await this.prisma.riskIntelligenceUnit.update({
        where: { id: disclosureId },
        data: {
          status: RiuStatus.PENDING_REVIEW,
          statusChangedAt: new Date(),
        },
      });

      await this.auditService.log({
        entityType: "DISCLOSURE",
        entityId: disclosureId,
        action: "auto_flagged_for_review",
        actionCategory: "UPDATE",
        actionDescription: `Auto-flagged for review by rule "${rule.name}"`,
        actorUserId: userId,
        actorType: "SYSTEM",
        organizationId,
      });

      return "FLAGGED_FOR_REVIEW";
    }

    if (rule.action === "AUTO_CLEAR") {
      await this.prisma.riskIntelligenceUnit.update({
        where: { id: disclosureId },
        data: {
          status: RiuStatus.COMPLETED,
          statusChangedAt: new Date(),
        },
      });

      await this.auditService.log({
        entityType: "DISCLOSURE",
        entityId: disclosureId,
        action: "auto_cleared",
        actionCategory: "UPDATE",
        actionDescription: `Auto-cleared by rule "${rule.name}" - nothing to disclose`,
        actorUserId: userId,
        actorType: "SYSTEM",
        organizationId,
      });

      this.eventEmitter.emit("disclosure.auto_cleared", {
        organizationId,
        disclosureId,
        ruleId: rule.id,
        ruleName: rule.name,
      });

      return "AUTO_CLEARED";
    }

    // AUTO_REJECT
    await this.prisma.riskIntelligenceUnit.update({
      where: { id: disclosureId },
      data: {
        status: RiuStatus.REJECTED,
        statusChangedAt: new Date(),
      },
    });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: disclosureId,
      action: "auto_rejected",
      actionCategory: "UPDATE",
      actionDescription: `Auto-rejected by rule "${rule.name}"`,
      actorUserId: userId,
      actorType: "SYSTEM",
      organizationId,
    });

    this.eventEmitter.emit("disclosure.auto_rejected", {
      organizationId,
      disclosureId,
      ruleId: rule.id,
      ruleName: rule.name,
    });

    return "AUTO_REJECTED";
  }
}
```

```typescript
import {
  Controller,
  Get,
  Post,
  Patch,
  Delete,
  Body,
  Param,
  Query,
  UseGuards,
  HttpCode,
  HttpStatus,
} from "@nestjs/common";
import { JwtAuthGuard } from "../../auth/guards/jwt-auth.guard";
import { Roles } from "@common/decorators/roles.decorator";
import { RolesGuard } from "@common/guards/roles.guard";
import { CurrentUser } from "@common/decorators/current-user.decorator";
import { TenantId } from "@common/decorators/tenant-id.decorator";
import { User, DisclosureType } from "@prisma/client";
import { AutoClearRejectRuleService } from "./auto-clear-reject-rule.service";
import {
  CreateAutoClearRejectRuleDto,
  UpdateAutoClearRejectRuleDto,
} from "./auto-clear-reject-rule.dto";

@Controller("api/v1/disclosures/auto-rules")
@UseGuards(JwtAuthGuard, RolesGuard)
export class AutoClearRejectRuleController {
  constructor(private readonly ruleService: AutoClearRejectRuleService) {}

  @Get()
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN")
  async findMany(
    @TenantId() organizationId: string,
    @Query("disclosureType") disclosureType?: DisclosureType,
  ) {
    return this.ruleService.findActive(organizationId, disclosureType);
  }

  @Get(":id")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN")
  async findOne(@Param("id") id: string, @TenantId() organizationId: string) {
    return this.ruleService.findOne(id, organizationId);
  }

  @Post()
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN")
  async create(
    @Body() dto: CreateAutoClearRejectRuleDto,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.ruleService.create(dto, organizationId, user.id);
  }

  @Patch(":id")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN")
  async update(
    @Param("id") id: string,
    @Body() dto: UpdateAutoClearRejectRuleDto,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.ruleService.update(id, dto, organizationId, user.id);
  }

  @Delete(":id")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN")
  @HttpCode(HttpStatus.NO_CONTENT)
  async delete(
    @Param("id") id: string,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.ruleService.delete(id, organizationId, user.id);
  }
}
```
