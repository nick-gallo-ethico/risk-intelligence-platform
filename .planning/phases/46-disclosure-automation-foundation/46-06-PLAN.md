```typescript
export enum ApproverType {
  SPECIFIC_USER = "SPECIFIC_USER",
  ROLE = "ROLE",
  MANAGER_OF_DISCLOSER = "MANAGER_OF_DISCLOSER",
  MANAGER_CHAIN = "MANAGER_CHAIN",
}

export class ApproverConfigDto {
  @IsString() @IsOptional() userId?: string;
  @IsString() @IsOptional() roleId?: string;
  @IsInt() @Min(1) @Max(5) @IsOptional() managerLevel?: number; // 1 = direct, 2 = skip-level
}

export class ApprovalStageDto {
  @IsInt() @Min(1) @Max(4) stageNumber: number;
  @IsString() @IsNotEmpty() name: string;
  @IsEnum(ApproverType) approverType: ApproverType;
  @ValidateNested()
  @Type(() => ApproverConfigDto)
  @IsOptional()
  approverConfig?: ApproverConfigDto;
  @IsInt() @Min(1) @Max(90) @IsOptional() slaDays?: number;
  @IsBoolean() @IsOptional() isRequired?: boolean = true;
  @IsBoolean() @IsOptional() canSkip?: boolean = false;
}

export class CreateDisclosureApprovalConfigDto {
  @IsString() @IsNotEmpty() name: string;
  @IsString() @IsOptional() description?: string;
  @IsArray()
  @IsEnum(DisclosureType, { each: true })
  disclosureTypes: DisclosureType[];
  @IsArray()
  @ArrayMaxSize(4)
  @ValidateNested({ each: true })
  @Type(() => ApprovalStageDto)
  stages: ApprovalStageDto[];
}

export class UpdateDisclosureApprovalConfigDto extends PartialType(
  CreateDisclosureApprovalConfigDto,
```

```typescript
@Injectable()
export class DisclosureApprovalConfigService {
  private readonly logger = new Logger(DisclosureApprovalConfigService.name);
  private readonly MAX_STAGES = 4;

  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
  ) {}

  async create(
    dto: CreateDisclosureApprovalConfigDto,
    organizationId: string,
    userId: string,
  ) {
    this.validateStages(dto.stages);

    const config = await this.prisma.disclosureApprovalConfig.create({
      data: {
        organizationId,
        name: dto.name,
        description: dto.description,
        disclosureTypes: dto.disclosureTypes,
        stages: dto.stages as Prisma.InputJsonValue,
        createdById: userId,
      },
    });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: config.id,
      action: "approval_config_created",
      actionCategory: "CREATE",
      actionDescription: `Created ${dto.stages.length}-stage approval workflow "${config.name}"`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
    });

    return config;
  }

  async update(
    id: string,
    dto: UpdateDisclosureApprovalConfigDto,
    organizationId: string,
    userId: string,
  );
  async findOne(id: string, organizationId: string);
  async findMany(organizationId: string);
  async findActiveForType(
    organizationId: string,
    disclosureType: DisclosureType,
  );
  async delete(id: string, organizationId: string, userId: string);

  private validateStages(stages: ApprovalStageDto[]): void {
    if (stages.length > this.MAX_STAGES) {
      throw new BadRequestException(
        `Maximum ${this.MAX_STAGES} stages allowed`,
      );
    }

    // Verify stage numbers are sequential 1, 2, 3, 4
    const stageNumbers = stages.map((s) => s.stageNumber).sort((a, b) => a - b);
    for (let i = 0; i < stageNumbers.length; i++) {
      if (stageNumbers[i] !== i + 1) {
        throw new BadRequestException(
          "Stage numbers must be sequential starting from 1",
        );
      }
    }
  }
}
```

```typescript
@Injectable()
export class DisclosureApprovalWorkflowService {
  private readonly logger = new Logger(DisclosureApprovalWorkflowService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  /**
   * Starts the approval workflow for a disclosure.
   * Finds applicable config and creates approval state.
   */
  async startWorkflow(
    disclosureId: string,
    disclosureType: DisclosureType,
    organizationId: string,
    userId: string,
  ): Promise<DisclosureApprovalState | null> {
    // Find applicable approval config
    const config = await this.prisma.disclosureApprovalConfig.findFirst({
      where: {
        organizationId,
        isActive: true,
        disclosureTypes: { has: disclosureType },
      },
      orderBy: { createdAt: "desc" }, // Most recently created takes precedence
    });

    if (!config) {
      this.logger.debug(
        `No approval config for disclosure type ${disclosureType} in org ${organizationId}`,
      );
      return null;
    }

    const state = await this.prisma.disclosureApprovalState.create({
      data: {
        organizationId,
        disclosureId,
        configId: config.id,
        currentStage: 1,
        stageHistory: [],
        status: "IN_PROGRESS",
      },
    });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: disclosureId,
      action: "approval_workflow_started",
      actionCategory: "UPDATE",
      actionDescription: `Started approval workflow "${config.name}" at stage 1`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
    });

    this.eventEmitter.emit("disclosure.approval.started", {
      organizationId,
      disclosureId,
      configId: config.id,
      stageNumber: 1,
    });

    return state;
  }

  /**
   * Approves the current stage and advances to next stage or completes.
   */
  async approveStage(
    disclosureId: string,
    approverUserId: string,
    approvalNote: string | null,
    organizationId: string,
  ): Promise<DisclosureApprovalState> {
    const state = await this.getOrThrowState(disclosureId, organizationId);
    const config = await this.prisma.disclosureApprovalConfig.findUnique({
      where: { id: state.configId },
    });

    const stages = config!.stages as ApprovalStageDto[];
    const currentStageConfig = stages.find(
      (s) => s.stageNumber === state.currentStage,
    );

    // Record stage completion
    const stageHistory = state.stageHistory as StageHistoryEntry[];
    stageHistory.push({
      stageNumber: state.currentStage,
      stageName: currentStageConfig?.name || `Stage ${state.currentStage}`,
      action: "APPROVED",
      actorUserId: approverUserId,
      note: approvalNote,
      timestamp: new Date().toISOString(),
    });

    // Check if this was the last stage
    if (state.currentStage >= stages.length) {
      // Workflow complete
      const updated = await this.prisma.disclosureApprovalState.update({
        where: { id: state.id },
        data: {
          status: "APPROVED",
          stageHistory: stageHistory as Prisma.InputJsonValue,
          completedAt: new Date(),
        },
      });

      await this.finalizeDisclosure(
        disclosureId,
        "APPROVED",
        organizationId,
        approverUserId,
      );
      return updated;
    }

    // Advance to next stage
    const updated = await this.prisma.disclosureApprovalState.update({
      where: { id: state.id },
      data: {
        currentStage: state.currentStage + 1,
        stageHistory: stageHistory as Prisma.InputJsonValue,
      },
    });

    this.eventEmitter.emit("disclosure.approval.stage_advanced", {
      organizationId,
      disclosureId,
      fromStage: state.currentStage,
      toStage: state.currentStage + 1,
    });

    return updated;
  }

  /**
   * Rejects at current stage, ending the workflow.
   */
  async rejectStage(
    disclosureId: string,
    rejecterUserId: string,
    rejectionReason: string,
    organizationId: string,
  ): Promise<DisclosureApprovalState> {
    const state = await this.getOrThrowState(disclosureId, organizationId);

    const stageHistory = state.stageHistory as StageHistoryEntry[];
    stageHistory.push({
      stageNumber: state.currentStage,
      stageName: `Stage ${state.currentStage}`,
      action: "REJECTED",
      actorUserId: rejecterUserId,
      note: rejectionReason,
      timestamp: new Date().toISOString(),
    });

    const updated = await this.prisma.disclosureApprovalState.update({
      where: { id: state.id },
      data: {
        status: "REJECTED",
        stageHistory: stageHistory as Prisma.InputJsonValue,
        completedAt: new Date(),
      },
    });

    await this.finalizeDisclosure(
      disclosureId,
      "REJECTED",
      organizationId,
      rejecterUserId,
    );
    return updated;
  }

  /**
   * Gets current approval state for a disclosure.
   */
  async getState(
    disclosureId: string,
    organizationId: string,
  ): Promise<DisclosureApprovalState | null> {
    return this.prisma.disclosureApprovalState.findFirst({
      where: { disclosureId, organizationId },
    });
  }

  private async getOrThrowState(disclosureId: string, organizationId: string) {
    const state = await this.getState(disclosureId, organizationId);
    if (!state) {
      throw new NotFoundException("Disclosure has no active approval workflow");
    }
    if (state.status !== "IN_PROGRESS") {
      throw new BadRequestException("Workflow is already completed");
    }
    return state;
  }

  private async finalizeDisclosure(
    disclosureId: string,
    outcome: "APPROVED" | "REJECTED",
    organizationId: string,
    userId: string,
  ): Promise<void> {
    const status =
      outcome === "APPROVED" ? RiuStatus.COMPLETED : RiuStatus.REJECTED;

    await this.prisma.riskIntelligenceUnit.update({
      where: { id: disclosureId },
      data: { status, statusChangedAt: new Date(), statusChangedById: userId },
    });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: disclosureId,
      action: `approval_workflow_${outcome.toLowerCase()}`,
      actionCategory: "UPDATE",
      actionDescription: `Approval workflow completed with outcome: ${outcome}`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
    });

    this.eventEmitter.emit("disclosure.approval.completed", {
      organizationId,
      disclosureId,
      outcome,
    });
  }
}

interface StageHistoryEntry {
  stageNumber: number;
  stageName: string;
  action: "APPROVED" | "REJECTED" | "SKIPPED";
  actorUserId: string;
  note: string | null;
  timestamp: string;
}
```
