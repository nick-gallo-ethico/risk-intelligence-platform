Create CampaignPauseService:

```typescript
import {
  Injectable,
  Logger,
  NotFoundException,
  BadRequestException,
} from "@nestjs/common";
import { InjectQueue } from "@nestjs/bullmq";
import { Queue } from "bullmq";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { PrismaService } from "../../prisma/prisma.service";
import { AuditService } from "../../audit/audit.service";
import { CampaignReminderService } from "../campaign-reminder.service";
import { Campaign, CampaignStatus, Prisma } from "@prisma/client";
import {
  PauseCampaignDto,
  ResumeCampaignDto,
  PauseState,
} from "./campaign-pause.dto";

@Injectable()
export class CampaignPauseService {
  private readonly logger = new Logger(CampaignPauseService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
    private readonly eventEmitter: EventEmitter2,
    private readonly reminderService: CampaignReminderService,
    @InjectQueue("campaign") private readonly campaignQueue: Queue,
  ) {}

  /**
   * Pauses an active campaign.
   * - Sets status to PAUSED
   * - Records pause state (for deadline restoration on resume)
   * - Cancels pending reminder jobs
   */
  async pauseCampaign(
    campaignId: string,
    dto: PauseCampaignDto,
    userId: string,
    organizationId: string,
  ): Promise<Campaign> {
    const campaign = await this.prisma.campaign.findFirst({
      where: { id: campaignId, organizationId, status: CampaignStatus.ACTIVE },
    });

    if (!campaign) {
      throw new NotFoundException("Active campaign not found");
    }

    // Record pause state for restoration
    const pauseState: PauseState = {
      pausedAt: new Date().toISOString(),
      pausedBy: userId,
      originalDueDate: campaign.dueDate.toISOString(),
      reason: dto.reason,
    };

    const updated = await this.prisma.campaign.update({
      where: { id: campaignId },
      data: {
        status: CampaignStatus.PAUSED,
        statusNote: dto.reason,
        pauseState: pauseState as unknown as Prisma.InputJsonValue,
      },
    });

    // Cancel pending reminder jobs for this campaign
    await this.cancelCampaignReminders(campaignId);

    await this.auditService.log({
      entityType: "CAMPAIGN",
      entityId: campaignId,
      action: "campaign_paused",
      actionCategory: "UPDATE",
      actionDescription: `Paused campaign: ${dto.reason}`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
    });

    this.eventEmitter.emit("campaign.paused", {
      organizationId,
      campaignId,
      reason: dto.reason,
      userId,
    });

    this.logger.log(`Paused campaign ${campaignId}: ${dto.reason}`);

    return updated;
  }

  /**
   * Resumes a paused campaign.
   * - Sets status back to ACTIVE
   * - Extends deadline by pause duration + optional extra days
   * - Extends all assignment due dates
   * - Re-schedules reminder jobs
   */
  async resumeCampaign(
    campaignId: string,
    dto: ResumeCampaignDto,
    userId: string,
    organizationId: string,
  ): Promise<Campaign> {
    const campaign = await this.prisma.campaign.findFirst({
      where: { id: campaignId, organizationId, status: CampaignStatus.PAUSED },
    });

    if (!campaign) {
      throw new NotFoundException("Paused campaign not found");
    }

    const pauseState = campaign.pauseState as PauseState | null;
    if (!pauseState) {
      throw new BadRequestException("Campaign has no pause state recorded");
    }

    // Calculate pause duration
    const pausedAt = new Date(pauseState.pausedAt);
    const now = new Date();
    const pauseDurationMs = now.getTime() - pausedAt.getTime();

    // Calculate new due date: original + pause duration + optional extension
    const extendDaysMs = (dto.extendDeadlineDays ?? 0) * 24 * 60 * 60 * 1000;
    const newDueDate = new Date(
      campaign.dueDate.getTime() + pauseDurationMs + extendDaysMs,
    );

    const updated = await this.prisma.campaign.update({
      where: { id: campaignId },
      data: {
        status: CampaignStatus.ACTIVE,
        statusNote: null,
        dueDate: newDueDate,
        pauseState: Prisma.JsonNull,
      },
    });

    // Extend all assignment due dates
    await this.prisma.campaignAssignment.updateMany({
      where: { campaignId, organizationId },
      data: { dueDate: newDueDate },
    });

    // Re-schedule reminders
    await this.rescheduleReminders(campaignId, organizationId);

    await this.auditService.log({
      entityType: "CAMPAIGN",
      entityId: campaignId,
      action: "campaign_resumed",
      actionCategory: "UPDATE",
      actionDescription: `Resumed campaign with new due date ${newDueDate.toISOString()}`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
      changes: {
        pauseDurationDays: Math.floor(pauseDurationMs / (24 * 60 * 60 * 1000)),
        extendDays: dto.extendDeadlineDays ?? 0,
        newDueDate: newDueDate.toISOString(),
      },
    });

    this.eventEmitter.emit("campaign.resumed", {
      organizationId,
      campaignId,
      newDueDate,
      userId,
    });

    this.logger.log(
      `Resumed campaign ${campaignId} with new due date ${newDueDate.toISOString()}`,
    );

    return updated;
  }

  /**
   * Gets the pause state for a campaign.
   */
  async getPauseState(
    campaignId: string,
    organizationId: string,
  ): Promise<PauseState | null> {
    const campaign = await this.prisma.campaign.findFirst({
      where: { id: campaignId, organizationId },
      select: { pauseState: true, status: true },
    });

    if (campaign?.status !== CampaignStatus.PAUSED) {
      return null;
    }

    return campaign.pauseState as PauseState | null;
  }

  /**
   * Cancels all pending reminder jobs for a campaign.
   */
  private async cancelCampaignReminders(campaignId: string): Promise<void> {
    try {
      // Get all jobs that match the campaign
      const jobs = await this.campaignQueue.getJobs(["waiting", "delayed"]);

      const campaignJobs = jobs.filter(
        (job) =>
          job.data?.campaignId === campaignId && job.name === "send-reminder",
      );

      for (const job of campaignJobs) {
        await job.remove();
      }

      this.logger.debug(
        `Cancelled ${campaignJobs.length} reminder jobs for campaign ${campaignId}`,
      );
    } catch (error) {
      this.logger.warn(
        `Failed to cancel reminders for campaign ${campaignId}: ` +
          `${error instanceof Error ? error.message : String(error)}`,
      );
    }
  }

  /**
   * Re-schedules reminder jobs for a campaign after resume.
   */
  private async rescheduleReminders(
    campaignId: string,
    organizationId: string,
  ): Promise<void> {
    try {
      // The reminder service will pick up pending assignments on next batch run
      // For immediate rescheduling, we could trigger a one-time check
      const assignments = await this.prisma.campaignAssignment.findMany({
        where: { campaignId, organizationId, status: "PENDING" },
        select: { id: true },
        take: 100,
      });

      if (assignments.length > 0) {
        this.logger.debug(
          `Campaign ${campaignId} has ${assignments.length} pending assignments - reminders will be processed in next batch`,
        );
      }
    } catch (error) {
      this.logger.warn(
        `Failed to check assignments for campaign ${campaignId}: ` +
          `${error instanceof Error ? error.message : String(error)}`,
      );
    }
  }
}
```

```typescript
import { CampaignPauseService } from "./pause";
import {
  PauseCampaignDto,
  ResumeCampaignDto,
} from "./pause/campaign-pause.dto";

@Controller("api/v1/campaigns")
@UseGuards(JwtAuthGuard, RolesGuard)
export class CampaignsController {
  constructor(
    private readonly campaignsService: CampaignsService,
    private readonly pauseService: CampaignPauseService,
    // ... other dependencies
  ) {}

  // ... existing methods ...

  /**
   * Pause an active campaign.
   */
  @Post(":id/pause")
  @Roles("SYSTEM_ADMIN", "COMPLIANCE_OFFICER")
  @HttpCode(HttpStatus.OK)
  async pauseCampaign(
    @Param("id") id: string,
    @Body() dto: PauseCampaignDto,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.pauseService.pauseCampaign(id, dto, user.id, organizationId);
  }

  /**
   * Resume a paused campaign.
   */
  @Post(":id/resume")
  @Roles("SYSTEM_ADMIN", "COMPLIANCE_OFFICER")
  @HttpCode(HttpStatus.OK)
  async resumeCampaign(
    @Param("id") id: string,
    @Body() dto: ResumeCampaignDto,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.pauseService.resumeCampaign(id, dto, user.id, organizationId);
  }

  /**
   * Get pause state for a campaign.
   */
  @Get(":id/pause-state")
  async getPauseState(
    @Param("id") id: string,
    @TenantId() organizationId: string,
  ) {
    return this.pauseService.getPauseState(id, organizationId);
  }
}
```
