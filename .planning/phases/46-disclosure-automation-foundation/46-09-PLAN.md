    disclosureIds: string[],
    reason: string,
  ): Promise<BulkOperationResult> {
    return apiClient.post("/api/v1/disclosures/bulk/reject", {
      disclosureIds,
      rejectionReason: reason,
    });
  },

"use client";

import { useState } from "react";
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import {
  disclosureAutomationService,
  RollingCampaignConfig,
  CreateRollingCampaignConfigDto,
} from "@/services/disclosure-automation.service";
import { RollingCampaignConfigTable } from "@/components/campaigns/rolling-campaign-config-table";
import { RollingCampaignConfigForm } from "@/components/campaigns/rolling-campaign-config-form";
import { Button } from "@/components/ui/button";
import {
  Sheet,
  SheetContent,
  SheetHeader,
  SheetTitle,
} from "@/components/ui/sheet";
import { toast } from "sonner";
import { Plus } from "lucide-react";

export default function RollingCampaignsPage() {
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingConfig, setEditingConfig] =
    useState<RollingCampaignConfig | null>(null);
  const queryClient = useQueryClient();

  const { data: configs, isLoading } = useQuery({
    queryKey: ["rolling-campaign-configs"],
    queryFn: () => disclosureAutomationService.getRollingConfigs(),
  });

  const createMutation = useMutation({
    mutationFn: disclosureAutomationService.createRollingConfig,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["rolling-campaign-configs"] });
      toast.success("Rolling campaign config created");
      setIsFormOpen(false);
    },
    onError: (error) => {
      toast.error("Failed to create rolling campaign config");
    },
  });

  const updateMutation = useMutation({
    mutationFn: ({
      id,
      data,
    }: {
      id: string;
      data: Partial<CreateRollingCampaignConfigDto>;
    }) => disclosureAutomationService.updateRollingConfig(id, data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["rolling-campaign-configs"] });
      toast.success("Rolling campaign config updated");
      setIsFormOpen(false);
      setEditingConfig(null);
    },
  });

  const deleteMutation = useMutation({
    mutationFn: disclosureAutomationService.deleteRollingConfig,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["rolling-campaign-configs"] });
      toast.success("Rolling campaign config deleted");
    },
  });

  const handleEdit = (config: RollingCampaignConfig) => {
    setEditingConfig(config);
    setIsFormOpen(true);
  };

  const handleSubmit = (data: CreateRollingCampaignConfigDto) => {
    if (editingConfig) {
      updateMutation.mutate({ id: editingConfig.id, data });
    } else {
      createMutation.mutate(data);
    }
  };

  return (
    <div className="p-6 space-y-6">
      <div className="flex justify-between items-center">
        <div>
          <h1 className="text-2xl font-semibold">Rolling Campaigns</h1>
          <p className="text-muted-foreground">
            Configure automatic campaign triggers based on HRIS events
          </p>
        </div>
        <Button
          onClick={() => {
            setEditingConfig(null);
            setIsFormOpen(true);
          }}
        >
          <Plus className="w-4 h-4 mr-2" />
          New Rolling Campaign
        </Button>
      </div>

      <RollingCampaignConfigTable
        configs={configs ?? []}
        isLoading={isLoading}
        onEdit={handleEdit}
        onDelete={(id) => deleteMutation.mutate(id)}
      />

      <Sheet open={isFormOpen} onOpenChange={setIsFormOpen}>
        <SheetContent className="sm:max-w-lg">
          <SheetHeader>
            <SheetTitle>
              {editingConfig ? "Edit Rolling Campaign" : "New Rolling Campaign"}
            </SheetTitle>
          </SheetHeader>
          <RollingCampaignConfigForm
            config={editingConfig}
            onSubmit={handleSubmit}
            isLoading={createMutation.isPending || updateMutation.isPending}
          />
        </SheetContent>
      </Sheet>
    </div>
  );
}

"use client";

import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { z } from "zod";
import { useQuery } from "@tanstack/react-query";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Textarea } from "@/components/ui/textarea";
import { Checkbox } from "@/components/ui/checkbox";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import {
  Form,
  FormControl,
  FormDescription,
  FormField,
  FormItem,
  FormLabel,
  FormMessage,
} from "@/components/ui/form";
import {
  RollingCampaignConfig,
  CreateRollingCampaignConfigDto,
} from "@/services/disclosure-automation.service";
import { campaignsService } from "@/services/campaigns.service";

const TRIGGER_EVENTS = [
  { value: "NEW_HIRE", label: "New Hire" },
  { value: "ROLE_CHANGE", label: "Role Change" },
  { value: "PROMOTION", label: "Promotion" },
  { value: "ANNUAL_ANNIVERSARY", label: "Annual Anniversary" },
] as const;

const schema = z.object({
  name: z.string().min(1, "Name is required").max(100),
  description: z.string().max(500).optional(),
  campaignTemplateId: z.string().uuid("Select a campaign template"),
  triggerEvents: z
    .array(z.string())
    .min(1, "Select at least one trigger event"),
  dueDaysAfterTrigger: z.number().min(1).max(365).default(30),
  segmentId: z.string().uuid().optional().nullable(),
});

interface RollingCampaignConfigFormProps {
  config?: RollingCampaignConfig | null;
  onSubmit: (data: CreateRollingCampaignConfigDto) => void;
  isLoading?: boolean;
}

export function RollingCampaignConfigForm({
  config,
  onSubmit,
  isLoading,
}: RollingCampaignConfigFormProps) {
  const { data: campaigns } = useQuery({
    queryKey: ["campaigns-templates"],
    queryFn: () => campaignsService.getTemplates(),
  });

  const form = useForm<z.infer<typeof schema>>({
    resolver: zodResolver(schema),
    defaultValues: {
      name: config?.name ?? "",
      description: config?.description ?? "",
      campaignTemplateId: config?.campaignTemplateId ?? "",
      triggerEvents: config?.triggerEvents ?? [],
      dueDaysAfterTrigger: config?.dueDaysAfterTrigger ?? 30,
      segmentId: config?.segmentId ?? null,
    },
  });

  const handleSubmit = (data: z.infer<typeof schema>) => {
    onSubmit({
      name: data.name,
      description: data.description,
      campaignTemplateId: data.campaignTemplateId,
      triggerEvents: data.triggerEvents,
      dueDaysAfterTrigger: data.dueDaysAfterTrigger,
      segmentId: data.segmentId ?? undefined,
    });
  };

  return (
    <Form {...form}>
      <form
        onSubmit={form.handleSubmit(handleSubmit)}
        className="space-y-4 mt-4"
      >
        <FormField
          control={form.control}
          name="name"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Name</FormLabel>
              <FormControl>
                <Input placeholder="New Hire COI Disclosure" {...field} />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />

        <FormField
          control={form.control}
          name="description"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Description</FormLabel>
              <FormControl>
                <Textarea
                  placeholder="Describe when this rolling campaign should trigger..."
                  {...field}
                />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />

        <FormField
          control={form.control}
          name="campaignTemplateId"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Campaign Template</FormLabel>
              <Select onValueChange={field.onChange} defaultValue={field.value}>
                <FormControl>
                  <SelectTrigger>
                    <SelectValue placeholder="Select a campaign template" />
                  </SelectTrigger>
                </FormControl>
                <SelectContent>
                  {campaigns?.map((campaign) => (
                    <SelectItem key={campaign.id} value={campaign.id}>
                      {campaign.name}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
              <FormMessage />
            </FormItem>
          )}
        />

        <FormField
          control={form.control}
          name="triggerEvents"
          render={() => (
            <FormItem>
              <FormLabel>Trigger Events</FormLabel>
              <FormDescription>
                Select HRIS events that will trigger this campaign
              </FormDescription>
              <div className="grid grid-cols-2 gap-2 mt-2">
                {TRIGGER_EVENTS.map((event) => (
                  <FormField
                    key={event.value}
                    control={form.control}
                    name="triggerEvents"
                    render={({ field }) => (
                      <FormItem className="flex items-center space-x-2">
                        <FormControl>
                          <Checkbox
                            checked={field.value?.includes(event.value)}
                            onCheckedChange={(checked) => {
                              const updated = checked
                                ? [...(field.value || []), event.value]
                                : field.value?.filter((v) => v !== event.value);
                              field.onChange(updated);
                            }}
                          />
                        </FormControl>
                        <FormLabel className="font-normal">
                          {event.label}
                        </FormLabel>
                      </FormItem>
                    )}
                  />
                ))}
              </div>
              <FormMessage />
            </FormItem>
          )}
        />

        <FormField
          control={form.control}
          name="dueDaysAfterTrigger"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Due Days After Trigger</FormLabel>
              <FormControl>
                <Input
                  type="number"
                  min={1}
                  max={365}
                  {...field}
                  onChange={(e) => field.onChange(parseInt(e.target.value, 10))}
                />
              </FormControl>
              <FormDescription>
                Number of days after trigger event for disclosure due date
              </FormDescription>
              <FormMessage />
            </FormItem>
          )}
        />

        <Button type="submit" disabled={isLoading} className="w-full">
          {config ? "Update Rolling Campaign" : "Create Rolling Campaign"}
        </Button>
      </form>
    </Form>
  );
}

"use client";

import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from "@/components/ui/table";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Switch } from "@/components/ui/switch";
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
} from "@/components/ui/dropdown-menu";
import { MoreHorizontal, Pencil, Trash2 } from "lucide-react";
import { RollingCampaignConfig } from "@/services/disclosure-automation.service";
import { Skeleton } from "@/components/ui/skeleton";

interface RollingCampaignConfigTableProps {
  configs: RollingCampaignConfig[];
  isLoading: boolean;
  onEdit: (config: RollingCampaignConfig) => void;
  onDelete: (id: string) => void;
}

const TRIGGER_EVENT_LABELS: Record<string, string> = {
  NEW_HIRE: "New Hire",
  ROLE_CHANGE: "Role Change",
  PROMOTION: "Promotion",
  ANNUAL_ANNIVERSARY: "Anniversary",
};

export function RollingCampaignConfigTable({
  configs,
  isLoading,
  onEdit,
  onDelete,
}: RollingCampaignConfigTableProps) {
  if (isLoading) {
    return (
      <div className="space-y-2">
        {[1, 2, 3].map((i) => (
          <Skeleton key={i} className="h-12 w-full" />
        ))}
      </div>
    );
  }

  if (configs.length === 0) {
    return (
      <div className="text-center py-10 text-muted-foreground">
        No rolling campaign configurations yet. Create one to get started.
      </div>
    );
  }

  return (
    <Table>
      <TableHeader>
        <TableRow>
          <TableHead>Name</TableHead>
          <TableHead>Trigger Events</TableHead>
          <TableHead>Due Days</TableHead>
          <TableHead>Status</TableHead>
          <TableHead className="w-[50px]"></TableHead>
        </TableRow>
      </TableHeader>
      <TableBody>
        {configs.map((config) => (
          <TableRow key={config.id}>
            <TableCell className="font-medium">{config.name}</TableCell>
            <TableCell>
              <div className="flex gap-1 flex-wrap">
                {config.triggerEvents.map((event) => (
                  <Badge key={event} variant="secondary">
                    {TRIGGER_EVENT_LABELS[event] ?? event}
                  </Badge>
                ))}
              </div>
            </TableCell>
            <TableCell>{config.dueDaysAfterTrigger} days</TableCell>
            <TableCell>
              <Badge variant={config.isActive ? "default" : "outline"}>
                {config.isActive ? "Active" : "Inactive"}
              </Badge>
            </TableCell>
            <TableCell>
              <DropdownMenu>
                <DropdownMenuTrigger asChild>
                  <Button variant="ghost" size="icon">
                    <MoreHorizontal className="h-4 w-4" />
                  </Button>
                </DropdownMenuTrigger>
                <DropdownMenuContent align="end">
                  <DropdownMenuItem onClick={() => onEdit(config)}>
                    <Pencil className="mr-2 h-4 w-4" />
                    Edit
                  </DropdownMenuItem>
                  <DropdownMenuItem
                    onClick={() => onDelete(config.id)}
                    className="text-destructive"
                  >
                    <Trash2 className="mr-2 h-4 w-4" />
                    Delete
                  </DropdownMenuItem>
                </DropdownMenuContent>
              </DropdownMenu>
            </TableCell>
          </TableRow>
        ))}
      </TableBody>
    </Table>
  );
}
