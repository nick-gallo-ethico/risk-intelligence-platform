```prisma
enum ProxyScope {
  FULL               // All delegator's authority
  DISCLOSURE_TYPE    // Specific disclosure types only
  VALUE_THRESHOLD    // Up to certain value
  APPROVAL_STAGE     // Specific approval stages only

  @@map("proxy_scope")
}

model ProxyDelegation {
  id              String     @id @default(uuid())
  organizationId  String     @map("organization_id")
  delegatorId     String     @map("delegator_id")
  delegateId      String     @map("delegate_id")
  scope           ProxyScope
  scopeConfig     Json?      @map("scope_config") // { disclosureTypes?, approvalStages?, maxValue? }
  validFrom       DateTime   @map("valid_from")
  validUntil      DateTime   @map("valid_until")
  reason          String
  isActive        Boolean    @default(true) @map("is_active")
  createdAt       DateTime   @default(now()) @map("created_at")
  revokedAt       DateTime?  @map("revoked_at")
  revokedBy       String?    @map("revoked_by")
  revokeReason    String?    @map("revoke_reason")

  organization    Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
  delegator       User         @relation("ProxyDelegator", fields: [delegatorId], references: [id])
  delegate        User         @relation("ProxyDelegate", fields: [delegateId], references: [id])
  revokedByUser   User?        @relation("ProxyRevoker", fields: [revokedBy], references: [id])

  @@index([organizationId, isActive])
  @@index([delegatorId, isActive])
  @@index([delegateId, isActive])
  @@map("proxy_delegations")
}
```

```typescript
import {
  IsString,
  IsNotEmpty,
  IsOptional,
  IsArray,
  IsEnum,
  IsBoolean,
  IsInt,
  IsUUID,
  IsDate,
  Min,
  ValidateNested,
  IsDateString,
} from "class-validator";
import { Type, Transform } from "class-transformer";
import { DisclosureType } from "@prisma/client";

export enum ProxyScopeDto {
  FULL = "FULL",
  DISCLOSURE_TYPE = "DISCLOSURE_TYPE",
  VALUE_THRESHOLD = "VALUE_THRESHOLD",
  APPROVAL_STAGE = "APPROVAL_STAGE",
}

export class ScopeConfigDto {
  @IsArray()
  @IsEnum(DisclosureType, { each: true })
  @IsOptional()
  disclosureTypes?: DisclosureType[];

  @IsArray()
  @IsInt({ each: true })
  @IsOptional()
  approvalStages?: number[];

  @IsInt()
  @Min(0)
  @IsOptional()
  maxValue?: number;
}

export class CreateProxyDelegationDto {
  @IsUUID() delegateId: string;
  @IsEnum(ProxyScopeDto) scope: ProxyScopeDto;
  @ValidateNested()
  @Type(() => ScopeConfigDto)
  @IsOptional()
  scopeConfig?: ScopeConfigDto;
  @IsDateString() validFrom: string;
  @IsDateString() validUntil: string;
  @IsString() @IsNotEmpty() reason: string;
}

export class RevokeProxyDelegationDto {
  @IsString() @IsNotEmpty() revokeReason: string;
}

export interface DelegationAuthorizationContext {
  disclosureType?: DisclosureType;
  approvalStage?: number;
  value?: number;
```

Create ProxyDelegationService:

```typescript
@Injectable()
export class ProxyDelegationService {
  private readonly logger = new Logger(ProxyDelegationService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
  ) {}

  /**
   * Creates a new proxy delegation.
   */
  async create(
    dto: CreateProxyDelegationDto,
    delegatorId: string,
    organizationId: string,
  ): Promise<ProxyDelegation> {
    // Validate date range
    const validFrom = new Date(dto.validFrom);
    const validUntil = new Date(dto.validUntil);

    if (validUntil <= validFrom) {
      throw new BadRequestException("validUntil must be after validFrom");
    }

    if (validUntil <= new Date()) {
      throw new BadRequestException("validUntil must be in the future");
    }

    // Cannot delegate to self
    if (dto.delegateId === delegatorId) {
      throw new BadRequestException("Cannot delegate to yourself");
    }

    const delegation = await this.prisma.proxyDelegation.create({
      data: {
        organizationId,
        delegatorId,
        delegateId: dto.delegateId,
        scope: dto.scope,
        scopeConfig: dto.scopeConfig as Prisma.InputJsonValue,
        validFrom,
        validUntil,
        reason: dto.reason,
      },
      include: { delegator: true, delegate: true },
    });

    await this.auditService.log({
      entityType: "USER",
      entityId: delegatorId,
      action: "proxy_delegation_created",
      actionCategory: "CREATE",
      actionDescription: `Delegated ${dto.scope} authority to user until ${validUntil.toISOString()}`,
      actorUserId: delegatorId,
      actorType: "USER",
      organizationId,
    });

    return delegation;
  }

  /**
   * Revokes an active delegation.
   */
  async revoke(
    delegationId: string,
    dto: RevokeProxyDelegationDto,
    revokerId: string,
    organizationId: string,
  ): Promise<ProxyDelegation> {
    const delegation = await this.prisma.proxyDelegation.findFirst({
      where: { id: delegationId, organizationId, isActive: true },
    });

    if (!delegation) {
      throw new NotFoundException("Active delegation not found");
    }

    const updated = await this.prisma.proxyDelegation.update({
      where: { id: delegationId },
      data: {
        isActive: false,
        revokedAt: new Date(),
        revokedBy: revokerId,
        revokeReason: dto.revokeReason,
      },
    });

    await this.auditService.log({
      entityType: "USER",
      entityId: delegation.delegatorId,
      action: "proxy_delegation_revoked",
      actionCategory: "UPDATE",
      actionDescription: `Revoked delegation: ${dto.revokeReason}`,
      actorUserId: revokerId,
      actorType: "USER",
      organizationId,
    });

    return updated;
  }

  /**
   * Checks if a user has delegated authority to act on behalf of another user.
   * Returns the applicable delegation or null.
   */
  async checkAuthorization(
    actingUserId: string,
    onBehalfOfUserId: string,
    context: DelegationAuthorizationContext,
    organizationId: string,
  ): Promise<ProxyDelegation | null> {
    const now = new Date();

    // Find active, valid delegation
    const delegation = await this.prisma.proxyDelegation.findFirst({
      where: {
        organizationId,
        delegatorId: onBehalfOfUserId,
        delegateId: actingUserId,
        isActive: true,
        validFrom: { lte: now },
        validUntil: { gte: now },
      },
    });

    if (!delegation) {
      return null;
    }

    // Check scope matches context
    if (!this.scopeMatchesContext(delegation, context)) {
      return null;
    }

    return delegation;
  }

  private scopeMatchesContext(
    delegation: ProxyDelegation,
    context: DelegationAuthorizationContext,
  ): boolean {
    if (delegation.scope === "FULL") {
      return true;
    }

    const config = delegation.scopeConfig as ScopeConfigDto | null;

    if (delegation.scope === "DISCLOSURE_TYPE") {
      if (!context.disclosureType || !config?.disclosureTypes) {
        return false;
      }
      return config.disclosureTypes.includes(context.disclosureType);
    }

    if (delegation.scope === "APPROVAL_STAGE") {
      if (context.approvalStage === undefined || !config?.approvalStages) {
        return false;
      }
      return config.approvalStages.includes(context.approvalStage);
    }

    if (delegation.scope === "VALUE_THRESHOLD") {
      if (context.value === undefined || config?.maxValue === undefined) {
        return false;
      }
      return context.value <= config.maxValue;
    }

    return false;
  }

  // CRUD methods
  async findOne(id: string, organizationId: string);
  async findByDelegator(delegatorId: string, organizationId: string);
  async findByDelegate(delegateId: string, organizationId: string);
  async findActive(organizationId: string);
}
```

```typescript
import {
  Controller,
  Get,
  Post,
  Delete,
  Body,
  Param,
  UseGuards,
  Query,
} from "@nestjs/common";
import { JwtAuthGuard } from "../../auth/guards/jwt-auth.guard";
import { CurrentUser } from "@common/decorators/current-user.decorator";
import { TenantId } from "@common/decorators/tenant-id.decorator";
import { User } from "@prisma/client";
import { ProxyDelegationService } from "./proxy-delegation.service";
import {
  CreateProxyDelegationDto,
  RevokeProxyDelegationDto,
} from "./proxy-delegation.dto";

@Controller("api/v1/proxy-delegations")
@UseGuards(JwtAuthGuard)
export class ProxyDelegationController {
  constructor(
    private readonly proxyDelegationService: ProxyDelegationService,
  ) {}

  /**
   * Create a new proxy delegation (current user as delegator).
   */
  @Post()
  async create(
    @Body() dto: CreateProxyDelegationDto,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.proxyDelegationService.create(dto, user.id, organizationId);
  }

  /**
   * Get delegations given by current user.
   */
  @Get("given")
  async getGiven(
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.proxyDelegationService.findByDelegator(user.id, organizationId);
  }

  /**
   * Get delegations received by current user.
   */
  @Get("received")
  async getReceived(
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.proxyDelegationService.findByDelegate(user.id, organizationId);
  }

  /**
   * Revoke a delegation.
   */
  @Delete(":id/revoke")
  async revoke(
    @Param("id") id: string,
    @Body() dto: RevokeProxyDelegationDto,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ) {
    return this.proxyDelegationService.revoke(id, dto, user.id, organizationId);
  }

  /**
   * Get a specific delegation.
   */
  @Get(":id")
  async getOne(@Param("id") id: string, @TenantId() organizationId: string) {
    return this.proxyDelegationService.findOne(id, organizationId);
  }
}
```
