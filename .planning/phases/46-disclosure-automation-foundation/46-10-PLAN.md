```typescript
import {
  PrismaClient,
  DisclosureType,
  CampaignStatus,
  ProxyScope,
} from "@prisma/client";

const prisma = new PrismaClient();

export async function seedAcmePhase46() {
  console.log("Seeding Acme Co. Phase 46: Disclosure Automation...");

  // Get Acme organization
  const acme = await prisma.organization.findFirst({
    where: { name: { contains: "Acme" } },
  });

  if (!acme) {
    console.log("Acme organization not found, skipping Phase 46 seed");
    return;
  }

  // Get users for seeding
  const complianceOfficer = await prisma.user.findFirst({
    where: { organizationId: acme.id, roles: { has: "COMPLIANCE_OFFICER" } },
  });

  const adminUser = await prisma.user.findFirst({
    where: { organizationId: acme.id, roles: { has: "SYSTEM_ADMIN" } },
  });

  if (!complianceOfficer || !adminUser) {
    console.log("Required users not found, skipping Phase 46 seed");
    return;
  }

  // Get a campaign template for rolling configs
  const disclosureCampaign = await prisma.campaign.findFirst({
    where: {
      organizationId: acme.id,
      type: "DISCLOSURE",
      status: CampaignStatus.ACTIVE,
    },
  });

  // 1. Create Rolling Campaign Configs (DISC-01)
  console.log("Creating rolling campaign configs...");

  await prisma.rollingCampaignConfig.upsert({
    where: { id: "acme-rolling-new-hire" },
    update: {},
    create: {
      id: "acme-rolling-new-hire",
      organizationId: acme.id,
      name: "New Hire COI Disclosure",
      description:
        "Automatically assigns COI disclosure to new hires within 30 days of start date",
      campaignTemplateId: disclosureCampaign?.id ?? "placeholder",
      triggerEvents: ["NEW_HIRE"],
      dueDaysAfterTrigger: 30,
      isActive: true,
      createdById: adminUser.id,
    },
  });

  await prisma.rollingCampaignConfig.upsert({
    where: { id: "acme-rolling-promotion" },
    update: {},
    create: {
      id: "acme-rolling-promotion",
      organizationId: acme.id,
      name: "Promotion COI Update",
      description:
        "Triggers COI update when employee is promoted to manager or above",
      campaignTemplateId: disclosureCampaign?.id ?? "placeholder",
      triggerEvents: ["PROMOTION", "ROLE_CHANGE"],
      dueDaysAfterTrigger: 14,
      isActive: true,
      createdById: adminUser.id,
    },
  });

  // 2. Create Auto-Clear/Reject Rules (DISC-02, DISC-03)
  console.log("Creating auto-clear/reject rules...");

  await prisma.autoClearRejectRule.upsert({
    where: { id: "acme-auto-clear-nothing" },
    update: {},
    create: {
      id: "acme-auto-clear-nothing",
      organizationId: acme.id,
      name: "Nothing to Disclose Auto-Clear",
      description:
        "Auto-completes disclosures where employee confirms nothing to disclose",
      disclosureTypes: [
        DisclosureType.COI,
        DisclosureType.GIFT,
        DisclosureType.OUTSIDE_EMPLOYMENT,
      ],
      conditions: {
        all: [
          { fact: "hasItemsToDisclose", operator: "equal", value: false },
          { fact: "confirmNoDisclosure", operator: "equal", value: true },
        ],
      },
      action: "AUTO_CLEAR",
      requiresReview: false,
      priority: 100,
      isActive: true,
      createdById: adminUser.id,
    },
  });

  await prisma.autoClearRejectRule.upsert({
    where: { id: "acme-auto-reject-govt" },
    update: {},
    create: {
      id: "acme-auto-reject-govt",
      organizationId: acme.id,
      name: "Government Official Auto-Reject",
      description: "Auto-rejects gifts to government officials over $25",
      disclosureTypes: [DisclosureType.GIFT],
      conditions: {
        all: [
          {
            fact: "recipientType",
            operator: "equal",
            value: "GOVERNMENT_OFFICIAL",
          },
          { fact: "disclosureValue", operator: "greaterThan", value: 25 },
        ],
      },
      action: "AUTO_REJECT",
      requiresReview: true, // Flag for review instead of auto-reject
      priority: 50,
      isActive: true,
      createdById: adminUser.id,
    },
  });

  // 3. Create Disclosure Approval Config (DISC-06)
  console.log("Creating approval workflow configs...");

  await prisma.disclosureApprovalConfig.upsert({
    where: { id: "acme-coi-approval" },
    update: {},
    create: {
      id: "acme-coi-approval",
      organizationId: acme.id,
      name: "COI Two-Stage Approval",
      description: "Manager approval followed by compliance review",
      disclosureTypes: [DisclosureType.COI],
      stages: [
        {
          stageNumber: 1,
          name: "Manager Review",
          approverType: "MANAGER_OF_DISCLOSER",
          approverConfig: { managerLevel: 1 },
          slaDays: 5,
          isRequired: true,
          canSkip: false,
        },
        {
          stageNumber: 2,
          name: "Compliance Review",
          approverType: "ROLE",
          approverConfig: { roleId: "COMPLIANCE_OFFICER" },
          slaDays: 7,
          isRequired: true,
          canSkip: false,
        },
      ],
      isActive: true,
      createdById: adminUser.id,
    },
  });

  // 4. Create Proxy Delegation (DISC-07)
  console.log("Creating proxy delegations...");

  const delegate = await prisma.user.findFirst({
    where: {
      organizationId: acme.id,
      id: { not: complianceOfficer.id },
      roles: { has: "COMPLIANCE_OFFICER" },
    },
  });

  if (delegate) {
    const nextWeek = new Date();
    nextWeek.setDate(nextWeek.getDate() + 7);

    await prisma.proxyDelegation.upsert({
      where: { id: "acme-proxy-vacation" },
      update: {},
      create: {
        id: "acme-proxy-vacation",
        organizationId: acme.id,
        delegatorId: complianceOfficer.id,
        delegateId: delegate.id,
        scope: ProxyScope.FULL,
        validFrom: new Date(),
        validUntil: nextWeek,
        reason: "Vacation coverage - full authority delegation",
        isActive: true,
      },
    });
  }

  // 5. Create a paused campaign for demo (DISC-08)
  console.log("Creating paused campaign for demo...");

  const pausedCampaign = await prisma.campaign.findFirst({
    where: { organizationId: acme.id, status: CampaignStatus.ACTIVE },
    orderBy: { createdAt: "desc" },
  });

  if (pausedCampaign) {
    await prisma.campaign.update({
      where: { id: pausedCampaign.id },
      data: {
        status: CampaignStatus.PAUSED,
        statusNote: "Demo: Paused for year-end processing",
        pauseState: {
          pausedAt: new Date().toISOString(),
          pausedBy: adminUser.id,
          originalDueDate: pausedCampaign.dueDate.toISOString(),
          reason: "Demo: Paused for year-end processing",
        },
      },
    });
  }

  console.log("Phase 46 demo data seeded successfully!");
}

export default seedAcmePhase46;
```
