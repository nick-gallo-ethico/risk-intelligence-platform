```typescript
/**
 * Gets the appropriate reminder sequence based on campaign type.
 * Disclosure campaigns use the CONDITION_REMINDER_SEQUENCE (14/7/3/1 days).
 * Other campaigns use DEFAULT_REMINDER_SEQUENCE or custom config.
 */
async getReminderSequenceForCampaign(campaignId: string): Promise<ReminderStep[]> {
  const campaign = await this.prisma.campaign.findUnique({
    where: { id: campaignId },
    select: { reminderConfig: true, type: true },
  });

  // Use custom config if set
  if (campaign?.reminderConfig) {
    return campaign.reminderConfig as unknown as ReminderStep[];
  }

  // Use condition sequence for disclosure campaigns
  if (campaign?.type === 'DISCLOSURE') {
    return CONDITION_REMINDER_SEQUENCE;
  }

  return DEFAULT_REMINDER_SEQUENCE;
}
```

```typescript
@Processor(CAMPAIGN_QUEUE_NAME)
export class CampaignReminderProcessor extends WorkerHost {
  private readonly logger = new Logger(CampaignReminderProcessor.name);

  constructor(
    private readonly reminderService: CampaignReminderService,
    private readonly notificationService: NotificationService,
    private readonly prisma: PrismaService,
  ) {
    super();
  }

  async process(job: Job<PendingReminder>): Promise<void> {
    if (job.name !== "send-reminder") {
      return;
    }

    const reminder = job.data;
    this.logger.debug(
      `Processing reminder for assignment ${reminder.assignmentId}, ` +
        `step ${reminder.reminderStep}, dueDate ${reminder.dueDate.toISOString()}`,
    );

    try {
      // Get employee details
      const employee = await this.prisma.employee.findUnique({
        where: { id: reminder.employeeId },
        select: { email: true, firstName: true, lastName: true },
      });

      if (!employee || !employee.email) {
        this.logger.warn(
          `Cannot send reminder: employee ${reminder.employeeId} has no email`,
        );
        return;
      }

      // Determine reminder urgency based on days from due
      const daysFromDue = this.calculateDaysFromDue(reminder.dueDate);
      const urgency = this.determineUrgency(daysFromDue);

      // Send notification
      await this.notificationService.send({
        type: "CAMPAIGN_REMINDER",
        recipientEmail: employee.email,
        recipientName: `${employee.firstName} ${employee.lastName}`,
        data: {
          campaignId: reminder.campaignId,
          campaignName: reminder.campaignName,
          dueDate: reminder.dueDate.toISOString(),
          daysRemaining: Math.abs(daysFromDue),
          urgency,
          assignmentId: reminder.assignmentId,
        },
        ccEmails:
          reminder.ccManager && reminder.managerEmail
            ? [reminder.managerEmail]
            : undefined,
      });

      // Record that reminder was sent
      await this.reminderService.recordReminderSent(
        reminder.assignmentId,
        reminder.ccManager,
      );

      this.logger.log(
        `Sent ${urgency} reminder to ${employee.email} for campaign ${reminder.campaignName}`,
      );
    } catch (error) {
      this.logger.error(
        `Failed to send reminder for assignment ${reminder.assignmentId}: ` +
          `${error instanceof Error ? error.message : String(error)}`,
      );
      throw error; // Allow retry
    }
  }

  private calculateDaysFromDue(dueDate: Date): number {
    const now = new Date();
    now.setHours(0, 0, 0, 0);
    const due = new Date(dueDate);
    due.setHours(0, 0, 0, 0);
    return Math.floor((now.getTime() - due.getTime()) / (1000 * 60 * 60 * 24));
  }

  private determineUrgency(
    daysFromDue: number,
  ): "low" | "medium" | "high" | "critical" {
    if (daysFromDue >= 0) return "critical"; // Overdue
    if (daysFromDue >= -1) return "high"; // 1 day or less
    if (daysFromDue >= -3) return "medium"; // 2-3 days
    return "low"; // 4+ days
  }
}
```

```typescript
/**
 * Filters assignments to find those needing reminders for the given date.
 * Uses CONDITION_REMINDER_SEQUENCE for disclosure campaigns.
 */
private filterAssignmentsNeedingReminders(
  assignments: Array<{
    id: string;
    campaignId: string;
    employeeId: string;
    reminderCount: number;
    campaign: {
      id: string;
      name: string;
      type: CampaignType;
      dueDate: Date | null;
      reminderConfig: unknown;
    };
    employee: {
      id: string;
      managerId: string | null;
      manager: { id: string; email: string } | null;
    };
  }>,
  today: Date,
): PendingReminder[] {
  const pendingReminders: PendingReminder[] = [];

  for (const assignment of assignments) {
    const dueDate = assignment.campaign.dueDate;
    if (!dueDate) continue;

    const daysFromDue = Math.floor(
      (today.getTime() - dueDate.getTime()) / (1000 * 60 * 60 * 24),
    );

    // Determine which reminder sequence to use
    let reminderSequence: ReminderStep[];
    if (assignment.campaign.reminderConfig) {
      reminderSequence = assignment.campaign.reminderConfig as unknown as ReminderStep[];
    } else if (assignment.campaign.type === CampaignType.DISCLOSURE) {
      reminderSequence = CONDITION_REMINDER_SEQUENCE;
    } else {
      reminderSequence = DEFAULT_REMINDER_SEQUENCE;
    }

    // Find if there's a reminder for today
    const todaysReminder = reminderSequence.find(
      (step) => step.daysFromDue === daysFromDue,
    );

    if (todaysReminder) {
      // Check if this reminder was already sent
      const reminderIndex = reminderSequence.indexOf(todaysReminder);
      if (assignment.reminderCount >= reminderIndex + 1) {
        continue; // Already sent this reminder
      }

      pendingReminders.push({
        assignmentId: assignment.id,
        employeeId: assignment.employeeId,
        campaignId: assignment.campaignId,
        campaignName: assignment.campaign.name,
        dueDate,
        reminderStep: reminderIndex + 1,
        ccManager: todaysReminder.ccManager,
        ccHR: todaysReminder.ccHR ?? false,
        managerId: assignment.employee.managerId,
        managerEmail: assignment.employee.manager?.email ?? null,
      });
    }
  }

  return pendingReminders;
}
```
