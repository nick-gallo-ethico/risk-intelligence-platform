```typescript
import {
  IsArray,
  IsUUID,
  ArrayMaxSize,
  ArrayMinSize,
  IsString,
  IsOptional,
  IsEnum,
} from "class-validator";

export class BulkDisclosureIdsDto {
  @IsArray()
  @IsUUID("4", { each: true })
  @ArrayMinSize(1)
  @ArrayMaxSize(100)
  disclosureIds: string[];
}

export class BulkApproveDto extends BulkDisclosureIdsDto {
  @IsString()
  @IsOptional()
  approvalNote?: string;
}

export class BulkRejectDto extends BulkDisclosureIdsDto {
  @IsString()
  rejectionReason: string;
}

export interface BulkOperationItem {
  id: string;
  success: boolean;
  error?: string;
}

export interface BulkOperationResult {
  total: number;
  succeeded: number;
  failed: number;
  results: BulkOperationItem[];
}
```

Create BulkDisclosureService:

```typescript
import { Injectable, Logger, BadRequestException } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";
import { AuditService } from "../../audit/audit.service";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { RiuStatus, Prisma } from "@prisma/client";
import {
  BulkApproveDto,
  BulkRejectDto,
  BulkOperationResult,
  BulkOperationItem,
} from "./bulk-disclosure.dto";

@Injectable()
export class BulkDisclosureService {
  private readonly logger = new Logger(BulkDisclosureService.name);
  private readonly CHUNK_SIZE = 50; // Process in chunks of 50 for DB efficiency
  private readonly MAX_ITEMS = 100;

  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  /**
   * Bulk approve disclosures with chunked transaction processing.
   */
  async bulkApprove(
    dto: BulkApproveDto,
    organizationId: string,
    userId: string,
  ): Promise<BulkOperationResult> {
    this.validateItemCount(dto.disclosureIds);

    const results: BulkOperationItem[] = [];

    // Process in chunks
    for (let i = 0; i < dto.disclosureIds.length; i += this.CHUNK_SIZE) {
      const chunk = dto.disclosureIds.slice(i, i + this.CHUNK_SIZE);

      const chunkResults = await this.prisma.$transaction(async (tx) => {
        return Promise.all(
          chunk.map(async (id) => {
            try {
              // Verify disclosure exists and belongs to org
              const disclosure = await tx.riskIntelligenceUnit.findFirst({
                where: { id, organizationId },
                select: { id: true, status: true },
              });

              if (!disclosure) {
                return { id, success: false, error: "Disclosure not found" };
              }

              if (disclosure.status === RiuStatus.COMPLETED) {
                return { id, success: false, error: "Already approved" };
              }

              // Update status
              await tx.riskIntelligenceUnit.update({
                where: { id },
                data: {
                  status: RiuStatus.COMPLETED,
                  statusChangedAt: new Date(),
                  statusChangedById: userId,
                },
              });

              return { id, success: true };
            } catch (error) {
              const message =
                error instanceof Error ? error.message : String(error);
              this.logger.error(
                `Failed to approve disclosure ${id}: ${message}`,
              );
              return { id, success: false, error: message };
            }
          }),
        );
      });

      results.push(...chunkResults);
    }

    // Audit log the bulk operation
    const succeeded = results.filter((r) => r.success).length;
    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: "bulk",
      action: "bulk_approved",
      actionCategory: "UPDATE",
      actionDescription: `Bulk approved ${succeeded} of ${dto.disclosureIds.length} disclosures`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
      changes: {
        disclosureIds: dto.disclosureIds,
        approvalNote: dto.approvalNote,
        results,
      },
    });

    this.eventEmitter.emit("disclosures.bulk_approved", {
      organizationId,
      userId,
      disclosureIds: results.filter((r) => r.success).map((r) => r.id),
      count: succeeded,
    });

    this.logger.log(
      `Bulk approve complete: ${succeeded}/${dto.disclosureIds.length} succeeded`,
    );

    return {
      total: dto.disclosureIds.length,
      succeeded,
      failed: results.filter((r) => !r.success).length,
      results,
    };
  }

  /**
   * Bulk reject disclosures with chunked transaction processing.
   */
  async bulkReject(
    dto: BulkRejectDto,
    organizationId: string,
    userId: string,
  ): Promise<BulkOperationResult> {
    this.validateItemCount(dto.disclosureIds);

    const results: BulkOperationItem[] = [];

    for (let i = 0; i < dto.disclosureIds.length; i += this.CHUNK_SIZE) {
      const chunk = dto.disclosureIds.slice(i, i + this.CHUNK_SIZE);

      const chunkResults = await this.prisma.$transaction(async (tx) => {
        return Promise.all(
          chunk.map(async (id) => {
            try {
              const disclosure = await tx.riskIntelligenceUnit.findFirst({
                where: { id, organizationId },
                select: { id: true, status: true },
              });

              if (!disclosure) {
                return { id, success: false, error: "Disclosure not found" };
              }

              if (disclosure.status === RiuStatus.REJECTED) {
                return { id, success: false, error: "Already rejected" };
              }

              await tx.riskIntelligenceUnit.update({
                where: { id },
                data: {
                  status: RiuStatus.REJECTED,
                  statusChangedAt: new Date(),
                  statusChangedById: userId,
                  statusNote: dto.rejectionReason,
                },
              });

              return { id, success: true };
            } catch (error) {
              const message =
                error instanceof Error ? error.message : String(error);
              this.logger.error(
                `Failed to reject disclosure ${id}: ${message}`,
              );
              return { id, success: false, error: message };
            }
          }),
        );
      });

      results.push(...chunkResults);
    }

    const succeeded = results.filter((r) => r.success).length;
    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: "bulk",
      action: "bulk_rejected",
      actionCategory: "UPDATE",
      actionDescription: `Bulk rejected ${succeeded} of ${dto.disclosureIds.length} disclosures: ${dto.rejectionReason}`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
      changes: {
        disclosureIds: dto.disclosureIds,
        rejectionReason: dto.rejectionReason,
        results,
      },
    });

    this.eventEmitter.emit("disclosures.bulk_rejected", {
      organizationId,
      userId,
      disclosureIds: results.filter((r) => r.success).map((r) => r.id),
      count: succeeded,
      reason: dto.rejectionReason,
    });

    this.logger.log(
      `Bulk reject complete: ${succeeded}/${dto.disclosureIds.length} succeeded`,
    );

    return {
      total: dto.disclosureIds.length,
      succeeded,
      failed: results.filter((r) => !r.success).length,
      results,
    };
  }

  private validateItemCount(ids: string[]): void {
    if (ids.length > this.MAX_ITEMS) {
      throw new BadRequestException(
        `Maximum ${this.MAX_ITEMS} disclosures per bulk operation`,
      );
    }
  }
}
```

Create BulkDisclosureController:

```typescript
import {
  Controller,
  Post,
  Body,
  UseGuards,
  HttpCode,
  HttpStatus,
} from "@nestjs/common";
import { JwtAuthGuard } from "../../auth/guards/jwt-auth.guard";
import { Roles } from "@common/decorators/roles.decorator";
import { RolesGuard } from "@common/guards/roles.guard";
import { CurrentUser } from "@common/decorators/current-user.decorator";
import { TenantId } from "@common/decorators/tenant-id.decorator";
import { User } from "@prisma/client";
import { BulkDisclosureService } from "./bulk-disclosure.service";
import {
  BulkApproveDto,
  BulkRejectDto,
  BulkOperationResult,
} from "./bulk-disclosure.dto";

@Controller("api/v1/disclosures/bulk")
@UseGuards(JwtAuthGuard, RolesGuard)
export class BulkDisclosureController {
  constructor(private readonly bulkDisclosureService: BulkDisclosureService) {}

  /**
   * Bulk approve disclosures (up to 100).
   * Requires COMPLIANCE_OFFICER or SYSTEM_ADMIN role.
   */
  @Post("approve")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN")
  @HttpCode(HttpStatus.OK)
  async bulkApprove(
    @Body() dto: BulkApproveDto,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ): Promise<BulkOperationResult> {
    return this.bulkDisclosureService.bulkApprove(dto, organizationId, user.id);
  }

  /**
   * Bulk reject disclosures (up to 100).
   * Requires COMPLIANCE_OFFICER or SYSTEM_ADMIN role.
   */
  @Post("reject")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN")
  @HttpCode(HttpStatus.OK)
  async bulkReject(
    @Body() dto: BulkRejectDto,
    @CurrentUser() user: User,
    @TenantId() organizationId: string,
  ): Promise<BulkOperationResult> {
    return this.bulkDisclosureService.bulkReject(dto, organizationId, user.id);
  }
}
```
