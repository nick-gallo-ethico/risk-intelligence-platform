```typescript
async evaluateAndAssign(
  configId: string,
  employeeIds: string[],
  organizationId: string,
): Promise<{ assigned: number; skipped: number }> {
  const config = await this.prisma.rollingCampaignConfig.findUnique({
    where: { id: configId },
    include: { segment: true },
  });

  if (!config || !config.isActive) {
    return { assigned: 0, skipped: employeeIds.length };
  }

  // Filter by segment if configured
  let eligibleEmployeeIds = employeeIds;
  if (config.segmentId && config.segment) {
    eligibleEmployeeIds = await this.segmentService.evaluateEmployees(
      config.segmentId,
      employeeIds,
      organizationId,
    );
  }

  // Calculate due date
  const dueDate = new Date();
  dueDate.setDate(dueDate.getDate() + config.dueDaysAfterTrigger);

  // Generate assignments for each eligible employee
  let assigned = 0;
  for (const employeeId of eligibleEmployeeIds) {
    await this.campaignAssignmentService.generateAssignments({
      campaignId: config.campaignTemplateId,
      employeeIds: [employeeId],
      dueDate,
      organizationId,
    });
    assigned++;
  }

  return {
    assigned,
    skipped: employeeIds.length - assigned,
  };
}
```

Create RollingCampaignListener that responds to hris.sync.completed events:

```typescript
@Injectable()
export class RollingCampaignListener {
  private readonly logger = new Logger(RollingCampaignListener.name);

  constructor(
    private readonly rollingCampaignService: RollingCampaignService,
    private readonly prisma: PrismaService,
  ) {}

  @OnEvent("hris.sync.completed")
  async handleHrisSyncCompleted(event: {
    organizationId: string;
    userId: string;
    result: {
      created: number;
      updated: number;
      skipped: number;
      errorCount: number;
    };
  }) {
    // Skip if no changes
    if (event.result.created === 0 && event.result.updated === 0) {
      this.logger.debug(
        `No HRIS changes for org ${event.organizationId}, skipping rolling campaign check`,
      );
      return;
    }

    // Find new hires (created employees with recent syncedAt)
    const recentEmployees = await this.prisma.employee.findMany({
      where: {
        organizationId: event.organizationId,
        syncedAt: { gte: new Date(Date.now() - 5 * 60 * 1000) }, // Last 5 minutes
      },
      select: {
        id: true,
        hrisEmployeeId: true,
        createdAt: true,
        syncedAt: true,
      },
    });

    // Separate new hires vs updates
    const fiveMinutesAgo = new Date(Date.now() - 5 * 60 * 1000);
    const newHires = recentEmployees.filter(
      (e) => e.createdAt >= fiveMinutesAgo,
    );

    // Process NEW_HIRE rolling campaigns
    if (newHires.length > 0) {
      const configs = await this.rollingCampaignService.findActiveByTrigger(
        event.organizationId,
        "NEW_HIRE",
      );
      for (const config of configs) {
        await this.rollingCampaignService.evaluateAndAssign(
          config.id,
          newHires.map((e) => e.id),
          event.organizationId,
        );
      }
    }

    this.logger.log(
      `Rolling campaign check complete for org ${event.organizationId}: ` +
        `${newHires.length} new hires processed`,
    );
  }
}
```
