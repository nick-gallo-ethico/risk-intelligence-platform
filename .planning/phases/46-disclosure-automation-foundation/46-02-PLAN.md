Create HrisSyncFenceService that uses Redis to ensure exactly-once processing:

```typescript
import { Injectable, Logger, Inject } from "@nestjs/common";
import { CACHE_MANAGER } from "@nestjs/cache-manager";
import { Cache } from "cache-manager";

@Injectable()
export class HrisSyncFenceService {
  private readonly logger = new Logger(HrisSyncFenceService.name);
  private readonly DEFAULT_TTL_SECONDS = 300; // 5 minutes

  constructor(@Inject(CACHE_MANAGER) private cacheManager: Cache) {}

  /**
   * Attempts to acquire a fence for HRIS sync processing.
   * Returns true if this caller should process, false if already processed/in-progress.
   *
   * Uses Redis SET NX pattern for atomic acquisition.
   *
   * @param organizationId - Organization being synced
   * @param syncBatchId - Unique identifier for this sync batch (timestamp or UUID)
   * @param ttlSeconds - Time-to-live for the fence (default 5 minutes)
   */
  async acquireFence(
    organizationId: string,
    syncBatchId: string,
    ttlSeconds: number = this.DEFAULT_TTL_SECONDS,
  ): Promise<boolean> {
    const key = this.buildKey(organizationId, syncBatchId);

    try {
      // Try to set the key only if it doesn't exist (NX behavior)
      // cache-manager v5+ uses set with ttl option
      const existing = await this.cacheManager.get(key);
      if (existing) {
        this.logger.debug(`Fence already acquired for ${key}`);
        return false;
      }

      await this.cacheManager.set(
        key,
        Date.now().toString(),
        ttlSeconds * 1000,
      );
      this.logger.debug(`Acquired fence for ${key}`);
      return true;
    } catch (error) {
      this.logger.error(
        `Failed to acquire fence for ${key}: ${error instanceof Error ? error.message : String(error)}`,
      );
      // On error, allow processing (fail-open) but log for monitoring
      return true;
    }
  }

  /**
   * Explicitly releases a fence. Usually not needed since TTL handles cleanup.
   */
  async releaseFence(
    organizationId: string,
    syncBatchId: string,
  ): Promise<void> {
    const key = this.buildKey(organizationId, syncBatchId);
    try {
      await this.cacheManager.del(key);
      this.logger.debug(`Released fence for ${key}`);
    } catch (error) {
      this.logger.warn(
        `Failed to release fence for ${key}: ${error instanceof Error ? error.message : String(error)}`,
      );
    }
  }

  /**
   * Checks if a fence is currently held.
   */
  async isFenced(
    organizationId: string,
    syncBatchId: string,
  ): Promise<boolean> {
    const key = this.buildKey(organizationId, syncBatchId);
    const value = await this.cacheManager.get(key);
    return value !== null && value !== undefined;
  }

  private buildKey(organizationId: string, syncBatchId: string): string {
    return `hris:sync:fence:${organizationId}:${syncBatchId}`;
  }
}
```

Key implementation details:

- Uses CACHE_MANAGER (Redis) for distributed locking
- TTL ensures fence auto-expires for retry scenarios
- Fail-open pattern: if Redis fails, allow processing to avoid blocking
- Unique key per org + sync batch prevents cross-sync interference

@Injectable()
export class RollingCampaignListener {
  private readonly logger = new Logger(RollingCampaignListener.name);

  constructor(
    private readonly rollingCampaignService: RollingCampaignService,
    private readonly fenceService: HrisSyncFenceService,
    private readonly prisma: PrismaService,
  ) {}

  @OnEvent("hris.sync.completed")
  async handleHrisSyncCompleted(event: {
    organizationId: string;
    userId: string;
    result: {
      created: number;
      updated: number;
      skipped: number;
      errorCount: number;
      durationMs?: number;
    };
  }) {
    // Generate sync batch ID from event data
    const syncBatchId = `${event.organizationId}-${Date.now()}`;

    // Try to acquire fence - if another worker got it, skip
    const acquired = await this.fenceService.acquireFence(
      event.organizationId,
      syncBatchId,
    );

    if (!acquired) {
      this.logger.debug(
        `Skipping rolling campaign processing for org ${event.organizationId} - fence held by another worker`,
      );
      return;
    }

    try {
      // Skip if no changes
      if (event.result.created === 0 && event.result.updated === 0) {
        this.logger.debug(
          `No HRIS changes for org ${event.organizationId}, skipping rolling campaign check`,
        );
        return;
      }

      // ... rest of existing handleHrisSyncCompleted logic ...
      // Find new hires, process rolling campaigns, etc.
    } catch (error) {
      this.logger.error(
        `Error processing rolling campaigns for org ${event.organizationId}: ` +
          `${error instanceof Error ? error.message : String(error)}`,
      );
      // Don't release fence on error - let TTL expire to prevent immediate retry
      throw error;
    }
  }
}
