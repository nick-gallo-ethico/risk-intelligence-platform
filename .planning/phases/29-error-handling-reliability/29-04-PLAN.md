```typescript
/**
 * Decrypt sensitive fields after reading.
 * Returns _decryptionFailed: true if decryption fails (key may have changed).
 */
decryptDraft(draft: ReportDraft): DecryptedDraft {
  if (!this.encryptionKey) {
    throw new Error('Encryption not initialized. Call initEncryption() first.');
  }

  try {
    return {
      ...draft,
      content: JSON.parse(
        decryptValue(draft.content as unknown as string, this.encryptionKey)
      ),
      attachments: JSON.parse(
        decryptValue(draft.attachments as unknown as string, this.encryptionKey)
      ),
    };
  } catch (error) {
    // Log the failure for debugging
    console.error('Draft decryption failed:', error);
    // Return marker flag so UI can show error message
    return {
      ...draft,
      content: {},
      attachments: [],
      _decryptionFailed: true,
    };
  }
}
```

3. Export the DecryptedDraft type so consuming components can check for the flag.

Modify the getUser() method to log and clear corrupted localStorage entries:

1. Find the getUser method (around line 33-42) and update the catch block:

```typescript
getUser<T>(): T | null {
  if (typeof window === 'undefined') return null;

  const user = localStorage.getItem(TOKEN_KEYS.USER);
  if (!user) return null;

  try {
    return JSON.parse(user) as T;
  } catch (error) {
    // Log corruption and clean up
    console.warn(
      'Corrupted user data in localStorage. Clearing auth state.',
      { key: TOKEN_KEYS.USER, error: error instanceof Error ? error.message : error },
    );
    // Clean up corrupted entry to prevent repeated errors
    localStorage.removeItem(TOKEN_KEYS.USER);
    return null;
  }
}
```

2. Also add similar handling to any other methods that parse JSON from localStorage. Check for getAccessToken and getRefreshToken - if they parse JSON, add the same pattern. If they just return strings (no JSON.parse), they don't need this change.

Key point: The cleanup (removeItem) is important to prevent the same error on every page load. Log first, then clean up.
