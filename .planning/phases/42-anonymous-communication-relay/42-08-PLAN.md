/**
 * Phase 42: Anonymous Communication Relay - Demo Data
 *
 * Adds sample relay messages to existing Acme Co. cases:
 * - Active conversation with unread messages
 * - Closed conversation with full thread
 * - Conversation awaiting reporter response
 */
import {
  PrismaClient,
  MessageDirection,
  MessageSenderType,
  MessageDeliveryStatus,
} from "@prisma/client";

export async function seedPhase42(prisma: PrismaClient): Promise<void> {
  console.log("Seeding Phase 42: Anonymous Communication Relay...");

  // Get Acme organization
  const acme = await prisma.organization.findFirst({
    where: { slug: "acme" },
  });

  if (!acme) {
    console.log("Skipping Phase 42 seed: Acme organization not found");
    return;
  }

  // Configure relay settings for Acme
  await prisma.organization.update({
    where: { id: acme.id },
    data: {
      settings: {
        ...((acme.settings as object) || {}),
        reporterVisibilityLevel: "STANDARD",
        enableMessaging: true,
        autoNotifyOnMessage: true,
        notificationDelayMinHours: 1,
        notificationDelayMaxHours: 6,
      },
    },
  });
  console.log("  - Configured relay settings for Acme");

  // Find cases with linked RIUs that have reporter email
  const casesWithReporter = await prisma.case.findMany({
    where: {
      organizationId: acme.id,
      riuAssociations: {
        some: {
          riu: {
            reporterEmail: { not: null },
          },
        },
      },
    },
    take: 3,
    orderBy: { createdAt: "desc" },
    include: {
      riuAssociations: {
        include: { riu: true },
        take: 1,
      },
    },
  });

  if (casesWithReporter.length === 0) {
    console.log("  - No cases with reporter email found, skipping messages");
    return;
  }

  // Get an investigator
  const investigator = await prisma.user.findFirst({
    where: {
      organizationId: acme.id,
      role: { in: ["COMPLIANCE_OFFICER", "INVESTIGATOR", "SYSTEM_ADMIN"] },
    },
  });

  if (!investigator) {
    console.log("  - No investigator found, skipping messages");
    return;
  }

  // Create message threads
  const now = new Date();
  let messagesCreated = 0;

  for (let i = 0; i < casesWithReporter.length; i++) {
    const caseRecord = casesWithReporter[i];

    // Thread patterns
    const threadPatterns = [
      // Active conversation with unread
      [
        {
          direction: MessageDirection.OUTBOUND,
          content:
            "Thank you for reporting this concern. We take all reports seriously and have begun our investigation. Can you provide more details about the date this incident occurred?",
          daysAgo: 5,
          isRead: true,
        },
        {
          direction: MessageDirection.INBOUND,
          content:
            "It happened on January 15th around 2pm in the main conference room.",
          daysAgo: 4,
          isRead: true,
        },
        {
          direction: MessageDirection.OUTBOUND,
          content:
            "Thank you for the additional information. Were there any witnesses present?",
          daysAgo: 2,
          isRead: false,
        },
      ],
      // Closed conversation
      [
        {
          direction: MessageDirection.OUTBOUND,
          content:
            "We have reviewed your report and completed our investigation. Thank you for bringing this to our attention.",
          daysAgo: 30,
          isRead: true,
        },
        {
          direction: MessageDirection.INBOUND,
          content:
            "Thank you for looking into this. I appreciate the follow-up.",
          daysAgo: 29,
          isRead: true,
        },
        {
          direction: MessageDirection.OUTBOUND,
          content:
            "You are welcome. Your report helped us identify and address an important issue. If you have any other concerns, please do not hesitate to reach out.",
          daysAgo: 28,
          isRead: true,
        },
      ],
      // Awaiting response
      [
        {
          direction: MessageDirection.INBOUND,
          content:
            "I wanted to follow up on my report. Has there been any progress?",
          daysAgo: 3,
          isRead: true,
        },
        {
          direction: MessageDirection.OUTBOUND,
          content:
            "Thank you for following up. Our investigation is ongoing. We expect to have an update for you within the next week. Is there anything else you can share that might help?",
          daysAgo: 2,
          isRead: true,
        },
      ],
    ];

    const pattern = threadPatterns[i % threadPatterns.length];

    for (const msg of pattern) {
      const createdAt = new Date(
        now.getTime() - msg.daysAgo * 24 * 60 * 60 * 1000,
      );

      await prisma.caseMessage.create({
        data: {
          organizationId: acme.id,
          caseId: caseRecord.id,
          direction: msg.direction,
          senderType:
            msg.direction === MessageDirection.OUTBOUND
              ? MessageSenderType.INVESTIGATOR
              : MessageSenderType.REPORTER,
          content: msg.content,
          isRead: msg.isRead,
          readAt: msg.isRead
            ? new Date(createdAt.getTime() + 60 * 60 * 1000)
            : null,
          readById:
            msg.isRead && msg.direction === MessageDirection.INBOUND
              ? investigator.id
              : null,
          createdById:
            msg.direction === MessageDirection.OUTBOUND
              ? investigator.id
              : null,
          deliveryStatus:
            msg.direction === MessageDirection.OUTBOUND
              ? MessageDeliveryStatus.DELIVERED
              : null,
          createdAt,
        },
      });
      messagesCreated++;
    }
  }

  console.log(
    `  - Created ${messagesCreated} relay messages across ${casesWithReporter.length} cases`,
  );
  console.log("Phase 42 seed complete.");
}

/**
 * E2E Tests: Relay Tenant Isolation
 *
 * Verifies that relay messages and settings are properly isolated between tenants.
 */
import { INestApplication } from "@nestjs/common";
import * as request from "supertest";
import {
  setupE2ETest,
  createTestTenant,
  getAuthToken,
} from "../helpers/test-setup";

describe("Relay Tenant Isolation (e2e)", () => {
  let app: INestApplication;
  let tenant1: { id: string; token: string; caseId: string; userId: string };
  let tenant2: { id: string; token: string; caseId: string; userId: string };

  beforeAll(async () => {
    app = await setupE2ETest();

    // Create two tenants with cases
    tenant1 = await createTestTenant(app, "tenant1");
    tenant2 = await createTestTenant(app, "tenant2");
  });

  afterAll(async () => {
    await app.close();
  });

  describe("Message Isolation", () => {
    it("should not allow tenant1 to access tenant2 case messages", async () => {
      // Create message in tenant2
      await request(app.getHttpServer())
        .post(`/api/cases/${tenant2.caseId}/messages`)
        .set("Authorization", `Bearer ${tenant2.token}`)
        .send({ content: "Test message from tenant2" })
        .expect(201);

      // Try to access from tenant1
      const response = await request(app.getHttpServer())
        .get(`/api/cases/${tenant2.caseId}/messages`)
        .set("Authorization", `Bearer ${tenant1.token}`)
        .expect(404);

      expect(response.body.message).toContain("not found");
    });

    it("should not leak messages across tenants in listing", async () => {
      // Create messages in both tenants
      await request(app.getHttpServer())
        .post(`/api/cases/${tenant1.caseId}/messages`)
        .set("Authorization", `Bearer ${tenant1.token}`)
        .send({ content: "Tenant1 secret message" });

      await request(app.getHttpServer())
        .post(`/api/cases/${tenant2.caseId}/messages`)
        .set("Authorization", `Bearer ${tenant2.token}`)
        .send({ content: "Tenant2 secret message" });

      // Get tenant1 messages
      const response = await request(app.getHttpServer())
        .get(`/api/cases/${tenant1.caseId}/messages`)
        .set("Authorization", `Bearer ${tenant1.token}`)
        .expect(200);

      // Should not contain tenant2 messages
      const contents = response.body.map((m: any) => m.content);
      expect(contents).not.toContain("Tenant2 secret message");
    });
  });

  describe("Settings Isolation", () => {
    it("should not allow tenant1 to update tenant2 relay settings", async () => {
      await request(app.getHttpServer())
        .patch(`/api/organizations/${tenant2.id}/relay-settings`)
        .set("Authorization", `Bearer ${tenant1.token}`)
        .send({ reporterVisibilityLevel: "MINIMAL" })
        .expect(403);
    });

    it("should maintain separate settings per tenant", async () => {
      // Update tenant1 to MINIMAL
      await request(app.getHttpServer())
        .patch(`/api/organizations/${tenant1.id}/relay-settings`)
        .set("Authorization", `Bearer ${tenant1.token}`)
        .send({ reporterVisibilityLevel: "MINIMAL" })
        .expect(200);

      // Update tenant2 to TRANSPARENT
      await request(app.getHttpServer())
        .patch(`/api/organizations/${tenant2.id}/relay-settings`)
        .set("Authorization", `Bearer ${tenant2.token}`)
        .send({ reporterVisibilityLevel: "TRANSPARENT" })
        .expect(200);

      // Verify tenant1 is MINIMAL
      const t1Settings = await request(app.getHttpServer())
        .get(`/api/organizations/${tenant1.id}/relay-settings`)
        .set("Authorization", `Bearer ${tenant1.token}`)
        .expect(200);
      expect(t1Settings.body.reporterVisibilityLevel).toBe("MINIMAL");

      // Verify tenant2 is TRANSPARENT
      const t2Settings = await request(app.getHttpServer())
        .get(`/api/organizations/${tenant2.id}/relay-settings`)
        .set("Authorization", `Bearer ${tenant2.token}`)
        .expect(200);
      expect(t2Settings.body.reporterVisibilityLevel).toBe("TRANSPARENT");
    });
  });

  describe("Access Code Isolation", () => {
    it("should not allow access code from tenant1 to view tenant2 messages", async () => {
      // This tests the public reporter endpoint
      // Access codes are tenant-scoped via RIU
      const tenant1AccessCode = "ACME12345678";
      const tenant2AccessCode = "BETA12345678";

      // Get messages with tenant1 code should not return tenant2 data
      const response = await request(app.getHttpServer())
        .get(`/api/reporter/status/${tenant1AccessCode}/messages`)
        .expect(200); // Or 404 if code not found

      // Response should only contain tenant1 data (or empty)
      if (response.body.length > 0) {
        expect(
          response.body.every(
            (m: any) =>
              m.organizationId === tenant1.id || m.organizationId === undefined,
          ),
        ).toBe(true);
      }
    });
  });
});
