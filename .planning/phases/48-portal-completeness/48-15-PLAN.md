// apps/backend/prisma/seeders/acme-phase-48.ts

import { PrismaClient, DirectiveStage } from "@prisma/client";

const prisma = new PrismaClient();

/**
 * Phase 48: Portal Completeness - Demo Data
 *
 * Seeds:
 * - Opening/closing statements for operator scripts
 * - Category-specific intake questions
 * - Crisis banner configuration
 * - Transparency stats configuration
 * - Sample disclosure conditions
 */
export async function seedPhase48(organizationId: string) {
  console.log("Seeding Phase 48: Portal Completeness...");

  // Get or create Acme org
  const org = await prisma.organization.findUnique({
    where: { id: organizationId },
  });

  if (!org) {
    console.log("Organization not found, skipping Phase 48 seed");
    return;
  }

  // 1. Create Opening/Closing Statements (Directives)
  const openingStatements = [
    {
      title: "Welcome Statement",
      content:
        "Thank you for calling the Acme Ethics Hotline. This call may be recorded for quality assurance purposes. My name is [Operator Name], and I am here to help you today.",
      isReadAloud: true,
    },
    {
      title: "Confidentiality Assurance",
      content:
        "Before we begin, I want to assure you that Acme takes your confidentiality seriously. You may remain anonymous if you choose, and there will be no retaliation for making a good-faith report.",
      isReadAloud: true,
    },
  ];

  const closingStatements = [
    {
      title: "Summary Confirmation",
      content:
        "Before we conclude, let me summarize what you have shared with me today to ensure I have captured everything correctly.",
      isReadAloud: true,
    },
    {
      title: "Access Code Delivery",
      content:
        "Your report has been assigned the following access code: [ACCESS_CODE]. Please keep this code safe as you will need it to check the status of your report or provide additional information.",
      isReadAloud: true,
    },
    {
      title: "Thank You",
      content:
        "Thank you for bringing this matter to our attention. Acme is committed to maintaining the highest ethical standards, and reports like yours help us achieve that goal.",
      isReadAloud: true,
    },
  ];

  for (let i = 0; i < openingStatements.length; i++) {
    await prisma.clientDirective.upsert({
      where: {
        organizationId_stage_title: {
          organizationId,
          stage: DirectiveStage.OPENING,
          title: openingStatements[i].title,
        },
      },
      update: {},
      create: {
        organizationId,
        stage: DirectiveStage.OPENING,
        title: openingStatements[i].title,
        content: openingStatements[i].content,
        isReadAloud: openingStatements[i].isReadAloud,
        order: i,
        isActive: true,
      },
    });
  }

  for (let i = 0; i < closingStatements.length; i++) {
    await prisma.clientDirective.upsert({
      where: {
        organizationId_stage_title: {
          organizationId,
          stage: DirectiveStage.CLOSING,
          title: closingStatements[i].title,
        },
      },
      update: {},
      create: {
        organizationId,
        stage: DirectiveStage.CLOSING,
        title: closingStatements[i].title,
        content: closingStatements[i].content,
        isReadAloud: closingStatements[i].isReadAloud,
        order: i,
        isActive: true,
      },
    });
  }

  console.log("  - Created opening/closing statements");

  // 2. Update organization with crisis banner and transparency settings
  await prisma.organization.update({
    where: { id: organizationId },
    data: {
      // Crisis banner (inactive by default for demo)
      crisisBannerActive: false,
      crisisBannerMessage:
        "If you are experiencing a life-threatening emergency, please call 911 immediately.",
      crisisBannerSeverity: "critical",

      // Emergency phone
      emergencyPhone: "1-800-ACME-ETH",

      // Transparency stats
      showTransparencyStats: true,

      // Default language settings
      defaultLanguage: "en",
      supportedLanguages: ["en", "es", "fr"],
    },
  });

  console.log("  - Updated organization with portal settings");

  // 3. Create sample category-specific intake questions
  // (This assumes CategoryIntakeQuestion model exists; if not, skip this section)
  const categories = await prisma.category.findMany({
    where: { organizationId },
    take: 3,
  });

  for (const category of categories) {
    // Delete existing questions for this category first
    await prisma.categoryIntakeQuestion.deleteMany({
      where: { categoryId: category.id },
    });

    // Create category-specific questions
    const questions = getCategoryQuestions(category.code || category.name);
    for (let i = 0; i < questions.length; i++) {
      await prisma.categoryIntakeQuestion.create({
        data: {
          categoryId: category.id,
          organizationId,
          questionText: questions[i].text,
          questionType: questions[i].type,
          options: questions[i].options || null,
          isRequired: questions[i].required || false,
          order: i,
          helpText: questions[i].helpText || null,
          isActive: true,
        },
      });
    }
  }

  console.log("  - Created category-specific intake questions");

  console.log("Phase 48 seed complete.");
}

function getCategoryQuestions(categoryCode: string): Array<{
  text: string;
  type: string;
  options?: string[];
  required?: boolean;
  helpText?: string;
}> {
  const questionSets: Record<string, typeof questions> = {
    HARASSMENT: [
      { text: "When did this incident occur?", type: "date", required: true },
      {
        text: "Where did this incident take place?",
        type: "text",
        required: true,
      },
      { text: "Were there any witnesses?", type: "boolean" },
      {
        text: "How would you describe the frequency?",
        type: "select",
        options: ["One-time incident", "Multiple occasions", "Ongoing pattern"],
      },
    ],
    DISCRIMINATION: [
      {
        text: "What type of discrimination occurred?",
        type: "select",
        options: ["Race", "Gender", "Age", "Religion", "Disability", "Other"],
      },
      { text: "When did you first notice this behavior?", type: "date" },
      {
        text: "Has anyone else experienced similar treatment?",
        type: "boolean",
      },
    ],
    FRAUD: [
      { text: "What is the estimated monetary value involved?", type: "text" },
      { text: "How did you become aware of this activity?", type: "text" },
      { text: "Do you have documentation or evidence?", type: "boolean" },
    ],
  };

  // Default questions for unknown categories
  const defaultQuestions = [
    { text: "When did you first become aware of this issue?", type: "date" },
    { text: "Who is involved in this matter?", type: "text" },
    {
      text: "Has this been reported through any other channel?",
      type: "boolean",
    },
  ];

  return questionSets[categoryCode.toUpperCase()] || defaultQuestions;
}

// Run if executed directly
if (require.main === module) {
  const orgId = process.argv[2] || "acme-demo-org-id";
  seedPhase48(orgId)
    .then(() => process.exit(0))
    .catch((e) => {
      console.error(e);
      process.exit(1);
    });
}
