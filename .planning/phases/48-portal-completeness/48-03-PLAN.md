```typescript
async getMyReports(
  userId: string,
  organizationId: string,
): Promise<MyReportsResponseDto> {
  // Find employee by user email
  const user = await this.prisma.user.findUnique({
    where: { id: userId },
    select: { email: true },
  });

  // Find RIUs submitted by this user (via reporterEmail or createdById for proxy)
  const rius = await this.prisma.riskIntelligenceUnit.findMany({
    where: {
      organizationId,
      OR: [
        { reporterEmail: user.email },
        { createdById: userId },
      ],
    },
    include: {
      category: { select: { name: true } },
      caseAssociations: {
        include: {
          case: {
            select: {
              id: true,
              referenceNumber: true,
              status: true,
              closedAt: true,
            },
          },
        },
      },
    },
    orderBy: { createdAt: 'desc' },
    take: 100, // Reasonable limit
  });

  // Build timeline from RIUs and their linked Cases
  const timeline = this.buildReportTimeline(rius);

  return {
    reports: rius.map(riu => ({
      riuId: riu.id,
      referenceNumber: riu.referenceNumber,
      type: riu.type,
      status: riu.status,
      createdAt: riu.createdAt,
      category: riu.category?.name ?? null,
      linkedCases: riu.caseAssociations.map(assoc => ({
        caseId: assoc.case.id,
        caseReferenceNumber: assoc.case.referenceNumber,
        status: assoc.case.status,
        closedAt: assoc.case.closedAt,
      })),
    })),
    timeline,
  };
}

private buildReportTimeline(rius): ReportTimelineItemDto[] {
  const events: ReportTimelineItemDto[] = [];

  for (const riu of rius) {
    // RIU creation event
    events.push({
      id: `riu-created-${riu.id}`,
      type: 'riu_created',
      date: riu.createdAt,
      title: 'Report Submitted',
      description: `Report ${riu.referenceNumber} submitted`,
      riuId: riu.id,
      riuReferenceNumber: riu.referenceNumber,
    });

    // Case events
    for (const assoc of riu.caseAssociations) {
      const c = assoc.case;
      events.push({
        id: `case-linked-${c.id}`,
        type: 'case_created',
        date: assoc.createdAt,
        title: 'Case Created',
        description: `Case ${c.referenceNumber} created from your report`,
        riuId: riu.id,
        riuReferenceNumber: riu.referenceNumber,
        caseId: c.id,
        caseReferenceNumber: c.referenceNumber,
        status: c.status,
      });

      if (c.closedAt) {
        events.push({
          id: `case-closed-${c.id}`,
          type: 'case_closed',
          date: c.closedAt,
          title: 'Case Closed',
          description: `Case ${c.referenceNumber} has been closed`,
          caseId: c.id,
          caseReferenceNumber: c.referenceNumber,
          status: c.status,
        });
      }
    }
  }

  // Sort by date descending
  return events.sort((a, b) => b.date.getTime() - a.date.getTime());
}
````

```typescript
'use client';

import { useQuery } from '@tanstack/react-query';
import { apiClient } from '@/lib/api';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { Skeleton } from '@/components/ui/skeleton';
import { FileText, Folder, CheckCircle, Clock } from 'lucide-react';
import { formatDistanceToNow } from 'date-fns';

interface TimelineItem {
  id: string;
  type: 'riu_created' | 'case_created' | 'case_status_changed' | 'case_closed';
  date: string;
  title: string;
  description: string;
  riuReferenceNumber?: string;
  caseReferenceNumber?: string;
  status?: string;
}

interface MyReportsResponse {
  reports: Array<{
    riuId: string;
    referenceNumber: string;
    type: string;
    status: string;
    createdAt: string;
    category: string | null;
    linkedCases: Array<{
      caseId: string;
      caseReferenceNumber: string;
      status: string;
      closedAt: string | null;
    }>;
  }>;
  timeline: TimelineItem[];
}

export function MyReportsTimeline() {
  const { data, isLoading, error } = useQuery({
    queryKey: ['employee', 'my-reports'],
    queryFn: () => apiClient.get<MyReportsResponse>('/employee/my-reports'),
  });

  if (isLoading) {
    return <TimelineSkeleton />;
  }

  if (error || !data) {
    return (
      <div className="text-center py-12 text-muted-foreground">
        <p>Unable to load your reports.</p>
      </div>
    );
  }

  if (data.reports.length === 0) {
    return (
      <div className="text-center py-12 text-muted-foreground">
        <FileText className="h-12 w-12 mx-auto mb-4 opacity-50" />
        <p>No reports found.</p>
        <p className="text-sm mt-2">Reports you submit will appear here.</p>
      </div>
    );
  }

  return (
    <div className="space-y-6">
      {/* Summary cards */}
      <div className="grid gap-4 grid-cols-2 lg:grid-cols-4">
        <SummaryCard
          title="Total Reports"
          value={data.reports.length}
          icon={FileText}
        />
        <SummaryCard
          title="Open Cases"
          value={data.reports.filter(r => r.linkedCases.some(c => !c.closedAt)).length}
          icon={Folder}
        />
        {/* ... more cards */}
      </div>

      {/* Timeline */}
      <Card>
        <CardHeader>
          <CardTitle>Activity Timeline</CardTitle>
        </CardHeader>
        <CardContent>
          <div className="relative">
            {/* Vertical line */}
            <div className="absolute left-4 top-0 bottom-0 w-0.5 bg-border" />

            {data.timeline.map((item, index) => (
              <TimelineEvent key={item.id} item={item} isLast={index === data.timeline.length - 1} />
            ))}
          </div>
        </CardContent>
      </Card>
    </div>
  );
}

function TimelineEvent({ item, isLast }: { item: TimelineItem; isLast: boolean }) {
  const iconMap = {
    riu_created: FileText,
    case_created: Folder,
    case_status_changed: Clock,
    case_closed: CheckCircle,
  };
  const Icon = iconMap[item.type] || FileText;

  return (
    <div className="relative pl-10 pb-8">
      {/* Icon circle */}
      <div className="absolute left-0 w-8 h-8 rounded-full bg-background border-2 border-primary flex items-center justify-center">
        <Icon className="h-4 w-4" />
      </div>

      <div className="bg-muted/50 rounded-lg p-4">
        <div className="flex justify-between items-start">
          <div>
            <h4 className="font-medium">{item.title}</h4>
            <p className="text-sm text-muted-foreground">{item.description}</p>
          </div>
          <span className="text-xs text-muted-foreground">
            {formatDistanceToNow(new Date(item.date), { addSuffix: true })}
          </span>
        </div>
        {item.status && (
          <Badge variant="outline" className="mt-2">{item.status}</Badge>
        )}
      </div>
    </div>
  );
}
```
