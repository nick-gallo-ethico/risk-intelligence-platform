```typescript
'use client';

import { useTranslation } from 'react-i18next';
import { AlertTriangle, Phone, ExternalLink } from 'lucide-react';
import { cn } from '@/lib/utils';

interface CrisisBannerProps {
  /** Crisis message to display */
  message: string;
  /** Severity level affects styling */
  severity: 'warning' | 'critical';
  /** Emergency phone number (optional) */
  emergencyPhone?: string;
  /** Translated messages by language code */
  translations?: Record<string, string>;
  /** Additional CSS classes */
  className?: string;
}

/**
 * Non-dismissible crisis banner for the ethics portal.
 * Displayed prominently at the top of the page.
 *
 * Requirements:
 * - Cannot be dismissed (no close button)
 * - Sticky to top of viewport
 * - Emergency phone clickable on mobile
 * - Supports translations
 */
export function CrisisBanner({
  message,
  severity,
  emergencyPhone,
  translations,
  className,
}: CrisisBannerProps) {
  const { i18n } = useTranslation();

  // Get translated message if available, fallback to default message
  const displayMessage = translations?.[i18n.language] || message;

  return (
    <div
      role="alert"
      aria-live="assertive"
      className={cn(
        // Base styles
        'w-full py-3 px-4',
        'flex items-center justify-center gap-3',
        'sticky top-0 z-50',
        // No dismiss - using CSS to prevent hiding
        'pointer-events-auto select-text',
        // Severity-based colors
        severity === 'critical'
          ? 'bg-destructive text-destructive-foreground'
          : 'bg-amber-500 text-amber-950',
        className
      )}
    >
      {/* Warning icon */}
      <AlertTriangle className="h-5 w-5 flex-shrink-0" aria-hidden="true" />

      {/* Message */}
      <span className="text-sm font-medium text-center flex-1 max-w-2xl">
        {displayMessage}
      </span>

      {/* Emergency phone (if provided) */}
      {emergencyPhone && (
        <a
          href={`tel:${emergencyPhone.replace(/\D/g, '')}`}
          className={cn(
            'inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full',
            'text-sm font-semibold whitespace-nowrap',
            'bg-white/20 hover:bg-white/30 transition-colors',
            'focus:outline-none focus:ring-2 focus:ring-white/50'
          )}
        >
          <Phone className="h-4 w-4" aria-hidden="true" />
          <span>{emergencyPhone}</span>
        </a>
      )}
    </div>
  );
}
```

```typescript
'use client';

import { useTranslation } from 'react-i18next';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import {
  FileText,
  Users,
  Clock,
  TrendingUp,
  TrendingDown,
  Minus,
  ShieldCheck,
  Folder,
} from 'lucide-react';
import { cn } from '@/lib/utils';

interface TransparencyStatsProps {
  stats: {
    totalReportsReceived: number;
    reportsThisYear: number;
    averageResolutionDays: number;
    categoriesReported: number;
    anonymousPercentage: number;
    reportingTrend: 'increasing' | 'stable' | 'decreasing';
  };
  className?: string;
}

/**
 * Anonymized program statistics display for the ethics portal.
 * Shows aggregate metrics without any identifying information.
 */
export function TransparencyStats({ stats, className }: TransparencyStatsProps) {
  const { t } = useTranslation();

  const trendConfig = {
    increasing: { icon: TrendingUp, color: 'text-green-600', label: 'Increasing' },
    stable: { icon: Minus, color: 'text-gray-500', label: 'Stable' },
    decreasing: { icon: TrendingDown, color: 'text-amber-600', label: 'Decreasing' },
  };

  const trend = trendConfig[stats.reportingTrend];
  const TrendIcon = trend.icon;

  return (
    <Card className={cn('', className)}>
      <CardHeader className="pb-2">
        <CardTitle className="text-lg flex items-center gap-2">
          <ShieldCheck className="h-5 w-5 text-primary" />
          {t('ethics.transparency.title', 'Program Transparency')}
        </CardTitle>
        <p className="text-sm text-muted-foreground">
          {t(
            'ethics.transparency.description',
            'Anonymized statistics about our ethics and compliance program'
          )}
        </p>
      </CardHeader>
      <CardContent>
        <div className="grid gap-4 grid-cols-2 lg:grid-cols-3">
          {/* Total Reports */}
          <StatCard
            icon={FileText}
            label={t('ethics.transparency.totalReports', 'Total Reports')}
            value={stats.totalReportsReceived.toLocaleString()}
            sublabel={t('ethics.transparency.allTime', 'All time')}
          />

          {/* Reports This Year */}
          <StatCard
            icon={Folder}
            label={t('ethics.transparency.thisYear', 'This Year')}
            value={stats.reportsThisYear.toLocaleString()}
            sublabel={
              <div className="flex items-center gap-1">
                <TrendIcon className={cn('h-3 w-3', trend.color)} />
                <span className={trend.color}>{trend.label}</span>
              </div>
            }
          />

          {/* Average Resolution */}
          <StatCard
            icon={Clock}
            label={t('ethics.transparency.avgResolution', 'Avg Resolution')}
            value={`${stats.averageResolutionDays}`}
            sublabel={t('ethics.transparency.days', 'days')}
          />

          {/* Categories */}
          <StatCard
            icon={Folder}
            label={t('ethics.transparency.categories', 'Categories Reported')}
            value={stats.categoriesReported.toString()}
            sublabel={t('ethics.transparency.differentTypes', 'Different types')}
          />

          {/* Anonymous Percentage */}
          <StatCard
            icon={Users}
            label={t('ethics.transparency.anonymous', 'Anonymous Reports')}
            value={`${stats.anonymousPercentage}%`}
            sublabel={t('ethics.transparency.ofTotal', 'of total reports')}
          />
        </div>

        <p className="text-xs text-muted-foreground mt-4 text-center">
          {t(
            'ethics.transparency.disclaimer',
            'Statistics are anonymized and updated periodically. No individual reports are identifiable.'
          )}
        </p>
      </CardContent>
    </Card>
  );
}

interface StatCardProps {
  icon: React.ComponentType<{ className?: string }>;
  label: string;
  value: string;
  sublabel: React.ReactNode;
}

function StatCard({ icon: Icon, label, value, sublabel }: StatCardProps) {
  return (
    <div className="flex flex-col items-center text-center p-3 rounded-lg bg-muted/30">
      <Icon className="h-5 w-5 text-muted-foreground mb-1" />
      <span className="text-xs text-muted-foreground">{label}</span>
      <span className="text-2xl font-bold">{value}</span>
      <span className="text-xs text-muted-foreground">{sublabel}</span>
    </div>
  );
}
````

```typescript
'use client';

import { useParams } from 'next/navigation';
import { useQuery } from '@tanstack/react-query';
import { apiClient } from '@/lib/api';
import { CrisisBanner } from '@/components/ethics/crisis-banner';
import { EmergencyPhone } from '@/components/ethics/emergency-phone';
import { TransparencyStats } from '@/components/ethics/transparency-stats';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Skeleton } from '@/components/ui/skeleton';
import { FileText, MessageSquare, Shield } from 'lucide-react';
import Link from 'next/link';

interface EthicsPortalConfig {
  organizationId: string;
  organizationName: string;
  logoUrl?: string;
  primaryColor?: string;
  emergencyPhone?: string;
  crisisBanner?: {
    isActive: boolean;
    message: string;
    severity: 'warning' | 'critical';
    emergencyPhone?: string;
    translations?: Record<string, string>;
  };
  showTransparencyStats: boolean;
  transparencyStats?: {
    totalReportsReceived: number;
    reportsThisYear: number;
    averageResolutionDays: number;
    categoriesReported: number;
    anonymousPercentage: number;
    reportingTrend: 'increasing' | 'stable' | 'decreasing';
  };
}

export default function EthicsPortalLandingPage() {
  const params = useParams();
  const tenantSlug = params?.tenant as string;

  const { data: config, isLoading } = useQuery({
    queryKey: ['ethics', 'config', tenantSlug],
    queryFn: () => apiClient.get<EthicsPortalConfig>(`/ethics/config/${tenantSlug}`),
    enabled: !!tenantSlug,
  });

  if (isLoading) {
    return <LandingSkeleton />;
  }

  if (!config) {
    return <div>Portal not found</div>;
  }

  return (
    <div className="min-h-screen flex flex-col">
      {/* Crisis Banner - Always at top, non-dismissible */}
      {config.crisisBanner?.isActive && (
        <CrisisBanner
          message={config.crisisBanner.message}
          severity={config.crisisBanner.severity}
          emergencyPhone={config.crisisBanner.emergencyPhone}
          translations={config.crisisBanner.translations}
        />
      )}

      {/* Main content */}
      <main className="flex-1 container mx-auto px-4 py-8">
        {/* Header */}
        <div className="text-center mb-8">
          {config.logoUrl && (
            <img
              src={config.logoUrl}
              alt={config.organizationName}
              className="h-16 mx-auto mb-4"
            />
          )}
          <h1 className="text-3xl font-bold">{config.organizationName}</h1>
          <p className="text-lg text-muted-foreground mt-2">
            Ethics & Compliance Reporting
          </p>
        </div>

        {/* Emergency Phone (if configured and no crisis banner with phone) */}
        {config.emergencyPhone && !config.crisisBanner?.emergencyPhone && (
          <div className="max-w-md mx-auto mb-8">
            <EmergencyPhone phoneNumber={config.emergencyPhone} />
          </div>
        )}

        {/* Action Cards */}
        <div className="grid gap-6 md:grid-cols-2 lg:grid-cols-3 max-w-4xl mx-auto mb-8">
          <ActionCard
            icon={FileText}
            title="Submit a Report"
            description="Report an ethics or compliance concern anonymously or confidentially"
            href={`/ethics/${tenantSlug}/report`}
            primary
          />

          <ActionCard
            icon={MessageSquare}
            title="Check Report Status"
            description="Check the status of a previously submitted report using your access code"
            href={`/ethics/${tenantSlug}/status`}
          />

          <ActionCard
            icon={Shield}
            title="Our Commitment"
            description="Learn about our non-retaliation policy and confidentiality protections"
            href={`/ethics/${tenantSlug}/commitment`}
          />
        </div>

        {/* Transparency Stats (if enabled) */}
        {config.showTransparencyStats && config.transparencyStats && (
          <div className="max-w-3xl mx-auto">
            <TransparencyStats stats={config.transparencyStats} />
          </div>
        )}
      </main>

      {/* Footer */}
      <footer className="border-t py-4 text-center text-sm text-muted-foreground">
        <p>Powered by Ethico</p>
      </footer>
    </div>
  );
}

function ActionCard({
  icon: Icon,
  title,
  description,
  href,
  primary,
}: {
  icon: React.ComponentType<{ className?: string }>;
  title: string;
  description: string;
  href: string;
  primary?: boolean;
}) {
  return (
    <Link href={href}>
      <Card
        className={`h-full transition-shadow hover:shadow-md cursor-pointer ${
          primary ? 'border-primary' : ''
        }`}
      >
        <CardContent className="pt-6 text-center">
          <Icon className={`h-10 w-10 mx-auto mb-4 ${primary ? 'text-primary' : 'text-muted-foreground'}`} />
          <h3 className="font-semibold mb-2">{title}</h3>
          <p className="text-sm text-muted-foreground">{description}</p>
        </CardContent>
      </Card>
    </Link>
  );
}

function LandingSkeleton() {
  return (
    <div className="min-h-screen flex flex-col">
      <div className="container mx-auto px-4 py-8">
        <div className="text-center mb-8">
          <Skeleton className="h-16 w-32 mx-auto mb-4" />
          <Skeleton className="h-8 w-64 mx-auto" />
        </div>
        <div className="grid gap-6 md:grid-cols-3 max-w-4xl mx-auto">
          <Skeleton className="h-40" />
          <Skeleton className="h-40" />
          <Skeleton className="h-40" />
        </div>
      </div>
    </div>
  );
}
```
