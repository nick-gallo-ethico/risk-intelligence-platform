"use client";

import { useEffect, useState, useCallback } from "react";
import { useSearchParams } from "next/navigation";
import { useTranslation } from "react-i18next";
import Cookies from "js-cookie";

/**
 * Supported languages for the ethics portal.
 * Add new languages here when translation files are available.
 */
export const SUPPORTED_LANGUAGES = [
  "en", // English
  "es", // Spanish
  "fr", // French
  "de", // German
  "pt", // Portuguese
  "zh", // Chinese
  "ar", // Arabic (RTL)
  "he", // Hebrew (RTL)
  "ja", // Japanese
  "ko", // Korean
];

/**
 * RTL (right-to-left) languages.
 */
const RTL_LANGUAGES = ["ar", "he"];

/**
 * Cookie name for storing language preference.
 */
const LANGUAGE_COOKIE = "ethics-language";

interface LanguageDetectionOptions {
  /** URL parameter name for language override */
  urlParamName?: string;
  /** Language from HRIS employee profile */
  hrisLanguage?: string;
  /** Organization's default language */
  orgDefaultLanguage?: string;
}

interface UseLanguageDetectionResult {
  /** Currently detected/active language */
  detectedLanguage: string | null;
  /** Function to manually set language */
  setLanguage: (lang: string) => void;
  /** List of supported languages */
  supportedLanguages: typeof SUPPORTED_LANGUAGES;
  /** Whether current language is RTL */
  isRtl: boolean;
  /** Whether detection is complete */
  isDetecting: boolean;
}

/**
 * Hook for auto-detecting user language with priority chain:
 * 1. URL parameter (?lang=es)
 * 2. Cookie (previously saved preference)
 * 3. Browser Accept-Language header
 * 4. HRIS employee profile language
 * 5. Organization default language
 *
 * Usage:
 * ```tsx
 * const { detectedLanguage, setLanguage, isRtl } = useLanguageDetection({
 *   orgDefaultLanguage: 'en',
 *   hrisLanguage: employee?.language,
 * });
 * ```
 */
export function useLanguageDetection(
  options: LanguageDetectionOptions = {},
): UseLanguageDetectionResult {
  const {
    urlParamName = "lang",
    hrisLanguage,
    orgDefaultLanguage = "en",
  } = options;

  const searchParams = useSearchParams();
  const { i18n } = useTranslation();
  const [detectedLanguage, setDetectedLanguage] = useState<string | null>(null);
  const [isDetecting, setIsDetecting] = useState(true);

  // Run detection on mount and when dependencies change
  useEffect(() => {
    const detectLanguage = (): string => {
      // 1. URL parameter (highest priority)
      const urlLang = searchParams?.get(urlParamName);
      if (urlLang && SUPPORTED_LANGUAGES.includes(urlLang)) {
        return urlLang;
      }

      // 2. Cookie (user's saved preference)
      const cookieLang = Cookies.get(LANGUAGE_COOKIE);
      if (cookieLang && SUPPORTED_LANGUAGES.includes(cookieLang)) {
        return cookieLang;
      }

      // 3. Browser Accept-Language
      if (typeof navigator !== "undefined") {
        // Get primary language from navigator.languages or navigator.language
        const browserLanguages = navigator.languages || [navigator.language];
        for (const lang of browserLanguages) {
          const shortLang = lang.split("-")[0].toLowerCase();
          if (SUPPORTED_LANGUAGES.includes(shortLang)) {
            return shortLang;
          }
        }
      }

      // 4. HRIS employee profile language
      if (hrisLanguage && SUPPORTED_LANGUAGES.includes(hrisLanguage)) {
        return hrisLanguage;
      }

      // 5. Organization default
      if (SUPPORTED_LANGUAGES.includes(orgDefaultLanguage)) {
        return orgDefaultLanguage;
      }

      // Fallback to English
      return "en";
    };

    const language = detectLanguage();

    // Apply language if different from current
    if (language !== i18n.language) {
      i18n.changeLanguage(language);
    }

    // Persist preference in cookie (1 year expiry)
    Cookies.set(LANGUAGE_COOKIE, language, { expires: 365, sameSite: "lax" });

    // Set document direction for RTL languages
    const isRtl = RTL_LANGUAGES.includes(language);
    document.documentElement.dir = isRtl ? "rtl" : "ltr";
    document.documentElement.lang = language;

    setDetectedLanguage(language);
    setIsDetecting(false);
  }, [searchParams, hrisLanguage, orgDefaultLanguage, i18n, urlParamName]);

  // Manual language setter
  const setLanguage = useCallback(
    (lang: string) => {
      if (!SUPPORTED_LANGUAGES.includes(lang)) {
        console.warn(`Language "${lang}" is not supported`);
        return;
      }

      i18n.changeLanguage(lang);
      Cookies.set(LANGUAGE_COOKIE, lang, { expires: 365, sameSite: "lax" });

      const isRtl = RTL_LANGUAGES.includes(lang);
      document.documentElement.dir = isRtl ? "rtl" : "ltr";
      document.documentElement.lang = lang;

      setDetectedLanguage(lang);
    },
    [i18n],
  );

  return {
    detectedLanguage,
    setLanguage,
    supportedLanguages: SUPPORTED_LANGUAGES,
    isRtl: detectedLanguage ? RTL_LANGUAGES.includes(detectedLanguage) : false,
    isDetecting,
  };
}
````

'use client';

import { Suspense } from 'react';
import { useParams } from 'next/navigation';
import { useQuery } from '@tanstack/react-query';
import { apiClient } from '@/lib/api';
import { useLanguageDetection } from '@/hooks/use-language-detection';
import { I18nextProvider } from 'react-i18next';
import i18n from '@/lib/i18n';
import { Skeleton } from '@/components/ui/skeleton';

interface EthicsPortalConfig {
  organizationId: string;
  organizationName: string;
  defaultLanguage: string;
  supportedLanguages: string[];
  // ... other fields
}

function EthicsPortalLayoutContent({ children }: { children: React.ReactNode }) {
  const params = useParams();
  const tenantSlug = params?.tenant as string;

  // Fetch portal config to get org default language
  const { data: config, isLoading } = useQuery({
    queryKey: ['ethics', 'config', tenantSlug],
    queryFn: () => apiClient.get<EthicsPortalConfig>(`/ethics/config/${tenantSlug}`),
    enabled: !!tenantSlug,
    staleTime: 5 * 60 * 1000, // 5 minutes
  });

  // Detect and apply language
  const { isDetecting, isRtl } = useLanguageDetection({
    orgDefaultLanguage: config?.defaultLanguage || 'en',
  });

  // Show loading while detecting language to prevent flash
  if (isLoading || isDetecting) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <Skeleton className="h-12 w-48" />
      </div>
    );
  }

  return (
    <I18nextProvider i18n={i18n}>
      <div className={isRtl ? 'rtl' : 'ltr'}>
        {children}
      </div>
    </I18nextProvider>
  );
}

export default function EthicsPortalLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <Suspense
      fallback={
        <div className="min-h-screen flex items-center justify-center">
          <Skeleton className="h-12 w-48" />
        </div>
      }
    >
      <EthicsPortalLayoutContent>{children}</EthicsPortalLayoutContent>
    </Suspense>
  );
}
```

Key points:

- Load org config to get default language before detection
- Show skeleton during detection to prevent language flash
- Apply RTL class wrapper when needed
- Use Suspense boundary for searchParams (Next.js requirement)
