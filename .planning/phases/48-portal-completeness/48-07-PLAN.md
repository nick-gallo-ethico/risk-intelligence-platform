```typescript
// Request DTO for getting AI suggestions
export class GetAiSuggestionsDto {
  @IsUUID("4")
  categoryId: string;

  @IsString()
  @MaxLength(10000)
  narrativeContent: string;

  @IsOptional()
  @IsArray()
  @IsString({ each: true })
  answeredQuestionIds?: string[];
}

// Response DTO for a single suggested question
export class SuggestedQuestionDto {
  id: string;
  question: string;
  reasoning: string;
  priority: "high" | "medium" | "low";
  source: "ai" | "category_template";
  categoryId?: string;
}

// Response DTO for AI suggestions endpoint
export class AiSuggestionsResponseDto {
  suggestions: SuggestedQuestionDto[];
  generatedAt: Date;
  fromCache: boolean;
}

// DTO for category-specific intake question
export class CategoryQuestionDto {
  id: string;
  categoryId: string;
  questionText: string;
  questionType: "text" | "select" | "multiselect" | "boolean" | "date";
  options?: string[]; // For select/multiselect
  isRequired: boolean;
  order: number;
  helpText?: string;
}

// Response for category questions
export class CategoryQuestionsResponseDto {
  categoryId: string;
  categoryName: string;
  questions: CategoryQuestionDto[];
}
```

```typescript
import { Injectable, Logger, NotFoundException } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";
import { Cache } from "cache-manager";
import { Inject } from "@nestjs/common";
import { CACHE_MANAGER } from "@nestjs/cache-manager";
import {
  CategoryQuestionDto,
  CategoryQuestionsResponseDto,
} from "./dto/ai-suggestions.dto";

@Injectable()
export class CategoryQuestionsService {
  private readonly logger = new Logger(CategoryQuestionsService.name);

  constructor(
    private readonly prisma: PrismaService,
    @Inject(CACHE_MANAGER) private readonly cacheManager: Cache,
  ) {}

  /**
   * Get intake questions for a specific category.
   * Questions are defined per-category and loaded dynamically when operator selects category.
   */
  async getQuestionsForCategory(
    categoryId: string,
    organizationId: string,
  ): Promise<CategoryQuestionsResponseDto> {
    const cacheKey = `cat-questions:${organizationId}:${categoryId}`;

    // Check cache first
    const cached =
      await this.cacheManager.get<CategoryQuestionsResponseDto>(cacheKey);
    if (cached) {
      return cached;
    }

    // Get category with its questions
    const category = await this.prisma.category.findFirst({
      where: {
        id: categoryId,
        organizationId,
      },
      include: {
        intakeQuestions: {
          where: { isActive: true },
          orderBy: { order: "asc" },
        },
      },
    });

    if (!category) {
      throw new NotFoundException(`Category not found: ${categoryId}`);
    }

    const response: CategoryQuestionsResponseDto = {
      categoryId: category.id,
      categoryName: category.name,
      questions: category.intakeQuestions.map((q) => ({
        id: q.id,
        categoryId: q.categoryId,
        questionText: q.questionText,
        questionType: q.questionType as CategoryQuestionDto["questionType"],
        options: q.options as string[] | undefined,
        isRequired: q.isRequired,
        order: q.order,
        helpText: q.helpText ?? undefined,
      })),
    };

    // Cache for 5 minutes
    await this.cacheManager.set(cacheKey, response, 300);

    return response;
  }

  /**
   * Get all categories with their question counts for UI display.
   */
  async getCategoriesWithQuestionCounts(
    organizationId: string,
  ): Promise<
    Array<{ categoryId: string; categoryName: string; questionCount: number }>
  > {
    const categories = await this.prisma.category.findMany({
      where: { organizationId, isActive: true },
      select: {
        id: true,
        name: true,
        _count: {
          select: { intakeQuestions: { where: { isActive: true } } },
        },
      },
      orderBy: { name: "asc" },
    });

    return categories.map((c) => ({
      categoryId: c.id,
      categoryName: c.name,
      questionCount: c._count.intakeQuestions,
    }));
  }
}
```

```typescript
import { Injectable, Logger } from '@nestjs/common';
import { Inject } from '@nestjs/common';
import { Cache } from 'cache-manager';
import { CACHE_MANAGER } from '@nestjs/cache-manager';
import * as crypto from 'crypto';
import { AiClientService } from '../../ai/services/ai-client.service';
import { CategoryQuestionsService } from './category-questions.service';
import {
  GetAiSuggestionsDto,
  SuggestedQuestionDto,
  AiSuggestionsResponseDto,
} from './dto/ai-suggestions.dto';

const OPERATOR_SUGGESTION_SYSTEM_PROMPT = `You are an expert hotline operator assistant. Based on the report category and narrative, suggest the most relevant follow-up questions to ask the reporter.

For each suggestion, provide:

1. The question text
2. Brief reasoning why this question is important
3. Priority (high/medium/low)

Focus on:

- Clarifying unclear details
- Gathering specific dates, names, or locations
- Understanding the impact or scope of the issue
- Identifying witnesses or evidence
- Determining urgency or safety concerns

Return suggestions as JSON array.`;

@Injectable()
export class AiSuggestionsService {
private readonly logger = new Logger(AiSuggestionsService.name);

constructor(
private readonly aiClientService: AiClientService,
private readonly categoryQuestionsService: CategoryQuestionsService,
@Inject(CACHE_MANAGER) private readonly cacheManager: Cache,
) {}

/\*\*

- Get AI-suggested follow-up questions based on category and narrative.
- Combines category-specific template questions with AI-generated suggestions.
  \*/
  async getSuggestedQuestions(
  dto: GetAiSuggestionsDto,
  organizationId: string,
  ): Promise<AiSuggestionsResponseDto> {
  const { categoryId, narrativeContent, answeredQuestionIds = [] } = dto;

  // Hash content for cache key (avoid storing PII in cache keys)
  const contentHash = this.hashContent(narrativeContent);
  const cacheKey = `ai-suggestions:${organizationId}:${categoryId}:${contentHash}`;

  // Check cache
  const cached = await this.cacheManager.get<AiSuggestionsResponseDto>(cacheKey);
  if (cached) {
  this.logger.debug(`Cache hit for suggestions: ${cacheKey}`);
  return { ...cached, fromCache: true };
  }

  // Get category-specific base questions
  let categoryQuestions: SuggestedQuestionDto[] = [];
  try {
  const catResponse = await this.categoryQuestionsService.getQuestionsForCategory(
  categoryId,
  organizationId,
  );
  categoryQuestions = catResponse.questions
  .filter(q => !answeredQuestionIds.includes(q.id))
  .map(q => ({
  id: q.id,
  question: q.questionText,
  reasoning: 'Standard question for this category',
  priority: q.isRequired ? 'high' as const : 'medium' as const,
  source: 'category_template' as const,
  categoryId,
  }));
  } catch (error) {
  this.logger.warn(`Failed to load category questions: ${error.message}`);
  }

  // Generate AI suggestions (only if narrative is substantial)
  let aiSuggestions: SuggestedQuestionDto[] = [];
  if (narrativeContent.length > 50) {
  try {
  aiSuggestions = await this.generateAiSuggestions(categoryId, narrativeContent);
  } catch (error) {
  this.logger.error(`AI suggestion generation failed: ${error.message}`);
  // Continue with category questions only
  }
  }

  // Combine and deduplicate
  const allSuggestions = [...categoryQuestions, ...aiSuggestions];

  // Sort by priority (high first) and limit
  const sortedSuggestions = allSuggestions
  .sort((a, b) => {
  const priorityOrder = { high: 0, medium: 1, low: 2 };
  return priorityOrder[a.priority] - priorityOrder[b.priority];
  })
  .slice(0, 8); // Max 8 suggestions

  const response: AiSuggestionsResponseDto = {
  suggestions: sortedSuggestions,
  generatedAt: new Date(),
  fromCache: false,
  };

  // Cache for 5 minutes
  await this.cacheManager.set(cacheKey, response, 300);

  return response;

}

private async generateAiSuggestions(
categoryId: string,
narrative: string,
): Promise<SuggestedQuestionDto[]> {
const prompt = `Category: ${categoryId}

Report narrative:
"${narrative.substring(0, 2000)}"

Suggest 3-5 follow-up questions. Return as JSON array with fields: question, reasoning, priority.`;

    const response = await this.aiClientService.createMessage({
      model: 'claude-sonnet-4-5',
      maxTokens: 1024,
      system: OPERATOR_SUGGESTION_SYSTEM_PROMPT,
      messages: [{ role: 'user', content: prompt }],
    });

    // Parse AI response
    try {
      const content = response.content[0];
      if (content.type !== 'text') return [];

      // Extract JSON from response (may be wrapped in markdown)
      const jsonMatch = content.text.match(/\[[\s\S]*\]/);
      if (!jsonMatch) return [];

      const parsed = JSON.parse(jsonMatch[0]) as Array<{
        question: string;
        reasoning: string;
        priority: string;
      }>;

      return parsed.map((item, index) => ({
        id: `ai-${Date.now()}-${index}`,
        question: item.question,
        reasoning: item.reasoning,
        priority: this.normalizePriority(item.priority),
        source: 'ai' as const,
      }));
    } catch (error) {
      this.logger.error(`Failed to parse AI suggestions: ${error.message}`);
      return [];
    }

}

private hashContent(content: string): string {
return crypto.createHash('sha256').update(content).digest('hex').substring(0, 16);
}

private normalizePriority(priority: string): 'high' | 'medium' | 'low' {
const normalized = priority?.toLowerCase();
if (normalized === 'high') return 'high';
if (normalized === 'low') return 'low';
return 'medium';
}
}
````
