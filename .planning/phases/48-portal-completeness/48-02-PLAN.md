2. **Add sendBulkReminders() to ManagerTeamService:**
```typescript
async sendBulkReminders(
  managerId: string,
  organizationId: string,
  employeeIds: string[],
  customMessage?: string,
): Promise<BulkReminderResultDto> {
  // Validate manager has authority over these employees
  const directReports = await this.getTeamMembers(managerId, organizationId);
  const authorizedIds = new Set(directReports.map(e => e.employeeId));

  const validIds = employeeIds.filter(id => authorizedIds.has(id));
  const invalidCount = employeeIds.length - validIds.length;

  // Use EventEmitter to send reminders (NotificationsModule handles delivery)
  let sent = 0;
  for (const employeeId of validIds) {
    this.eventEmitter.emit('employee.compliance_reminder', {
      organizationId,
      employeeId,
      sentByManagerId: managerId,
      customMessage,
    });
    sent++;
  }

  return {
    sent,
    failed: invalidCount,
    errors: invalidCount > 0 ? ['Some employees are not in your team'] : undefined,
  };
}
````

  const person = await this.getPersonForUser(user, organizationId);
  if (!person) {
    throw new NotFoundException('Person profile not found');
  }
  return this.managerTeamService.sendBulkReminders(
    person.id,
    organizationId,
    dto.employeeIds,
    dto.customMessage,
  );
}

2. **Add submitProxyReport() to ManagerTeamService:**
```typescript
async submitProxyReport(
  managerId: string,
  organizationId: string,
  dto: ProxyReportDto,
): Promise<{ riuId: string }> {
  // Validate manager has authority over this employee
  const directReports = await this.getTeamMembers(managerId, organizationId);
  const isAuthorized = directReports.some(e => e.employeeId === dto.employeeId);
  if (!isAuthorized) {
    throw new ForbiddenException('Employee is not a direct report');
  }

  // Create RIU with proxy attribution
  const riu = await this.prisma.riskIntelligenceUnit.create({
    data: {
      organizationId,
      type: 'PROXY_REPORT',
      sourceChannel: 'PROXY',
      details: dto.details,
      categoryId: dto.categoryId,
      severity: dto.severity || 'MEDIUM',
      reporterType: 'IDENTIFIED',
      // Attribution fields
      submittedById: managerId,
      submittedOnBehalfOfId: dto.employeeId,
      status: 'PENDING_REVIEW',
    },
  });

  // Log activity with attribution
  await this.activityService.log({
    entityType: 'RIU',
    entityId: riu.id,
    action: 'proxy_report_submitted',
    actionDescription: `Manager submitted proxy report on behalf of employee`,
    actorUserId: managerId,
    organizationId,
    metadata: { onBehalfOfEmployeeId: dto.employeeId },
  });

  return { riuId: riu.id };
}
```
