```typescript
@Get('manager/team-compliance')
@UseGuards(JwtAuthGuard)
@ApiOperation({ summary: 'Get team compliance dashboard for managers' })
@ApiResponse({ type: TeamComplianceResponseDto })
async getTeamCompliance(
  @CurrentUser() user: User,
  @TenantId() organizationId: string,
): Promise<TeamComplianceResponseDto> {
  // Get the user's Person record to find manager relationship
  const person = await this.getPersonForUser(user, organizationId);
  if (!person) {
    throw new NotFoundException('Person profile not found');
  }

  return this.managerTeamService.getTeamCompliance(person.id, organizationId);
}
```
