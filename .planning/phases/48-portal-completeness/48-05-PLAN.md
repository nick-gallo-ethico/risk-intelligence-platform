Create DisclosureExportService that generates Excel files:

```typescript
import { Injectable, Logger } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";
import {
  ExcelExportService,
  ColumnDefinition,
} from "../../analytics/exports/excel-export.service";

@Injectable()
export class DisclosureExportService {
  private readonly logger = new Logger(DisclosureExportService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly excelExport: ExcelExportService,
  ) {}

  /**
   * Export disclosure history for an employee to Excel buffer.
   */
  async exportDisclosureHistory(
    userId: string,
    organizationId: string,
  ): Promise<Buffer> {
    // Get user email to find their disclosures
    const user = await this.prisma.user.findUnique({
      where: { id: userId },
      select: { email: true },
    });

    if (!user) {
      throw new Error("User not found");
    }

    // Find all disclosures submitted by this user
    const disclosures = await this.prisma.disclosure.findMany({
      where: {
        organizationId,
        submittedById: userId,
      },
      include: {
        disclosureType: { select: { name: true } },
        category: { select: { name: true } },
        conditions: {
          select: {
            conditionText: true,
            completedAt: true,
          },
        },
      },
      orderBy: { submittedAt: "desc" },
      take: 1000, // Reasonable limit
    });

    // Define Excel columns
    const columns: ColumnDefinition[] = [
      { key: "referenceNumber", label: "Reference #", type: "string" },
      { key: "disclosureType", label: "Type", type: "string" },
      { key: "category", label: "Category", type: "string" },
      { key: "status", label: "Status", type: "string" },
      { key: "submittedAt", label: "Submitted Date", type: "date" },
      { key: "relatedCompany", label: "Related Company", type: "string" },
      { key: "relatedPersonName", label: "Related Person", type: "string" },
      { key: "relationship", label: "Relationship", type: "string" },
      { key: "disclosureValue", label: "Value", type: "currency" },
      { key: "conflictDetected", label: "Conflict Detected", type: "boolean" },
      { key: "conditionsCount", label: "Conditions", type: "number" },
      {
        key: "conditionsComplete",
        label: "Conditions Complete",
        type: "number",
      },
    ];

    // Map disclosures to rows
    const rows = disclosures.map((d) => ({
      referenceNumber: d.referenceNumber ?? "",
      disclosureType: d.disclosureType?.name ?? "",
      category: d.category?.name ?? "",
      status: d.status,
      submittedAt: d.submittedAt,
      relatedCompany: d.relatedCompany ?? "",
      relatedPersonName: d.relatedPersonName ?? "",
      relationship: d.relationship ?? "",
      disclosureValue: d.disclosureValue ?? 0,
      conflictDetected: d.conflictDetected ?? false,
      conditionsCount: d.conditions.length,
      conditionsComplete: d.conditions.filter((c) => c.completedAt).length,
    }));

    // Generate Excel buffer
    return this.excelExport.generateBuffer(rows, columns, {
      sheetName: "Disclosure History",
      includeMetadata: true,
      metadata: {
        exportedAt: new Date(),
        exportedBy: user.email,
        recordCount: rows.length,
      },
    });
  }
}
```

```typescript
import { Get, Res, StreamableFile } from '@nestjs/common';
import { Response } from 'express';

// Inject DisclosureExportService in constructor

@Get('disclosures/export')
@UseGuards(JwtAuthGuard)
@ApiOperation({ summary: 'Export disclosure history to Excel' })
@ApiProduces('application/vnd.openxmlformats-officedocument.spreadsheetml.sheet')
async exportDisclosures(
  @CurrentUser() user: User,
  @TenantId() organizationId: string,
  @Res({ passthrough: true }) res: Response,
): Promise<StreamableFile> {
  const buffer = await this.disclosureExportService.exportDisclosureHistory(
    user.id,
    organizationId,
  );

  const filename = `disclosure-history-${new Date().toISOString().split('T')[0]}.xlsx`;

  res.set({
    'Content-Type': 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    'Content-Disposition': `attachment; filename="${filename}"`,
    'Content-Length': buffer.length,
  });

  return new StreamableFile(buffer);
}
```

Create a button component that triggers the export download:

```typescript
'use client';

import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { useToast } from '@/components/ui/use-toast';
import { Download, Loader2 } from 'lucide-react';
import { getAuthToken } from '@/lib/auth';

interface DisclosureExportButtonProps {
  className?: string;
}

export function DisclosureExportButton({ className }: DisclosureExportButtonProps) {
  const [isExporting, setIsExporting] = useState(false);
  const { toast } = useToast();

  const handleExport = async () => {
    setIsExporting(true);

    try {
      const token = getAuthToken();
      const response = await fetch('/api/employee/disclosures/export', {
        headers: {
          Authorization: `Bearer ${token}`,
        },
      });

      if (!response.ok) {
        throw new Error('Export failed');
      }

      // Get filename from Content-Disposition header
      const contentDisposition = response.headers.get('Content-Disposition');
      const filenameMatch = contentDisposition?.match(/filename="(.+)"/);
      const filename = filenameMatch?.[1] ?? 'disclosure-history.xlsx';

      // Download the file
      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      window.URL.revokeObjectURL(url);
      document.body.removeChild(a);

      toast({
        title: 'Export Complete',
        description: 'Your disclosure history has been downloaded.',
      });
    } catch (error) {
      toast({
        title: 'Export Failed',
        description: 'Unable to export your disclosure history. Please try again.',
        variant: 'destructive',
      });
    } finally {
      setIsExporting(false);
    }
  };

  return (
    <Button
      variant="outline"
      onClick={handleExport}
      disabled={isExporting}
      className={className}
    >
      {isExporting ? (
        <Loader2 className="h-4 w-4 mr-2 animate-spin" />
      ) : (
        <Download className="h-4 w-4 mr-2" />
      )}
      Export History
    </Button>
  );
}
```

This component:

- Shows loading state during export
- Handles file download via blob/URL
- Shows toast on success or failure
- Can be placed anywhere in employee portal (e.g., My Reports tab header)
