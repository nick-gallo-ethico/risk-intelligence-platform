```typescript
'use client';

import { useState, useEffect, useCallback } from 'react';
import { useMutation } from '@tanstack/react-query';
import { apiClient } from '@/lib/api';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { Skeleton } from '@/components/ui/skeleton';
import { Sparkles, Copy, Check, RefreshCw, ChevronDown, ChevronUp } from 'lucide-react';
import { useDebounce } from '@/hooks/use-debounce';

interface SuggestedQuestion {
  id: string;
  question: string;
  reasoning: string;
  priority: 'high' | 'medium' | 'low';
  source: 'ai' | 'category_template';
}

interface AiSuggestionsResponse {
  suggestions: SuggestedQuestion[];
  generatedAt: string;
  fromCache: boolean;
}

interface AiSuggestedQuestionsProps {
  categoryId: string | null;
  narrativeContent: string;
  answeredQuestionIds?: string[];
  onQuestionClick?: (question: string) => void;
}

export function AiSuggestedQuestions({
  categoryId,
  narrativeContent,
  answeredQuestionIds = [],
  onQuestionClick,
}: AiSuggestedQuestionsProps) {
  const [copiedId, setCopiedId] = useState<string | null>(null);
  const [expanded, setExpanded] = useState(true);

  // Debounce narrative content to avoid excessive API calls
  const debouncedNarrative = useDebounce(narrativeContent, 1000);

  // Fetch suggestions
  const {
    mutate: fetchSuggestions,
    data: suggestions,
    isPending,
    reset,
  } = useMutation({
    mutationFn: async (params: {
      categoryId: string;
      narrativeContent: string;
      answeredQuestionIds: string[];
    }) => {
      return apiClient.post<AiSuggestionsResponse>('/operator/intake/ai-suggestions', params);
    },
  });

  // Fetch when category or narrative changes (debounced)
  useEffect(() => {
    if (categoryId && debouncedNarrative.length > 50) {
      fetchSuggestions({
        categoryId,
        narrativeContent: debouncedNarrative,
        answeredQuestionIds,
      });
    } else {
      reset();
    }
  }, [categoryId, debouncedNarrative, fetchSuggestions, reset]);

  const handleCopy = useCallback((id: string, question: string) => {
    navigator.clipboard.writeText(question);
    setCopiedId(id);
    setTimeout(() => setCopiedId(null), 2000);
  }, []);

  const handleRefresh = useCallback(() => {
    if (categoryId && narrativeContent.length > 50) {
      fetchSuggestions({
        categoryId,
        narrativeContent,
        answeredQuestionIds,
      });
    }
  }, [categoryId, narrativeContent, answeredQuestionIds, fetchSuggestions]);

  // Don't show if no category selected
  if (!categoryId) {
    return null;
  }

  const priorityColors = {
    high: 'bg-red-100 text-red-800 dark:bg-red-950 dark:text-red-200',
    medium: 'bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-200',
    low: 'bg-blue-100 text-blue-800 dark:bg-blue-950 dark:text-blue-200',
  };

  return (
    <Card className="border-dashed">
      <CardHeader className="pb-2 cursor-pointer" onClick={() => setExpanded(!expanded)}>
        <div className="flex items-center justify-between">
          <CardTitle className="text-sm flex items-center gap-2">
            <Sparkles className="h-4 w-4 text-purple-500" />
            Suggested Follow-Up Questions
            {suggestions?.fromCache && (
              <Badge variant="outline" className="text-xs">cached</Badge>
            )}
          </CardTitle>
          <div className="flex items-center gap-2">
            <Button
              variant="ghost"
              size="icon"
              className="h-6 w-6"
              onClick={(e) => {
                e.stopPropagation();
                handleRefresh();
              }}
              disabled={isPending || narrativeContent.length < 50}
            >
              <RefreshCw className={`h-3 w-3 ${isPending ? 'animate-spin' : ''}`} />
            </Button>
            {expanded ? (
              <ChevronUp className="h-4 w-4" />
            ) : (
              <ChevronDown className="h-4 w-4" />
            )}
          </div>
        </div>
      </CardHeader>

      {expanded && (
        <CardContent className="pt-0">
          {isPending && (
            <div className="space-y-2">
              <Skeleton className="h-12" />
              <Skeleton className="h-12" />
              <Skeleton className="h-12" />
            </div>
          )}

          {!isPending && narrativeContent.length < 50 && (
            <p className="text-sm text-muted-foreground text-center py-4">
              Enter more narrative content to see AI suggestions
            </p>
          )}

          {!isPending && suggestions?.suggestions && (
            <div className="space-y-2">
              {suggestions.suggestions.map((suggestion) => (
                <SuggestionItem
                  key={suggestion.id}
                  suggestion={suggestion}
                  isCopied={copiedId === suggestion.id}
                  priorityColors={priorityColors}
                  onCopy={() => handleCopy(suggestion.id, suggestion.question)}
                  onClick={() => onQuestionClick?.(suggestion.question)}
                />
              ))}

              {suggestions.suggestions.length === 0 && (
                <p className="text-sm text-muted-foreground text-center py-4">
                  No additional questions suggested
                </p>
              )}
            </div>
          )}
        </CardContent>
      )}
    </Card>
  );
}

function SuggestionItem({
  suggestion,
  isCopied,
  priorityColors,
  onCopy,
  onClick,
}: {
  suggestion: SuggestedQuestion;
  isCopied: boolean;
  priorityColors: Record<string, string>;
  onCopy: () => void;
  onClick: () => void;
}) {
  return (
    <div
      className="p-3 rounded-lg border bg-muted/30 hover:bg-muted/50 transition-colors cursor-pointer"
      onClick={onClick}
    >
      <div className="flex items-start gap-2">
        <div className="flex-1">
          <p className="text-sm font-medium">{suggestion.question}</p>
          <p className="text-xs text-muted-foreground mt-1">{suggestion.reasoning}</p>
        </div>
        <div className="flex items-center gap-1 flex-shrink-0">
          <Badge
            variant="secondary"
            className={`text-xs ${priorityColors[suggestion.priority]}`}
          >
            {suggestion.priority}
          </Badge>
          {suggestion.source === 'ai' && (
            <Badge variant="outline" className="text-xs">AI</Badge>
          )}
          <Button
            variant="ghost"
            size="icon"
            className="h-6 w-6"
            onClick={(e) => {
              e.stopPropagation();
              onCopy();
            }}
          >
            {isCopied ? (
              <Check className="h-3 w-3 text-green-500" />
            ) : (
              <Copy className="h-3 w-3" />
            )}
          </Button>
        </div>
      </div>
    </div>
  );
}
```

```typescript
'use client';

import { useQuery } from '@tanstack/react-query';
import { apiClient } from '@/lib/api';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Input } from '@/components/ui/input';
import { Textarea } from '@/components/ui/textarea';
import { Checkbox } from '@/components/ui/checkbox';
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select';
import { Label } from '@/components/ui/label';
import { Skeleton } from '@/components/ui/skeleton';
import { Badge } from '@/components/ui/badge';
import { HelpCircle, ListChecks } from 'lucide-react';
import {
  Tooltip,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from '@/components/ui/tooltip';

interface CategoryQuestion {
  id: string;
  questionText: string;
  questionType: 'text' | 'select' | 'multiselect' | 'boolean' | 'date';
  options?: string[];
  isRequired: boolean;
  order: number;
  helpText?: string;
}

interface CategoryQuestionsResponse {
  categoryId: string;
  categoryName: string;
  questions: CategoryQuestion[];
}

interface CategoryQuestionsProps {
  categoryId: string | null;
  answers: Record<string, string | string[] | boolean>;
  onAnswerChange: (questionId: string, value: string | string[] | boolean) => void;
}

export function CategoryQuestions({
  categoryId,
  answers,
  onAnswerChange,
}: CategoryQuestionsProps) {
  const { data, isLoading } = useQuery({
    queryKey: ['operator', 'category-questions', categoryId],
    queryFn: () =>
      apiClient.get<CategoryQuestionsResponse>(`/operator/intake/categories/${categoryId}/questions`),
    enabled: !!categoryId,
  });

  if (!categoryId) {
    return null;
  }

  if (isLoading) {
    return (
      <Card>
        <CardHeader>
          <Skeleton className="h-5 w-48" />
        </CardHeader>
        <CardContent className="space-y-4">
          <Skeleton className="h-10" />
          <Skeleton className="h-10" />
          <Skeleton className="h-10" />
        </CardContent>
      </Card>
    );
  }

  if (!data?.questions || data.questions.length === 0) {
    return null;
  }

  return (
    <Card>
      <CardHeader className="pb-3">
        <CardTitle className="text-sm flex items-center gap-2">
          <ListChecks className="h-4 w-4" />
          {data.categoryName} - Intake Questions
          <Badge variant="outline" className="text-xs">
            {data.questions.length} question{data.questions.length !== 1 ? 's' : ''}
          </Badge>
        </CardTitle>
      </CardHeader>
      <CardContent className="space-y-4">
        {data.questions.map((question) => (
          <QuestionField
            key={question.id}
            question={question}
            value={answers[question.id]}
            onChange={(value) => onAnswerChange(question.id, value)}
          />
        ))}
      </CardContent>
    </Card>
  );
}

function QuestionField({
  question,
  value,
  onChange,
}: {
  question: CategoryQuestion;
  value: string | string[] | boolean | undefined;
  onChange: (value: string | string[] | boolean) => void;
}) {
  const labelId = `question-${question.id}`;

  return (
    <div className="space-y-2">
      <div className="flex items-center gap-2">
        <Label htmlFor={labelId} className="text-sm">
          {question.questionText}
          {question.isRequired && <span className="text-red-500 ml-1">*</span>}
        </Label>
        {question.helpText && (
          <TooltipProvider>
            <Tooltip>
              <TooltipTrigger asChild>
                <HelpCircle className="h-3 w-3 text-muted-foreground cursor-help" />
              </TooltipTrigger>
              <TooltipContent>
                <p className="max-w-xs">{question.helpText}</p>
              </TooltipContent>
            </Tooltip>
          </TooltipProvider>
        )}
      </div>

      {question.questionType === 'text' && (
        <Textarea
          id={labelId}
          value={(value as string) || ''}
          onChange={(e) => onChange(e.target.value)}
          placeholder="Enter response..."
          rows={2}
        />
      )}

      {question.questionType === 'select' && (
        <Select value={(value as string) || ''} onValueChange={onChange}>
          <SelectTrigger id={labelId}>
            <SelectValue placeholder="Select an option" />
          </SelectTrigger>
          <SelectContent>
            {question.options?.map((option) => (
              <SelectItem key={option} value={option}>
                {option}
              </SelectItem>
            ))}
          </SelectContent>
        </Select>
      )}

      {question.questionType === 'multiselect' && (
        <div className="space-y-2">
          {question.options?.map((option) => {
            const currentValue = (value as string[]) || [];
            const isChecked = currentValue.includes(option);
            return (
              <div key={option} className="flex items-center gap-2">
                <Checkbox
                  id={`${labelId}-${option}`}
                  checked={isChecked}
                  onCheckedChange={(checked) => {
                    if (checked) {
                      onChange([...currentValue, option]);
                    } else {
                      onChange(currentValue.filter((v) => v !== option));
                    }
                  }}
                />
                <Label htmlFor={`${labelId}-${option}`} className="text-sm font-normal">
                  {option}
                </Label>
              </div>
            );
          })}
        </div>
      )}

      {question.questionType === 'boolean' && (
        <div className="flex items-center gap-2">
          <Checkbox
            id={labelId}
            checked={(value as boolean) || false}
            onCheckedChange={onChange}
          />
          <Label htmlFor={labelId} className="text-sm font-normal">
            Yes
          </Label>
        </div>
      )}

      {question.questionType === 'date' && (
        <Input
          id={labelId}
          type="date"
          value={(value as string) || ''}
          onChange={(e) => onChange(e.target.value)}
        />
      )}
    </div>
  );
}
```

```typescript
'use client';

import { useState } from 'react';
import { AiSuggestedQuestions } from './ai-suggested-questions';
import { CategoryQuestions } from './category-questions';
import { ScrollArea } from '@/components/ui/scroll-area';
import { Separator } from '@/components/ui/separator';

interface IntakeSidebarProps {
  categoryId: string | null;
  narrativeContent: string;
  onInsertQuestion?: (question: string) => void;
}

export function IntakeSidebar({
  categoryId,
  narrativeContent,
  onInsertQuestion,
}: IntakeSidebarProps) {
  const [categoryAnswers, setCategoryAnswers] = useState<Record<string, string | string[] | boolean>>({});
  const [answeredQuestionIds, setAnsweredQuestionIds] = useState<string[]>([]);

  const handleCategoryAnswer = (questionId: string, value: string | string[] | boolean) => {
    setCategoryAnswers((prev) => ({ ...prev, [questionId]: value }));
    // Track as answered for AI filtering
    if (value && !answeredQuestionIds.includes(questionId)) {
      setAnsweredQuestionIds((prev) => [...prev, questionId]);
    }
  };

  return (
    <ScrollArea className="h-full">
      <div className="p-4 space-y-4">
        {/* AI Suggestions Panel */}
        <AiSuggestedQuestions
          categoryId={categoryId}
          narrativeContent={narrativeContent}
          answeredQuestionIds={answeredQuestionIds}
          onQuestionClick={onInsertQuestion}
        />

        {categoryId && (
          <>
            <Separator />

            {/* Category-Specific Questions */}
            <CategoryQuestions
              categoryId={categoryId}
              answers={categoryAnswers}
              onAnswerChange={handleCategoryAnswer}
            />
          </>
        )}
      </div>
    </ScrollArea>
  );
}
```
