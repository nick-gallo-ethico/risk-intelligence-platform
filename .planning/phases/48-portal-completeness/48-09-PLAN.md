```typescript
import { Injectable, Logger } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { Inject } from '@nestjs/common';
import { Cache } from 'cache-manager';
import { CACHE_MANAGER } from '@nestjs/cache-manager';
import { QaTeamMetricsDto, QaOperatorMetricsDto } from './dto/qa-metrics.dto';

@Injectable()
export class QaMetricsService {
  private readonly logger = new Logger(QaMetricsService.name);

  constructor(
    private readonly prisma: PrismaService,
    @Inject(CACHE_MANAGER) private readonly cacheManager: Cache,
  ) {}

  /**
   * Get QA team metrics for the dashboard.
   * Cached for 60 seconds to reduce query load.
   */
  async getQaTeamMetrics(
    organizationId: string,
    period: '7d' | '30d' | '90d' = '7d',
  ): Promise<QaTeamMetricsDto> {
    const cacheKey = `qa-metrics:${organizationId}:${period}`;
    const cached = await this.cacheManager.get<QaTeamMetricsDto>(cacheKey);
    if (cached) {
      return cached;
    }

    const now = new Date();
    const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate());

    // Calculate period start
    const periodDays = period === '7d' ? 7 : period === '30d' ? 30 : 90;
    const periodStart = new Date(now.getTime() - periodDays * 24 * 60 * 60 * 1000);

    // Run all queries in parallel
    const [
      pendingCount,
      pendingItems,
      reviewedToday,
      returnedToday,
      operatorMetrics,
      byHourData,
    ] = await Promise.all([
      // Total pending in queue
      this.prisma.riuHotlineExtension.count({
        where: {
          organizationId,
          qaStatus: 'PENDING',
        },
      }),

      // Pending items for wait time calculation
      this.prisma.riuHotlineExtension.findMany({
        where: {
          organizationId,
          qaStatus: 'PENDING',
        },
        select: { createdAt: true },
        take: 100,
      }),

      // Reviewed today
      this.prisma.riuHotlineExtension.count({
        where: {
          organizationId,
          qaStatus: { in: ['APPROVED', 'REJECTED'] },
          qaReviewedAt: { gte: todayStart },
        },
      }),

      // Returned today
      this.prisma.riuHotlineExtension.count({
        where: {
          organizationId,
          qaStatus: 'REJECTED',
          qaReviewedAt: { gte: todayStart },
        },
      }),

      // Per-operator metrics
      this.getOperatorMetrics(organizationId, periodStart),

      // Reviews by hour today
      this.getReviewsByHour(organizationId, todayStart),
    ]);

    // Calculate average wait time
    const avgWaitSeconds =
      pendingItems.length > 0
        ? pendingItems.reduce(
            (sum, item) =>
              sum + (now.getTime() - item.createdAt.getTime()) / 1000,
            0,
          ) / pendingItems.length
        : 0;

    const metrics: QaTeamMetricsDto = {
      totalPending: pendingCount,
      averageWaitTimeSeconds: Math.round(avgWaitSeconds),
      reviewedToday,
      returnedToday,
      todayReturnRate:
        reviewedToday > 0 ? Math.round((returnedToday / reviewedToday) * 100) : 0,
      operators: operatorMetrics,
      byHour: byHourData,
      summaryPeriod: period,
    };

    // Cache for 60 seconds
    await this.cacheManager.set(cacheKey, metrics, 60);

    return metrics;
  }

  private async getOperatorMetrics(
    organizationId: string,
    periodStart: Date,
  ): Promise<QaOperatorMetricsDto[]> {
    // Get operators who have reviewed items in this period
    const reviewsByOperator = await this.prisma.riuHotlineExtension.groupBy({
      by: ['qaReviewerId'],
      where: {
        organizationId,
        qaStatus: { in: ['APPROVED', 'REJECTED'] },
        qaReviewedAt: { gte: periodStart },
        qaReviewerId: { not: null },
      },
      _count: { id: true },
    });

    const operatorIds = reviewsByOperator
      .map((r) => r.qaReviewerId)
      .filter((id): id is string => id !== null);

    if (operatorIds.length === 0) return [];

    // Get operator names
    const operators = await this.prisma.user.findMany({
      where: { id: { in: operatorIds } },
      select: { id: true, firstName: true, lastName: true },
    });

    const operatorMap = new Map(
      operators.map((o) => [o.id, `${o.firstName} ${o.lastName}`])
    );

    // Get detailed metrics per operator
    const metrics: QaOperatorMetricsDto[] = [];

    for (const operatorId of operatorIds) {
      const [reviewed, returned, avgReviewTime] = await Promise.all([
        this.prisma.riuHotlineExtension.count({
          where: {
            organizationId,
            qaReviewerId: operatorId,
            qaStatus: { in: ['APPROVED', 'REJECTED'] },
            qaReviewedAt: { gte: periodStart },
          },
        }),
        this.prisma.riuHotlineExtension.count({
          where: {
            organizationId,
            qaReviewerId: operatorId,
            qaStatus: 'REJECTED',
            qaReviewedAt: { gte: periodStart },
          },
        }),
        // Average review time (from submission to review)
        this.calculateAverageReviewTime(organizationId, operatorId, periodStart),
      ]);

      // Last 7 days specifically
      const sevenDaysAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
      const [reviewed7d, returned7d] = await Promise.all([
        this.prisma.riuHotlineExtension.count({
          where: {
            organizationId,
            qaReviewerId: operatorId,
            qaStatus: { in: ['APPROVED', 'REJECTED'] },
            qaReviewedAt: { gte: sevenDaysAgo },
          },
        }),
        this.prisma.riuHotlineExtension.count({
          where: {
            organizationId,
            qaReviewerId: operatorId,
            qaStatus: 'REJECTED',
            qaReviewedAt: { gte: sevenDaysAgo },
          },
        }),
      ]);

      metrics.push({
        operatorId,
        operatorName: operatorMap.get(operatorId) || 'Unknown',
        totalReviewed: reviewed,
        totalReturned: returned,
        returnRate: reviewed > 0 ? Math.round((returned / reviewed) * 100) : 0,
        averageReviewTimeSeconds: avgReviewTime,
        last7Days: {
          reviewed: reviewed7d,
          returned: returned7d,
          returnRate:
            reviewed7d > 0 ? Math.round((returned7d / reviewed7d) * 100) : 0,
        },
      });
    }

    // Sort by total reviewed descending
    return metrics.sort((a, b) => b.totalReviewed - a.totalReviewed);
  }

  private async calculateAverageReviewTime(
    organizationId: string,
    operatorId: string,
    periodStart: Date,
  ): Promise<number> {
    const items = await this.prisma.riuHotlineExtension.findMany({
      where: {
        organizationId,
        qaReviewerId: operatorId,
        qaStatus: { in: ['APPROVED', 'REJECTED'] },
        qaReviewedAt: { gte: periodStart },
      },
      select: {
        createdAt: true,
        qaReviewedAt: true,
      },
      take: 100,
    });

    if (items.length === 0) return 0;

    const totalSeconds = items.reduce((sum, item) => {
      if (!item.qaReviewedAt) return sum;
      return sum + (item.qaReviewedAt.getTime() - item.createdAt.getTime()) / 1000;
    }, 0);

    return Math.round(totalSeconds / items.length);
  }

  private async getReviewsByHour(
    organizationId: string,
    dayStart: Date,
  ): Promise<Array<{ hour: number; count: number }>> {
    // This is a simplified version - in production might use raw SQL for grouping
    const reviews = await this.prisma.riuHotlineExtension.findMany({
      where: {
        organizationId,
        qaStatus: { in: ['APPROVED', 'REJECTED'] },
        qaReviewedAt: { gte: dayStart },
      },
      select: { qaReviewedAt: true },
    });

    const byHour = Array.from({ length: 24 }, (_, i) => ({ hour: i, count: 0 }));

    for (const review of reviews) {
      if (review.qaReviewedAt) {
        const hour = review.qaReviewedAt.getHours();
        byHour[hour].count++;
      }
    }

    return byHour;
  }
}
````

```typescript
function SummaryCard({
title,
value,
icon: Icon,

function ReturnRateBadge({ rate }: { rate: number }) {
if (rate <= 10) return <Badge variant="outline" className="bg-green-50">{rate}%</Badge>;
if (rate <= 20) return <Badge variant="outline" className="bg-amber-50">{rate}%</Badge>;
return <Badge variant="destructive">{rate}%</Badge>;
}

function TrendIndicator({ current, previous }: { current: number; previous: number }) {
if (current < previous) return <span className="text-green-600">↓ Improving</span>;
if (current > previous) return <span className="text-red-600">↑ Needs attention</span>;
return <span className="text-muted-foreground">→ Stable</span>;
}

function formatSeconds(seconds: number): string {
if (seconds < 60) return `${seconds}s`;
if (seconds < 3600) return `${Math.floor(seconds / 60)}m`;
return `${Math.floor(seconds / 3600)}h ${Math.floor((seconds % 3600) / 60)}m`;
}

);
}
````
