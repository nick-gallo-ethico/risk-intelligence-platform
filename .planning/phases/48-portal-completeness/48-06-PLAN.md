```typescript
'use client';

import { useState, useCallback, useEffect } from 'react';
import { useIdleTimer } from 'react-idle-timer';
import { useRouter } from 'next/navigation';
import { signOut } from 'next-auth/react';
import {
  AlertDialog,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
} from '@/components/ui/alert-dialog';
import { Button } from '@/components/ui/button';
import { Progress } from '@/components/ui/progress';
import { Clock } from 'lucide-react';

interface SessionTimeoutModalProps {
  /** Warning duration in seconds (default 60) */
  warningDurationSeconds?: number;
  /** Idle timeout before showing warning in minutes (default 15) */
  idleTimeoutMinutes?: number;
  /** Callback to extend session (e.g., refresh token) */
  onExtendSession?: () => Promise<void>;
  /** Enable cross-tab synchronization (default true) */
  crossTab?: boolean;
}

export function SessionTimeoutModal({
  warningDurationSeconds = 60,
  idleTimeoutMinutes = 15,
  onExtendSession,
  crossTab = true,
}: SessionTimeoutModalProps) {
  const router = useRouter();
  const [showWarning, setShowWarning] = useState(false);
  const [remainingSeconds, setRemainingSeconds] = useState(warningDurationSeconds);

  // Handle when user becomes idle
  const onIdle = useCallback(() => {
    setShowWarning(true);
    setRemainingSeconds(warningDurationSeconds);
  }, [warningDurationSeconds]);

  // Configure idle timer
  const { reset, pause, start } = useIdleTimer({
    timeout: idleTimeoutMinutes * 60 * 1000, // Convert to milliseconds
    onIdle,
    debounce: 500,
    crossTab,
    syncTimers: 200, // Sync across tabs every 200ms
    events: [
      'mousemove',
      'keydown',
      'wheel',
      'DOMMouseScroll',
      'mousewheel',
      'mousedown',
      'touchstart',
      'touchmove',
      'MSPointerDown',
      'MSPointerMove',
      'visibilitychange',
    ],
  });

  // Countdown timer when warning is shown
  useEffect(() => {
    if (!showWarning) return;

    const interval = setInterval(() => {
      setRemainingSeconds((prev) => {
        if (prev <= 1) {
          clearInterval(interval);
          handleLogout();
          return 0;
        }
        return prev - 1;
      });
    }, 1000);

    return () => clearInterval(interval);
  }, [showWarning]);

  // Extend session and dismiss warning
  const handleExtendSession = useCallback(async () => {
    pause(); // Pause while extending
    setShowWarning(false);

    try {
      if (onExtendSession) {
        await onExtendSession();
      }
    } catch (error) {
      console.error('Failed to extend session:', error);
    }

    reset(); // Reset idle timer
    start(); // Resume timer
  }, [reset, pause, start, onExtendSession]);

  // Logout immediately
  const handleLogout = useCallback(async () => {
    setShowWarning(false);
    await signOut({ callbackUrl: '/login?reason=timeout' });
  }, []);

  // Calculate progress percentage
  const progressPercent = (remainingSeconds / warningDurationSeconds) * 100;

  // Format time display
  const minutes = Math.floor(remainingSeconds / 60);
  const seconds = remainingSeconds % 60;
  const timeDisplay = `${minutes}:${seconds.toString().padStart(2, '0')}`;

  return (
    <AlertDialog open={showWarning}>
      <AlertDialogContent className="max-w-md">
        <AlertDialogHeader>
          <AlertDialogTitle className="flex items-center gap-2">
            <Clock className="h-5 w-5 text-amber-500" />
            Session About to Expire
          </AlertDialogTitle>
          <AlertDialogDescription>
            You&apos;ve been inactive for {idleTimeoutMinutes} minutes. For your security,
            you&apos;ll be automatically logged out in:
          </AlertDialogDescription>
        </AlertDialogHeader>

        <div className="py-6">
          {/* Countdown display */}
          <div className="text-center mb-4">
            <span className="text-5xl font-mono font-bold tabular-nums">
              {timeDisplay}
            </span>
          </div>

          {/* Progress bar */}
          <Progress
            value={progressPercent}
            className="h-2"
            // Change color based on time remaining
            // Using inline style since Progress doesn't accept className for indicator
          />
          <p className="text-center text-sm text-muted-foreground mt-2">
            {remainingSeconds <= 10 ? 'Logging out soon...' : 'Click below to stay logged in'}
          </p>
        </div>

        <AlertDialogFooter className="flex-col sm:flex-row gap-2">
          <Button
            variant="outline"
            onClick={handleLogout}
            className="sm:flex-1"
          >
            Log Out Now
          </Button>
          <Button
            onClick={handleExtendSession}
            className="sm:flex-1"
            autoFocus
          >
            Stay Logged In
          </Button>
        </AlertDialogFooter>
      </AlertDialogContent>
    </AlertDialog>
  );
}
```

Key features:

- Uses react-idle-timer for reliable activity detection
- Cross-tab synchronization prevents logout in one tab while active in another
- Countdown with visual progress bar
- Prominent "Stay Logged In" button (autofocus for accessibility)
- Passes reason=timeout to login page for user feedback
