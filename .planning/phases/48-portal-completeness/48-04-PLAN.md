```typescript
import { Injectable, NotFoundException, ForbiddenException, BadRequestException, Logger } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { AttachmentsService } from '../../attachments/attachments.service';
import { AuditService } from '../../audit/audit.service';
import { CompleteConditionDto, DisclosureConditionDto } from '../dto/disclosure-condition.dto';

@Injectable()
export class DisclosureConditionService {
  private readonly logger = new Logger(DisclosureConditionService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly attachmentsService: AttachmentsService,
    private readonly auditService: AuditService,
  ) {}

  /**
   * Get conditions for a disclosure that the employee can see.
   */
  async getConditionsForDisclosure(
    disclosureId: string,
    userId: string,
    organizationId: string,
  ): Promise<DisclosureConditionDto[]> {
    // Verify disclosure belongs to user
    const disclosure = await this.verifyDisclosureOwnership(disclosureId, userId, organizationId);

    const conditions = await this.prisma.disclosureCondition.findMany({
      where: { disclosureId },
      include: {
        completedBy: { select: { firstName: true, lastName: true } },
        evidenceAttachment: { select: { fileName: true } },
      },
      orderBy: { requiredBy: 'asc' },
    });

    return conditions.map(c => ({
      id: c.id,
      disclosureId: c.disclosureId,
      conditionText: c.conditionText,
      requiredBy: c.requiredBy,
      completedAt: c.completedAt,
      completedById: c.completedById,
      completedByName: c.completedBy
        ? `${c.completedBy.firstName} ${c.completedBy.lastName}`
        : null,
      notes: c.notes,
      evidenceAttachmentId: c.evidenceAttachmentId,
      evidenceFileName: c.evidenceAttachment?.fileName ?? null,
    }));
  }

  /**
   * Mark a condition as complete with optional evidence.
   */
  async completeCondition(
    disclosureId: string,
    conditionId: string,
    userId: string,
    organizationId: string,
    dto: CompleteConditionDto,
  ): Promise<ConditionCompletionResultDto> {
    // Verify ownership
    await this.verifyDisclosureOwnership(disclosureId, userId, organizationId);

    // Find the condition
    const condition = await this.prisma.disclosureCondition.findFirst({
      where: { id: conditionId, disclosureId },
    });

    if (!condition) {
      throw new NotFoundException('Condition not found');
    }

    if (condition.completedAt) {
      throw new BadRequestException('Condition already completed');
    }

    // Verify evidence attachment if provided
    if (dto.evidenceAttachmentId) {
      const attachment = await this.prisma.attachment.findFirst({
        where: {
          id: dto.evidenceAttachmentId,
          organizationId,
          createdById: userId, // Must be uploaded by same user
        },
      });

      if (!attachment) {
        throw new BadRequestException('Evidence attachment not found or not owned by user');
      }
    }

    // Update condition
    const now = new Date();
    await this.prisma.disclosureCondition.update({
      where: { id: conditionId },
      data: {
        completedAt: now,
        completedById: userId,
        notes: dto.notes,
        evidenceAttachmentId: dto.evidenceAttachmentId,
      },
    });

    // Audit log
    await this.auditService.log({
      organizationId,
      entityType: 'DISCLOSURE',
      entityId: disclosureId,
      action: 'condition.completed',
      actionCategory: 'UPDATE',
      actionDescription: `Condition "${condition.conditionText.substring(0, 50)}..." marked complete`,
      actorUserId: userId,
      actorType: 'USER',
      context: { conditionId, hasEvidence: !!dto.evidenceAttachmentId },
    });

    this.logger.log(`Condition ${conditionId} completed for disclosure ${disclosureId}`);

    return {
      conditionId,
      completedAt: now,
      success: true,
    };
  }

  private async verifyDisclosureOwnership(
    disclosureId: string,
    userId: string,
    organizationId: string,
  ) {
    const disclosure = await this.prisma.disclosure.findFirst({
      where: {
        id: disclosureId,
        organizationId,
        submittedById: userId,
      },
    });

    if (!disclosure) {
      throw new NotFoundException('Disclosure not found or access denied');
    }

    return disclosure;
  }
}
````

```typescript
  );

  @Param('conditionId') conditionId: string,
  @Body() dto: CompleteConditionDto,
  @CurrentUser() user: User,
  @TenantId() organizationId: string,
): Promise<ConditionCompletionResultDto> {
  return this.disclosureConditionService.completeCondition(
    disclosureId,
    conditionId,
    user.id,
    organizationId,
    dto,
  );
}
```

```typescript
'use client';

import { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { apiClient } from '@/lib/api';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Textarea } from '@/components/ui/textarea';
import { Badge } from '@/components/ui/badge';
import { Checkbox } from '@/components/ui/checkbox';
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { FileUpload } from '@/components/ui/file-upload';
import { useToast } from '@/components/ui/use-toast';
import { CheckCircle, Clock, Upload, FileText } from 'lucide-react';
import { formatDistanceToNow } from 'date-fns';

interface DisclosureCondition {
  id: string;
  conditionText: string;
  requiredBy: string | null;
  completedAt: string | null;
  completedByName: string | null;
  notes: string | null;
  evidenceFileName: string | null;
}

interface DisclosureConditionsProps {
  disclosureId: string;
}

export function DisclosureConditions({ disclosureId }: DisclosureConditionsProps) {
  const [completeDialogOpen, setCompleteDialogOpen] = useState(false);
  const [selectedCondition, setSelectedCondition] = useState<DisclosureCondition | null>(null);
  const [notes, setNotes] = useState('');
  const [evidenceId, setEvidenceId] = useState<string | null>(null);
  const { toast } = useToast();
  const queryClient = useQueryClient();

  const { data: conditions, isLoading } = useQuery({
    queryKey: ['disclosures', disclosureId, 'conditions'],
    queryFn: () => apiClient.get<DisclosureCondition[]>(`/disclosures/${disclosureId}/conditions`),
  });

  const completeCondition = useMutation({
    mutationFn: ({ conditionId, notes, evidenceAttachmentId }: {
      conditionId: string;
      notes?: string;
      evidenceAttachmentId?: string;
    }) => apiClient.post(
      `/disclosures/${disclosureId}/conditions/${conditionId}/complete`,
      { notes, evidenceAttachmentId }
    ),
    onSuccess: () => {
      toast({ title: 'Condition Completed', description: 'The condition has been marked as complete.' });
      queryClient.invalidateQueries({ queryKey: ['disclosures', disclosureId, 'conditions'] });
      setCompleteDialogOpen(false);
      setSelectedCondition(null);
      setNotes('');
      setEvidenceId(null);
    },
  });

  const handleOpenComplete = (condition: DisclosureCondition) => {
    setSelectedCondition(condition);
    setCompleteDialogOpen(true);
  };

  const handleComplete = () => {
    if (!selectedCondition) return;
    completeCondition.mutate({
      conditionId: selectedCondition.id,
      notes: notes || undefined,
      evidenceAttachmentId: evidenceId || undefined,
    });
  };

  if (isLoading) {
    return <ConditionsSkeleton />;
  }

  if (!conditions?.length) {
    return null; // No conditions to show
  }

  const pending = conditions.filter(c => !c.completedAt);
  const completed = conditions.filter(c => c.completedAt);

  return (
    <Card>
      <CardHeader>
        <CardTitle className="flex items-center gap-2">
          <CheckCircle className="h-5 w-5" />
          Disclosure Conditions
        </CardTitle>
      </CardHeader>
      <CardContent className="space-y-4">
        {/* Pending conditions */}
        {pending.map(condition => (
          <ConditionItem
            key={condition.id}
            condition={condition}
            onComplete={() => handleOpenComplete(condition)}
          />
        ))}

        {/* Completed conditions */}
        {completed.map(condition => (
          <ConditionItem key={condition.id} condition={condition} />
        ))}

        {/* Complete dialog */}
        <Dialog open={completeDialogOpen} onOpenChange={setCompleteDialogOpen}>
          <DialogContent>
            <DialogHeader>
              <DialogTitle>Complete Condition</DialogTitle>
              <DialogDescription>
                {selectedCondition?.conditionText}
              </DialogDescription>
            </DialogHeader>

            <div className="space-y-4 py-4">
              <div>
                <label className="text-sm font-medium">Notes (optional)</label>
                <Textarea
                  value={notes}
                  onChange={(e) => setNotes(e.target.value)}
                  placeholder="Add any notes about how this condition was met..."
                  rows={3}
                />
              </div>

              <div>
                <label className="text-sm font-medium">Evidence (optional)</label>
                <FileUpload
                  onUploadComplete={(attachmentId) => setEvidenceId(attachmentId)}
                  accept=".pdf,.doc,.docx,.jpg,.png"
                  maxSize={10 * 1024 * 1024} // 10MB
                />
              </div>
            </div>

            <DialogFooter>
              <Button variant="outline" onClick={() => setCompleteDialogOpen(false)}>
                Cancel
              </Button>
              <Button onClick={handleComplete} disabled={completeCondition.isPending}>
                Mark Complete
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>
      </CardContent>
    </Card>
  );
}

function ConditionItem({
  condition,
  onComplete,
}: {
  condition: DisclosureCondition;
  onComplete?: () => void;
}) {
  const isComplete = !!condition.completedAt;

  return (
    <div className={`flex items-start gap-3 p-3 rounded-lg border ${
      isComplete ? 'bg-muted/30' : 'bg-background'
    }`}>
      <div className="flex-shrink-0 mt-0.5">
        {isComplete ? (
          <CheckCircle className="h-5 w-5 text-green-500" />
        ) : (
          <Clock className="h-5 w-5 text-amber-500" />
        )}
      </div>

      <div className="flex-1 min-w-0">
        <p className={`text-sm ${isComplete ? 'line-through text-muted-foreground' : ''}`}>
          {condition.conditionText}
        </p>

        {condition.requiredBy && !isComplete && (
          <p className="text-xs text-muted-foreground mt-1">
            Due: {formatDistanceToNow(new Date(condition.requiredBy), { addSuffix: true })}
          </p>
        )}

        {isComplete && (
          <p className="text-xs text-muted-foreground mt-1">
            Completed by {condition.completedByName} {formatDistanceToNow(new Date(condition.completedAt!), { addSuffix: true })}
            {condition.evidenceFileName && (
              <span className="ml-2">
                <FileText className="h-3 w-3 inline" /> {condition.evidenceFileName}
              </span>
            )}
          </p>
        )}
      </div>

      {!isComplete && onComplete && (
        <Button size="sm" variant="outline" onClick={onComplete}>
          Complete
        </Button>
      )}
    </div>
  );
}
```
