import { Injectable, Logger, NotFoundException } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";
import { Inject } from "@nestjs/common";
import { Cache } from "cache-manager";
import { CACHE_MANAGER } from "@nestjs/cache-manager";
import {
  CrisisBannerConfigDto,
  UpdateCrisisBannerDto,
  CrisisBannerSeverity,
} from "./dto/ethics-config.dto";

@Injectable()
export class CrisisBannerService {
  private readonly logger = new Logger(CrisisBannerService.name);

  constructor(
    private readonly prisma: PrismaService,
    @Inject(CACHE_MANAGER) private readonly cacheManager: Cache,
  ) {}

  /**
   * Get current crisis banner configuration for a tenant.
   * Returns null if no active banner.
   */
  async getCrisisBannerConfig(
    organizationId: string,
  ): Promise<CrisisBannerConfigDto | null> {
    const cacheKey = `ethics:crisis:${organizationId}`;

    // Check cache
    const cached = await this.cacheManager.get<CrisisBannerConfigDto | null>(
      cacheKey,
    );
    if (cached !== undefined) {
      return cached;
    }

    // Query organization config
    const org = await this.prisma.organization.findUnique({
      where: { id: organizationId },
      select: {
        crisisBannerActive: true,
        crisisBannerMessage: true,
        crisisBannerSeverity: true,
        crisisBannerStartDate: true,
        crisisBannerEndDate: true,
        emergencyPhone: true,
        crisisBannerTranslations: true,
      },
    });

    if (!org) {
      return null;
    }

    // Check if banner is active and within date range
    if (!org.crisisBannerActive || !org.crisisBannerMessage) {
      await this.cacheManager.set(cacheKey, null, 300); // 5 min cache
      return null;
    }

    const now = new Date();
    const inRange =
      (!org.crisisBannerStartDate || now >= org.crisisBannerStartDate) &&
      (!org.crisisBannerEndDate || now <= org.crisisBannerEndDate);

    if (!inRange) {
      await this.cacheManager.set(cacheKey, null, 300);
      return null;
    }

    const config: CrisisBannerConfigDto = {
      isActive: true,
      message: org.crisisBannerMessage,
      severity:
        (org.crisisBannerSeverity as CrisisBannerSeverity) ||
        CrisisBannerSeverity.WARNING,
      startDate: org.crisisBannerStartDate ?? undefined,
      endDate: org.crisisBannerEndDate ?? undefined,
      emergencyPhone: org.emergencyPhone ?? undefined,
      translations: org.crisisBannerTranslations as
        | Record<string, string>
        | undefined,
    };

    await this.cacheManager.set(cacheKey, config, 300); // 5 min cache

    return config;
  }

  /**
   * Update crisis banner configuration.
   * Admin-only operation.
   */
  async updateCrisisBannerConfig(
    organizationId: string,
    dto: UpdateCrisisBannerDto,
  ): Promise<CrisisBannerConfigDto> {
    await this.prisma.organization.update({
      where: { id: organizationId },
      data: {
        crisisBannerActive: dto.isActive,
        crisisBannerMessage: dto.message,
        crisisBannerSeverity: dto.severity,
        crisisBannerStartDate: dto.startDate ?? null,
        crisisBannerEndDate: dto.endDate ?? null,
        emergencyPhone: dto.emergencyPhone ?? null,
        crisisBannerTranslations: dto.translations ?? null,
      },
    });

    // Invalidate cache
    await this.cacheManager.del(`ethics:crisis:${organizationId}`);

    this.logger.log(`Updated crisis banner for org ${organizationId}`);

    return {
      isActive: dto.isActive,
      message: dto.message,
      severity: dto.severity,
      startDate: dto.startDate,
      endDate: dto.endDate,
      emergencyPhone: dto.emergencyPhone,
      translations: dto.translations,
    };
  }

  /**
   * Get emergency phone number for a tenant.
   */
  async getEmergencyPhone(organizationId: string): Promise<string | null> {
    const org = await this.prisma.organization.findUnique({
      where: { id: organizationId },
      select: { emergencyPhone: true },
    });

    return org?.emergencyPhone ?? null;
  }

  /**
   * Update emergency phone number.
   */
  async updateEmergencyPhone(
    organizationId: string,
    phone: string | null,
  ): Promise<void> {
    await this.prisma.organization.update({
      where: { id: organizationId },
      data: { emergencyPhone: phone },
    });

    // Invalidate crisis banner cache (includes phone)
    await this.cacheManager.del(`ethics:crisis:${organizationId}`);
  }
}

import { Injectable, Logger } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { Inject } from '@nestjs/common';
import { Cache } from 'cache-manager';
import { CACHE_MANAGER } from '@nestjs/cache-manager';
import { TransparencyStatsDto } from './dto/ethics-config.dto';

@Injectable()
export class TransparencyStatsService {
private readonly logger = new Logger(TransparencyStatsService.name);

constructor(
private readonly prisma: PrismaService,
@Inject(CACHE_MANAGER) private readonly cacheManager: Cache,
) {}

/\*\*

- Get anonymized transparency statistics for the ethics portal.
- Returns null if organization has disabled transparency display.
  \*/
  async getTransparencyStats(
  organizationId: string,
  ): Promise<TransparencyStatsDto | null> {
  // Check if transparency display is enabled
  const org = await this.prisma.organization.findUnique({
  where: { id: organizationId },
  select: { showTransparencyStats: true },
  });

  if (!org?.showTransparencyStats) {
  return null;
  }

  // Check cache
  const cacheKey = `ethics:transparency:${organizationId}`;
  const cached = await this.cacheManager.get<TransparencyStatsDto>(cacheKey);
  if (cached) {
  return cached;
  }

  // Calculate stats
  const now = new Date();
  const yearStart = new Date(now.getFullYear(), 0, 1);
  const lastYear = new Date(now.getFullYear() - 1, 0, 1);
  const lastYearEnd = new Date(now.getFullYear() - 1, 11, 31);

  // Run all queries in parallel
  const [
  totalReports,
  reportsThisYear,
  reportsLastYear,
  categories,
  anonymousCount,
  avgResolution,
  ] = await Promise.all([
  // Total reports received (all time)
  this.prisma.riskIntelligenceUnit.count({
  where: {
  organizationId,
  type: { in: ['HOTLINE_REPORT', 'WEB_FORM_SUBMISSION'] },
  },
  }),

      // Reports this year
      this.prisma.riskIntelligenceUnit.count({
        where: {
          organizationId,
          type: { in: ['HOTLINE_REPORT', 'WEB_FORM_SUBMISSION'] },
          createdAt: { gte: yearStart },
        },
      }),

      // Reports last year (for trend calculation)
      this.prisma.riskIntelligenceUnit.count({
        where: {
          organizationId,
          type: { in: ['HOTLINE_REPORT', 'WEB_FORM_SUBMISSION'] },
          createdAt: { gte: lastYear, lte: lastYearEnd },
        },
      }),

      // Distinct categories reported
      this.prisma.riskIntelligenceUnit.findMany({
        where: { organizationId },
        select: { categoryId: true },
        distinct: ['categoryId'],
      }),

      // Anonymous reports
      this.prisma.riskIntelligenceUnit.count({
        where: {
          organizationId,
          reporterType: 'ANONYMOUS',
        },
      }),

      // Average resolution time (from linked cases)
      this.calculateAverageResolutionDays(organizationId),

  ]);

  // Calculate trend
  let trend: TransparencyStatsDto['reportingTrend'] = 'stable';
  if (reportsLastYear > 0) {
  const changePercent = ((reportsThisYear - reportsLastYear) / reportsLastYear) \* 100;
  if (changePercent > 10) trend = 'increasing';
  else if (changePercent < -10) trend = 'decreasing';
  }

  const stats: TransparencyStatsDto = {
  totalReportsReceived: totalReports,
  reportsThisYear,
  averageResolutionDays: avgResolution,
  categoriesReported: categories.length,
  anonymousPercentage:
  totalReports > 0 ? Math.round((anonymousCount / totalReports) \* 100) : 0,
  reportingTrend: trend,
  };

  // Cache for 1 hour
  await this.cacheManager.set(cacheKey, stats, 3600);

  return stats;

}

private async calculateAverageResolutionDays(organizationId: string): Promise<number> {
// Get closed cases with their RIU creation dates
const closedCases = await this.prisma.case.findMany({
where: {
organizationId,
status: 'CLOSED',
closedAt: { not: null },
},
select: {
closedAt: true,
createdAt: true,
},
take: 1000, // Sample for performance
orderBy: { closedAt: 'desc' },
});

    if (closedCases.length === 0) return 0;

    const totalDays = closedCases.reduce((sum, c) => {
      if (!c.closedAt) return sum;
      const days = Math.ceil(
        (c.closedAt.getTime() - c.createdAt.getTime()) / (1000 * 60 * 60 * 24)
      );
      return sum + days;
    }, 0);

    return Math.round(totalDays / closedCases.length);

}
}

// Public endpoint - no auth required for ethics portal visitors
@Get('config/:tenantSlug')
@ApiOperation({ summary: 'Get ethics portal configuration' })
async getPortalConfig(
  @Param('tenantSlug') tenantSlug: string,
): Promise<EthicsPortalConfigDto> {
  const org = await this.ethicsPortalService.getOrganizationBySlug(tenantSlug);
  if (!org) {
    throw new NotFoundException('Organization not found');
  }

  const [crisisBanner, transparencyStats] = await Promise.all([
    this.crisisBannerService.getCrisisBannerConfig(org.id),
    this.transparencyStatsService.getTransparencyStats(org.id),
  ]);

  return {
    organizationId: org.id,
    organizationName: org.name,
    logoUrl: org.logoUrl,
    primaryColor: org.primaryColor,
    crisisBanner: crisisBanner ?? undefined,
    emergencyPhone: org.emergencyPhone ?? undefined,
    showTransparencyStats: org.showTransparencyStats ?? false,
    transparencyStats: transparencyStats ?? undefined,
    supportedLanguages: org.supportedLanguages ?? ['en'],
    defaultLanguage: org.defaultLanguage ?? 'en',
  };
}
