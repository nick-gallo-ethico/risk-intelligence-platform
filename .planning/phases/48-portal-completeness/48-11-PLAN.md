```typescript
// Add to existing directives.controller.ts

/**
 * Statement management endpoints (for OPENING and CLOSING stages only)
 */

@Get('statements')
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles('SYSTEM_ADMIN', 'QA_MANAGER')
@ApiOperation({ summary: 'Get all opening/closing statements' })
async getStatements(
  @TenantId() organizationId: string,
  @Query('includeInactive') includeInactive?: boolean,
): Promise<StatementsByStageDto> {
  const all = await this.directivesService.getAllDirectives(organizationId, {
    includeInactive: includeInactive === true,
  });

  return {
    opening: all['OPENING'].map(this.mapToStatementResponse),
    closing: all['CLOSING'].map(this.mapToStatementResponse),
  };
}

@Post('statements')
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles('SYSTEM_ADMIN', 'QA_MANAGER')
@ApiOperation({ summary: 'Create a new statement' })
async createStatement(
  @Body() dto: CreateStatementDto,
  @TenantId() organizationId: string,
): Promise<StatementResponseDto> {
  // Validate stage is OPENING or CLOSING
  if (dto.stage !== 'OPENING' && dto.stage !== 'CLOSING') {
    throw new BadRequestException('Statement stage must be OPENING or CLOSING');
  }

  const directive = await this.directivesService.create(organizationId, {
    stage: dto.stage,
    title: dto.title,
    content: dto.content,
    isReadAloud: dto.isReadAloud ?? false,
  });

  return this.mapToStatementResponse(directive);
}

@Patch('statements/:id')
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles('SYSTEM_ADMIN', 'QA_MANAGER')
@ApiOperation({ summary: 'Update a statement' })
async updateStatement(
  @Param('id') id: string,
  @Body() dto: UpdateStatementDto,
  @TenantId() organizationId: string,
): Promise<StatementResponseDto> {
  const directive = await this.directivesService.update(id, organizationId, dto);
  return this.mapToStatementResponse(directive);
}

@Delete('statements/:id')
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles('SYSTEM_ADMIN', 'QA_MANAGER')
@ApiOperation({ summary: 'Deactivate a statement' })
@HttpCode(204)
async deleteStatement(
  @Param('id') id: string,
  @TenantId() organizationId: string,
): Promise<void> {
  await this.directivesService.delete(id, organizationId);
}

@Post('statements/reorder')
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles('SYSTEM_ADMIN', 'QA_MANAGER')
@ApiOperation({ summary: 'Reorder statements within a stage' })
@HttpCode(204)
async reorderStatements(
  @Body() dto: ReorderStatementsDto,
  @TenantId() organizationId: string,
): Promise<void> {
  await this.directivesService.reorder(organizationId, dto.ids);
}

private mapToStatementResponse(directive: DirectiveWithCategory): StatementResponseDto {
  return {
    id: directive.id,
    stage: directive.stage,
    title: directive.title,
    content: directive.content,
    isReadAloud: directive.isReadAloud,
    isActive: directive.isActive,
    order: directive.order,
    createdAt: directive.createdAt,
    updatedAt: directive.updatedAt,
  };
}
```

```typescript
'use client';

import { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { apiClient } from '@/lib/api';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Textarea } from '@/components/ui/textarea';
import { Checkbox } from '@/components/ui/checkbox';
import { Badge } from '@/components/ui/badge';
import { Label } from '@/components/ui/label';
import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
import {
Dialog,
DialogContent,
DialogFooter,
DialogHeader,
DialogTitle,
} from '@/components/ui/dialog';
import {
DndContext,
closestCenter,
PointerSensor,
useSensor,
useSensors,
DragEndEvent,
} from '@dnd-kit/core';
import {
SortableContext,
useSortable,
verticalListSortingStrategy,
arrayMove,
} from '@dnd-kit/sortable';
import { CSS } from '@dnd-kit/utilities';
import { useToast } from '@/components/ui/use-toast';
import { Plus, GripVertical, Pencil, Trash2, Mic, Eye, EyeOff } from 'lucide-react';

interface Statement {
id: string;
stage: string;
title: string;
content: string;
isReadAloud: boolean;
isActive: boolean;
order: number;
}

interface StatementsByStage {
opening: Statement[];
closing: Statement[];
}

export function StatementManager() {
const [editingStatement, setEditingStatement] = useState<Statement | null>(null);
const [isCreateDialogOpen, setIsCreateDialogOpen] = useState(false);
const [createStage, setCreateStage] = useState<'OPENING' | 'CLOSING'>('OPENING');
const { toast } = useToast();
const queryClient = useQueryClient();

// Fetch statements
const { data: statements, isLoading } = useQuery({
queryKey: ['operator', 'statements'],
queryFn: () =>
apiClient.get<StatementsByStage>('/operator/directives/statements', {
params: { includeInactive: true },
}),
});

// Create statement
const createStatement = useMutation({
mutationFn: (data: {
stage: 'OPENING' | 'CLOSING';
title: string;
content: string;
isReadAloud: boolean;
}) => apiClient.post('/operator/directives/statements', data),
onSuccess: () => {
toast({ title: 'Statement Created' });
queryClient.invalidateQueries({ queryKey: ['operator', 'statements'] });
setIsCreateDialogOpen(false);
},
});

// Update statement
const updateStatement = useMutation({
mutationFn: ({ id, ...data }: { id: string } & Partial<Statement>) =>
apiClient.patch(`/operator/directives/statements/${id}`, data),
onSuccess: () => {
toast({ title: 'Statement Updated' });
queryClient.invalidateQueries({ queryKey: ['operator', 'statements'] });
setEditingStatement(null);
},
});

// Delete (deactivate) statement
const deleteStatement = useMutation({
mutationFn: (id: string) =>
apiClient.delete(`/operator/directives/statements/${id}`),
onSuccess: () => {
toast({ title: 'Statement Deactivated' });
queryClient.invalidateQueries({ queryKey: ['operator', 'statements'] });
},
});

// Reorder statements
const reorderStatements = useMutation({
mutationFn: (ids: string[]) =>
apiClient.post('/operator/directives/statements/reorder', { ids }),
onSuccess: () => {
queryClient.invalidateQueries({ queryKey: ['operator', 'statements'] });
},
});

const handleDragEnd = (event: DragEndEvent, stage: 'opening' | 'closing') => {
const { active, over } = event;
if (!over || active.id === over.id) return;

    const items = statements?.[stage] || [];
    const oldIndex = items.findIndex((i) => i.id === active.id);
    const newIndex = items.findIndex((i) => i.id === over.id);

    const newOrder = arrayMove(items, oldIndex, newIndex);
    reorderStatements.mutate(newOrder.map((i) => i.id));

};

const sensors = useSensors(useSensor(PointerSensor));

if (isLoading) {
return <div>Loading...</div>;
}

return (

<div className="space-y-6">
<div className="flex justify-between items-center">
<h2 className="text-2xl font-bold">Operator Scripts</h2>
</div>

      <Tabs defaultValue="opening">
        <TabsList>
          <TabsTrigger value="opening">
            Opening Statements ({statements?.opening?.length || 0})
          </TabsTrigger>
          <TabsTrigger value="closing">
            Closing Statements ({statements?.closing?.length || 0})
          </TabsTrigger>
        </TabsList>

        <TabsContent value="opening" className="mt-4">
          <StatementsTab
            statements={statements?.opening || []}
            stage="OPENING"
            onEdit={setEditingStatement}
            onDelete={(id) => deleteStatement.mutate(id)}
            onReorder={(ids) => reorderStatements.mutate(ids)}
            onCreate={() => {
              setCreateStage('OPENING');
              setIsCreateDialogOpen(true);
            }}
            sensors={sensors}
            onDragEnd={(e) => handleDragEnd(e, 'opening')}
          />
        </TabsContent>

        <TabsContent value="closing" className="mt-4">
          <StatementsTab
            statements={statements?.closing || []}
            stage="CLOSING"
            onEdit={setEditingStatement}
            onDelete={(id) => deleteStatement.mutate(id)}
            onReorder={(ids) => reorderStatements.mutate(ids)}
            onCreate={() => {
              setCreateStage('CLOSING');
              setIsCreateDialogOpen(true);
            }}
            sensors={sensors}
            onDragEnd={(e) => handleDragEnd(e, 'closing')}
          />
        </TabsContent>
      </Tabs>

      {/* Create Dialog */}
      <StatementDialog
        open={isCreateDialogOpen}
        onOpenChange={setIsCreateDialogOpen}
        title={`Create ${createStage === 'OPENING' ? 'Opening' : 'Closing'} Statement`}
        onSubmit={(data) =>
          createStatement.mutate({ ...data, stage: createStage })
        }
        isSubmitting={createStatement.isPending}
      />

      {/* Edit Dialog */}
      {editingStatement && (
        <StatementDialog
          open={true}
          onOpenChange={() => setEditingStatement(null)}
          title="Edit Statement"
          initialData={editingStatement}
          onSubmit={(data) =>
            updateStatement.mutate({ id: editingStatement.id, ...data })
          }
          isSubmitting={updateStatement.isPending}
        />
      )}
    </div>

);
}

function StatementsTab({
statements,
stage,
onEdit,
onDelete,
onReorder,
onCreate,
sensors,
onDragEnd,
}: {
statements: Statement[];
stage: string;
onEdit: (s: Statement) => void;
onDelete: (id: string) => void;
onReorder: (ids: string[]) => void;
onCreate: () => void;
sensors: any;
onDragEnd: (event: DragEndEvent) => void;
}) {
return (
<Card>
<CardHeader className="flex flex-row items-center justify-between">
<CardTitle className="text-sm">
{stage === 'OPENING' ? 'Opening' : 'Closing'} Statements
</CardTitle>
<Button size="sm" onClick={onCreate}>
<Plus className="h-4 w-4 mr-1" />
Add Statement
</Button>
</CardHeader>
<CardContent>
{statements.length === 0 ? (

<p className="text-center py-8 text-muted-foreground">
No statements configured. Add one to get started.
</p>
) : (
<DndContext
            sensors={sensors}
            collisionDetection={closestCenter}
            onDragEnd={onDragEnd}
          >
<SortableContext
items={statements.map((s) => s.id)}
strategy={verticalListSortingStrategy} >
<div className="space-y-2">
{statements.map((statement) => (
<SortableStatementItem
key={statement.id}
statement={statement}
onEdit={() => onEdit(statement)}
onDelete={() => onDelete(statement.id)}
/>
))}
</div>
</SortableContext>
</DndContext>
)}
</CardContent>
</Card>
);
}

function SortableStatementItem({
statement,
onEdit,
onDelete,
}: {
statement: Statement;
onEdit: () => void;
onDelete: () => void;
}) {
const { attributes, listeners, setNodeRef, transform, transition } = useSortable({
id: statement.id,
});

const style = {
transform: CSS.Transform.toString(transform),
transition,
};

return (

<div
ref={setNodeRef}
style={style}
className={`flex items-center gap-3 p-3 border rounded-lg ${
        statement.isActive ? 'bg-background' : 'bg-muted/30 opacity-60'
      }`} >
<button {...attributes} {...listeners} className="cursor-grab">
<GripVertical className="h-4 w-4 text-muted-foreground" />
</button>

      <div className="flex-1 min-w-0">
        <div className="flex items-center gap-2">
          <span className="font-medium text-sm">{statement.title}</span>
          {statement.isReadAloud && (
            <Badge variant="outline" className="text-xs gap-1">
              <Mic className="h-3 w-3" />
              Read Aloud
            </Badge>
          )}
          {!statement.isActive && (
            <Badge variant="secondary" className="text-xs">
              <EyeOff className="h-3 w-3 mr-1" />
              Inactive
            </Badge>
          )}
        </div>
        <p className="text-xs text-muted-foreground line-clamp-1">
          {statement.content}
        </p>
      </div>

      <div className="flex items-center gap-1">
        <Button variant="ghost" size="icon" onClick={onEdit}>
          <Pencil className="h-4 w-4" />
        </Button>
        <Button variant="ghost" size="icon" onClick={onDelete}>
          <Trash2 className="h-4 w-4" />
        </Button>
      </div>
    </div>

);
}

// StatementDialog component for create/edit
function StatementDialog({
open,
onOpenChange,
title,
initialData,
onSubmit,
isSubmitting,
}: {
open: boolean;
onOpenChange: (open: boolean) => void;
title: string;
initialData?: Statement;
onSubmit: (data: { title: string; content: string; isReadAloud: boolean }) => void;
isSubmitting: boolean;
}) {
const [formTitle, setFormTitle] = useState(initialData?.title || '');
const [content, setContent] = useState(initialData?.content || '');
const [isReadAloud, setIsReadAloud] = useState(initialData?.isReadAloud || false);

const handleSubmit = () => {
onSubmit({ title: formTitle, content, isReadAloud });
};

return (

<Dialog open={open} onOpenChange={onOpenChange}>
<DialogContent>
<DialogHeader>
<DialogTitle>{title}</DialogTitle>
</DialogHeader>

        <div className="space-y-4 py-4">
          <div>
            <Label>Title</Label>
            <Input
              value={formTitle}
              onChange={(e) => setFormTitle(e.target.value)}
              placeholder="Statement title"
            />
          </div>

          <div>
            <Label>Content</Label>
            <Textarea
              value={content}
              onChange={(e) => setContent(e.target.value)}
              placeholder="Statement content..."
              rows={5}
            />
          </div>

          <div className="flex items-center gap-2">
            <Checkbox
              id="isReadAloud"
              checked={isReadAloud}
              onCheckedChange={(c) => setIsReadAloud(c === true)}
            />
            <Label htmlFor="isReadAloud">Must be read aloud verbatim</Label>
          </div>
        </div>

        <DialogFooter>
          <Button variant="outline" onClick={() => onOpenChange(false)}>
            Cancel
          </Button>
          <Button onClick={handleSubmit} disabled={isSubmitting || !formTitle || !content}>
            Save
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>

);
}
````
