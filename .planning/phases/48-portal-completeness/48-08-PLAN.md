```typescript
'use client';

import { useState, useMemo, useCallback } from 'react';
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Checkbox } from '@/components/ui/checkbox';
import { ScrollArea } from '@/components/ui/scroll-area';
import { Alert, AlertDescription } from '@/components/ui/alert';
import { Badge } from '@/components/ui/badge';
import { AlertTriangle, FileText, Mic } from 'lucide-react';

export interface Directive {
  id: string;
  title: string;
  content: string;
  stage: 'OPENING' | 'INTAKE' | 'CATEGORY_SPECIFIC' | 'CLOSING';
  isReadAloud: boolean;
  isRequired?: boolean; // All shown directives should be acknowledged
  order: number;
}

interface DirectiveAcknowledgmentProps {
  directives: Directive[];
  onConfirm: (acknowledgedIds: string[]) => void;
  onCancel: () => void;
  open: boolean;
}

export function DirectiveAcknowledgment({
  directives,
  onConfirm,
  onCancel,
  open,
}: DirectiveAcknowledgmentProps) {
  const [acknowledged, setAcknowledged] = useState<Set<string>>(new Set());

  // All directives in this modal are required
  const requiredDirectives = directives;

  const allRequiredAcknowledged = useMemo(
    () => requiredDirectives.every((d) => acknowledged.has(d.id)),
    [requiredDirectives, acknowledged]
  );

  const handleToggle = useCallback((directiveId: string) => {
    setAcknowledged((prev) => {
      const next = new Set(prev);
      if (next.has(directiveId)) {
        next.delete(directiveId);
      } else {
        next.add(directiveId);
      }
      return next;
    });
  }, []);

  const handleConfirm = useCallback(() => {
    if (allRequiredAcknowledged) {
      onConfirm(Array.from(acknowledged));
    }
  }, [allRequiredAcknowledged, acknowledged, onConfirm]);

  // Reset state when dialog opens
  const handleOpenChange = useCallback(
    (isOpen: boolean) => {
      if (!isOpen) {
        onCancel();
      }
    },
    [onCancel]
  );

  // Group by stage for organized display
  const groupedDirectives = useMemo(() => {
    const groups: Record<string, Directive[]> = {
      OPENING: [],
      INTAKE: [],
      CATEGORY_SPECIFIC: [],
      CLOSING: [],
    };
    directives.forEach((d) => {
      groups[d.stage]?.push(d);
    });
    return groups;
  }, [directives]);

  const stageLabels: Record<string, string> = {
    OPENING: 'Opening Statement',
    INTAKE: 'During Intake',
    CATEGORY_SPECIFIC: 'Category-Specific',
    CLOSING: 'Closing Statement',
  };

  return (
    <Dialog open={open} onOpenChange={handleOpenChange}>
      <DialogContent className="max-w-2xl max-h-[90vh]">
        <DialogHeader>
          <DialogTitle className="flex items-center gap-2">
            <FileText className="h-5 w-5" />
            Review Required Directives
          </DialogTitle>
          <DialogDescription>
            Please confirm you have reviewed and followed all required directives
            before submitting this report. Directives marked with{' '}
            <Mic className="h-3 w-3 inline" /> must be read aloud.
          </DialogDescription>
        </DialogHeader>

        <ScrollArea className="h-[400px] pr-4">
          <div className="space-y-6">
            {Object.entries(groupedDirectives).map(([stage, stageDirectives]) =>
              stageDirectives.length > 0 ? (
                <div key={stage}>
                  <h4 className="text-sm font-medium text-muted-foreground mb-3">
                    {stageLabels[stage]}
                  </h4>
                  <div className="space-y-3">
                    {stageDirectives.map((directive) => (
                      <DirectiveItem
                        key={directive.id}
                        directive={directive}
                        isAcknowledged={acknowledged.has(directive.id)}
                        onToggle={() => handleToggle(directive.id)}
                      />
                    ))}
                  </div>
                </div>
              ) : null
            )}
          </div>
        </ScrollArea>

        {!allRequiredAcknowledged && (
          <Alert variant="destructive" className="mt-4">
            <AlertTriangle className="h-4 w-4" />
            <AlertDescription>
              You must acknowledge all {requiredDirectives.length} directive(s) before
              submitting.
            </AlertDescription>
          </Alert>
        )}

        <div className="flex items-center justify-between pt-4 border-t">
          <span className="text-sm text-muted-foreground">
            {acknowledged.size} of {requiredDirectives.length} acknowledged
          </span>
          <DialogFooter className="flex-row gap-2">
            <Button variant="outline" onClick={onCancel}>
              Cancel
            </Button>
            <Button
              onClick={handleConfirm}
              disabled={!allRequiredAcknowledged}
            >
              Confirm & Submit Report
            </Button>
          </DialogFooter>
        </div>
      </DialogContent>
    </Dialog>
  );
}

interface DirectiveItemProps {
  directive: Directive;
  isAcknowledged: boolean;
  onToggle: () => void;
}

function DirectiveItem({ directive, isAcknowledged, onToggle }: DirectiveItemProps) {
  return (
    <div
      className={`p-4 border rounded-lg transition-colors ${
        isAcknowledged
          ? 'bg-green-50 border-green-200 dark:bg-green-950/20 dark:border-green-800'
          : 'bg-amber-50 border-amber-200 dark:bg-amber-950/20 dark:border-amber-800'
      }`}
    >
      <div className="flex items-start gap-3">
        <Checkbox
          id={directive.id}
          checked={isAcknowledged}
          onCheckedChange={onToggle}
          className="mt-0.5"
        />
        <div className="flex-1 min-w-0">
          <label
            htmlFor={directive.id}
            className="text-sm font-medium cursor-pointer flex items-center gap-2"
          >
            {directive.title}
            {directive.isReadAloud && (
              <Badge variant="outline" className="text-xs gap-1">
                <Mic className="h-3 w-3" />
                Read Aloud
              </Badge>
            )}
          </label>
          <p className="text-sm text-muted-foreground mt-1 whitespace-pre-wrap">
            {directive.content}
          </p>
        </div>
      </div>
    </div>
  );
}
```

```typescript
const handleSubmitClick = () => {
  // If directives exist and not yet acknowledged, show gate
  const allDirectives = [
    ...(directives?.opening || []),
    ...(directives?.intake || []),
    ...(directives?.categorySpecific || []),
    ...(directives?.closing || []),
  ];

  if (allDirectives.length > 0 && acknowledgedDirectiveIds.length === 0) {
    setShowDirectiveGate(true);
    return;
  }

  // Proceed with submission
  submitRiu();
};

const handleDirectiveConfirm = (ids: string[]) => {
  setAcknowledgedDirectiveIds(ids);
  setShowDirectiveGate(false);
  // Now submit
  submitRiu();
};

const handleDirectiveCancel = () => {
  setShowDirectiveGate(false);
};
```
