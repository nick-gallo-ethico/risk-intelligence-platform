import { PrismaClient } from "@prisma/client";

/**
 * Phase 45: Enhanced AI Features Demo Data
 *
 * Seeds:
 * - Pattern alerts for repeat subjects (3+ cases)
 * - Historical case data to show trends
 */
export async function seedPhase45(prisma: PrismaClient): Promise<void> {
  console.log("Seeding Phase 45: Enhanced AI Features...");

  // Get Acme org
  const acmeOrg = await prisma.organization.findFirst({
    where: { slug: "acme-corp" },
  });

  if (!acmeOrg) {
    console.log("Acme Corp not found, skipping Phase 45 seed");
    return;
  }

  // Find a person who is a repeat subject (or create one)
  const repeatSubjectPerson = await prisma.person.findFirst({
    where: {
      organizationId: acmeOrg.id,
      firstName: "John",
    },
  });

  if (repeatSubjectPerson) {
    // Count how many cases this person is involved in
    const involvementCount = await prisma.personCaseAssociation.count({
      where: {
        personId: repeatSubjectPerson.id,
        organizationId: acmeOrg.id,
        label: "SUBJECT",
      },
    });

    // Create pattern alert if they're in 3+ cases
    if (involvementCount >= 3) {
      const caseAssociations = await prisma.personCaseAssociation.findMany({
        where: {
          personId: repeatSubjectPerson.id,
          organizationId: acmeOrg.id,
          label: "SUBJECT",
        },
        select: { caseId: true },
        take: 10,
      });

      await prisma.patternAlert.upsert({
        where: {
          organizationId_patternType_entityId: {
            organizationId: acmeOrg.id,
            patternType: "REPEAT_SUBJECT",
            entityId: repeatSubjectPerson.id,
          },
        },
        update: {
          actualCount: involvementCount,
          caseIds: caseAssociations.map((a) => a.caseId),
        },
        create: {
          organizationId: acmeOrg.id,
          patternType: "REPEAT_SUBJECT",
          entityType: "PERSON",
          entityId: repeatSubjectPerson.id,
          entityName: `${repeatSubjectPerson.firstName} ${repeatSubjectPerson.lastName}`,
          threshold: 3,
          actualCount: involvementCount,
          caseIds: caseAssociations.map((a) => a.caseId),
          severity: involvementCount >= 5 ? "HIGH" : "MEDIUM",
          status: "ACTIVE",
        },
      });

      console.log(
        `Created pattern alert for ${repeatSubjectPerson.firstName} (${involvementCount} cases)`,
      );
    }
  }

  // Create a few more pattern alerts for demo variety
  const otherSubjects = await prisma.personCaseAssociation.groupBy({
    by: ["personId"],
    where: {
      organizationId: acmeOrg.id,
      label: "SUBJECT",
    },
    _count: true,
    having: {
      personId: {
        _count: {
          gte: 3,
        },
      },
    },
    take: 5,
  });

  for (const subject of otherSubjects) {
    const person = await prisma.person.findUnique({
      where: { id: subject.personId },
    });

    if (person) {
      const caseAssociations = await prisma.personCaseAssociation.findMany({
        where: {
          personId: subject.personId,
          organizationId: acmeOrg.id,
          label: "SUBJECT",
        },
        select: { caseId: true },
      });

      await prisma.patternAlert.upsert({
        where: {
          organizationId_patternType_entityId: {
            organizationId: acmeOrg.id,
            patternType: "REPEAT_SUBJECT",
            entityId: subject.personId,
          },
        },
        update: {
          actualCount: subject._count,
        },
        create: {
          organizationId: acmeOrg.id,
          patternType: "REPEAT_SUBJECT",
          entityType: "PERSON",
          entityId: subject.personId,
          entityName: `${person.firstName} ${person.lastName}`,
          threshold: 3,
          actualCount: subject._count,
          caseIds: caseAssociations.map((a) => a.caseId),
          severity: subject._count >= 5 ? "HIGH" : "MEDIUM",
          status: "ACTIVE",
        },
      });
    }
  }

  console.log("Phase 45 seed complete");
}
