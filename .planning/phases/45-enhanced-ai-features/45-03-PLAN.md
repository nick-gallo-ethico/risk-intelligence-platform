import { Processor, WorkerHost, OnWorkerEvent } from "@nestjs/bullmq";
import { Logger } from "@nestjs/common";
import { Job } from "bullmq";
import { PersonCaseLabel } from "@prisma/client";
import { PATTERN_DETECTION_QUEUE } from "../queues/pattern-detection.queue";
import { PatternDetectionJobData } from "../types/job-data.types";
import { PatternDetectionService } from "../../associations/pattern-detection/pattern-detection.service";
import { PatternAlertService } from "../../associations/pattern-detection/pattern-alert.service";
import {
  PatternType,
  PatternEntityType,
  PatternAlertSeverity,
} from "../../associations/pattern-detection/dto/pattern-alert.dto";

/**
 * PatternDetectionProcessor handles scheduled pattern detection jobs.
 *
 * Per CONTEXT.md: Pattern detection runs per-tenant (organizationId in job data).
 * Uses existing PatternDetectionService.findRepeatInvolvements() for queries.
 * Creates alerts via PatternAlertService.createOrUpdate() with deduplication.
 */
@Processor(PATTERN_DETECTION_QUEUE, { concurrency: 2 })
export class PatternDetectionProcessor extends WorkerHost {
  private readonly logger = new Logger(PatternDetectionProcessor.name);

  constructor(
    private readonly patternDetection: PatternDetectionService,
    private readonly alertService: PatternAlertService,
  ) {
    super();
  }

  async process(
    job: Job<PatternDetectionJobData>,
  ): Promise<{ alertsCreated: number; alertsUpdated: number }> {
    const { organizationId, thresholds = {} } = job.data;

    this.logger.log(
      `Processing pattern detection job ${job.id} for org: ${organizationId}`,
    );

    const results = { alertsCreated: 0, alertsUpdated: 0 };

    // Default thresholds from RESEARCH.md
    const {
      repeatSubjectMin = 3,
      velocityDays = 90,
      velocityCases = 5,
    } = thresholds;

    // Detect repeat subjects using existing PatternDetectionService
    try {
      const repeatSubjects = await this.patternDetection.findRepeatInvolvements(
        PersonCaseLabel.SUBJECT,
        repeatSubjectMin,
        organizationId,
      );

      this.logger.log(
        `Found ${repeatSubjects.length} repeat subjects in org ${organizationId}`,
      );

      // Create alerts for each pattern
      for (const subject of repeatSubjects) {
        const { alert, isNew } = await this.alertService.createOrUpdate(
          organizationId,
          {
            patternType: PatternType.REPEAT_SUBJECT,
            entityType: PatternEntityType.PERSON,
            entityId: subject.personId,
            entityName: subject.personName,
            threshold: repeatSubjectMin,
            actualCount: subject.caseCount,
            caseIds: [], // Would need additional query to get specific case IDs
            severity: this.calculateSeverity(
              subject.caseCount,
              repeatSubjectMin,
            ),
          },
        );

        if (isNew) {
          results.alertsCreated++;
        } else {
          results.alertsUpdated++;
        }
      }
    } catch (error) {
      this.logger.error(
        `Pattern detection failed for org ${organizationId}: ${(error as Error).message}`,
      );
      throw error;
    }

    this.logger.log(
      `Pattern detection complete for org ${organizationId}: ${results.alertsCreated} created, ${results.alertsUpdated} updated`,
    );

    return results;
  }

  /**
   * Calculate severity based on how much count exceeds threshold.
   */
  private calculateSeverity(
    actualCount: number,
    threshold: number,
  ): PatternAlertSeverity {
    const ratio = actualCount / threshold;
    if (ratio >= 3) return PatternAlertSeverity.HIGH;
    if (ratio >= 2) return PatternAlertSeverity.MEDIUM;
    return PatternAlertSeverity.LOW;
  }

  @OnWorkerEvent("completed")
  onCompleted(job: Job<PatternDetectionJobData>) {
    this.logger.log(
      `Pattern detection job ${job.id} completed for org ${job.data.organizationId}`,
    );
  }

  @OnWorkerEvent("failed")
  onFailed(job: Job<PatternDetectionJobData>, error: Error) {
    this.logger.error(
      `Pattern detection job ${job.id} failed for org ${job.data.organizationId}: ${error.message}`,
    );
  }
}
