```typescript
import {
  Controller,
  Get,
  Post,
  Patch,
  Param,
  Query,
  Body,
  UseGuards,
} from "@nestjs/common";
import { JwtAuthGuard } from "../../auth/guards/jwt-auth.guard";
import { TenantGuard } from "@common/guards/tenant.guard";
import { Roles } from "@common/decorators/roles.decorator";
import { RolesGuard } from "@common/guards/roles.guard";
import { CurrentUser } from "@common/decorators/current-user.decorator";
import { TenantId } from "@common/decorators/tenant-id.decorator";
import { PatternAlertService } from "./pattern-alert.service";
import {
  PatternAlertFilterDto,
  DismissPatternAlertDto,
} from "./dto/pattern-alert.dto";

/**
 * PatternAlertController provides REST API for pattern alert management.
 *
 * Endpoints:
 * GET /api/pattern-alerts - List alerts with filters
 * GET /api/pattern-alerts/:id - Get single alert
 * POST /api/pattern-alerts/:id/acknowledge - Mark alert as acknowledged
 * POST /api/pattern-alerts/:id/dismiss - Dismiss alert
 */
@Controller("api/pattern-alerts")
@UseGuards(JwtAuthGuard, TenantGuard, RolesGuard)
export class PatternAlertController {
  constructor(private readonly alertService: PatternAlertService) {}

  /**
   * List pattern alerts with optional filters.
   */
  @Get()
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN", "CCO")
  async findMany(
    @TenantId() organizationId: string,
    @Query() filters: PatternAlertFilterDto,
  ) {
    return this.alertService.findMany(organizationId, filters);
  }

  /**
   * Get single alert by ID.
   */
  @Get(":id")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN", "CCO")
  async findOne(
    @TenantId() organizationId: string,
    @Param("id") alertId: string,
  ) {
    return this.alertService.findOne(organizationId, alertId);
  }

  /**
   * Get count of active alerts for badge display.
   */
  @Get("count/active")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN", "CCO")
  async getActiveCount(@TenantId() organizationId: string) {
    const count = await this.alertService.getActiveCount(organizationId);
    return { count };
  }

  /**
   * Acknowledge an alert (marks as reviewed).
   */
  @Post(":id/acknowledge")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN", "CCO")
  async acknowledge(
    @TenantId() organizationId: string,
    @Param("id") alertId: string,
    @CurrentUser("id") userId: string,
  ) {
    return this.alertService.acknowledge(organizationId, alertId, userId);
  }

  /**
   * Dismiss an alert (no longer relevant).
   */
  @Post(":id/dismiss")
  @Roles("COMPLIANCE_OFFICER", "SYSTEM_ADMIN", "CCO")
  async dismiss(
    @TenantId() organizationId: string,
    @Param("id") alertId: string,
    @CurrentUser("id") userId: string,
    @Body() dto: DismissPatternAlertDto,
  ) {
    return this.alertService.dismiss(organizationId, alertId, userId, dto);
  }
}
```

```typescript
"use client";

import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { useSession } from "next-auth/react";
import { toast } from "sonner";

export type PatternAlertStatus = "ACTIVE" | "ACKNOWLEDGED" | "DISMISSED";
export type PatternType = "REPEAT_SUBJECT" | "VELOCITY" | "CATEGORY_SPIKE";
export type PatternAlertSeverity = "LOW" | "MEDIUM" | "HIGH";

export interface PatternAlert {
  id: string;
  patternType: PatternType;
  entityType: string;
  entityId: string;
  entityName: string | null;
  threshold: number;
  actualCount: number;
  caseIds: string[];
  severity: PatternAlertSeverity;
  status: PatternAlertStatus;
  acknowledgedAt: string | null;
  acknowledgedBy: { id: string; firstName: string; lastName: string } | null;
  dismissedAt: string | null;
  dismissedBy: { id: string; firstName: string; lastName: string } | null;
  dismissReason: string | null;
  createdAt: string;
}

interface PatternAlertsResponse {
  alerts: PatternAlert[];
  total: number;
}

interface FilterParams {
  status?: PatternAlertStatus;
  patternType?: PatternType;
  limit?: number;
  offset?: number;
}

async function fetchAlerts(
  params: FilterParams = {},
): Promise<PatternAlertsResponse> {
  const searchParams = new URLSearchParams();
  if (params.status) searchParams.set("status", params.status);
  if (params.patternType) searchParams.set("patternType", params.patternType);
  if (params.limit) searchParams.set("limit", params.limit.toString());
  if (params.offset) searchParams.set("offset", params.offset.toString());

  const response = await fetch(`/api/pattern-alerts?${searchParams}`, {
    credentials: "include",
  });

  if (!response.ok) throw new Error("Failed to fetch pattern alerts");
  return response.json();
}

async function acknowledgeAlert(alertId: string): Promise<PatternAlert> {
  const response = await fetch(`/api/pattern-alerts/${alertId}/acknowledge`, {
    method: "POST",
    credentials: "include",
  });

  if (!response.ok) throw new Error("Failed to acknowledge alert");
  return response.json();
}

async function dismissAlert(
  alertId: string,
  reason?: string,
): Promise<PatternAlert> {
  const response = await fetch(`/api/pattern-alerts/${alertId}/dismiss`, {
    method: "POST",
    credentials: "include",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ reason }),
  });

  if (!response.ok) throw new Error("Failed to dismiss alert");
  return response.json();
}

async function fetchActiveCount(): Promise<number> {
  const response = await fetch("/api/pattern-alerts/count/active", {
    credentials: "include",
  });
  if (!response.ok) return 0;
  const data = await response.json();
  return data.count;
}

/**
 * Hook for fetching pattern alerts.
 */
export function usePatternAlerts(params: FilterParams = {}) {
  const { data: session } = useSession();

  return useQuery({
    queryKey: ["patternAlerts", params],
    queryFn: () => fetchAlerts(params),
    enabled: !!session,
  });
}

/**
 * Hook for acknowledging an alert.
 */
export function useAcknowledgeAlert() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: acknowledgeAlert,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["patternAlerts"] });
      toast.success("Alert acknowledged");
    },
    onError: () => {
      toast.error("Failed to acknowledge alert");
    },
  });
}

/**
 * Hook for dismissing an alert.
 */
export function useDismissAlert() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: ({ alertId, reason }: { alertId: string; reason?: string }) =>
      dismissAlert(alertId, reason),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["patternAlerts"] });
      toast.success("Alert dismissed");
    },
    onError: () => {
      toast.error("Failed to dismiss alert");
    },
  });
}

/**
 * Hook for getting active alert count (for badge display).
 */
export function useActiveAlertCount() {
  const { data: session } = useSession();

  return useQuery({
    queryKey: ["patternAlerts", "activeCount"],
    queryFn: fetchActiveCount,
    enabled: !!session,
    staleTime: 60 * 1000, // 1 minute
  });
}
```

Create table component for displaying and managing pattern alerts:

```typescript
'use client';

import { useState } from 'react';
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@/components/ui/table';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
} from '@/components/ui/dropdown-menu';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogFooter,
} from '@/components/ui/dialog';
import { Textarea } from '@/components/ui/textarea';
import { MoreHorizontal, Check, X, ExternalLink, User } from 'lucide-react';
import { formatDistanceToNow } from 'date-fns';
import Link from 'next/link';
import {
  PatternAlert,
  useAcknowledgeAlert,
  useDismissAlert,
} from '@/hooks/usePatternAlerts';

interface PatternAlertTableProps {
  alerts: PatternAlert[];
  isLoading?: boolean;
}

export function PatternAlertTable({ alerts, isLoading }: PatternAlertTableProps) {
  const [dismissDialogOpen, setDismissDialogOpen] = useState(false);
  const [selectedAlert, setSelectedAlert] = useState<PatternAlert | null>(null);
  const [dismissReason, setDismissReason] = useState('');

  const acknowledgeMutation = useAcknowledgeAlert();
  const dismissMutation = useDismissAlert();

  const handleAcknowledge = (alert: PatternAlert) => {
    acknowledgeMutation.mutate(alert.id);
  };

  const handleDismissClick = (alert: PatternAlert) => {
    setSelectedAlert(alert);
    setDismissReason('');
    setDismissDialogOpen(true);
  };

  const handleDismissConfirm = () => {
    if (selectedAlert) {
      dismissMutation.mutate(
        { alertId: selectedAlert.id, reason: dismissReason || undefined },
        { onSuccess: () => setDismissDialogOpen(false) },
      );
    }
  };

  if (isLoading) {
    return <div className="p-4 text-center text-muted-foreground">Loading alerts...</div>;
  }

  if (alerts.length === 0) {
    return (
      <div className="p-8 text-center text-muted-foreground">
        No pattern alerts found
      </div>
    );
  }

  return (
    <>
      <Table>
        <TableHeader>
          <TableRow>
            <TableHead>Pattern</TableHead>
            <TableHead>Entity</TableHead>
            <TableHead>Count</TableHead>
            <TableHead>Severity</TableHead>
            <TableHead>Status</TableHead>
            <TableHead>Detected</TableHead>
            <TableHead className="w-[100px]">Actions</TableHead>
          </TableRow>
        </TableHeader>
        <TableBody>
          {alerts.map((alert) => (
            <TableRow key={alert.id}>
              <TableCell>
                <Badge variant="outline" className="font-mono text-xs">
                  {formatPatternType(alert.patternType)}
                </Badge>
              </TableCell>
              <TableCell>
                <div className="flex items-center gap-2">
                  <User className="h-4 w-4 text-muted-foreground" />
                  <span>{alert.entityName || alert.entityId}</span>
                </div>
              </TableCell>
              <TableCell>
                <span className="font-medium">{alert.actualCount}</span>
                <span className="text-muted-foreground"> / {alert.threshold} threshold</span>
              </TableCell>
              <TableCell>
                <SeverityBadge severity={alert.severity} />
              </TableCell>
              <TableCell>
                <StatusBadge status={alert.status} />
              </TableCell>
              <TableCell className="text-muted-foreground text-sm">
                {formatDistanceToNow(new Date(alert.createdAt), { addSuffix: true })}
              </TableCell>
              <TableCell>
                <DropdownMenu>
                  <DropdownMenuTrigger asChild>
                    <Button variant="ghost" size="sm">
                      <MoreHorizontal className="h-4 w-4" />
                    </Button>
                  </DropdownMenuTrigger>
                  <DropdownMenuContent align="end">
                    <DropdownMenuItem asChild>
                      <Link href={`/cases?subjectId=${alert.entityId}`}>
                        <ExternalLink className="h-4 w-4 mr-2" />
                        View Cases
                      </Link>
                    </DropdownMenuItem>
                    {alert.status === 'ACTIVE' && (
                      <>
                        <DropdownMenuItem onClick={() => handleAcknowledge(alert)}>
                          <Check className="h-4 w-4 mr-2" />
                          Acknowledge
                        </DropdownMenuItem>
                        <DropdownMenuItem onClick={() => handleDismissClick(alert)}>
                          <X className="h-4 w-4 mr-2" />
                          Dismiss
                        </DropdownMenuItem>
                      </>
                    )}
                  </DropdownMenuContent>
                </DropdownMenu>
              </TableCell>
            </TableRow>
          ))}
        </TableBody>
      </Table>

      <Dialog open={dismissDialogOpen} onOpenChange={setDismissDialogOpen}>
        <DialogContent>
          <DialogHeader>
            <DialogTitle>Dismiss Alert</DialogTitle>
          </DialogHeader>
          <div className="py-4">
            <p className="text-sm text-muted-foreground mb-4">
              Dismissing this alert will remove it from your active alerts.
              Optionally provide a reason for dismissal.
            </p>
            <Textarea
              placeholder="Reason for dismissal (optional)"
              value={dismissReason}
              onChange={(e) => setDismissReason(e.target.value)}
              rows={3}
            />
          </div>
          <DialogFooter>
            <Button variant="outline" onClick={() => setDismissDialogOpen(false)}>
              Cancel
            </Button>
            <Button onClick={handleDismissConfirm} disabled={dismissMutation.isPending}>
              Dismiss Alert
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </>
  );
}

function formatPatternType(type: string): string {
  return type.replace(/_/g, ' ').toLowerCase().replace(/\b\w/g, (c) => c.toUpperCase());
}

function SeverityBadge({ severity }: { severity: string }) {
  const variants: Record<string, 'default' | 'secondary' | 'destructive'> = {
    LOW: 'secondary',
    MEDIUM: 'default',
    HIGH: 'destructive',
  };
  return <Badge variant={variants[severity] || 'secondary'}>{severity}</Badge>;
}

function StatusBadge({ status }: { status: string }) {
  const variants: Record<string, 'default' | 'secondary' | 'outline'> = {
    ACTIVE: 'default',
    ACKNOWLEDGED: 'secondary',
    DISMISSED: 'outline',
  };
  return <Badge variant={variants[status] || 'secondary'}>{status}</Badge>;
}

export { PatternAlertTable };
```

```typescript
'use client';

import { useState } from 'react';
import { Tabs, TabsList, TabsTrigger, TabsContent } from '@/components/ui/tabs';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { PatternAlertTable } from '@/components/alerts/pattern-alert-table';
import { usePatternAlerts, PatternAlertStatus } from '@/hooks/usePatternAlerts';
import { AlertTriangle } from 'lucide-react';

export default function PatternAlertsPage() {
  const [statusFilter, setStatusFilter] = useState<PatternAlertStatus | undefined>(undefined);

  const { data, isLoading } = usePatternAlerts({
    status: statusFilter,
    limit: 50,
  });

  const handleTabChange = (value: string) => {
    setStatusFilter(value === 'all' ? undefined : (value as PatternAlertStatus));
  };

  return (
    <div className="container mx-auto py-6 space-y-6">
      <div className="flex items-center gap-3">
        <AlertTriangle className="h-6 w-6 text-amber-500" />
        <h1 className="text-2xl font-semibold">Pattern Alerts</h1>
      </div>

      <p className="text-muted-foreground">
        Review detected patterns across cases, including repeat subjects and unusual trends.
      </p>

      <Card>
        <CardHeader className="pb-0">
          <Tabs defaultValue="all" onValueChange={handleTabChange}>
            <TabsList>
              <TabsTrigger value="all">All</TabsTrigger>
              <TabsTrigger value="ACTIVE">Active</TabsTrigger>
              <TabsTrigger value="ACKNOWLEDGED">Acknowledged</TabsTrigger>
              <TabsTrigger value="DISMISSED">Dismissed</TabsTrigger>
            </TabsList>
          </Tabs>
        </CardHeader>
        <CardContent className="pt-4">
          <PatternAlertTable
            alerts={data?.alerts || []}
            isLoading={isLoading}
          />
        </CardContent>
      </Card>
    </div>
  );
}
```
