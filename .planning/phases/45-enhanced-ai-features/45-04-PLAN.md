import { Injectable, Logger } from "@nestjs/common";
import { OnEvent } from "@nestjs/event-emitter";
import { PrismaService } from "../../prisma/prisma.service";
import { PatternDetectionService } from "./pattern-detection.service";
import { PatternAlertService } from "./pattern-alert.service";
import {
  PatternType,
  PatternEntityType,
  PatternAlertSeverity,
} from "./dto/pattern-alert.dto";

/**
 * Event payload for case.created event.
 * Matches existing CaseCreatedEvent from case management module.
 */
interface CaseCreatedEvent {
  caseId: string;
  organizationId: string;
  createdById: string;
}

/**
 * PatternAlertListener checks for repeat subject patterns on case creation.
 *
 * Per CONTEXT.md:
 * - Uses { async: true } to avoid blocking case creation
 * - Checks existing PersonCaseAssociations, not just this case
 * - Default threshold: 3 cases (configurable per org in future)
 */
@Injectable()
export class PatternAlertListener {
  private readonly logger = new Logger(PatternAlertListener.name);
  private readonly DEFAULT_THRESHOLD = 3;

  constructor(
    private readonly prisma: PrismaService,
    private readonly patternDetection: PatternDetectionService,
    private readonly alertService: PatternAlertService,
  ) {}

  /**
   * Handle case.created event to detect repeat subject patterns.
   * Runs asynchronously to avoid blocking case creation.
   */
  @OnEvent("case.created", { async: true })
  async handleCaseCreated(event: CaseCreatedEvent): Promise<void> {
    const { caseId, organizationId } = event;

    this.logger.debug(`Checking for repeat subject patterns on case ${caseId}`);

    try {
      // Get person associations for this case with SUBJECT label
      const associations = await this.prisma.personCaseAssociation.findMany({
        where: {
          caseId,
          organizationId,
          label: "SUBJECT",
        },
        select: {
          personId: true,
          person: {
            select: {
              firstName: true,
              lastName: true,
            },
          },
        },
      });

      if (associations.length === 0) {
        this.logger.debug(`No subjects found for case ${caseId}`);
        return;
      }

      // Check each subject for repeat involvement
      for (const assoc of associations) {
        const summary = await this.patternDetection.getPersonInvolvementSummary(
          assoc.personId,
          organizationId,
        );

        // Check if total cases meets threshold
        if (summary.totalCases >= this.DEFAULT_THRESHOLD) {
          const personName = assoc.person
            ? `${assoc.person.firstName || ""} ${assoc.person.lastName || ""}`.trim() ||
              "Unknown"
            : "Unknown";

          this.logger.log(
            `Repeat subject detected: ${personName} in ${summary.totalCases} cases`,
          );

          // Get case IDs from the involvement summary if available
          // For now, include at least the triggering case
          await this.alertService.createOrUpdate(organizationId, {
            patternType: PatternType.REPEAT_SUBJECT,
            entityType: PatternEntityType.PERSON,
            entityId: assoc.personId,
            entityName: personName,
            threshold: this.DEFAULT_THRESHOLD,
            actualCount: summary.totalCases,
            caseIds: [caseId], // Include triggering case, full list added by nightly job
            severity: this.calculateSeverity(summary.totalCases),
          });
        }
      }
    } catch (error) {
      // Log but don't throw - pattern detection failure shouldn't break case creation
      this.logger.error(
        `Pattern detection failed for case ${caseId}: ${(error as Error).message}`,
        (error as Error).stack,
      );
    }
  }

  /**
   * Calculate severity based on case count.
   */
  private calculateSeverity(caseCount: number): PatternAlertSeverity {
    if (caseCount >= 10) return PatternAlertSeverity.HIGH;
    if (caseCount >= 5) return PatternAlertSeverity.MEDIUM;
    return PatternAlertSeverity.LOW;
  }
}
