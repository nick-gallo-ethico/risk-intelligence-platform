```prisma
/// PatternAlert stores detected cross-case patterns for compliance review.
/// Part of AIEX-02: Pattern detection alerts for repeat subjects.
///
/// Pattern types:
/// - REPEAT_SUBJECT: Same person appears as subject in N+ cases
/// - VELOCITY: Same person involved in N+ cases within time window
/// - CATEGORY_SPIKE: Unusual increase in category volume
///
/// Alert lifecycle: ACTIVE -> ACKNOWLEDGED -> DISMISSED
/// Deduplication: Unique on (organizationId, patternType, entityId) to prevent spam
model PatternAlert {
  id               String       @id @default(uuid())
  organizationId   String       @map("organization_id")
  patternType      String       @map("pattern_type")  // REPEAT_SUBJECT, VELOCITY, CATEGORY_SPIKE
  entityType       String       @map("entity_type")   // PERSON, CATEGORY, LOCATION
  entityId         String       @map("entity_id")
  entityName       String?      @map("entity_name")   // Cached for display
  threshold        Int          // Minimum count that triggered alert
  actualCount      Int          @map("actual_count")
  caseIds          String[]     @map("case_ids")      // Cases involved in pattern
  severity         String       @default("MEDIUM")    // LOW, MEDIUM, HIGH
  status           String       @default("ACTIVE")    // ACTIVE, ACKNOWLEDGED, DISMISSED
  acknowledgedAt   DateTime?    @map("acknowledged_at")
  acknowledgedById String?      @map("acknowledged_by_id")
  dismissedAt      DateTime?    @map("dismissed_at")
  dismissedById    String?      @map("dismissed_by_id")
  dismissReason    String?      @map("dismiss_reason")
  createdAt        DateTime     @default(now()) @map("created_at")
  updatedAt        DateTime     @updatedAt @map("updated_at")

  organization     Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
  acknowledgedBy   User?        @relation("PatternAlertAcknowledger", fields: [acknowledgedById], references: [id])
  dismissedBy      User?        @relation("PatternAlertDismisser", fields: [dismissedById], references: [id])

  @@unique([organizationId, patternType, entityId])
  @@index([organizationId, status])
  @@index([organizationId, patternType])
  @@index([organizationId, entityType])
  @@index([organizationId, createdAt])
  @@map("pattern_alerts")
}
```

```typescript
  REPEAT_SUBJECT = "REPEAT_SUBJECT",
  VELOCITY = "VELOCITY",
  CATEGORY_SPIKE = "CATEGORY_SPIKE",
}

export enum PatternEntityType {
  PERSON = "PERSON",
  CATEGORY = "CATEGORY",
  LOCATION = "LOCATION",
}

export enum PatternAlertSeverity {
  LOW = "LOW",
  MEDIUM = "MEDIUM",
  HIGH = "HIGH",
}

export enum PatternAlertStatus {
  ACTIVE = "ACTIVE",
  ACKNOWLEDGED = "ACKNOWLEDGED",
  DISMISSED = "DISMISSED",
}

export class CreatePatternAlertDto {
  @IsEnum(PatternType)
  patternType: PatternType;

  @IsEnum(PatternEntityType)
  entityType: PatternEntityType;

  @IsString()
  entityId: string;

  @IsString()
  @IsOptional()
  entityName?: string;

  @IsInt()
  @Min(1)
  threshold: number;

  @IsInt()
  @Min(1)
  actualCount: number;

  @IsArray()
  @IsString({ each: true })
  caseIds: string[];

  @IsEnum(PatternAlertSeverity)
  @IsOptional()
  severity?: PatternAlertSeverity;
}

  // No additional fields needed - userId comes from request context

  @IsString()
```

Create service for managing pattern alerts with deduplication:

```typescript
import { Injectable, Logger } from "@nestjs/common";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { PrismaService } from "../../prisma/prisma.service";
import {
  CreatePatternAlertDto,
  DismissPatternAlertDto,
  PatternAlertFilterDto,
  PatternAlertStatus,
  PatternAlertSeverity,
} from "./dto/pattern-alert.dto";

@Injectable()
export class PatternAlertService {
  private readonly logger = new Logger(PatternAlertService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  /**
   * Create or update a pattern alert (upsert for deduplication).
   * If alert exists for same org+type+entity, updates count and caseIds.
   */
  async createOrUpdate(
    organizationId: string,
    dto: CreatePatternAlertDto,
  ): Promise<{ alert: any; isNew: boolean }> {
    const existingAlert = await this.prisma.patternAlert.findUnique({
      where: {
        organizationId_patternType_entityId: {
          organizationId,
          patternType: dto.patternType,
          entityId: dto.entityId,
        },
      },
    });

    if (existingAlert) {
      // Update existing alert with new count and cases
      const updatedCaseIds = Array.from(
        new Set([...existingAlert.caseIds, ...dto.caseIds]),
      );

      const alert = await this.prisma.patternAlert.update({
        where: { id: existingAlert.id },
        data: {
          actualCount: dto.actualCount,
          caseIds: updatedCaseIds,
          // Re-activate if dismissed but pattern persists
          status:
            existingAlert.status === PatternAlertStatus.DISMISSED
              ? PatternAlertStatus.ACTIVE
              : existingAlert.status,
        },
      });

      this.logger.log(
        `Updated pattern alert ${alert.id}: ${dto.patternType} for ${dto.entityId}`,
      );

      return { alert, isNew: false };
    }

    // Create new alert
    const alert = await this.prisma.patternAlert.create({
      data: {
        organizationId,
        patternType: dto.patternType,
        entityType: dto.entityType,
        entityId: dto.entityId,
        entityName: dto.entityName,
        threshold: dto.threshold,
        actualCount: dto.actualCount,
        caseIds: dto.caseIds,
        severity: dto.severity || PatternAlertSeverity.MEDIUM,
        status: PatternAlertStatus.ACTIVE,
      },
    });

    this.logger.log(
      `Created pattern alert ${alert.id}: ${dto.patternType} for ${dto.entityId}`,
    );

    // Emit event for notification delivery
    this.eventEmitter.emit("pattern_alert.created", {
      organizationId,
      alert,
    });

    return { alert, isNew: true };
  }

  /**
   * Acknowledge an alert (marks as reviewed but not dismissed).
   */
  async acknowledge(
    organizationId: string,
    alertId: string,
    userId: string,
  ): Promise<any> {
    const alert = await this.prisma.patternAlert.update({
      where: {
        id: alertId,
        organizationId, // Tenant isolation
      },
      data: {
        status: PatternAlertStatus.ACKNOWLEDGED,
        acknowledgedAt: new Date(),
        acknowledgedById: userId,
      },
    });

    this.logger.log(`Acknowledged pattern alert ${alertId} by user ${userId}`);
    return alert;
  }

  /**
   * Dismiss an alert (no longer relevant/actionable).
   */
  async dismiss(
    organizationId: string,
    alertId: string,
    userId: string,
    dto: DismissPatternAlertDto,
  ): Promise<any> {
    const alert = await this.prisma.patternAlert.update({
      where: {
        id: alertId,
        organizationId, // Tenant isolation
      },
      data: {
        status: PatternAlertStatus.DISMISSED,
        dismissedAt: new Date(),
        dismissedById: userId,
        dismissReason: dto.reason,
      },
    });

    this.logger.log(`Dismissed pattern alert ${alertId} by user ${userId}`);
    return alert;
  }

  /**
   * Get alerts for an organization with optional filters.
   */
  async findMany(
    organizationId: string,
    filters: PatternAlertFilterDto = {},
  ): Promise<{ alerts: any[]; total: number }> {
    const { status, patternType, limit = 50, offset = 0 } = filters;

    const where: any = { organizationId };
    if (status) where.status = status;
    if (patternType) where.patternType = patternType;

    const [alerts, total] = await Promise.all([
      this.prisma.patternAlert.findMany({
        where,
        orderBy: { createdAt: "desc" },
        take: limit,
        skip: offset,
        include: {
          acknowledgedBy: {
            select: { id: true, firstName: true, lastName: true },
          },
          dismissedBy: {
            select: { id: true, firstName: true, lastName: true },
          },
        },
      }),
      this.prisma.patternAlert.count({ where }),
    ]);

    return { alerts, total };
  }

  /**
   * Get single alert by ID.
   */
  async findOne(organizationId: string, alertId: string): Promise<any | null> {
    return this.prisma.patternAlert.findFirst({
      where: { id: alertId, organizationId },
      include: {
        acknowledgedBy: {
          select: { id: true, firstName: true, lastName: true },
        },
        dismissedBy: { select: { id: true, firstName: true, lastName: true } },
      },
    });
  }

  /**
   * Get count of active alerts for dashboard display.
   */
  async getActiveCount(organizationId: string): Promise<number> {
    return this.prisma.patternAlert.count({
      where: {
        organizationId,
        status: PatternAlertStatus.ACTIVE,
      },
    });
  }
}
```
