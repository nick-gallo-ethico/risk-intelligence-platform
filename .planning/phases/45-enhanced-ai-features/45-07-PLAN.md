```typescript
"use client";

import { useQuery } from "@tanstack/react-query";
import { useSession } from "next-auth/react";

export type TrendDimension =
  | "CATEGORY"
  | "LOCATION"
  | "BUSINESS_UNIT"
  | "SOURCE_CHANNEL"
  | "SEVERITY";
export type TrendDirection = "up" | "down" | "stable";
export type TrendSignificance = "notable" | "significant" | "critical";

export interface TrendResult {
  dimensionValue: string;
  dimensionLabel: string;
  dimensionId?: string;
  currentPeriodCount: number;
  previousPeriodCount: number;
  percentChange: number;
  direction: TrendDirection;
  significance: TrendSignificance;
}

export interface TrendAnalysisResponse {
  dimension: TrendDimension;
  periodDays: number;
  periodStart: string;
  periodEnd: string;
  previousPeriodStart: string;
  previousPeriodEnd: string;
  trends: TrendResult[];
  totalCurrentPeriod: number;
  totalPreviousPeriod: number;
  overallChange: number;
}

interface TrendAnalysisParams {
  dimension: TrendDimension;
  periodDays?: number;
  minChangePercent?: number;
  limit?: number;
}

async function fetchTrends(
  params: TrendAnalysisParams,
): Promise<TrendAnalysisResponse> {
  const searchParams = new URLSearchParams({
    dimension: params.dimension,
    ...(params.periodDays && { periodDays: params.periodDays.toString() }),
    ...(params.minChangePercent && {
      minChangePercent: params.minChangePercent.toString(),
    }),
    ...(params.limit && { limit: params.limit.toString() }),
  });

  const response = await fetch(`/api/analytics/trends?${searchParams}`, {
    credentials: "include",
  });

  if (!response.ok) {
    throw new Error("Failed to fetch trends");
  }

  return response.json();
}

/**
 * Hook for fetching trend analysis data.
 *
 * @param params - Analysis parameters (dimension, period, etc.)
 * @param options - React Query options
 */
export function useTrendAnalysis(
  params: TrendAnalysisParams,
  options?: { enabled?: boolean },
) {
  const { data: session } = useSession();

  return useQuery({
    queryKey: ["trends", params],
    queryFn: () => fetchTrends(params),
    enabled: options?.enabled !== false && !!session,
    staleTime: 5 * 60 * 1000, // 5 minutes
    refetchOnWindowFocus: false,
  });
}
```

```typescript
'use client';

import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { Skeleton } from '@/components/ui/skeleton';
import { TrendingUp, TrendingDown, Minus, AlertTriangle, ExternalLink } from 'lucide-react';
import { cn } from '@/lib/utils';
import { useTrendAnalysis, TrendResult, TrendDirection, TrendDimension } from '@/hooks/useTrendAnalysis';
import Link from 'next/link';

interface TrendWidgetProps {
  title?: string;
  dimension?: TrendDimension;
  periodDays?: number;
  limit?: number;
  className?: string;
}

/**
 * Dashboard widget showing statistical trends.
 *
 * Displays top changes by category/location with visual indicators.
 * Clicking a trend navigates to filtered case list.
 */
export function TrendWidget({
  title = 'Category Trends',
  dimension = 'CATEGORY',
  periodDays = 30,
  limit = 5,
  className,
}: TrendWidgetProps) {
  const { data, isLoading, error } = useTrendAnalysis({
    dimension,
    periodDays,
    minChangePercent: 25,
    limit,
  });

  if (isLoading) {
    return (
      <Card className={className}>
        <CardHeader>
          <CardTitle className="text-sm font-medium">{title}</CardTitle>
        </CardHeader>
        <CardContent>
          <div className="space-y-3">
            {[1, 2, 3].map((i) => (
              <Skeleton key={i} className="h-12 w-full" />
            ))}
          </div>
        </CardContent>
      </Card>
    );
  }

  if (error) {
    return (
      <Card className={className}>
        <CardHeader>
          <CardTitle className="text-sm font-medium">{title}</CardTitle>
        </CardHeader>
        <CardContent>
          <p className="text-sm text-muted-foreground">Failed to load trends</p>
        </CardContent>
      </Card>
    );
  }

  const trends = data?.trends || [];

  return (
    <Card className={className}>
      <CardHeader className="pb-2">
        <div className="flex items-center justify-between">
          <CardTitle className="text-sm font-medium">{title}</CardTitle>
          <span className="text-xs text-muted-foreground">
            Last {periodDays} days vs prior
          </span>
        </div>
      </CardHeader>
      <CardContent>
        {trends.length === 0 ? (
          <p className="text-sm text-muted-foreground py-4 text-center">
            No significant changes detected
          </p>
        ) : (
          <div className="space-y-2">
            {trends.map((trend) => (
              <TrendItem key={trend.dimensionValue} trend={trend} dimension={dimension} />
            ))}
          </div>
        )}
      </CardContent>
    </Card>
  );
}

interface TrendItemProps {
  trend: TrendResult;
  dimension: TrendDimension;
}

function TrendItem({ trend, dimension }: TrendItemProps) {
  const DirectionIcon = getDirectionIcon(trend.direction);
  const directionColor = getDirectionColor(trend.direction);

  // Build filter URL for case list
  const filterParam = dimension === 'CATEGORY'
    ? `categoryId=${trend.dimensionId}`
    : `${dimension.toLowerCase()}=${trend.dimensionValue}`;

  return (
    <Link
      href={`/cases?${filterParam}`}
      className="flex items-center justify-between p-2 rounded-lg hover:bg-muted/50 transition-colors group"
    >
      <div className="flex items-center gap-2 flex-1 min-w-0">
        <DirectionIcon className={cn('h-4 w-4 shrink-0', directionColor)} />
        <span className="text-sm truncate">{trend.dimensionLabel}</span>
        {trend.significance === 'critical' && (
          <AlertTriangle className="h-3.5 w-3.5 text-destructive shrink-0" />
        )}
      </div>
      <div className="flex items-center gap-2 shrink-0">
        <Badge
          variant={trend.direction === 'up' ? 'destructive' : trend.direction === 'down' ? 'secondary' : 'outline'}
          className="text-xs"
        >
          {trend.direction === 'up' ? '+' : ''}{trend.percentChange}%
        </Badge>
        <span className="text-xs text-muted-foreground">
          {trend.previousPeriodCount} → {trend.currentPeriodCount}
        </span>
        <ExternalLink className="h-3 w-3 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity" />
      </div>
    </Link>
  );
}

function getDirectionIcon(direction: TrendDirection) {
  switch (direction) {
    case 'up':
      return TrendingUp;
    case 'down':
      return TrendingDown;
    default:
      return Minus;
  }
}

function getDirectionColor(direction: TrendDirection): string {
  switch (direction) {
    case 'up':
      return 'text-red-500 dark:text-red-400';
    case 'down':
      return 'text-green-500 dark:text-green-400';
    default:
      return 'text-muted-foreground';
  }
}

export { TrendWidget };
```
