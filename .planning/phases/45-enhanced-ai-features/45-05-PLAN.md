```typescript
import { Engine, Almanac } from "json-rules-engine";
import { PatternDetectionService } from "../../../associations/pattern-detection/pattern-detection.service";

/**
 * Register pattern-related operators with json-rules-engine.
 *
 * Operators:
 * - repeatSubjectCountGte: Check if person appears as subject in N+ cases
 * - repeatInvolvementGte: Check if person appears in any role in N+ cases
 *
 * These operators are async and query the PatternDetectionService.
 *
 * @param engine - json-rules-engine instance
 * @param patternDetection - PatternDetectionService for queries
 */
export function registerPatternOperators(
  engine: Engine,
  patternDetection: PatternDetectionService,
): void {
  /**
   * repeatSubjectCountGte operator
   *
   * Usage in rule condition:
   * {
   *   fact: 'subjectPersonIds',
   *   operator: 'repeatSubjectCountGte',
   *   value: 5,
   *   params: { withinDays: 90 }  // Optional, for future time-windowed queries
   * }
   *
   * factValue: Array of person IDs from case subjects
   * jsonValue: Threshold count (e.g., 5)
   */
  engine.addOperator(
    "repeatSubjectCountGte",
    async (factValue, jsonValue, almanac: Almanac) => {
      // factValue is array of person IDs
      if (!Array.isArray(factValue) || factValue.length === 0) {
        return false;
      }

      const organizationId = (await almanac.factValue(
        "organizationId",
      )) as string;
      if (!organizationId) {
        return false;
      }

      // Check each subject for repeat involvement
      for (const personId of factValue) {
        try {
          const summary = await patternDetection.getPersonInvolvementSummary(
            personId,
            organizationId,
          );

          // Check subject role specifically
          const subjectRole = summary.byRole.find((r) => r.label === "SUBJECT");
          if (subjectRole && subjectRole.count >= jsonValue) {
            return true;
          }
        } catch {
          // Continue checking other subjects if one fails
        }
      }

      return false;
    },
  );

  /**
   * repeatInvolvementGte operator
   *
   * Similar to repeatSubjectCountGte but checks total involvement across all roles.
   */
  engine.addOperator(
    "repeatInvolvementGte",
    async (factValue, jsonValue, almanac: Almanac) => {
      if (!Array.isArray(factValue) || factValue.length === 0) {
        return false;
      }

      const organizationId = (await almanac.factValue(
        "organizationId",
      )) as string;
      if (!organizationId) {
        return false;
      }

      for (const personId of factValue) {
        try {
          const summary = await patternDetection.getPersonInvolvementSummary(
            personId,
            organizationId,
          );

          if (summary.totalCases >= jsonValue) {
            return true;
          }
        } catch {
          // Continue checking other persons
        }
      }

      return false;
    },
  );
}
```

```typescript
import { Injectable, Logger } from "@nestjs/common";
import { EventEmitter2 } from "@nestjs/event-emitter";
import { PrismaService } from "../../../prisma/prisma.service";
import { ActionExecutor, ActionContext, ActionResult } from "./action.types";

/**
 * Parameters for escalate action.
 */
export interface EscalateActionParams {
  /** Escalation level (e.g., 'CCO', 'LEGAL', 'EXECUTIVE') */
  level: string;
  /** Reason for escalation (logged to activity) */
  reason: string;
  /** Optional: Assign to specific user ID */
  assignToUserId?: string;
  /** Optional: Assign to specific team ID */
  assignToTeamId?: string;
  /** Optional: Increase priority by this amount (1-2) */
  priorityIncrease?: number;
}

/**
 * EscalateAction handles case escalation triggered by rules.
 *
 * Actions performed:
 * 1. Update case priority (if priorityIncrease specified)
 * 2. Assign to escalation user/team (if specified)
 * 3. Create escalation activity log entry
 * 4. Emit case.escalated event for notifications
 */
@Injectable()
export class EscalateAction implements ActionExecutor<EscalateActionParams> {
  readonly type = "escalate";
  private readonly logger = new Logger(EscalateAction.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  async execute(
    params: EscalateActionParams,
    context: ActionContext,
  ): Promise<ActionResult> {
    const { caseId, organizationId, userId } = context;

    if (!caseId) {
      return {
        success: false,
        error: "Escalate action requires caseId in context",
      };
    }

    this.logger.log(
      `Escalating case ${caseId} to level ${params.level}: ${params.reason}`,
    );

    try {
      // Build update data
      const updateData: Record<string, unknown> = {};

      if (params.assignToUserId) {
        updateData.assignedToId = params.assignToUserId;
      }

      if (params.priorityIncrease) {
        // Get current case to calculate new priority
        const currentCase = await this.prisma.case.findUnique({
          where: { id: caseId },
          select: { priority: true },
        });

        if (currentCase) {
          const currentPriority = currentCase.priority || 0;
          updateData.priority = Math.min(
            currentPriority + params.priorityIncrease,
            5,
          );
        }
      }

      // Update case if there are changes
      if (Object.keys(updateData).length > 0) {
        await this.prisma.case.update({
          where: { id: caseId, organizationId },
          data: updateData,
        });
      }

      // Create activity log entry
      await this.prisma.activity.create({
        data: {
          organizationId,
          entityType: "CASE",
          entityId: caseId,
          action: "ESCALATED",
          actionDescription: `Case escalated to ${params.level}: ${params.reason}`,
          metadata: {
            escalationLevel: params.level,
            reason: params.reason,
            triggeredBy: "rule_engine",
          },
          actorUserId: userId,
        },
      });

      // Emit event for notification delivery
      this.eventEmitter.emit("case.escalated", {
        caseId,
        organizationId,
        escalationLevel: params.level,
        reason: params.reason,
        assignedToUserId: params.assignToUserId,
      });

      return {
        success: true,
        message: `Case escalated to ${params.level}`,
        data: {
          caseId,
          escalationLevel: params.level,
          priorityIncrease: params.priorityIncrease,
        },
      };
    } catch (error) {
      this.logger.error(
        `Escalation failed for case ${caseId}: ${(error as Error).message}`,
      );
      return {
        success: false,
        error: `Escalation failed: ${(error as Error).message}`,
      };
    }
  }
}
```
