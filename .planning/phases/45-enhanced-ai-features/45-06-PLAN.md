```typescript
import { IsInt, IsEnum, IsOptional, Min, Max } from "class-validator";

/**
 * Dimension to analyze trends by.
 */
export enum TrendDimension {
  CATEGORY = "CATEGORY",
  LOCATION = "LOCATION",
  BUSINESS_UNIT = "BUSINESS_UNIT",
  SOURCE_CHANNEL = "SOURCE_CHANNEL",
  SEVERITY = "SEVERITY",
}

/**
 * Direction of trend change.
 */
export enum TrendDirection {
  UP = "up",
  DOWN = "down",
  STABLE = "stable",
}

/**
 * Significance level based on change magnitude.
 */
export enum TrendSignificance {
  NOTABLE = "notable", // 25-50% change
  SIGNIFICANT = "significant", // 50-100% change
  CRITICAL = "critical", // >100% change
}

/**
 * Parameters for trend analysis query.
 */
export class TrendAnalysisParams {
  @IsEnum(TrendDimension)
  dimension: TrendDimension;

  @IsInt()
  @Min(7)
  @Max(365)
  @IsOptional()
  periodDays?: number = 30;

  @IsInt()
  @Min(1)
  @Max(100)
  @IsOptional()
  minChangePercent?: number = 25;

  @IsInt()
  @Min(1)
  @Max(50)
  @IsOptional()
  limit?: number = 10;
}

/**
 * Individual trend result.
 */
export interface TrendResult {
  /** Dimension value (category name, location, etc.) */
  dimensionValue: string;
  /** Display-friendly dimension label */
  dimensionLabel: string;
  /** ID for linking (category ID, location ID, etc.) */
  dimensionId?: string;
  /** Count in current period */
  currentPeriodCount: number;
  /** Count in previous period */
  previousPeriodCount: number;
  /** Percentage change */
  percentChange: number;
  /** Direction of change */
  direction: TrendDirection;
  /** Significance level */
  significance: TrendSignificance;
}

/**
 * Full trend analysis response.
 */
export interface TrendAnalysisResponse {
  dimension: TrendDimension;
  periodDays: number;
  periodStart: Date;
  periodEnd: Date;
  previousPeriodStart: Date;
  previousPeriodEnd: Date;
  trends: TrendResult[];
  totalCurrentPeriod: number;
  totalPreviousPeriod: number;
  overallChange: number;
}
```

```typescript
import { Injectable, Logger } from "@nestjs/common";
import { PrismaService } from "../../../prisma/prisma.service";
import {
  TrendAnalysisParams,
  TrendAnalysisResponse,
  TrendResult,
  TrendDimension,
  TrendDirection,
  TrendSignificance,
} from "../dto/trend-analysis.dto";
import { subDays } from "date-fns";

/**
 * TrendAnalysisService computes period-over-period statistical trends.
 *
 * Per CONTEXT.md: This is DATABASE AGGREGATION, not AI.
 * AI is only for narrating the trends, not computing them.
 */
@Injectable()
export class TrendAnalysisService {
  private readonly logger = new Logger(TrendAnalysisService.name);

  constructor(private readonly prisma: PrismaService) {}

  /**
   * Detect trends by the specified dimension.
   *
   * Compares current period vs previous period of same length.
   * Filters to only show changes >= minChangePercent.
   */
  async detectTrends(
    organizationId: string,
    params: TrendAnalysisParams,
  ): Promise<TrendAnalysisResponse> {
    const {
      dimension,
      periodDays = 30,
      minChangePercent = 25,
      limit = 10,
    } = params;

    const now = new Date();
    const periodEnd = now;
    const periodStart = subDays(now, periodDays);
    const previousPeriodEnd = periodStart;
    const previousPeriodStart = subDays(previousPeriodEnd, periodDays);

    // Get the groupBy field based on dimension
    const groupByField = this.getGroupByField(dimension);

    // Aggregate current period
    const currentCounts = await this.aggregateByPeriod(
      organizationId,
      groupByField,
      periodStart,
      periodEnd,
    );

    // Aggregate previous period
    const previousCounts = await this.aggregateByPeriod(
      organizationId,
      groupByField,
      previousPeriodStart,
      previousPeriodEnd,
    );

    // Calculate trends
    const trends = this.calculateTrends(
      currentCounts,
      previousCounts,
      minChangePercent,
      limit,
    );

    // Calculate totals
    const totalCurrentPeriod = Object.values(currentCounts).reduce(
      (a, b) => a + b,
      0,
    );
    const totalPreviousPeriod = Object.values(previousCounts).reduce(
      (a, b) => a + b,
      0,
    );
    const overallChange = this.calculatePercentChange(
      totalCurrentPeriod,
      totalPreviousPeriod,
    );

    return {
      dimension,
      periodDays,
      periodStart,
      periodEnd,
      previousPeriodStart,
      previousPeriodEnd,
      trends,
      totalCurrentPeriod,
      totalPreviousPeriod,
      overallChange,
    };
  }

  /**
   * Get the Prisma field name for grouping by dimension.
   */
  private getGroupByField(dimension: TrendDimension): string {
    switch (dimension) {
      case TrendDimension.CATEGORY:
        return "primaryCategoryId";
      case TrendDimension.LOCATION:
        return "locationId";
      case TrendDimension.BUSINESS_UNIT:
        return "businessUnitId";
      case TrendDimension.SOURCE_CHANNEL:
        return "sourceChannel";
      case TrendDimension.SEVERITY:
        return "severity";
      default:
        return "primaryCategoryId";
    }
  }

  /**
   * Aggregate case counts by dimension for a time period.
   */
  private async aggregateByPeriod(
    organizationId: string,
    groupByField: string,
    start: Date,
    end: Date,
  ): Promise<Record<string, number>> {
    const result = await this.prisma.case.groupBy({
      by: [groupByField as "primaryCategoryId"],
      where: {
        organizationId,
        createdAt: { gte: start, lt: end },
      },
      _count: true,
    });

    const counts: Record<string, number> = {};
    for (const row of result) {
      const key = (row as Record<string, unknown>)[groupByField] as
        | string
        | null;
      if (key) {
        counts[key] = (row as { _count: number })._count;
      }
    }

    return counts;
  }

  /**
   * Calculate trend results from period counts.
   */
  private calculateTrends(
    currentCounts: Record<string, number>,
    previousCounts: Record<string, number>,
    minChangePercent: number,
    limit: number,
  ): TrendResult[] {
    const allKeys = new Set([
      ...Object.keys(currentCounts),
      ...Object.keys(previousCounts),
    ]);

    const trends: TrendResult[] = [];

    for (const key of allKeys) {
      const current = currentCounts[key] || 0;
      const previous = previousCounts[key] || 0;
      const percentChange = this.calculatePercentChange(current, previous);

      // Only include if meets minimum change threshold
      if (Math.abs(percentChange) < minChangePercent) {
        continue;
      }

      trends.push({
        dimensionValue: key,
        dimensionLabel: key, // Would be enriched with actual labels in real implementation
        dimensionId: key,
        currentPeriodCount: current,
        previousPeriodCount: previous,
        percentChange,
        direction: this.getDirection(percentChange),
        significance: this.getSignificance(Math.abs(percentChange)),
      });
    }

    // Sort by absolute change, descending
    trends.sort(
      (a, b) => Math.abs(b.percentChange) - Math.abs(a.percentChange),
    );

    return trends.slice(0, limit);
  }

  /**
   * Calculate percentage change between two values.
   */
  private calculatePercentChange(current: number, previous: number): number {
    if (previous === 0) {
      return current > 0 ? 100 : 0;
    }
    return Math.round(((current - previous) / previous) * 100);
  }

  /**
   * Determine trend direction.
   */
  private getDirection(percentChange: number): TrendDirection {
    if (percentChange > 5) return TrendDirection.UP;
    if (percentChange < -5) return TrendDirection.DOWN;
    return TrendDirection.STABLE;
  }

  /**
   * Determine trend significance.
   */
  private getSignificance(absChange: number): TrendSignificance {
    if (absChange > 100) return TrendSignificance.CRITICAL;
    if (absChange > 50) return TrendSignificance.SIGNIFICANT;
    return TrendSignificance.NOTABLE;
  }
}
```
