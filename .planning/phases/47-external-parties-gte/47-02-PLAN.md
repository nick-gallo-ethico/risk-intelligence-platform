```typescript
import { Injectable, Logger } from "@nestjs/common";
import Fuse from "fuse.js";
import { ExternalParty } from "@prisma/client";
import { PrismaService } from "../../prisma/prisma.service";
import {
  ExternalPartyMatchResultDto,
  ExternalPartyResponseDto,
} from "./dto/external-party.dto";

/**
 * ExternalPartyMatcherService provides fuzzy matching for external party deduplication.
 *
 * Uses fuse.js with configurable threshold (default 0.3 = 30% difference allowed).
 * Searches across name and aliases fields.
 */
@Injectable()
export class ExternalPartyMatcherService {
  private readonly logger = new Logger(ExternalPartyMatcherService.name);

  private readonly fuseOptions: Fuse.IFuseOptions<ExternalParty> = {
    keys: [
      { name: "name", weight: 0.7 },
      { name: "aliases", weight: 0.3 },
    ],
    threshold: 0.3, // Allow up to 30% difference
    includeScore: true,
    minMatchCharLength: 2,
    ignoreLocation: true,
  };

  constructor(private readonly prisma: PrismaService) {}

  /**
   * Finds potential matches for a search name within an organization.
   * Returns matches sorted by confidence (best first).
   */
  async findMatches(
    searchName: string,
    organizationId: string,
    limit: number = 5,
  ): Promise<ExternalPartyMatchResultDto[]> {
    // Load all external parties for the organization
    const parties = await this.prisma.externalParty.findMany({
      where: { organizationId },
      select: {
        id: true,
        organizationId: true,
        name: true,
        aliases: true,
        taxId: true,
        dunsNumber: true,
        website: true,
        partyType: true,
        isGovernment: true,
        isSanctioned: true,
        riskRating: true,
        country: true,
        state: true,
        city: true,
        notes: true,
        createdAt: true,
        updatedAt: true,
        createdById: true,
      },
    });

    if (parties.length === 0) {
      return [];
    }

    const fuse = new Fuse(parties, this.fuseOptions);
    const results = fuse.search(searchName, { limit });

    return results.map((result) => ({
      party: this.mapToResponse(result.item),
      score: result.score ?? 1,
      matchedOn: this.determineMatchField(result) as "name" | "alias" | "taxId",
    }));
  }

  /**
   * Suggests potential duplicates when creating a new external party.
   * Checks tax ID for exact match first, then falls back to fuzzy name matching.
   */
  async suggestDuplicates(
    newParty: { name: string; taxId?: string },
    organizationId: string,
  ): Promise<ExternalPartyMatchResultDto[]> {
    // Check for exact tax ID match first
    if (newParty.taxId) {
      const exactTaxMatch = await this.prisma.externalParty.findFirst({
        where: { organizationId, taxId: newParty.taxId },
      });

      if (exactTaxMatch) {
        this.logger.debug(`Found exact tax ID match for ${newParty.taxId}`);
        return [
          {
            party: this.mapToResponse(exactTaxMatch),
            score: 0, // Perfect match
            matchedOn: "taxId",
          },
        ];
      }
    }

    // Fall back to fuzzy name matching
    return this.findMatches(newParty.name, organizationId, 3);
  }

  /**
   * Determines which field the match was found on.
   */
  private determineMatchField(result: Fuse.FuseResult<ExternalParty>): string {
    if (result.matches && result.matches.length > 0) {
      const firstMatch = result.matches[0];
      if (firstMatch.key === "aliases") {
        return "alias";
      }
    }
    return "name";
  }

  /**
   * Maps Prisma model to response DTO.
   */
  private mapToResponse(party: ExternalParty): ExternalPartyResponseDto {
    return {
      id: party.id,
      organizationId: party.organizationId,
      name: party.name,
      partyType: party.partyType,
      aliases: party.aliases,
      taxId: party.taxId ?? undefined,
      dunsNumber: party.dunsNumber ?? undefined,
      website: party.website ?? undefined,
      isGovernment: party.isGovernment,
      isSanctioned: party.isSanctioned,
      riskRating: party.riskRating,
      country: party.country ?? undefined,
      state: party.state ?? undefined,
      city: party.city ?? undefined,
      notes: party.notes ?? undefined,
      createdAt: party.createdAt,
      updatedAt: party.updatedAt,
      createdById: party.createdById,
    };
  }
}
```

```typescript
import {
  Injectable,
  Logger,
  NotFoundException,
  ConflictException,
} from "@nestjs/common";
import { ExternalParty, Prisma } from "@prisma/client";
import { PrismaService } from "../../prisma/prisma.service";
import { AuditService } from "../../audit/audit.service";
import { ExternalPartyMatcherService } from "./external-party-matcher.service";
import {
  CreateExternalPartyDto,
  UpdateExternalPartyDto,
  ExternalPartyResponseDto,
  ExternalPartyMatchResultDto,
  LinkDisclosureToExternalPartyDto,
  SearchExternalPartiesDto,
} from "./dto/external-party.dto";
import { Decimal } from "@prisma/client/runtime/library";

/**
 * ExternalPartyService manages external party CRUD operations and disclosure linking.
 *
 * Features:
 * - Create/update/delete external parties with audit logging
 * - Link disclosures to external parties (DISC-09)
 * - Fuzzy matching for duplicate detection
 * - Search and filter by type, government status, risk rating
 */
@Injectable()
export class ExternalPartyService {
  private readonly logger = new Logger(ExternalPartyService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
    private readonly matcherService: ExternalPartyMatcherService,
  ) {}

  /**
   * Creates a new external party.
   * Logs warning if potential duplicates detected but does not block creation.
   */
  async create(
    dto: CreateExternalPartyDto,
    organizationId: string,
    userId: string,
  ): Promise<{
    party: ExternalPartyResponseDto;
    potentialDuplicates: ExternalPartyMatchResultDto[];
  }> {
    // Check for potential duplicates
    const potentialDuplicates = await this.matcherService.suggestDuplicates(
      { name: dto.name, taxId: dto.taxId },
      organizationId,
    );

    if (potentialDuplicates.length > 0 && potentialDuplicates[0].score === 0) {
      // Exact tax ID match - this is a real duplicate
      throw new ConflictException(
        `External party with tax ID ${dto.taxId} already exists: ${potentialDuplicates[0].party.name}`,
      );
    }

    // Check for exact name match
    const exactNameMatch = await this.prisma.externalParty.findFirst({
      where: { organizationId, name: dto.name },
    });

    if (exactNameMatch) {
      throw new ConflictException(
        `External party with name "${dto.name}" already exists`,
      );
    }

    const party = await this.prisma.externalParty.create({
      data: {
        organizationId,
        name: dto.name,
        partyType: dto.partyType,
        aliases: dto.aliases ?? [],
        taxId: dto.taxId,
        dunsNumber: dto.dunsNumber,
        website: dto.website,
        isGovernment: dto.isGovernment ?? false,
        isSanctioned: dto.isSanctioned ?? false,
        riskRating: dto.riskRating ?? "MEDIUM",
        country: dto.country,
        state: dto.state,
        city: dto.city,
        notes: dto.notes,
        createdById: userId,
      },
    });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: party.id,
      action: "external_party_created",
      actionCategory: "CREATE",
      actionDescription: `Created external party "${party.name}" (${party.partyType})`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
    });

    this.logger.log(
      `Created external party ${party.id} "${party.name}" in org ${organizationId}`,
    );

    return {
      party: this.mapToResponse(party),
      potentialDuplicates: potentialDuplicates.filter((d) => d.score > 0), // Exclude exact matches
    };
  }

  /**
   * Updates an existing external party.
   */
  async update(
    id: string,
    dto: UpdateExternalPartyDto,
    organizationId: string,
    userId: string,
  ): Promise<ExternalPartyResponseDto> {
    const existing = await this.findOneOrThrow(id, organizationId);

    // Check name uniqueness if name is being changed
    if (dto.name && dto.name !== existing.name) {
      const nameConflict = await this.prisma.externalParty.findFirst({
        where: { organizationId, name: dto.name, NOT: { id } },
      });
      if (nameConflict) {
        throw new ConflictException(
          `External party with name "${dto.name}" already exists`,
        );
      }
    }

    const party = await this.prisma.externalParty.update({
      where: { id },
      data: {
        ...(dto.name !== undefined && { name: dto.name }),
        ...(dto.partyType !== undefined && { partyType: dto.partyType }),
        ...(dto.aliases !== undefined && { aliases: dto.aliases }),
        ...(dto.taxId !== undefined && { taxId: dto.taxId }),
        ...(dto.dunsNumber !== undefined && { dunsNumber: dto.dunsNumber }),
        ...(dto.website !== undefined && { website: dto.website }),
        ...(dto.isGovernment !== undefined && {
          isGovernment: dto.isGovernment,
        }),
        ...(dto.isSanctioned !== undefined && {
          isSanctioned: dto.isSanctioned,
        }),
        ...(dto.riskRating !== undefined && { riskRating: dto.riskRating }),
        ...(dto.country !== undefined && { country: dto.country }),
        ...(dto.state !== undefined && { state: dto.state }),
        ...(dto.city !== undefined && { city: dto.city }),
        ...(dto.notes !== undefined && { notes: dto.notes }),
      },
    });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: party.id,
      action: "external_party_updated",
      actionCategory: "UPDATE",
      actionDescription: `Updated external party "${party.name}"`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
      changes: { old: existing, new: dto },
    });

    return this.mapToResponse(party);
  }

  /**
   * Deletes an external party.
   * Fails if the party has linked disclosures.
   */
  async delete(
    id: string,
    organizationId: string,
    userId: string,
  ): Promise<void> {
    const party = await this.findOneOrThrow(id, organizationId);

    // Check for linked disclosures
    const linkCount = await this.prisma.disclosureExternalPartyLink.count({
      where: { externalPartyId: id },
    });

    if (linkCount > 0) {
      throw new ConflictException(
        `Cannot delete external party "${party.name}" - it has ${linkCount} linked disclosure(s)`,
      );
    }

    await this.prisma.externalParty.delete({ where: { id } });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: id,
      action: "external_party_deleted",
      actionCategory: "DELETE",
      actionDescription: `Deleted external party "${party.name}"`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
    });

    this.logger.log(`Deleted external party ${id} "${party.name}"`);
  }

  /**
   * Finds an external party by ID.
   */
  async findOne(
    id: string,
    organizationId: string,
  ): Promise<ExternalPartyResponseDto | null> {
    const party = await this.prisma.externalParty.findFirst({
      where: { id, organizationId },
      include: {
        _count: { select: { disclosureLinks: true } },
      },
    });

    if (!party) return null;

    const response = this.mapToResponse(party);
    response.disclosureCount = party._count.disclosureLinks;
    return response;
  }

  /**
   * Searches external parties with filters.
   */
  async search(
    filters: SearchExternalPartiesDto,
    organizationId: string,
  ): Promise<{ items: ExternalPartyResponseDto[]; total: number }> {
    const where: Prisma.ExternalPartyWhereInput = {
      organizationId,
      ...(filters.partyType && { partyType: filters.partyType }),
      ...(filters.isGovernment !== undefined && {
        isGovernment: filters.isGovernment,
      }),
      ...(filters.isSanctioned !== undefined && {
        isSanctioned: filters.isSanctioned,
      }),
      ...(filters.riskRating && { riskRating: filters.riskRating }),
      ...(filters.search && {
        OR: [
          { name: { contains: filters.search, mode: "insensitive" as const } },
          { aliases: { has: filters.search } },
          { taxId: { contains: filters.search, mode: "insensitive" as const } },
        ],
      }),
    };

    const [parties, total] = await Promise.all([
      this.prisma.externalParty.findMany({
        where,
        include: { _count: { select: { disclosureLinks: true } } },
        orderBy: { name: "asc" },
        skip: filters.skip ?? 0,
        take: filters.take ?? 50,
      }),
      this.prisma.externalParty.count({ where }),
    ]);

    return {
      items: parties.map((p) => {
        const resp = this.mapToResponse(p);
        resp.disclosureCount = p._count.disclosureLinks;
        return resp;
      }),
      total,
    };
  }

  /**
   * Links a disclosure to an external party.
   */
  async linkDisclosure(
    dto: LinkDisclosureToExternalPartyDto,
    organizationId: string,
    userId: string,
  ): Promise<void> {
    // Verify external party exists
    await this.findOneOrThrow(dto.externalPartyId, organizationId);

    // Verify disclosure exists
    const disclosure = await this.prisma.riuDisclosureExtension.findFirst({
      where: { riuId: dto.disclosureId, organizationId },
    });

    if (!disclosure) {
      throw new NotFoundException(`Disclosure not found: ${dto.disclosureId}`);
    }

    // Create or update link
    await this.prisma.disclosureExternalPartyLink.upsert({
      where: {
        disclosureId_externalPartyId: {
          disclosureId: dto.disclosureId,
          externalPartyId: dto.externalPartyId,
        },
      },
      update: {
        relationship: dto.relationship,
        valueContributed: dto.valueContributed
          ? new Decimal(dto.valueContributed)
          : null,
      },
      create: {
        organizationId,
        disclosureId: dto.disclosureId,
        externalPartyId: dto.externalPartyId,
        relationship: dto.relationship,
        valueContributed: dto.valueContributed
          ? new Decimal(dto.valueContributed)
          : null,
      },
    });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: dto.disclosureId,
      action: "external_party_linked",
      actionCategory: "UPDATE",
      actionDescription: `Linked external party to disclosure as ${dto.relationship}`,
      actorUserId: userId,
      actorType: "USER",
      organizationId,
      context: { externalPartyId: dto.externalPartyId },
    });
  }

  /**
   * Unlinks a disclosure from an external party.
   */
  async unlinkDisclosure(
    disclosureId: string,
    externalPartyId: string,
    organizationId: string,
    userId: string,
  ): Promise<void> {
    const link = await this.prisma.disclosureExternalPartyLink.findFirst({
      where: { disclosureId, externalPartyId, organizationId },
    });

    if (!link) {
      throw new NotFoundException("Disclosure-external party link not found");
    }

    await this.prisma.disclosureExternalPartyLink.delete({
      where: { id: link.id },
    });

    await this.auditService.log({
      entityType: "DISCLOSURE",
      entityId: disclosureId,
      action: "external_party_unlinked",
      actionCategory: "UPDATE",
      actionDescription: "Unlinked external party from disclosure",
      actorUserId: userId,
      actorType: "USER",
      organizationId,
      context: { externalPartyId },
    });
  }

  /**
   * Finds potential matches for fuzzy searching.
   */
  async findMatches(
    searchName: string,
    organizationId: string,
    limit: number = 5,
  ): Promise<ExternalPartyMatchResultDto[]> {
    return this.matcherService.findMatches(searchName, organizationId, limit);
  }

  private async findOneOrThrow(
    id: string,
    organizationId: string,
  ): Promise<ExternalParty> {
    const party = await this.prisma.externalParty.findFirst({
      where: { id, organizationId },
    });
    if (!party) {
      throw new NotFoundException(`External party not found: ${id}`);
    }
    return party;
  }

  private mapToResponse(party: ExternalParty): ExternalPartyResponseDto {
```
