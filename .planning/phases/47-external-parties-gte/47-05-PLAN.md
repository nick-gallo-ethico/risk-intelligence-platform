```typescript
import { Injectable, Logger } from "@nestjs/common";
import { PrismaService } from "../../prisma/prisma.service";
import {
  LocationConditionDto,
  LocationConditionType,
} from "../dto/threshold-rule.dto";

/**
 * Result of location condition evaluation.
 */
export interface LocationEvaluationResult {
  matches: boolean;
  evaluatedLocations: {
    employee?: { country?: string; state?: string };
    externalParty?: {
      country?: string;
      state?: string;
      isGovernment?: boolean;
    };
  };
  reason?: string;
}

/**
 * LocationRulesService evaluates location-based threshold conditions.
 *
 * Features (DISC-12):
 * - Evaluate employee location (from Employee record)
 * - Evaluate external party location (from ExternalParty record)
 * - Check government official status
 * - Support ANY match (employee OR external party)
 */
@Injectable()
export class LocationRulesService {
  private readonly logger = new Logger(LocationRulesService.name);

  constructor(private readonly prisma: PrismaService) {}

  /**
   * Evaluates location conditions for a threshold rule.
   *
   * @param conditions - The location conditions to evaluate
   * @param employeeId - The employee making the disclosure
   * @param externalPartyId - The external party involved (optional)
   * @param organizationId - The organization ID for tenant scoping
   * @returns Whether the conditions are satisfied
   */
  async evaluateLocationConditions(
    conditions: LocationConditionDto | null,
    employeeId: string,
    externalPartyId: string | null,
    organizationId: string,
  ): Promise<LocationEvaluationResult> {
    // No location conditions = always matches
    if (!conditions) {
      return { matches: true, evaluatedLocations: {} };
    }

    const evaluatedLocations: LocationEvaluationResult["evaluatedLocations"] =
      {};

    // Evaluate based on condition type
    switch (conditions.type) {
      case LocationConditionType.EMPLOYEE_LOCATION:
        return this.evaluateEmployeeLocation(
          conditions,
          employeeId,
          organizationId,
          evaluatedLocations,
        );

      case LocationConditionType.EXTERNAL_PARTY_LOCATION:
        return this.evaluateExternalPartyLocation(
          conditions,
          externalPartyId,
          organizationId,
          evaluatedLocations,
        );

      case LocationConditionType.ANY:
        // Try employee first, then external party
        const employeeResult = await this.evaluateEmployeeLocation(
          conditions,
          employeeId,
          organizationId,
          evaluatedLocations,
        );
        if (employeeResult.matches) {
          return employeeResult;
        }

        return this.evaluateExternalPartyLocation(
          conditions,
          externalPartyId,
          organizationId,
          evaluatedLocations,
        );

      default:
        this.logger.warn(`Unknown location condition type: ${conditions.type}`);
        return { matches: true, evaluatedLocations: {} };
    }
  }

  /**
   * Evaluates conditions against employee location.
   */
  private async evaluateEmployeeLocation(
    conditions: LocationConditionDto,
    employeeId: string,
    organizationId: string,
    evaluatedLocations: LocationEvaluationResult["evaluatedLocations"],
  ): Promise<LocationEvaluationResult> {
    // Get employee location
    const employee = await this.prisma.employee.findFirst({
      where: { id: employeeId, organizationId },
      select: {
        locationCode: true,
        location: true,
      },
    });

    if (!employee) {
      this.logger.debug(`Employee ${employeeId} not found`);
      return {
        matches: false,
        evaluatedLocations,
        reason: "Employee not found",
      };
    }

    // Parse location (assumes format like "California, US" or state/country codes)
    const { country, state } = this.parseLocationString(
      employee.location,
      employee.locationCode,
    );
    evaluatedLocations.employee = { country, state };

    const locationMatches = this.matchesLocation(
      { country, state },
      conditions.countries,
      conditions.states,
    );

    if (locationMatches) {
      return { matches: true, evaluatedLocations };
    }

    return {
      matches: false,
      evaluatedLocations,
      reason: `Employee location (${state ?? country ?? "unknown"}) does not match required locations`,
    };
  }

  /**
   * Evaluates conditions against external party location.
   */
  private async evaluateExternalPartyLocation(
    conditions: LocationConditionDto,
    externalPartyId: string | null,
    organizationId: string,
    evaluatedLocations: LocationEvaluationResult["evaluatedLocations"],
  ): Promise<LocationEvaluationResult> {
    if (!externalPartyId) {
      return {
        matches: false,
        evaluatedLocations,
        reason: "No external party specified",
      };
    }

    const party = await this.prisma.externalParty.findFirst({
      where: { id: externalPartyId, organizationId },
      select: {
        country: true,
        state: true,
        isGovernment: true,
      },
    });

    if (!party) {
      this.logger.debug(`External party ${externalPartyId} not found`);
      return {
        matches: false,
        evaluatedLocations,
        reason: "External party not found",
      };
    }

    evaluatedLocations.externalParty = {
      country: party.country ?? undefined,
      state: party.state ?? undefined,
      isGovernment: party.isGovernment,
    };

    // Check government official requirement
    if (conditions.isGovernmentOfficial !== undefined) {
      if (party.isGovernment !== conditions.isGovernmentOfficial) {
        return {
          matches: false,
          evaluatedLocations,
          reason: conditions.isGovernmentOfficial
            ? "External party is not a government entity"
            : "External party is a government entity (rule excludes government)",
        };
      }
    }

    // Check location match
    const locationMatches = this.matchesLocation(
      { country: party.country ?? undefined, state: party.state ?? undefined },
      conditions.countries,
      conditions.states,
    );

    if (locationMatches) {
      return { matches: true, evaluatedLocations };
    }

    return {
      matches: false,
      evaluatedLocations,
      reason: `External party location (${party.state ?? party.country ?? "unknown"}) does not match required locations`,
    };
  }

  /**
   * Checks if an entity's location matches the required countries/states.
   */
  private matchesLocation(
    entity: { country?: string; state?: string },
    countries?: string[],
    states?: string[],
  ): boolean {
    // If no location constraints, matches everything
    if (
      (!countries || countries.length === 0) &&
      (!states || states.length === 0)
    ) {
      return true;
    }

    // Check country match
    if (countries && countries.length > 0) {
      if (
        !entity.country ||
        !countries.includes(entity.country.toUpperCase())
      ) {
        return false;
      }
    }

    // Check state match
    if (states && states.length > 0) {
      if (!entity.state || !states.includes(entity.state.toUpperCase())) {
        return false;
      }
    }

    return true;
  }

  /**
   * Parses a location string into country and state components.
   * Handles various formats:
   * - "California, US" -> { country: "US", state: "CA" }
   * - "CA" -> { state: "CA" }
   * - Location code like "US-CA" -> { country: "US", state: "CA" }
   */
  private parseLocationString(
    location?: string | null,
    locationCode?: string | null,
  ): { country?: string; state?: string } {
    // Try location code first (more reliable)
    if (locationCode) {
      if (locationCode.includes("-")) {
        const [country, state] = locationCode.split("-");
        return { country: country.toUpperCase(), state: state.toUpperCase() };
      }
      // Assume it's a state code if 2 chars
      if (locationCode.length === 2) {
        return { state: locationCode.toUpperCase() };
      }
    }

    // Try to parse location string
    if (location) {
      const parts = location.split(",").map((s) => s.trim());
      if (parts.length >= 2) {
        const state = parts[0];
        const country = parts[parts.length - 1];
        return {
          country: country.length === 2 ? country.toUpperCase() : undefined,
          state: this.stateNameToCode(state),
        };
      }
    }

    return {};
  }

  /**
   * Converts full state names to state codes (US states only for now).
   */
  private stateNameToCode(name: string): string | undefined {
    const stateMap: Record<string, string> = {
      california: "CA",
      "new york": "NY",
      texas: "TX",
      florida: "FL",
      illinois: "IL",
      // Add more as needed, or use country-state-city library for full mapping
    };

    const normalized = name.toLowerCase().trim();
    return (
      stateMap[normalized] ??
      (name.length === 2 ? name.toUpperCase() : undefined)
    );
  }
}
```
