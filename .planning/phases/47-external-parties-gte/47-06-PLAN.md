```typescript
import { apiClient } from "@/lib/api-client";

export interface ExternalParty {
  id: string;
  organizationId: string;
  name: string;
  partyType: "COMPANY" | "INDIVIDUAL" | "GOVERNMENT_ENTITY" | "NON_PROFIT";
  aliases: string[];
  taxId?: string;
  dunsNumber?: string;
  website?: string;
  isGovernment: boolean;
  isSanctioned: boolean;
  riskRating: "LOW" | "MEDIUM" | "HIGH" | "CRITICAL";
  country?: string;
  state?: string;
  city?: string;
  notes?: string;
  createdAt: string;
  updatedAt: string;
  createdById: string;
  disclosureCount?: number;
  totalGiftValue?: number;
}

export interface ExternalPartyMatch {
  party: ExternalParty;
  score: number;
  matchedOn: "name" | "alias" | "taxId";
}

export interface CreateExternalPartyRequest {
  name: string;
  partyType: ExternalParty["partyType"];
  aliases?: string[];
  taxId?: string;
  dunsNumber?: string;
  website?: string;
  isGovernment?: boolean;
  isSanctioned?: boolean;
  riskRating?: ExternalParty["riskRating"];
  country?: string;
  state?: string;
  city?: string;
  notes?: string;
}

export interface SearchExternalPartiesParams {
  search?: string;
  partyType?: ExternalParty["partyType"];
  isGovernment?: boolean;
  isSanctioned?: boolean;
  riskRating?: ExternalParty["riskRating"];
  skip?: number;
  take?: number;
}

export interface SearchExternalPartiesResponse {
  items: ExternalParty[];
  total: number;
}

export interface CreateExternalPartyResponse {
  party: ExternalParty;
  potentialDuplicates: ExternalPartyMatch[];
}

class ExternalPartiesService {
  private readonly basePath = "/api/v1/external-parties";

  async search(
    params: SearchExternalPartiesParams = {},
  ): Promise<SearchExternalPartiesResponse> {
    const queryParams = new URLSearchParams();
    if (params.search) queryParams.set("search", params.search);
    if (params.partyType) queryParams.set("partyType", params.partyType);
    if (params.isGovernment !== undefined)
      queryParams.set("isGovernment", String(params.isGovernment));
    if (params.isSanctioned !== undefined)
      queryParams.set("isSanctioned", String(params.isSanctioned));
    if (params.riskRating) queryParams.set("riskRating", params.riskRating);
    if (params.skip !== undefined) queryParams.set("skip", String(params.skip));
    if (params.take !== undefined) queryParams.set("take", String(params.take));

    const query = queryParams.toString();
    const url = query ? `${this.basePath}?${query}` : this.basePath;
    return apiClient.get<SearchExternalPartiesResponse>(url);
  }

  async getById(id: string): Promise<ExternalParty> {
    return apiClient.get<ExternalParty>(`${this.basePath}/${id}`);
  }

  async create(
    data: CreateExternalPartyRequest,
  ): Promise<CreateExternalPartyResponse> {
    return apiClient.post<CreateExternalPartyResponse>(this.basePath, data);
  }

  async update(
    id: string,
    data: Partial<CreateExternalPartyRequest>,
  ): Promise<ExternalParty> {
    return apiClient.put<ExternalParty>(`${this.basePath}/${id}`, data);
  }

  async delete(id: string): Promise<void> {
    return apiClient.delete(`${this.basePath}/${id}`);
  }

  async findMatches(
    name: string,
    limit: number = 5,
  ): Promise<ExternalPartyMatch[]> {
    const params = new URLSearchParams({ name, limit: String(limit) });
    return apiClient.get<ExternalPartyMatch[]>(
      `${this.basePath}/match?${params}`,
    );
  }

  async linkToDisclosure(
    disclosureId: string,
    externalPartyId: string,
    relationship: string,
    valueContributed?: number,
  ): Promise<void> {
    return apiClient.post(`${this.basePath}/link`, {
      disclosureId,
      externalPartyId,
      relationship,
      valueContributed,
    });
  }

  async unlinkFromDisclosure(
    disclosureId: string,
    externalPartyId: string,
  ): Promise<void> {
    return apiClient.delete(
      `${this.basePath}/link/${disclosureId}/${externalPartyId}`,
    );
  }
}

export const externalPartiesService = new ExternalPartiesService();
```

```tsx
import { useState, useEffect } from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { z } from "zod";
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogFooter,
} from "@/components/ui/dialog";
import {

  FormControl,
  FormDescription,
  FormField,
  FormItem,
  FormLabel,
  FormMessage,
} from "@/components/ui/form";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";
import { Switch } from "@/components/ui/switch";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import { Alert, AlertDescription, AlertTitle } from "@/components/ui/alert";
import { AlertTriangle, Loader2 } from "lucide-react";
import {
  ExternalParty,
  ExternalPartyMatch,
  CreateExternalPartyRequest,

import { useDebounce } from "@/hooks/use-debounce";

const formSchema = z.object({
  name: z.string().min(1, "Name is required").max(200),
  partyType: z.enum([
    "COMPANY",
    "INDIVIDUAL",
    "GOVERNMENT_ENTITY",
    "NON_PROFIT",
  ]),
  aliases: z.string().optional(),
  taxId: z.string().max(50).optional(),
  dunsNumber: z.string().max(20).optional(),
  website: z.string().url().optional().or(z.literal("")),
  isGovernment: z.boolean().default(false),
  isSanctioned: z.boolean().default(false),
  riskRating: z.enum(["LOW", "MEDIUM", "HIGH", "CRITICAL"]).default("MEDIUM"),
  country: z.string().max(2).optional(),
  state: z.string().max(50).optional(),
  city: z.string().max(100).optional(),
  notes: z.string().max(2000).optional(),
});

type FormValues = z.infer<typeof formSchema>;

interface ExternalPartyFormProps {
  open: boolean;
  onClose: () => void;
  party?: ExternalParty | null;
  onSuccess: (party: ExternalParty) => void;
}

export function ExternalPartyForm({
  open,
  onClose,
  party,
  onSuccess,
}: ExternalPartyFormProps) {
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [duplicates, setDuplicates] = useState<ExternalPartyMatch[]>([]);
  const [isCheckingDuplicates, setIsCheckingDuplicates] = useState(false);

  const isEditing = !!party;

  const form = useForm<FormValues>({
    resolver: zodResolver(formSchema),
    defaultValues: {
      name: "",
      partyType: "COMPANY",
      aliases: "",
      taxId: "",
      dunsNumber: "",
      website: "",
      isGovernment: false,
      isSanctioned: false,
      riskRating: "MEDIUM",
      country: "",
      state: "",
      city: "",
      notes: "",
    },
  });

  // Reset form when party changes
  useEffect(() => {
    if (party) {
      form.reset({
        name: party.name,
        partyType: party.partyType,
        aliases: party.aliases.join(", "),
        taxId: party.taxId ?? "",
        dunsNumber: party.dunsNumber ?? "",
        website: party.website ?? "",
        isGovernment: party.isGovernment,
        isSanctioned: party.isSanctioned,
        riskRating: party.riskRating,
        country: party.country ?? "",
        state: party.state ?? "",
        city: party.city ?? "",
        notes: party.notes ?? "",
      });
      setDuplicates([]);
    } else {
      form.reset();
      setDuplicates([]);
    }
  }, [party, form]);

  // Check for duplicates when name changes (only for new parties)
  const nameValue = form.watch("name");
  const debouncedName = useDebounce(nameValue, 500);

  useEffect(() => {
    if (!isEditing && debouncedName && debouncedName.length >= 3) {
      setIsCheckingDuplicates(true);
      externalPartiesService
        .findMatches(debouncedName, 3)
        .then(setDuplicates)
        .catch(() => setDuplicates([]))
        .finally(() => setIsCheckingDuplicates(false));
    } else {
      setDuplicates([]);
    }
  }, [debouncedName, isEditing]);

  const onSubmit = async (values: FormValues) => {
    setIsSubmitting(true);
    try {
      const data: CreateExternalPartyRequest = {
        name: values.name,
        partyType: values.partyType,
        aliases: values.aliases
          ? values.aliases
              .split(",")
              .map((a) => a.trim())
              .filter(Boolean)
          : undefined,
        taxId: values.taxId || undefined,
        dunsNumber: values.dunsNumber || undefined,
        website: values.website || undefined,
        isGovernment: values.isGovernment,
        isSanctioned: values.isSanctioned,
        riskRating: values.riskRating,
        country: values.country || undefined,
        state: values.state || undefined,
        city: values.city || undefined,
        notes: values.notes || undefined,
      };

      let result: ExternalParty;
      if (isEditing && party) {
        result = await externalPartiesService.update(party.id, data);
      } else {
        const response = await externalPartiesService.create(data);
        result = response.party;
      }

      onSuccess(result);
      onClose();
    } catch (error: any) {
      form.setError("root", {
        message: error.message || "Failed to save external party",
      });
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <Dialog open={open} onOpenChange={(isOpen) => !isOpen && onClose()}>
      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>
            {isEditing ? "Edit External Party" : "Add External Party"}
          </DialogTitle>
        </DialogHeader>

        <Form {...form}>
          <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-6">
            {/* Duplicate warning */}
            {duplicates.length > 0 && (
              <Alert variant="warning">
                <AlertTriangle className="h-4 w-4" />
                <AlertTitle>Potential duplicates found</AlertTitle>
                <AlertDescription>
                  <ul className="mt-2 space-y-1">
                    {duplicates.map((d) => (
                      <li key={d.party.id} className="text-sm">
                        <strong>{d.party.name}</strong> (
                        {d.party.partyType.toLowerCase()}){" - "}matched on{" "}
                        {d.matchedOn}
                        {d.score === 0 && " (exact match)"}
                      </li>
                    ))}
                  </ul>
                </AlertDescription>
              </Alert>
            )}

            {/* Basic Info */}
            <div className="grid grid-cols-2 gap-4">
              <FormField
                control={form.control}
                name="name"
                render={({ field }) => (
                  <FormItem className="col-span-2">
                    <FormLabel>Name *</FormLabel>
                    <FormControl>
                      <div className="relative">
                        <Input
                          placeholder="Company or individual name"
                          {...field}
                        />
                        {isCheckingDuplicates && (
                          <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 animate-spin text-muted-foreground" />
                        )}
                      </div>
                    </FormControl>
                    <FormMessage />
                  </FormItem>
                )}
              />

              <FormField
                control={form.control}
                name="partyType"
                render={({ field }) => (
                  <FormItem>
                    <FormLabel>Type *</FormLabel>
                    <Select onValueChange={field.onChange} value={field.value}>
                      <FormControl>
                        <SelectTrigger>
                          <SelectValue />
                        </SelectTrigger>
                      </FormControl>
                      <SelectContent>
                        <SelectItem value="COMPANY">Company</SelectItem>
                        <SelectItem value="INDIVIDUAL">Individual</SelectItem>
                        <SelectItem value="GOVERNMENT_ENTITY">
                          Government Entity
                        </SelectItem>
                        <SelectItem value="NON_PROFIT">Non-Profit</SelectItem>
                      </SelectContent>
                    </Select>
                    <FormMessage />
                  </FormItem>
                )}
              />

              <FormField
                control={form.control}
                name="riskRating"
                render={({ field }) => (
                  <FormItem>
                    <FormLabel>Risk Rating</FormLabel>
                    <Select onValueChange={field.onChange} value={field.value}>
                      <FormControl>
                        <SelectTrigger>
                          <SelectValue />
                        </SelectTrigger>
```

```tsx
"use client";

import { useState, useEffect, useCallback } from "react";
import { Button } from "@/components/ui/button";
import { Plus } from "lucide-react";
import { ExternalPartyTable } from "@/components/external-parties/external-party-table";
import { ExternalPartyForm } from "@/components/external-parties/external-party-form";
import {
  ExternalParty,
  SearchExternalPartiesParams,
  externalPartiesService,
} from "@/services/external-parties.service";
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
} from "@/components/ui/alert-dialog";
import { toast } from "sonner";

export default function ExternalPartiesPage() {
  const [parties, setParties] = useState<ExternalParty[]>([]);
  const [total, setTotal] = useState(0);
  const [isLoading, setIsLoading] = useState(true);
  const [filters, setFilters] = useState<SearchExternalPartiesParams>({
    take: 50,
    skip: 0,
  });

  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingParty, setEditingParty] = useState<ExternalParty | null>(null);
  const [deletingParty, setDeletingParty] = useState<ExternalParty | null>(
    null,
  );

  const fetchParties = useCallback(async () => {
    setIsLoading(true);
    try {
      const result = await externalPartiesService.search(filters);
      setParties(result.items);
      setTotal(result.total);
    } catch (error) {
      toast.error("Failed to load external parties");
    } finally {
      setIsLoading(false);
    }
  }, [filters]);

  useEffect(() => {
    fetchParties();
  }, [fetchParties]);

  const handleCreate = () => {
    setEditingParty(null);
    setIsFormOpen(true);
  };

  const handleEdit = (party: ExternalParty) => {
    setEditingParty(party);
    setIsFormOpen(true);
  };

  const handleDelete = async () => {
    if (!deletingParty) return;

    try {
      await externalPartiesService.delete(deletingParty.id);
      toast.success(`Deleted "${deletingParty.name}"`);
      setDeletingParty(null);
      fetchParties();
    } catch (error: any) {
      toast.error(error.message || "Failed to delete");
    }
  };

  const handleFormSuccess = () => {
    toast.success(editingParty ? "Party updated" : "Party created");
    fetchParties();
  };

  return (
    <div className="container mx-auto py-6 space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-bold">External Parties</h1>
          <p className="text-muted-foreground">
            Manage companies and individuals involved in gift and entertainment
            disclosures
          </p>
        </div>
        <Button onClick={handleCreate}>
          <Plus className="h-4 w-4 mr-2" />
          Add External Party
        </Button>
      </div>

      <ExternalPartyTable
        parties={parties}
        total={total}
        isLoading={isLoading}
        filters={filters}
        onFiltersChange={setFilters}
        onEdit={handleEdit}
        onDelete={(party) => setDeletingParty(party)}
      />

      <ExternalPartyForm
        open={isFormOpen}
        onClose={() => setIsFormOpen(false)}
        party={editingParty}
        onSuccess={handleFormSuccess}
      />

      <AlertDialog
        open={!!deletingParty}
        onOpenChange={() => setDeletingParty(null)}
      >
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Delete External Party</AlertDialogTitle>
            <AlertDialogDescription>
              Are you sure you want to delete "{deletingParty?.name}"? This
              action cannot be undone.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Cancel</AlertDialogCancel>
            <AlertDialogAction
              onClick={handleDelete}
              className="bg-destructive text-destructive-foreground"
            >
              Delete
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  );
}
```
