```prisma
/// ExternalParty represents companies, individuals, or organizations
/// involved in GT&E disclosures. Supports alias matching and government flagging.
model ExternalParty {
  id                String              @id @default(uuid())
  organizationId    String              @map("organization_id")
  name              String
  partyType         ExternalPartyType   @map("party_type")
  aliases           String[]            @default([])
  taxId             String?             @map("tax_id")
  dunsNumber        String?             @map("duns_number")
  website           String?
  isGovernment      Boolean             @default(false) @map("is_government")
  isSanctioned      Boolean             @default(false) @map("is_sanctioned")
  riskRating        RiskRating          @default(MEDIUM) @map("risk_rating")
  country           String?
  state             String?
  city              String?
  notes             String?
  createdAt         DateTime            @default(now()) @map("created_at")
  updatedAt         DateTime            @updatedAt @map("updated_at")
  createdById       String              @map("created_by_id")

  disclosureLinks   DisclosureExternalPartyLink[]
  createdBy         User                @relation("ExternalPartyCreatedBy", fields: [createdById], references: [id])
  organization      Organization        @relation(fields: [organizationId], references: [id], onDelete: Cascade)

  @@unique([organizationId, name])
  @@index([organizationId])
  @@index([organizationId, partyType])
  @@index([organizationId, isGovernment])
  @@index([organizationId, isSanctioned])
  @@index([organizationId, riskRating])
  @@map("external_parties")
}
```

4. Add DisclosureExternalPartyLink model:

```prisma
/// Links disclosures to external parties (many-to-many).
/// Tracks the relationship type (GIFT_GIVER, ENTERTAINMENT_HOST, etc.)
model DisclosureExternalPartyLink {
  id                String               @id @default(uuid())
  organizationId    String               @map("organization_id")
  disclosureId      String               @map("disclosure_id")
  externalPartyId   String               @map("external_party_id")
  relationship      String               // GIFT_GIVER, GIFT_RECEIVER, ENTERTAINMENT_HOST, TRAVEL_SPONSOR
  valueContributed  Decimal?             @map("value_contributed") @db.Decimal(12, 2)
  createdAt         DateTime             @default(now()) @map("created_at")

  disclosure        RiuDisclosureExtension @relation(fields: [disclosureId], references: [riuId], onDelete: Cascade)
  externalParty     ExternalParty          @relation(fields: [externalPartyId], references: [id], onDelete: Cascade)
  organization      Organization           @relation(fields: [organizationId], references: [id], onDelete: Cascade)

  @@unique([disclosureId, externalPartyId])
  @@index([organizationId, externalPartyId])
  @@index([organizationId, disclosureId])
  @@map("disclosure_external_party_links")
}
```

5. Add ExchangeRate model:

```prisma
/// Stores daily exchange rates for multi-currency GT&E threshold evaluation.
/// Rates are stored as USD per unit of foreign currency.
model ExchangeRate {
  id          String   @id @default(uuid())
  currency    String   // ISO 4217 code (EUR, GBP, JPY, etc.)
  rateDate    DateTime @map("rate_date") @db.Date
  rate        Decimal  @db.Decimal(18, 8) // High precision for FX
  source      String   // OPEN_EXCHANGE_RATES, MANUAL, etc.
  createdAt   DateTime @default(now()) @map("created_at")

  @@unique([currency, rateDate])
  @@index([currency, rateDate])
  @@map("exchange_rates")
}
```
