```typescript
import { Injectable, Logger, Inject } from "@nestjs/common";
import { CACHE_MANAGER } from "@nestjs/cache-manager";
import { Cache } from "cache-manager";
import { Decimal } from "@prisma/client/runtime/library";
import { PrismaService } from "../../prisma/prisma.service";

/**
 * CurrencyConversionService converts disclosure values to USD for threshold evaluation.
 *
 * Features (DISC-11):
 * - Convert any currency to USD using stored exchange rates
 * - Fall back to most recent rate if exact date not available
 * - Cache rates for 24 hours to minimize DB lookups
 * - Daily refresh from Open Exchange Rates API
 */
@Injectable()
export class CurrencyConversionService {
  private readonly logger = new Logger(CurrencyConversionService.name);
  private readonly CACHE_TTL = 86400; // 24 hours in seconds

  constructor(
    private readonly prisma: PrismaService,
    @Inject(CACHE_MANAGER) private cacheManager: Cache,
  ) {}

  /**
   * Converts an amount to USD using the exchange rate for the given date.
   * Falls back to most recent rate if exact date not available.
   *
   * @param amount - The amount to convert
   * @param fromCurrency - ISO 4217 currency code (EUR, GBP, etc.)
   * @param date - The date to use for the exchange rate
   * @returns Amount in USD
   */
  async convertToUsd(
    amount: number,
    fromCurrency: string,
    date: Date,
  ): Promise<number> {
    if (fromCurrency === "USD") {
      return amount;
    }

    const rate = await this.getRate(fromCurrency, date);
    if (!rate) {
      this.logger.warn(
        `No exchange rate found for ${fromCurrency} on ${date.toISOString()}, using 1:1 fallback`,
      );
      return amount; // Fallback: treat as USD
    }

    // Rate is stored as "1 USD = X foreign currency"
    // So to convert foreign to USD: amount / rate
    const usdAmount = amount / rate;
    this.logger.debug(
      `Converted ${amount} ${fromCurrency} to ${usdAmount.toFixed(2)} USD (rate: ${rate})`,
    );
    return usdAmount;
  }

  /**
   * Converts an amount from USD to a foreign currency.
   *
   * @param amountUsd - The amount in USD
   * @param toCurrency - Target ISO 4217 currency code
   * @param date - The date to use for the exchange rate
   * @returns Amount in target currency
   */
  async convertFromUsd(
    amountUsd: number,
    toCurrency: string,
    date: Date,
  ): Promise<number> {
    if (toCurrency === "USD") {
      return amountUsd;
    }

    const rate = await this.getRate(toCurrency, date);
    if (!rate) {
      this.logger.warn(
        `No exchange rate found for ${toCurrency} on ${date.toISOString()}, using 1:1 fallback`,
      );
      return amountUsd;
    }

    return amountUsd * rate;
  }

  /**
   * Gets the exchange rate for a currency on a given date.
   * Uses cached daily rates, falls back to most recent if date not found.
   *
   * @param currency - ISO 4217 currency code
   * @param date - The date for the rate lookup
   * @returns Exchange rate (1 USD = X foreign currency), or null if not found
   */
  async getRate(currency: string, date: Date): Promise<number | null> {
    const dateKey = date.toISOString().split("T")[0]; // YYYY-MM-DD
    const cacheKey = `exchange_rate:${currency}:${dateKey}`;

    // Check cache first
    const cached = await this.cacheManager.get<number>(cacheKey);
    if (cached !== undefined && cached !== null) {
      return cached;
    }

    // Look up from database - find rate for exact date or most recent before
    const rate = await this.prisma.exchangeRate.findFirst({
      where: {
        currency,
        rateDate: {
          lte: date,
        },
      },
      orderBy: { rateDate: "desc" },
    });

    if (rate) {
      const rateValue = Number(rate.rate);
      await this.cacheManager.set(cacheKey, rateValue, this.CACHE_TTL);
      return rateValue;
    }

    return null;
  }

  /**
   * Refreshes exchange rates from Open Exchange Rates API.
   * Called daily by the ExchangeRateProcessor.
   *
   * @returns Number of rates stored
   */
  async refreshRates(): Promise<number> {
    const appId = process.env.OPEN_EXCHANGE_RATES_APP_ID;

    if (!appId) {
      this.logger.error(
        "OPEN_EXCHANGE_RATES_APP_ID environment variable not set",
      );
      throw new Error("Missing OPEN_EXCHANGE_RATES_APP_ID configuration");
    }

    this.logger.log(
      "Refreshing exchange rates from Open Exchange Rates API...",
    );

    try {
      const response = await fetch(
        `https://openexchangerates.org/api/latest.json?app_id=${appId}`,
      );

      if (!response.ok) {
        const errorText = await response.text();
        throw new Error(`API returned ${response.status}: ${errorText}`);
      }

      const data = await response.json();
      const rates = data.rates as Record<string, number>;
      const today = new Date();
      today.setHours(0, 0, 0, 0);

      let storedCount = 0;

      // Store rates in database using upsert
      for (const [currency, rate] of Object.entries(rates)) {
        try {
          await this.prisma.exchangeRate.upsert({
            where: {
              currency_rateDate: {
                currency,
                rateDate: today,
              },
            },
            update: {
              rate: new Decimal(rate),
              source: "OPEN_EXCHANGE_RATES",
            },
            create: {
              currency,
              rate: new Decimal(rate),
              rateDate: today,
              source: "OPEN_EXCHANGE_RATES",
            },
          });
          storedCount++;

          // Clear cache for this currency/date
          const dateKey = today.toISOString().split("T")[0];
          await this.cacheManager.del(`exchange_rate:${currency}:${dateKey}`);
        } catch (error) {
          this.logger.warn(`Failed to store rate for ${currency}: ${error}`);
        }
      }

      this.logger.log(`Refreshed ${storedCount} exchange rates`);
      return storedCount;
    } catch (error) {
      this.logger.error(`Failed to refresh exchange rates: ${error}`);
      throw error;
    }
  }

  /**
   * Gets the most recent rate date in the database.
   * Useful for health checks and debugging.
   */
  async getMostRecentRateDate(): Promise<Date | null> {
    const latestRate = await this.prisma.exchangeRate.findFirst({
      orderBy: { rateDate: "desc" },
      select: { rateDate: true },
    });
    return latestRate?.rateDate ?? null;
  }

  /**
   * Gets all supported currencies (currencies with rates in the database).
   */
  async getSupportedCurrencies(): Promise<string[]> {
    const currencies = await this.prisma.exchangeRate.findMany({
      distinct: ["currency"],
      select: { currency: true },
      orderBy: { currency: "asc" },
    });
    return currencies.map((c) => c.currency);
  }
}
```

```typescript
import {
  Processor,
  Process,
  OnQueueCompleted,
  OnQueueFailed,
} from "@nestjs/bull";
import { Injectable, Logger } from "@nestjs/common";
import { Job } from "bull";
import { CurrencyConversionService } from "./currency-conversion.service";

export const EXCHANGE_RATE_QUEUE = "exchange-rates";

export interface ExchangeRateJobData {
  triggeredBy: "scheduler" | "manual";
}

/**
 * ExchangeRateProcessor handles scheduled exchange rate refreshes.
 *
 * Runs daily at 6 AM to fetch latest rates from Open Exchange Rates API.
 * Supports manual trigger for testing and recovery.
 */
@Injectable()
@Processor(EXCHANGE_RATE_QUEUE)
export class ExchangeRateProcessor {
  private readonly logger = new Logger(ExchangeRateProcessor.name);

  constructor(private readonly currencyService: CurrencyConversionService) {}

  @Process("refresh-daily-rates")
  async handleRefresh(job: Job<ExchangeRateJobData>): Promise<number> {
    this.logger.log(
      `Starting exchange rate refresh (triggered by: ${job.data.triggeredBy})`,
    );

    try {
      const count = await this.currencyService.refreshRates();
      this.logger.log(`Exchange rate refresh completed: ${count} rates stored`);
      return count;
    } catch (error) {
      this.logger.error(`Exchange rate refresh failed: ${error}`);
      throw error; // Let Bull handle retry
    }
  }

  @OnQueueCompleted()
  onCompleted(job: Job<ExchangeRateJobData>, result: number) {
    this.logger.log(
      `Exchange rate job ${job.id} completed: ${result} rates refreshed`,
    );
  }

  @OnQueueFailed()
  onFailed(job: Job<ExchangeRateJobData>, error: Error) {
    this.logger.error(
      `Exchange rate job ${job.id} failed: ${error.message}`,
      error.stack,
    );
  }
}
```

```typescript
import { Module, OnModuleInit, Logger } from "@nestjs/common";
import { BullModule, InjectQueue } from "@nestjs/bull";
import { CacheModule } from "@nestjs/cache-manager";
import { Queue } from "bull";
import { PrismaModule } from "../../prisma/prisma.module";
import { CurrencyConversionService } from "./currency-conversion.service";
import {
  ExchangeRateProcessor,
  EXCHANGE_RATE_QUEUE,
  ExchangeRateJobData,
} from "./exchange-rate.processor";

/**
 * GteModule provides GT&E-specific services:
 * - Currency conversion for multi-currency disclosures
 * - Exchange rate management with daily refresh
 *
 * NOTE: The exchange-rates queue is registered HERE in GteModule (not in JobsModule)
 * because the processor lives in this module. This follows NestJS best practices
 * of keeping queue registration co-located with its processor.
 *
 * Phase 47: External Parties & GT&E
 */
@Module({
  imports: [
    PrismaModule,
    CacheModule.register({
      ttl: 86400000, // 24 hours in ms
    }),
    // Queue registration is HERE - co-located with the processor
    BullModule.registerQueue({
      name: EXCHANGE_RATE_QUEUE,
    }),
  ],
  providers: [CurrencyConversionService, ExchangeRateProcessor],
  exports: [CurrencyConversionService],
})
export class GteModule implements OnModuleInit {
  private readonly logger = new Logger(GteModule.name);

  constructor(
    @InjectQueue(EXCHANGE_RATE_QUEUE)
    private readonly exchangeRateQueue: Queue<ExchangeRateJobData>,
  ) {}

  async onModuleInit() {
    // Remove any existing repeatable jobs to avoid duplicates
    const existingJobs = await this.exchangeRateQueue.getRepeatableJobs();
    for (const job of existingJobs) {
      await this.exchangeRateQueue.removeRepeatableByKey(job.key);
    }

    // Schedule daily refresh at 6 AM
    await this.exchangeRateQueue.add(
      "refresh-daily-rates",
      { triggeredBy: "scheduler" },
      {
        repeat: {
          cron: "0 6 * * *", // 6 AM daily
        },
        removeOnComplete: 100, // Keep last 100 completed jobs
        removeOnFail: 50, // Keep last 50 failed jobs
      },
    );

    this.logger.log("Exchange rate refresh scheduled for 6 AM daily");
  }

  /**
   * Manually triggers an exchange rate refresh.
   * Useful for initial setup, testing, or recovery.
   */
  async triggerManualRefresh(): Promise<void> {
    await this.exchangeRateQueue.add(
      "refresh-daily-rates",
      { triggeredBy: "manual" },
      {
        removeOnComplete: true,
      },
    );
    this.logger.log("Manual exchange rate refresh triggered");
  }
}
```
