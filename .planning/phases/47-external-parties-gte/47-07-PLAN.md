import { Injectable, Logger } from "@nestjs/common";
import { DisclosureType } from "@prisma/client";
import { PrismaService } from "../../prisma/prisma.service";
import { CurrencyConversionService } from "./currency-conversion.service";
import {
  GteAggregationService,
  GteAggregateResult,
} from "./gte-aggregation.service";

/**
 * Summary of GT&E activity for an external party.
 */
export interface ExternalPartySummary {
  externalPartyId: string;
  externalPartyName: string;
  partyType: string;
  isGovernment: boolean;
  isSanctioned: boolean;
  riskRating: string;
  totalValueUsd: number;
  disclosureCount: number;
  employeeCount: number;
  lastDisclosureDate: Date | null;
}

/**
 * Summary of GT&E activity for an employee.
 */
export interface EmployeeGteSummary {
  employeeId: string;
  employeeName: string;
  department: string | null;
  totalValueUsd: number;
  disclosureCount: number;
  externalPartyCount: number;
  topParties: Array<{
    partyId: string;
    partyName: string;
    totalValueUsd: number;
  }>;
}

/**
 * Time series data point for trending.
 */
export interface GteTrendPoint {
  period: string; // YYYY-MM or YYYY-Q#
  totalValueUsd: number;
  disclosureCount: number;
  giftCount: number;
  entertainmentCount: number;
  travelCount: number;
}

/**
 * Filters for GT&E reports.
 */
export interface GteReportFilters {
  startDate?: Date;
  endDate?: Date;
  employeeIds?: string[];
  externalPartyIds?: string[];
  partyTypes?: string[];
  disclosureTypes?: DisclosureType[];
  isGovernment?: boolean;
  minValueUsd?: number;
  limit?: number;
}

/**
 * GteReportingService provides aggregated GT&E reports for compliance dashboards.
 *
 * Features (DISC-10):
 * - Top external parties by total gift value
 * - Employee GT&E summaries
 * - Trend analysis over time
 * - High-risk relationship identification
 */
@Injectable()
export class GteReportingService {
  private readonly logger = new Logger(GteReportingService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly currencyService: CurrencyConversionService,
    private readonly aggregationService: GteAggregationService,
  ) {}

  /**
   * Gets top external parties by total GT&E value.
   */
  async getTopExternalParties(
    organizationId: string,
    filters: GteReportFilters = {},
  ): Promise<ExternalPartySummary[]> {
    const startDate = filters.startDate ?? this.defaultStartDate();
    const endDate = filters.endDate ?? new Date();

    // Query disclosure links with aggregation
    const partyStats = await this.prisma.disclosureExternalPartyLink.groupBy({
      by: ["externalPartyId"],
      where: {
        organizationId,
        createdAt: {
          gte: startDate,
          lte: endDate,
        },
        ...(filters.externalPartyIds && {
          externalPartyId: { in: filters.externalPartyIds },
        }),
        ...(filters.disclosureTypes && {
          disclosure: {
            disclosureType: { in: filters.disclosureTypes },
          },
        }),
        ...(filters.isGovernment !== undefined && {
          externalParty: {
            isGovernment: filters.isGovernment,
          },
        }),
      },
      _count: { id: true },
    });

    // Get full party details and calculate totals
    const summaries: ExternalPartySummary[] = [];

    for (const stat of partyStats) {
      const party = await this.prisma.externalParty.findUnique({
        where: { id: stat.externalPartyId },
      });

      if (!party) continue;

      // Get all disclosures for this party to calculate USD total
      const disclosures =
        await this.prisma.disclosureExternalPartyLink.findMany({
          where: {
            organizationId,
            externalPartyId: stat.externalPartyId,
            createdAt: {
              gte: startDate,
              lte: endDate,
            },
          },
          include: {
            disclosure: {
              select: {
                disclosureValue: true,
                disclosureCurrency: true,
                relatedPersonId: true,
                createdAt: true,
              },
            },
          },
        });

      // Calculate total in USD
      let totalValueUsd = 0;
      const employeeIds = new Set<string>();
      let lastDate: Date | null = null;

      for (const link of disclosures) {
        const value = link.valueContributed
          ? Number(link.valueContributed)
          : link.disclosure.disclosureValue
            ? Number(link.disclosure.disclosureValue)
            : 0;

        const currency = link.disclosure.disclosureCurrency ?? "USD";
        const usdValue = await this.currencyService.convertToUsd(
          value,
          currency,
          link.disclosure.createdAt,
        );
        totalValueUsd += usdValue;

        if (link.disclosure.relatedPersonId) {
          employeeIds.add(link.disclosure.relatedPersonId);
        }

        if (!lastDate || link.disclosure.createdAt > lastDate) {
          lastDate = link.disclosure.createdAt;
        }
      }

      if (filters.minValueUsd && totalValueUsd < filters.minValueUsd) {
        continue;
      }

      summaries.push({
        externalPartyId: party.id,
        externalPartyName: party.name,
        partyType: party.partyType,
        isGovernment: party.isGovernment,
        isSanctioned: party.isSanctioned,
        riskRating: party.riskRating,
        totalValueUsd,
        disclosureCount: stat._count.id,
        employeeCount: employeeIds.size,
        lastDisclosureDate: lastDate,
      });
    }

    // Sort by total value descending
    summaries.sort((a, b) => b.totalValueUsd - a.totalValueUsd);

    return summaries.slice(0, filters.limit ?? 20);
  }

  /**
   * Gets GT&E summaries for employees.
   */
  async getEmployeeGteSummaries(
    organizationId: string,
    filters: GteReportFilters = {},
  ): Promise<EmployeeGteSummary[]> {
    const startDate = filters.startDate ?? this.defaultStartDate();
    const endDate = filters.endDate ?? new Date();

    // Query disclosures grouped by employee
    const employeeStats = await this.prisma.riuDisclosureExtension.groupBy({
      by: ["relatedPersonId"],
      where: {
        organizationId,
        disclosureType: { in: filters.disclosureTypes ?? ["GIFT", "TRAVEL"] },
        createdAt: {
          gte: startDate,
          lte: endDate,
        },
        relatedPersonId: { not: null },
        ...(filters.employeeIds && {
          relatedPersonId: { in: filters.employeeIds },
        }),
      },
      _count: { riuId: true },
    });

    const summaries: EmployeeGteSummary[] = [];

    for (const stat of employeeStats) {
      if (!stat.relatedPersonId) continue;

      // Get employee details
      const employee = await this.prisma.employee.findFirst({
        where: { id: stat.relatedPersonId, organizationId },
        select: {
          firstName: true,
          lastName: true,
          department: true,
        },
      });

      if (!employee) continue;

      // Get aggregated data for this employee
      const partyAggregates =
        await this.aggregationService.aggregateAllPartiesForEmployee(
          stat.relatedPersonId,
          organizationId,
          {
            windowDays: Math.ceil(
              (endDate.getTime() - startDate.getTime()) / (1000 * 60 * 60 * 24),
            ),
            disclosureTypes: filters.disclosureTypes,
          },
        );

      const totalValueUsd = partyAggregates.reduce(
        (sum, p) => sum + p.totalValueUsd,
        0,
      );

      if (filters.minValueUsd && totalValueUsd < filters.minValueUsd) {
        continue;
      }

      summaries.push({
        employeeId: stat.relatedPersonId,
        employeeName: `${employee.firstName} ${employee.lastName}`,
        department: employee.department,
        totalValueUsd,
        disclosureCount: stat._count.riuId,
        externalPartyCount: partyAggregates.length,
        topParties: partyAggregates.slice(0, 3).map((p) => ({
          partyId: p.externalPartyId,
          partyName: p.externalPartyName,
          totalValueUsd: p.totalValueUsd,
        })),
      });
    }

    // Sort by total value descending
    summaries.sort((a, b) => b.totalValueUsd - a.totalValueUsd);

    return summaries.slice(0, filters.limit ?? 20);
  }

  /**
   * Gets GT&E trend data over time.
   */
  async getGteTrends(
    organizationId: string,
    period: "monthly" | "quarterly" = "monthly",
    filters: GteReportFilters = {},
  ): Promise<GteTrendPoint[]> {
    const startDate = filters.startDate ?? this.defaultStartDate();
    const endDate = filters.endDate ?? new Date();

    // Get all disclosures in range
    const disclosures = await this.prisma.riuDisclosureExtension.findMany({
      where: {
        organizationId,
        disclosureType: { in: filters.disclosureTypes ?? ["GIFT", "TRAVEL"] },
        createdAt: {
          gte: startDate,
          lte: endDate,
        },
      },
      select: {
        riuId: true,
        disclosureType: true,
        disclosureValue: true,
        disclosureCurrency: true,
        createdAt: true,
      },
      orderBy: { createdAt: "asc" },
    });

    // Group by period
    const periodMap = new Map<string, GteTrendPoint>();

    for (const disclosure of disclosures) {
      const periodKey = this.getPeriodKey(disclosure.createdAt, period);

      if (!periodMap.has(periodKey)) {
        periodMap.set(periodKey, {
          period: periodKey,
          totalValueUsd: 0,
          disclosureCount: 0,
          giftCount: 0,
          entertainmentCount: 0,
          travelCount: 0,
        });
      }

      const point = periodMap.get(periodKey)!;
      point.disclosureCount++;

      // Convert to USD
      const value = disclosure.disclosureValue
        ? Number(disclosure.disclosureValue)
        : 0;
      const currency = disclosure.disclosureCurrency ?? "USD";
      const usdValue = await this.currencyService.convertToUsd(
        value,
        currency,
        disclosure.createdAt,
      );
      point.totalValueUsd += usdValue;

      // Count by type
      switch (disclosure.disclosureType) {
        case "GIFT":
          point.giftCount++;
          break;
        case "TRAVEL":
          point.travelCount++;
          break;
      }
    }

    return Array.from(periodMap.values());
  }

  /**
   * Gets high-risk GT&E relationships.
   * Identifies employee-party pairs with high gift volumes.
   */
  async getHighRiskRelationships(
    organizationId: string,
    thresholdUsd: number = 500,
    filters: GteReportFilters = {},
  ): Promise<GteAggregateResult[]> {
    const startDate = filters.startDate ?? this.defaultStartDate();
    const endDate = filters.endDate ?? new Date();

    // Get all unique employee-party pairs
    const links = await this.prisma.disclosureExternalPartyLink.findMany({
      where: {
        organizationId,
        createdAt: {
          gte: startDate,
          lte: endDate,
        },
      },
      include: {
        disclosure: {
          select: { relatedPersonId: true },
        },
      },
      distinct: ["externalPartyId"],
    });

    const highRiskPairs: GteAggregateResult[] = [];

    for (const link of links) {
      if (!link.disclosure.relatedPersonId) continue;

      const aggregate = await this.aggregationService.aggregateByExternalParty(
        link.disclosure.relatedPersonId,
        link.externalPartyId,
        organizationId,
        {
          windowDays: Math.ceil(
            (endDate.getTime() - startDate.getTime()) / (1000 * 60 * 60 * 24),
          ),
        },
      );

      if (aggregate.totalValueUsd >= thresholdUsd) {
        highRiskPairs.push(aggregate);
      }
    }

    // Sort by total value descending
    highRiskPairs.sort((a, b) => b.totalValueUsd - a.totalValueUsd);

    return highRiskPairs.slice(0, filters.limit ?? 50);
  }

  /**
   * Gets the period key for a date.
   */
  private getPeriodKey(date: Date, period: "monthly" | "quarterly"): string {
    const year = date.getFullYear();
    const month = date.getMonth() + 1;

    if (period === "monthly") {
      return `${year}-${month.toString().padStart(2, "0")}`;
    } else {
      const quarter = Math.ceil(month / 3);
      return `${year}-Q${quarter}`;
    }
  }

  /**
   * Returns default start date (12 months ago).
   */
  private defaultStartDate(): Date {
    const date = new Date();
    date.setFullYear(date.getFullYear() - 1);
    return date;
  }
}

class GteReportsService {

  async getTopParties(

    filters: GteReportFilters = {},
  ): Promise<any[]> {
    const params = new URLSearchParams({ thresholdUsd: String(thresholdUsd) });
    if (filters.startDate) params.set("startDate", filters.startDate);
    if (filters.endDate) params.set("endDate", filters.endDate);
    if (filters.limit) params.set("limit", String(filters.limit));

    return apiClient.get(`${this.basePath}/high-risk?${params}`);
  }
}

export const gteReportsService = new GteReportsService();
