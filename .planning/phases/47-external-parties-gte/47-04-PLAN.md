import { Injectable, Logger } from "@nestjs/common";
import { DisclosureType } from "@prisma/client";
import { PrismaService } from "../../prisma/prisma.service";
import { CurrencyConversionService } from "./currency-conversion.service";

/**
 * Breakdown of a single disclosure in the aggregation.
 */
export interface DisclosureBreakdownItem {
  id: string;
  date: Date;
  valueUsd: number;
  originalValue: number;
  originalCurrency: string;
  disclosureType: string;
  relationship: string;
}

/**
 * Result of aggregating disclosures for an external party.
 */
export interface GteAggregateResult {
  externalPartyId: string;
  externalPartyName: string;
  employeeId: string;
  totalValueUsd: number;
  disclosureCount: number;
  windowStart: Date;
  windowEnd: Date;
  disclosures: DisclosureBreakdownItem[];
}

/**
 * Options for aggregation query.
 */
export interface GteAggregateOptions {
  windowDays?: number; // Default: 365
  disclosureTypes?: DisclosureType[]; // Default: GIFT, ENTERTAINMENT, TRAVEL
}

/**
 * GteAggregationService calculates cumulative gift values from external parties.
 *
 * Features (DISC-10):
 * - Aggregate by employee + external party over rolling window
 * - Convert all values to USD for consistent threshold evaluation
 * - Return detailed breakdown for audit trail
 */
@Injectable()
export class GteAggregationService {
  private readonly logger = new Logger(GteAggregationService.name);

  private readonly DEFAULT_WINDOW_DAYS = 365;
  private readonly DEFAULT_DISCLOSURE_TYPES: DisclosureType[] = [
    "GIFT",
    "TRAVEL",
  ];

  constructor(
    private readonly prisma: PrismaService,
    private readonly currencyService: CurrencyConversionService,
  ) {}

  /**
   * Aggregates gift values from a specific external party over a rolling window.
   *
   * @param employeeId - The employee receiving the gifts
   * @param externalPartyId - The external party giving the gifts
   * @param organizationId - The organization ID for tenant scoping
   * @param options - Window size and disclosure type filters
   * @returns Aggregated total in USD with breakdown
   */
  async aggregateByExternalParty(
    employeeId: string,
    externalPartyId: string,
    organizationId: string,
    options: GteAggregateOptions = {},
  ): Promise<GteAggregateResult> {
    const windowDays = options.windowDays ?? this.DEFAULT_WINDOW_DAYS;
    const disclosureTypes =
      options.disclosureTypes ?? this.DEFAULT_DISCLOSURE_TYPES;

    const windowEnd = new Date();
    const windowStart = new Date();
    windowStart.setDate(windowStart.getDate() - windowDays);

    // Query disclosures linked to this external party
    const disclosureLinks =
      await this.prisma.disclosureExternalPartyLink.findMany({
        where: {
          organizationId,
          externalPartyId,
          disclosure: {
            relatedPersonId: employeeId,
            disclosureType: { in: disclosureTypes },
            createdAt: {
              gte: windowStart,
              lte: windowEnd,
            },
          },
        },
        include: {
          disclosure: {
            select: {
              riuId: true,
              disclosureType: true,
              disclosureValue: true,
              disclosureCurrency: true,
              createdAt: true,
            },
          },
          externalParty: {
            select: {
              name: true,
            },
          },
        },
        orderBy: { createdAt: "desc" },
      });

    // Convert all values to USD and build breakdown
    let totalValueUsd = 0;
    const disclosures: DisclosureBreakdownItem[] = [];

    for (const link of disclosureLinks) {
      const originalValue = link.disclosure.disclosureValue
        ? Number(link.disclosure.disclosureValue)
        : 0;
      const originalCurrency = link.disclosure.disclosureCurrency ?? "USD";

      // Use the disclosure date for exchange rate lookup
      const valueUsd = await this.currencyService.convertToUsd(
        originalValue,
        originalCurrency,
        link.disclosure.createdAt,
      );

      // If link has specific value contributed, use that instead
      const effectiveValueUsd = link.valueContributed
        ? await this.currencyService.convertToUsd(
            Number(link.valueContributed),
            originalCurrency,
            link.disclosure.createdAt,
          )
        : valueUsd;

      totalValueUsd += effectiveValueUsd;

      disclosures.push({
        id: link.disclosure.riuId,
        date: link.disclosure.createdAt,
        valueUsd: effectiveValueUsd,
        originalValue,
        originalCurrency,
        disclosureType: link.disclosure.disclosureType,
        relationship: link.relationship,
      });
    }

    const externalPartyName =
      disclosureLinks[0]?.externalParty?.name ?? "Unknown";

    this.logger.debug(
      `Aggregated ${disclosures.length} disclosures for employee ${employeeId} ` +
        `from party ${externalPartyId}: $${totalValueUsd.toFixed(2)} USD`,
    );

    return {
      externalPartyId,
      externalPartyName,
      employeeId,
      totalValueUsd,
      disclosureCount: disclosures.length,
      windowStart,
      windowEnd,
      disclosures,
    };
  }

  /**
   * Aggregates gifts from ALL external parties for an employee.
   * Returns an array sorted by total value (highest first).
   *
   * @param employeeId - The employee receiving the gifts
   * @param organizationId - The organization ID for tenant scoping
   * @param options - Window size and disclosure type filters
   * @returns Array of aggregations per external party
   */
  async aggregateAllPartiesForEmployee(
    employeeId: string,
    organizationId: string,
    options: GteAggregateOptions = {},
  ): Promise<GteAggregateResult[]> {
    const windowDays = options.windowDays ?? this.DEFAULT_WINDOW_DAYS;
    const disclosureTypes =
      options.disclosureTypes ?? this.DEFAULT_DISCLOSURE_TYPES;

    const windowEnd = new Date();
    const windowStart = new Date();
    windowStart.setDate(windowStart.getDate() - windowDays);

    // Get all unique external parties linked to this employee's disclosures
    const links = await this.prisma.disclosureExternalPartyLink.findMany({
      where: {
        organizationId,
        disclosure: {
          relatedPersonId: employeeId,
          disclosureType: { in: disclosureTypes },
          createdAt: {
            gte: windowStart,
            lte: windowEnd,
          },
        },
      },
      select: {
        externalPartyId: true,
      },
      distinct: ["externalPartyId"],
    });

    const uniquePartyIds = links.map((l) => l.externalPartyId);

    // Aggregate for each party
    const results: GteAggregateResult[] = [];

    for (const partyId of uniquePartyIds) {
      const result = await this.aggregateByExternalParty(
        employeeId,
        partyId,
        organizationId,
        options,
      );
      results.push(result);
    }

    // Sort by total value descending
    results.sort((a, b) => b.totalValueUsd - a.totalValueUsd);

    return results;
  }

  /**
   * Checks if adding a new disclosure would exceed a threshold.
   * Useful for pre-submission warning.
   *
   * @param employeeId - The employee receiving the gift
   * @param externalPartyId - The external party giving the gift
   * @param newValueUsd - The value of the new gift in USD
   * @param thresholdUsd - The threshold to check against
   * @param organizationId - The organization ID
   * @param options - Window and type options
   * @returns Object indicating if threshold would be exceeded and by how much
   */
  async wouldExceedThreshold(
    employeeId: string,
    externalPartyId: string,
    newValueUsd: number,
    thresholdUsd: number,
    organizationId: string,
    options: GteAggregateOptions = {},
  ): Promise<{
    wouldExceed: boolean;
    currentTotal: number;
    projectedTotal: number;
    thresholdUsd: number;
    overage: number;
    breakdown: GteAggregateResult;
  }> {
    const aggregate = await this.aggregateByExternalParty(
      employeeId,
      externalPartyId,
      organizationId,
      options,
    );

    const projectedTotal = aggregate.totalValueUsd + newValueUsd;
    const wouldExceed = projectedTotal > thresholdUsd;

    return {
      wouldExceed,
      currentTotal: aggregate.totalValueUsd,
      projectedTotal,
      thresholdUsd,
      overage: wouldExceed ? projectedTotal - thresholdUsd : 0,
      breakdown: aggregate,
    };
  }
}

/**
 * Calculates aggregate value across a time window (RS.38).
 * Extended in Phase 47 to support external party aggregation (DISC-10).
 */
private async calculateAggregate(
  personId: string,
  organizationId: string,
  config: AggregateConfigDto,
  currentDisclosure: Record<string, unknown>,
): Promise<AggregateBreakdown> {
  // Calculate time window
  const windowEnd = new Date();
  let windowStart = new Date();

  if (config.timeWindow) {
    const { period, value, type } = config.timeWindow;
    if (type === 'rolling') {
      switch (period) {
        case 'days':
          windowStart.setDate(windowStart.getDate() - value);
          break;
        case 'months':
          windowStart.setMonth(windowStart.getMonth() - value);
          break;
        case 'years':
          windowStart.setFullYear(windowStart.getFullYear() - value);
          break;
      }
    } else {
      // Calendar year (year-to-date)
      windowStart = new Date(windowEnd.getFullYear(), 0, 1);
    }
  } else {
    // Default: rolling 12 months
    windowStart.setFullYear(windowStart.getFullYear() - 1);
  }

  // Check if we should use external party aggregation
  const externalPartyId = currentDisclosure.externalPartyId as string | undefined;

  if (config.dimensions?.includes('entity') && externalPartyId) {
    // Use GteAggregationService for external party aggregation
    const windowDays = Math.ceil(
      (windowEnd.getTime() - windowStart.getTime()) / (1000 * 60 * 60 * 24)
    );

    const aggregate = await this.gteAggregationService.aggregateByExternalParty(
      personId,
      externalPartyId,
      organizationId,
      { windowDays },
    );

    // Add current disclosure value
    const currentValue = this.extractNumericValue(currentDisclosure, 'disclosureValue');
    const totalValue = aggregate.totalValueUsd + currentValue;

    return {
      relatedDisclosures: aggregate.disclosures.map(d => ({
        id: d.id,
        date: d.date,
        value: d.valueUsd,
      })),
      totalValue,
      windowStart: aggregate.windowStart,
      windowEnd: aggregate.windowEnd,
    };
  }

  // Fall back to existing string-based entity matching (legacy support)
  const whereClause: Prisma.RiuDisclosureExtensionWhereInput = {
    organizationId,
    createdAt: {
      gte: windowStart,
      lte: windowEnd,
    },
  };

  // Add dimension filters
  if (config.dimensions?.includes('person')) {
    whereClause.relatedPersonId = personId;
  }

  if (
    config.dimensions?.includes('entity') &&
    currentDisclosure.relatedCompany
  ) {
    whereClause.relatedCompany = {
      contains: currentDisclosure.relatedCompany as string,
      mode: 'insensitive',
    };
  }

  // ... rest of existing implementation
  const relatedDisclosures = await this.prisma.riuDisclosureExtension.findMany({
    where: whereClause,
    select: {
      riuId: true,
      disclosureValue: true,
      createdAt: true,
    },
    orderBy: { createdAt: 'desc' },
  });

  // Calculate aggregate
  const func = config.aggregateFunction || 'SUM';
  const values: number[] = relatedDisclosures
    .map(d => (d.disclosureValue ? Number(d.disclosureValue) : 0))
    .filter(v => v > 0);

  // Add current disclosure value
  const currentValue = this.extractNumericValue(currentDisclosure, 'disclosureValue');
  values.push(currentValue);

  let totalValue = 0;
  switch (func) {
    case 'SUM':
      totalValue = values.reduce((sum: number, v: number) => sum + v, 0);
      break;
    case 'COUNT':
      totalValue = values.length;
      break;
    case 'AVG':
      totalValue = values.length > 0
        ? values.reduce((sum: number, v: number) => sum + v, 0) / values.length
        : 0;
      break;
    case 'MAX':
      totalValue = Math.max(...values, 0);
      break;
  }

  return {
    relatedDisclosures: relatedDisclosures.map(d => ({
      id: d.riuId,
      date: d.createdAt,
      value: d.disclosureValue ? Number(d.disclosureValue) : 0,
    })),
    totalValue,
    windowStart,
    windowEnd,
  };
}
