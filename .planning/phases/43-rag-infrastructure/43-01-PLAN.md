Add the DocumentEmbedding model to the Prisma schema. This model stores chunked document embeddings for semantic search.

Key design decisions per RESEARCH.md:

- NO RLS policies (explicit organizationId filtering in queries)
- 1024 dimensions for Voyage AI voyage-3 embeddings
- sourceType enum for different document sources
- chunkMetadata JSON for flexible metadata storage

Add before the MigrationJob model:

```prisma
/// SourceType for document embeddings - identifies the source entity type.
enum EmbeddingSourceType {
  POLICY_VERSION
  KNOWLEDGE_BASE
  CASE
  INVESTIGATION
}

/// DocumentEmbedding stores chunked document embeddings for semantic search.
/// CRITICAL: This table does NOT use RLS - all queries MUST filter by organizationId explicitly.
/// This is because pgvector similarity queries don't work reliably with RLS.
model DocumentEmbedding {
  id             String              @id @default(uuid())
  organizationId String              @map("organization_id")
  sourceType     EmbeddingSourceType @map("source_type")
  sourceId       String              @map("source_id")
  chunkIndex     Int                 @map("chunk_index")
  chunkText      String              @map("chunk_text")
  chunkMetadata  Json                @default("{}") @map("chunk_metadata")
  embedding      Unsupported("vector(1024)")
  modelVersion   String              @map("model_version")
  createdAt      DateTime            @default(now()) @map("created_at")

  organization   Organization        @relation(fields: [organizationId], references: [id], onDelete: Cascade)

  @@unique([sourceType, sourceId, chunkIndex])
  @@index([organizationId, sourceType])
  @@map("document_embeddings")
}
```

```sql
-- Enable pgvector extension for vector similarity search
CREATE EXTENSION IF NOT EXISTS vector;

-- CreateEnum
CREATE TYPE "EmbeddingSourceType" AS ENUM ('POLICY_VERSION', 'KNOWLEDGE_BASE', 'CASE', 'INVESTIGATION');

-- CreateTable
CREATE TABLE "document_embeddings" (
    "id" UUID NOT NULL DEFAULT gen_random_uuid(),
    "organization_id" UUID NOT NULL,
    "source_type" "EmbeddingSourceType" NOT NULL,
    "source_id" UUID NOT NULL,
    "chunk_index" INTEGER NOT NULL,
    "chunk_text" TEXT NOT NULL,
    "chunk_metadata" JSONB NOT NULL DEFAULT '{}',
    "embedding" vector(1024) NOT NULL,
    "model_version" VARCHAR(50) NOT NULL,
    "created_at" TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT "document_embeddings_pkey" PRIMARY KEY ("id")
);

-- CreateIndex: Unique constraint on source chunk
CREATE UNIQUE INDEX "document_embeddings_source_type_source_id_chunk_index_key"
ON "document_embeddings"("source_type", "source_id", "chunk_index");

-- CreateIndex: B-tree for tenant filtering (CRITICAL for query performance)
CREATE INDEX "document_embeddings_organization_id_source_type_idx"
ON "document_embeddings"("organization_id", "source_type");

-- CreateIndex: HNSW for vector similarity search
-- Parameters: m=16 (graph degree), ef_construction=64 (build quality)
CREATE INDEX "document_embeddings_embedding_idx"
ON "document_embeddings"
USING hnsw (embedding vector_cosine_ops)
WITH (m = 16, ef_construction = 64);

-- Add foreign key constraint
ALTER TABLE "document_embeddings"
ADD CONSTRAINT "document_embeddings_organization_id_fkey"
FOREIGN KEY ("organization_id") REFERENCES "organizations"("id") ON DELETE CASCADE ON UPDATE CASCADE;

-- NOTE: NO RLS POLICIES - this is intentional per CRIT-01
-- All vector queries MUST include explicit WHERE organization_id = $1 clause
```

After editing, run: `cd apps/backend && npx prisma migrate dev`
