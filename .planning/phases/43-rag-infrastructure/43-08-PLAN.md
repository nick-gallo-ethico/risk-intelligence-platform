import { Test, TestingModule } from "@nestjs/testing";
import { EmbeddingService } from "./embedding.service";
import { VoyageProvider } from "../providers/voyage.provider";

describe("EmbeddingService", () => {
  let service: EmbeddingService;
  let voyageProvider: jest.Mocked<VoyageProvider>;

  beforeEach(async () => {
    const mockVoyageProvider = {
      name: "voyage",
      dimensions: 1024,
      maxTokens: 32000,
      maxBatchSize: 128,
      isReady: jest.fn().mockReturnValue(true),
      embed: jest.fn(),
      embedSingle: jest.fn(),
    };

    const module: TestingModule = await Test.createTestingModule({
      providers: [
        EmbeddingService,
        { provide: VoyageProvider, useValue: mockVoyageProvider },
      ],
    }).compile();

    service = module.get<EmbeddingService>(EmbeddingService);
    voyageProvider = module.get(VoyageProvider);
  });

  describe("isReady", () => {
    it("should return true when provider is ready", () => {
      expect(service.isReady()).toBe(true);
    });

    it("should return false when provider is not ready", () => {
      voyageProvider.isReady.mockReturnValue(false);
      expect(service.isReady()).toBe(false);
    });
  });

  describe("embedBatch", () => {
    it("should embed multiple texts", async () => {
      const texts = ["text1", "text2"];
      const embeddings = [
        [0.1, 0.2],
        [0.3, 0.4],
      ];
      voyageProvider.embed.mockResolvedValue(embeddings);

      const result = await service.embedBatch(texts, "document");

      expect(voyageProvider.embed).toHaveBeenCalledWith(texts, "document");
      expect(result).toEqual(embeddings);
    });

    it("should handle empty input", async () => {
      const result = await service.embedBatch([], "document");
      expect(result).toEqual([]);
      expect(voyageProvider.embed).not.toHaveBeenCalled();
    });

    it("should batch large inputs", async () => {
      // Create more texts than batch size
      const texts = Array(200).fill("text");
      voyageProvider.embed.mockResolvedValue(Array(128).fill([0.1]));

      await service.embedBatch(texts, "document");

      // Should be called twice (128 + 72)
      expect(voyageProvider.embed).toHaveBeenCalledTimes(2);
    });
  });

  describe("embedQuery", () => {
    it("should embed with query input type", async () => {
      const embedding = [0.1, 0.2, 0.3];
      voyageProvider.embedSingle.mockResolvedValue(embedding);

      const result = await service.embedQuery("search query");

      expect(voyageProvider.embedSingle).toHaveBeenCalledWith(
        "search query",
        "query",
      );
      expect(result).toEqual(embedding);
    });
  });
});

import { Test, TestingModule } from "@nestjs/testing";
import { ChunkingService } from "./chunking.service";

describe("ChunkingService", () => {
  let service: ChunkingService;

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [ChunkingService],
    }).compile();

    service = module.get<ChunkingService>(ChunkingService);
  });

  describe("chunkPolicy", () => {
    it("should chunk content by sections", async () => {
      const content = `# Introduction

This is the introduction section.

## Section 1

This is section 1 content.

## Section 2

This is section 2 content.`;

      const result = await service.chunkPolicy(
        content,
        "policy-1",
        "version-1",
      );

      expect(result.strategy).toBe("section");
      expect(result.chunkCount).toBeGreaterThan(0);
      expect(
        result.chunks.every((c) => c.metadata.parentId === "policy-1"),
      ).toBe(true);
    });

    it("should fall back to recursive for content without sections", async () => {
      const content = "This is plain text without any section headers. ".repeat(
        100,
      );

      const result = await service.chunkPolicy(
        content,
        "policy-1",
        "version-1",
      );

      expect(result.strategy).toBe("recursive");
    });

    it("should split large sections", async () => {
      const largeSection = "## Large Section\n\n" + "Content ".repeat(1000);

      const result = await service.chunkPolicy(
        largeSection,
        "policy-1",
        "version-1",
        {
          chunkSize: 500,
        },
      );

      expect(result.chunkCount).toBeGreaterThan(1);
    });
  });

  describe("chunkCaseActivities", () => {
    it("should chunk activities with timestamps", async () => {
      const activities = [
        {
          type: "NOTE",
          content: "First note",
          timestamp: new Date("2024-01-01"),
        },
        {
          type: "STATUS_CHANGE",
          content: "Status changed",
          timestamp: new Date("2024-01-02"),
        },
      ];

      const result = await service.chunkCaseActivities(activities, "case-1");

      expect(result.strategy).toBe("activity");
      expect(result.chunkCount).toBe(2);
      expect(result.chunks[0].metadata.activityType).toBe("NOTE");
      expect(result.chunks[1].metadata.timestamp).toBeDefined();
    });
  });

  describe("chunkKnowledgeBase", () => {
    it("should chunk knowledge base content", async () => {
      const content = "Knowledge base content. ".repeat(200);

      const result = await service.chunkKnowledgeBase(content, "doc-1");

      expect(result.strategy).toBe("passage");
      expect(result.chunkCount).toBeGreaterThan(0);
      expect(result.chunks[0].metadata.parentId).toBe("doc-1");
    });
  });

  describe("estimateTokens", () => {
    it("should estimate tokens (1 token ≈ 4 chars)", () => {
      const text = "Hello world!"; // 12 chars
      expect(service.estimateTokens(text)).toBe(3);
    });
  });
});

      $executeRaw: jest.fn(),
      $queryRaw: jest.fn(),

      providers: [
        VectorStoreService,

    prisma = module.get(PrismaService);

    it("should delete existing and insert new chunks", async () => {
      prisma.$executeRaw.mockResolvedValue(0);

      const chunks = [
        {
          chunkIndex: 0,
          text: "Chunk 1",
          metadata: {},
          embedding: Array(1024).fill(0.1),
        },
      ];

      const result = await service.upsertChunks(
        "org-1",
        "POLICY_VERSION",
        "source-1",
        chunks,
        "voyage-3",
      );

      expect(result.chunksInserted).toBe(1);
      expect(prisma.$executeRaw).toHaveBeenCalled();
    });

    it("should handle empty chunks array", async () => {
      const result = await service.upsertChunks(
        "org-1",
        "POLICY_VERSION",
        "source-1",
        [],
        "voyage-3",
      );

      expect(result.chunksInserted).toBe(0);
      expect(result.chunksDeleted).toBe(0);
    });
  });

  describe("semanticSearch", () => {
    it("should return search results with similarity scores", async () => {
      const mockResults = [
        {
          id: "emb-1",
          source_type: "POLICY_VERSION",
          source_id: "policy-1",
          chunk_index: 0,
          chunk_text: "Test content",
          chunk_metadata: { sectionTitle: "Test" },
          distance: 0.2,
        },
      ];

        "org-1",
        Array(1024).fill(0.1),

      expect(result[0].sourceType).toBe("POLICY_VERSION");
    });

    it("should filter by minimum similarity", async () => {
      const mockResults = [
        { id: "emb-1", distance: 0.2, chunk_text: "High sim" },
        { id: "emb-2", distance: 1.5, chunk_text: "Low sim" },
      ].map((r) => ({
        ...r,
        source_type: "POLICY_VERSION",
        source_id: "p1",
        chunk_index: 0,
        chunk_metadata: {},
      }));
      prisma.$queryRaw.mockResolvedValue(mockResults);

        "org-1",
        Array(1024).fill(0.1),
        { minSimilarity: 0.5 },
      );

      expect(result.length).toBe(1);
      expect(result[0].id).toBe("emb-1");

        "POLICY_VERSION",
        "source-1",
      );

      expect(prisma.$executeRaw).toHaveBeenCalled();
    });
  });

import { PrismaClient } from "@prisma/client";
import { Logger } from "@nestjs/common";

const logger = new Logger("AcmePhase43Seed");

/**
 * Phase 43: RAG Infrastructure Demo Data
 *
 * Seeds knowledge base documents for the Acme Co. demo tenant.
 * These documents will be chunked and embedded when the embedding
 * service processes them.
 */
export async function seedAcmePhase43(prisma: PrismaClient): Promise<void> {
  logger.log("Seeding Phase 43: RAG Infrastructure demo data...");

  // Get Acme organization
  const acme = await prisma.organization.findFirst({
    where: { slug: "acme-co" },
  });

  if (!acme) {
    logger.warn("Acme organization not found, skipping Phase 43 seed");
    return;
  }

  // Get admin user for createdById
  const admin = await prisma.user.findFirst({
    where: { organizationId: acme.id, role: "SYSTEM_ADMIN" },
  });

  if (!admin) {
    logger.warn("Admin user not found, skipping Phase 43 seed");
    return;
  }

  // Create knowledge base documents (without actual file uploads)
  // These represent documents that would be uploaded and embedded
  const knowledgeBaseDocs = [
    {
      title: "Compliance Training Manual 2024",
      description: "Annual compliance training requirements and procedures",
      category: "Training",
      fileName: "compliance-training-manual-2024.pdf",
      fileType: "application/pdf",
      fileSize: 2048576, // 2MB
      storagePath: `knowledge-base/${acme.id}/compliance-training-manual-2024.pdf`,
      status: "EMBEDDED",
      chunkCount: 45,
    },
    {
      title: "Ethics Hotline FAQ",
      description:
        "Frequently asked questions about the ethics reporting process",
      category: "FAQ",
      fileName: "ethics-hotline-faq.docx",
      fileType:
        "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
      fileSize: 524288, // 512KB
      storagePath: `knowledge-base/${acme.id}/ethics-hotline-faq.docx`,
      status: "EMBEDDED",
      chunkCount: 12,
    },
    {
      title: "Gift and Entertainment Policy Guide",
      description:
        "Detailed guidance on gift acceptance thresholds and reporting",
      category: "Policy Guide",
      fileName: "gte-policy-guide.pdf",
      fileType: "application/pdf",
      fileSize: 1572864, // 1.5MB
      storagePath: `knowledge-base/${acme.id}/gte-policy-guide.pdf`,
      status: "EMBEDDED",
      chunkCount: 28,
    },
    {
      title: "Conflict of Interest Disclosure Instructions",
      description: "Step-by-step instructions for completing COI disclosures",
      category: "Instructions",
      fileName: "coi-disclosure-instructions.txt",
      fileType: "text/plain",
      fileSize: 32768, // 32KB
      storagePath: `knowledge-base/${acme.id}/coi-disclosure-instructions.txt`,
      status: "EMBEDDED",
      chunkCount: 8,
    },
    {
      title: "Healthcare Compliance Regulations Summary",
      description:
        "Summary of key healthcare compliance regulations (HIPAA, Stark, Anti-Kickback)",
      category: "Regulations",
      fileName: "healthcare-compliance-summary.pdf",
      fileType: "application/pdf",
      fileSize: 3145728, // 3MB
      storagePath: `knowledge-base/${acme.id}/healthcare-compliance-summary.pdf`,
      status: "EMBEDDED",
      chunkCount: 67,
    },
  ];

  // Insert knowledge base documents
  for (const doc of knowledgeBaseDocs) {
    await prisma.knowledgeBaseDocument.upsert({
      where: {
        id: `kb-${acme.id}-${doc.fileName.replace(/[^a-z0-9]/gi, "-")}`,
      },
      create: {
        id: `kb-${acme.id}-${doc.fileName.replace(/[^a-z0-9]/gi, "-")}`,
        organizationId: acme.id,
        ...doc,
        createdById: admin.id,
      },
      update: {
        ...doc,
      },
    });
  }

  logger.log(
    `Seeded ${knowledgeBaseDocs.length} knowledge base documents for Acme Co.`,
  );

  // Note: Actual embeddings would be generated by the embedding service
  // In a real scenario, these documents would be processed by the queue
  // For demo purposes, we mark them as EMBEDDED with mock chunk counts

  logger.log("Phase 43 seed complete");
}
