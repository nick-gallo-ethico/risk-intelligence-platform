/**
 * Result of a semantic similarity search.
 */
export interface SemanticSearchResult {
  /** Embedding record ID */
  id: string;

  /** Source document type */
  sourceType: string;

  /** Source document ID */
  sourceId: string;

  /** Chunk index within source */
  chunkIndex: number;

  /** Chunk text content */
  text: string;

  /** Chunk metadata */
  metadata: ChunkMetadata;

  /** Cosine distance (0 = identical, 2 = opposite) */
  distance: number;

  /** Similarity score (1 - distance/2, range 0-1) */
  similarity: number;
}

/**
 * Options for semantic search queries.
 */
export interface SemanticSearchOptions {
  /** Maximum results to return (default: 10) */
  limit?: number;

  /** Minimum similarity threshold (0-1, default: 0) */
  minSimilarity?: number;

  /** Filter by source types */
  sourceTypes?: string[];

  /** Filter by source IDs */
  sourceIds?: string[];
}

/**
 * Chunk with embedding for storage.
 */
export interface EmbeddedChunk {
  chunkIndex: number;
  text: string;
  metadata: ChunkMetadata;
  embedding: number[];
}

/**
 * Upsert result statistics.
 */
export interface UpsertResult {
  sourceType: string;
  sourceId: string;
  chunksInserted: number;
  chunksDeleted: number;
}

import { Injectable, Logger } from "@nestjs/common";
import { Prisma } from "@prisma/client";
import pgvector from "pgvector";
import { PrismaService } from "../../prisma/prisma.service";
import { ChunkMetadata } from "../dto/chunk.dto";
import {
  SemanticSearchResult,
  SemanticSearchOptions,
  EmbeddedChunk,
  UpsertResult,
} from "../dto/search.dto";

/**
 * VectorStoreService handles pgvector CRUD operations.
 *
 * CRITICAL: All queries MUST include explicit organizationId filter.
 * This table does NOT use RLS (per CRIT-01 in STATE.md).
 */
@Injectable()
export class VectorStoreService {
  private readonly logger = new Logger(VectorStoreService.name);

  constructor(private readonly prisma: PrismaService) {}

  /**
   * Store or replace chunks for a source document.
   *
   * Deletes existing chunks for the source, then inserts new ones.
   * This ensures clean replacement on re-embedding.
   */
  async upsertChunks(
    organizationId: string,
    sourceType: string,
    sourceId: string,
    chunks: EmbeddedChunk[],
    modelVersion: string,
  ): Promise<UpsertResult> {
    if (chunks.length === 0) {
      return {
        sourceType,
        sourceId,
        chunksInserted: 0,
        chunksDeleted: 0,
      };
    }

    // Delete existing chunks for this source
    const deleted = await this.deleteBySource(
      organizationId,
      sourceType,
      sourceId,
    );

    // Insert new chunks using raw SQL for vector support
    let inserted = 0;
    for (const chunk of chunks) {
      const vectorSql = pgvector.toSql(chunk.embedding);

      await this.prisma.$executeRaw`
        INSERT INTO document_embeddings (
          id, organization_id, source_type, source_id,
          chunk_index, chunk_text, chunk_metadata,
          embedding, model_version, created_at
        ) VALUES (
          gen_random_uuid(),
          ${organizationId}::uuid,
          ${sourceType}::"EmbeddingSourceType",
          ${sourceId}::uuid,
          ${chunk.chunkIndex},
          ${chunk.text},
          ${JSON.stringify(chunk.metadata)}::jsonb,
          ${vectorSql}::vector,
          ${modelVersion},
          now()
        )
      `;
      inserted++;
    }

    this.logger.debug(
      `Upserted ${inserted} chunks for ${sourceType}:${sourceId} (deleted ${deleted})`,
    );

    return {
      sourceType,
      sourceId,
      chunksInserted: inserted,
      chunksDeleted: deleted,
    };
  }

  /**
   * Delete all chunks for a source document.
   */
  async deleteBySource(
    organizationId: string,
    sourceType: string,
    sourceId: string,
  ): Promise<number> {
    // CRITICAL: Always filter by organizationId
    const result = await this.prisma.$executeRaw`
      DELETE FROM document_embeddings
      WHERE organization_id = ${organizationId}::uuid
        AND source_type = ${sourceType}::"EmbeddingSourceType"
        AND source_id = ${sourceId}::uuid
    `;

    return Number(result);
  }

  /**
   * Delete all embeddings for an organization.
   * Used for tenant cleanup.
   */
  async deleteByOrganization(organizationId: string): Promise<number> {
    const result = await this.prisma.$executeRaw`
      DELETE FROM document_embeddings
      WHERE organization_id = ${organizationId}::uuid
    `;

    this.logger.log(
      `Deleted all embeddings for organization ${organizationId}`,
    );
    return Number(result);
  }

  /**
   * Perform semantic similarity search.
   *
   * Uses pgvector cosine distance operator (<=>).
   * Returns results ordered by similarity (closest first).
   */
  async semanticSearch(
    organizationId: string,
    queryEmbedding: number[],
    options?: SemanticSearchOptions,
  ): Promise<SemanticSearchResult[]> {
    const limit = options?.limit || 10;
    const vectorSql = pgvector.toSql(queryEmbedding);

    // Build source type filter
    let sourceTypeFilter = Prisma.empty;
    if (options?.sourceTypes && options.sourceTypes.length > 0) {
      const types = options.sourceTypes.map((t) => `'${t}'`).join(",");
      sourceTypeFilter = Prisma.sql`AND source_type IN (${Prisma.raw(types)}::"EmbeddingSourceType")`;
    }

    // Build source ID filter
    let sourceIdFilter = Prisma.empty;
    if (options?.sourceIds && options.sourceIds.length > 0) {
      const ids = options.sourceIds.map((id) => `'${id}'::uuid`).join(",");
      sourceIdFilter = Prisma.sql`AND source_id IN (${Prisma.raw(ids)})`;
    }

    const results = await this.prisma.$queryRaw<
      Array<{
        id: string;
        source_type: string;
        source_id: string;
        chunk_index: number;
        chunk_text: string;
        chunk_metadata: Record<string, unknown>;
        distance: number;
      }>
    >`
      SELECT
        id,
        source_type,
        source_id,
        chunk_index,
        chunk_text,
        chunk_metadata,
        embedding <=> ${vectorSql}::vector AS distance
      FROM document_embeddings
      WHERE organization_id = ${organizationId}::uuid
        ${sourceTypeFilter}
        ${sourceIdFilter}
      ORDER BY embedding <=> ${vectorSql}::vector
      LIMIT ${limit}
    `;

    // Transform to SemanticSearchResult with similarity score
    const searchResults: SemanticSearchResult[] = results.map((row) => ({
      id: row.id,
      sourceType: row.source_type,
      sourceId: row.source_id,
      chunkIndex: row.chunk_index,
      text: row.chunk_text,
      metadata: row.chunk_metadata as ChunkMetadata,
      distance: row.distance,
      similarity: 1 - row.distance / 2, // Convert cosine distance to similarity (0-1)
    }));

    // Apply minimum similarity filter if specified
    if (options?.minSimilarity && options.minSimilarity > 0) {
      return searchResults.filter(
        (r) => r.similarity >= options.minSimilarity!,
      );
    }

    return searchResults;
  }

  /**
   * Get chunk count for an organization.
   * Useful for monitoring and quotas.
   */
  async getChunkCount(
    organizationId: string,
    sourceType?: string,
  ): Promise<number> {
    let sourceTypeFilter = Prisma.empty;
    if (sourceType) {
      sourceTypeFilter = Prisma.sql`AND source_type = ${sourceType}::"EmbeddingSourceType"`;
    }

    const result = await this.prisma.$queryRaw<Array<{ count: bigint }>>`
      SELECT COUNT(*) as count
      FROM document_embeddings
      WHERE organization_id = ${organizationId}::uuid
        ${sourceTypeFilter}
    `;

    return Number(result[0]?.count || 0);
  }

  /**
   * Get sources that have embeddings.
   * Useful for checking embedding status.
   */
  async getEmbeddedSources(
    organizationId: string,
    sourceType: string,
  ): Promise<Array<{ sourceId: string; chunkCount: number; createdAt: Date }>> {
    const results = await this.prisma.$queryRaw<
      Array<{ source_id: string; chunk_count: bigint; created_at: Date }>
    >`
      SELECT
        source_id,
        COUNT(*) as chunk_count,
        MIN(created_at) as created_at
      FROM document_embeddings
      WHERE organization_id = ${organizationId}::uuid
        AND source_type = ${sourceType}::"EmbeddingSourceType"
      GROUP BY source_id
    `;

    return results.map((r) => ({
      sourceId: r.source_id,
      chunkCount: Number(r.chunk_count),
      createdAt: r.created_at,
    }));
  }
}
