import {
  IsString,
  IsOptional,
  IsNumber,
  IsArray,
  IsEnum,
  Min,
  Max,
} from "class-validator";
import { ApiProperty, ApiPropertyOptional } from "@nestjs/swagger";
import { ChunkMetadata } from "../../embeddings/dto/chunk.dto";

/**
 * Source types for hybrid search.
 */
export enum HybridSearchSourceType {
  POLICY = "policy",
  KNOWLEDGE_BASE = "knowledge_base",
  CASE = "case",
  ALL = "all",
}

/**
 * Method that found the result.
 */
export type SearchMethod = "keyword" | "semantic" | "both";

/**
 * DTO for hybrid search request.
 */
export class HybridSearchRequestDto {
  @ApiProperty({ description: "Search query" })
  @IsString()
  query: string;

  @ApiPropertyOptional({
    description: "Source types to search",
    default: ["all"],
  })
  @IsArray()
  @IsOptional()
  sourceTypes?: HybridSearchSourceType[] = [HybridSearchSourceType.ALL];

  @ApiPropertyOptional({ description: "Maximum results", default: 10 })
  @IsNumber()
  @IsOptional()
  @Min(1)
  @Max(100)
  limit?: number = 10;

  @ApiPropertyOptional({
    description: "Minimum similarity score (0-1)",
    default: 0.3,
  })
  @IsNumber()
  @IsOptional()
  @Min(0)
  @Max(1)
  minScore?: number = 0.3;

  @ApiPropertyOptional({
    description: "Weight for keyword results (0-1)",
    default: 0.5,
  })
  @IsNumber()
  @IsOptional()
  @Min(0)
  @Max(1)
  keywordWeight?: number = 0.5;

  @ApiPropertyOptional({
    description: "Weight for semantic results (0-1)",
    default: 0.5,
  })
  @IsNumber()
  @IsOptional()
  @Min(0)
  @Max(1)
  semanticWeight?: number = 0.5;
}

/**
 * Individual hybrid search result.
 */
export interface HybridSearchResult {
  /** Unique identifier for the result */
  id: string;

  /** Source type (policy, knowledge_base, case) */
  sourceType: string;

  /** Source document ID */
  sourceId: string;

  /** Title or reference for display */
  title: string;

  /** Text content (excerpt or chunk) */
  text: string;

  /** Combined score from RRF (0-1) */
  score: number;

  /** Which search method found this result */
  method: SearchMethod;

  /** Keyword match score (if found via keyword) */
  keywordScore?: number;

  /** Semantic similarity score (if found via semantic) */
  semanticScore?: number;

  /** Additional metadata */
  metadata?: ChunkMetadata & {
    policyTitle?: string;
    caseReferenceNumber?: string;
    [key: string]: unknown;
  };
}

/**
 * Hybrid search response.
 */
export interface HybridSearchResponse {
  /** Search results */
  results: HybridSearchResult[];

  /** Total results before limit */
  total: number;

  /** Time taken in milliseconds */
  took: number;

  /** Query that was searched */
  query: string;

  /** Breakdown of results by method */
  methodCounts: {
    keyword: number;
    semantic: number;
    both: number;
  };
}

import { Injectable, Logger } from "@nestjs/common";
import { ElasticsearchService } from "@nestjs/elasticsearch";
import { EmbeddingService } from "../embeddings/services/embedding.service";
import { VectorStoreService } from "../embeddings/services/vector-store.service";
import { IndexingService } from "./indexing/indexing.service";
import {
  HybridSearchRequestDto,
  HybridSearchResponse,
  HybridSearchResult,
  HybridSearchSourceType,
  SearchMethod,
} from "./dto/hybrid-search.dto";
import { SemanticSearchResult } from "../embeddings/dto/search.dto";

/**
 * HybridSearchService combines Elasticsearch keyword search with
 * pgvector semantic search using Reciprocal Rank Fusion.
 *
 * Best for user-facing search where both exact matches
 * and conceptual understanding matter.
 */
@Injectable()
export class HybridSearchService {
  private readonly logger = new Logger(HybridSearchService.name);

  // RRF constant (standard value from literature)
  private readonly RRF_K = 60;

  constructor(
    private readonly esService: ElasticsearchService,
    private readonly embeddingService: EmbeddingService,
    private readonly vectorStore: VectorStoreService,
    private readonly indexingService: IndexingService,
  ) {}

  /**
   * Perform hybrid search combining keyword and semantic results.
   */
  async search(
    organizationId: string,
    request: HybridSearchRequestDto,
  ): Promise<HybridSearchResponse> {
    const startTime = Date.now();
    const limit = request.limit || 10;

    // Determine source types to search
    const sourceTypes = this.resolveSourceTypes(request.sourceTypes || []);

    // Run keyword and semantic searches in parallel
    const [keywordResults, semanticResults] = await Promise.all([
      this.keywordSearch(organizationId, request.query, sourceTypes, limit * 2),
      this.semanticSearch(
        organizationId,
        request.query,
        sourceTypes,
        limit * 2,
      ),
    ]);

    // Merge results using Reciprocal Rank Fusion
    const mergedResults = this.reciprocalRankFusion(
      keywordResults,
      semanticResults,
      request.keywordWeight || 0.5,
      request.semanticWeight || 0.5,
    );

    // Apply minimum score filter and limit
    const filteredResults = mergedResults
      .filter((r) => r.score >= (request.minScore || 0.3))
      .slice(0, limit);

    // Calculate method counts
    const methodCounts = {
      keyword: filteredResults.filter((r) => r.method === "keyword").length,
      semantic: filteredResults.filter((r) => r.method === "semantic").length,
      both: filteredResults.filter((r) => r.method === "both").length,
    };

    return {
      results: filteredResults,
      total: mergedResults.length,
      took: Date.now() - startTime,
      query: request.query,
      methodCounts,
    };
  }

  /**
   * Perform keyword search via Elasticsearch.
   */
  private async keywordSearch(
    organizationId: string,
    query: string,
    sourceTypes: string[],
    limit: number,
  ): Promise<Array<{ id: string; score: number; result: HybridSearchResult }>> {
    const results: Array<{
      id: string;
      score: number;
      result: HybridSearchResult;
    }> = [];

    // Search each source type index
    for (const sourceType of sourceTypes) {
      const indexName = this.getIndexName(organizationId, sourceType);

      try {
        const response = await this.esService.search({
          index: indexName,
          query: {
            bool: {
              should: [
                { match: { title: { query, boost: 2 } } },
                { match: { content: query } },
                { match: { text: query } },
                { match: { summary: query } },
                { match: { details: query } },
              ],
              minimum_should_match: 1,
            },
          },
          size: limit,
          _source: [
            "id",
            "title",
            "content",
            "text",
            "summary",
            "details",
            "referenceNumber",
          ],
        });

        for (const hit of response.hits.hits) {
          const source = hit._source as Record<string, unknown>;
          const id = `${sourceType}:${hit._id}`;

          results.push({
            id,
            score: hit._score || 0,
            result: {
              id,
              sourceType,
              sourceId: hit._id || "",
              title: (source.title ||
                source.referenceNumber ||
                "Untitled") as string,
              text: (source.content ||
                source.text ||
                source.summary ||
                source.details ||
                "") as string,
              score: 0, // Will be calculated by RRF
              method: "keyword" as SearchMethod,
              keywordScore: hit._score || 0,
            },
          });
        }
      } catch (error) {
        // Index might not exist, skip
        this.logger.debug(`Keyword search skipped for ${indexName}: ${error}`);
      }
    }

    // Sort by score descending
    return results.sort((a, b) => b.score - a.score);
  }

  /**
   * Perform semantic search via pgvector.
   */
  private async semanticSearch(
    organizationId: string,
    query: string,
    sourceTypes: string[],
    limit: number,
  ): Promise<Array<{ id: string; score: number; result: HybridSearchResult }>> {
    // Check if embedding service is ready
    if (!this.embeddingService.isReady()) {
      this.logger.debug(
        "Embedding service not ready, skipping semantic search",
      );
      return [];
    }

    try {
      // Generate query embedding
      const queryEmbedding = await this.embeddingService.embedQuery(query);

      // Map source types to embedding source types
      const embeddingSourceTypes = sourceTypes
        .map((t) => this.mapToEmbeddingSourceType(t))
        .filter(Boolean) as string[];

      // Perform semantic search
      const semanticResults = await this.vectorStore.semanticSearch(
        organizationId,
        queryEmbedding,
        {
          limit,
          sourceTypes:
            embeddingSourceTypes.length > 0 ? embeddingSourceTypes : undefined,
        },
      );

      return semanticResults.map((r) => {
        const id = `${r.sourceType}:${r.sourceId}:${r.chunkIndex}`;
        return {
          id,
          score: r.similarity,
          result: {
            id,
            sourceType: this.mapFromEmbeddingSourceType(r.sourceType),
            sourceId: r.sourceId,
            title: (r.metadata?.policyTitle ||
              r.metadata?.sectionTitle ||
              "Document") as string,
            text: r.text,
            score: 0, // Will be calculated by RRF
            method: "semantic" as SearchMethod,
            semanticScore: r.similarity,
            metadata: r.metadata,
          },
        };
      });
    } catch (error) {
      this.logger.error(`Semantic search failed: ${error}`);
      return [];
    }
  }

  /**
   * Merge results using Reciprocal Rank Fusion (RRF).
   *
   * RRF formula: score = 1 / (k + rank)
   * Where k is a constant (60) and rank is the position in results.
   *
   * When a result appears in both lists, its scores are combined
   * with optional weighting.
   */
  private reciprocalRankFusion(
    keywordResults: Array<{
      id: string;
      score: number;
      result: HybridSearchResult;
    }>,
    semanticResults: Array<{
      id: string;
      score: number;
      result: HybridSearchResult;
    }>,
    keywordWeight: number,
    semanticWeight: number,
  ): HybridSearchResult[] {
    const scores = new Map<
      string,
      {
        score: number;
        methods: Set<SearchMethod>;
        result: HybridSearchResult;
        keywordScore?: number;
        semanticScore?: number;
      }
    >();

    // Score keyword results by rank
    keywordResults.forEach((item, rank) => {
      const rrfScore = keywordWeight / (this.RRF_K + rank + 1);
      const existing = scores.get(item.id);

      if (existing) {
        existing.score += rrfScore;
        existing.methods.add("keyword");
        existing.keywordScore = item.result.keywordScore;
      } else {
        scores.set(item.id, {
          score: rrfScore,
          methods: new Set(["keyword"]),
          result: item.result,
          keywordScore: item.result.keywordScore,
        });
      }
    });

    // Score semantic results by rank
    semanticResults.forEach((item, rank) => {
      const rrfScore = semanticWeight / (this.RRF_K + rank + 1);

      // Use sourceId for matching (semantic results have chunk index in id)
      const baseId = `${item.result.sourceType}:${item.result.sourceId}`;
      const existingKey = Array.from(scores.keys()).find((k) =>
        k.startsWith(baseId),
      );

      if (existingKey) {
        const existing = scores.get(existingKey)!;
        existing.score += rrfScore;
        existing.methods.add("semantic");
        existing.semanticScore = item.result.semanticScore;
        // Prefer semantic result's text (more specific chunk)
        if (item.result.text && item.result.text.length > 0) {
          existing.result.text = item.result.text;
        }
      } else {
        scores.set(item.id, {
          score: rrfScore,
          methods: new Set(["semantic"]),
          result: item.result,
          semanticScore: item.result.semanticScore,
        });
      }
    });

    // Convert to sorted array
    return Array.from(scores.values())
      .sort((a, b) => b.score - a.score)
      .map(({ score, methods, result, keywordScore, semanticScore }) => ({
        ...result,
        score,
        method:
          methods.size === 2
            ? "both"
            : methods.has("keyword")
              ? "keyword"
              : "semantic",
        keywordScore,
        semanticScore,
      }));
  }

  /**
   * Resolve source types, expanding 'all' to specific types.
   */
  private resolveSourceTypes(sourceTypes: HybridSearchSourceType[]): string[] {
    if (
      sourceTypes.includes(HybridSearchSourceType.ALL) ||
      sourceTypes.length === 0
    ) {
      return ["policies", "knowledge_base", "cases"];
    }

    return sourceTypes.map((t) => {
      switch (t) {
        case HybridSearchSourceType.POLICY:
          return "policies";
        case HybridSearchSourceType.KNOWLEDGE_BASE:
          return "knowledge_base";
        case HybridSearchSourceType.CASE:
          return "cases";
        default:
          return t;
      }
    });
  }

  /**
   * Map display source type to ES index name.
   */
  private getIndexName(organizationId: string, sourceType: string): string {
    return this.indexingService.getIndexName(organizationId, sourceType);
  }

  /**
   * Map source type to embedding source type enum.
   */
  private mapToEmbeddingSourceType(sourceType: string): string | null {
    switch (sourceType) {
      case "policies":
        return "POLICY_VERSION";
      case "knowledge_base":
        return "KNOWLEDGE_BASE";
      case "cases":
        return "CASE";
      default:
        return null;
    }
  }

  /**
   * Map embedding source type back to display type.
   */
  private mapFromEmbeddingSourceType(sourceType: string): string {
    switch (sourceType) {
      case "POLICY_VERSION":
        return "policy";
      case "KNOWLEDGE_BASE":
        return "knowledge_base";
      case "CASE":
        return "case";
      default:
        return sourceType.toLowerCase();
    }
  }
}

    UnifiedSearchService,
    HybridSearchService,
  ],
  exports: [
    SearchService,
    IndexingService,
    UnifiedSearchService,
